1//! Type-state builder for constructing a [`Hopr`] node.
2//!
3//! The builder guides construction through a series of mandatory phases:
4//!
5//! 1. **Identity** — `HoprBuilder` → `HoprBuilder::with_identity`
6//! 2. **Configuration** — `HoprBuilderWithIdentity::with_config`
7//! 3. **Component factories** — chain API, graph, network, and cover-traffic
8//! 4. **Session server** (when the `session-server` feature is enabled) — `HoprBuilderConfigured::with_session_server`
9//! 5. **Build** — `build_edge` for an entry/exit node or `build_full` for a relay node with ticket management.
10//!
11//! # Example
12//!
13//! ```rust,ignore
14//! use hopr_lib::{config::HoprLibConfig, builder::{HoprBuilder, ChainKeypair, OffchainKeypair, Keypair}};
15//!
16//! let chain_key = ChainKeypair::random();
17//! let offchain_key = OffchainKeypair::random();
18//! let config = HoprLibConfig::default();
19//!
20//! let builder = HoprBuilder
21//!     .with_identity(&chain_key, &offchain_key)
22//!     .with_config(config)
23//!     .with_chain_api(|_ctx| { /* ... */ todo!() })
24//!     .with_graph(|_ctx| { /* ... */ todo!() })
25//!     .with_network(|_ctx| Box::pin(async { /* ... */ Ok(todo!()) }))
26//!     .with_cover_traffic(|_ctx| { /* ... */ todo!() });
27//! ```

322    /// Attaches a session server for handling incoming sessions.
323    ///
324    /// Eagerly spawns the server task and returns a [`HoprBuilderWithSession`]
325    /// that has the `build_edge` / `build_full` methods.
326    ///
327    /// Two tasks are spawned, not one, because the server is consulted at two different points in a
328    /// Session's life. `admit` is asked while the peer's request is still being negotiated and no
329    /// Session exists yet; `process` is handed the established byte-stream. Sharing one channel
330    /// would put a long-lived `process` in front of the admission decisions queued behind it.
331    #[cfg(feature = "session-server")]
332    pub fn with_session_server(
333        self,
334        // `Sync` on top of what `process` alone needed: the server is now driven from two
335        // independent tasks, and `admit` carries a default body, which `async_trait` can only make
336        // `Send` by requiring `&Self` to cross threads.
337        server: impl hopr_api::node::HoprSessionServer<Session = IncomingSession, Error: std::fmt::Display>
338        + Clone
339        + Send
340        + Sync
341        + 'static,
342    ) -> HoprBuilderWithSession<Chain, Graph, Net, Ct, Dump> {
343        let incoming_session_capacity = self.ctx.cfg.incoming_session_capacity.max(1);
344
345        let (session_tx, session_rx) = futures::channel::mpsc::channel::<IncomingSession>(incoming_session_capacity);
346
347        // Sized like the incoming-session channel: an admission request is in flight only for as
348        // long as one Start negotiation, so this bounds concurrent negotiations rather than live
349        // Sessions. Overflowing refuses the Session as busy rather than queueing its peer.
350        let (admission_tx, admission_rx) = futures::channel::mpsc::channel(incoming_session_capacity);
351
352        let admission_server = server.clone();
353        let admission_diag = hopr_utils::runtime::diagnostics::ConcurrentDiagnostics::new(
354            "session_admission_for_each_concurrent",
355            module_path!(),
356            file!(),
357            line!(),
358        );
359        let admission_handle = hopr_utils::spawn_as_abortable_named!(
360            "hopr_lib_session_admission",
361            admission_rx
362                .for_each_concurrent(
363                    // The transport releases its admission permit when its own deadline fires, not
364                    // when `admit` returns, so this pool needs its own bound or a stalled server
365                    // accumulates work without limit.
366                    Some(hopr_transport::session::START_SESSION_ADMISSION_CONCURRENCY),
367                    move |(request, reply): (_, hopr_transport::SessionAdmissionReply)| {
368                        let server = admission_server.clone();
369                        let admission_diag = admission_diag.clone();
370                        admission_diag.wrap(|| async move {
371                            // Concurrently rather than in sequence: the transport gives each request its
372                            // own deadline, so one slow decision must not spend another's.
373                            let decision = server.admit(request).await.map_err(|error| error.to_string());
374                            if reply.send(decision).is_err() {
375                                // The transport stopped waiting — its own timeout fired, or the peer
376                                // went away. Nothing to do but note it; it already refused the Session.
377                                tracing::debug!("session admission decided after the transport stopped waiting");
378                            }
379                        })
380                    }
381                )
382                .inspect(|_| tracing::warn!(
383                    task = %HoprLibProcess::SessionAdmission,
384                    "long-running background task finished"
385                ))
386        );
387
388        tracing::debug!(capacity = incoming_session_capacity, "spawning session server");
389        let session_diag = hopr_utils::runtime::diagnostics::ConcurrentDiagnostics::new(
390            "session_server_for_each_concurrent",
391            module_path!(),
392            file!(),
393            line!(),
394        );
395        let handle = hopr_utils::spawn_as_abortable_named!(
396            "hopr_lib_session_server",
397            session_rx
398                .for_each_concurrent(None, move |session| {
399                    let server = server.clone();
400                    let session_diag = session_diag.clone();
401                    session_diag.wrap(|| async move {
402                        let session_id = *session.session.id();
403                        match server.process(session).await {
404                            Ok(()) => tracing::debug!(?session_id, "session processed successfully"),
405                            Err(error) => {
406                                tracing::error!(?session_id, %error, "session processing failed")
407                            }
408                        }
409                    })
410                })
411                .inspect(|_| tracing::warn!(
412                    task = %HoprLibProcess::SessionServer,
413                    "long-running background task finished"
414                ))
415        );
416
417        HoprBuilderWithSession {
418            inner: self,
419            session_tx,
420            admission_tx,
421            session_handle: handle,
422            admission_handle,
423        }
424    }
425}

475/// Drains a HoprSocket reader, discarding all packets and logging throughput every ~60 seconds.
476/// Runs until the stream ends (sender side dropped).
477async fn drain_incoming_data<S: futures::Stream + Unpin>(mut reader: S) {
478    let mut received: u64 = 0;
479    let mut last_report = std::time::Instant::now();
480    while reader.next().await.is_some() {
481        received += 1;
482        if last_report.elapsed().as_secs() >= 60 {
483            tracing::info!(
484                received,
485                "incoming-data drain: unrelated packets discarded in last ~1 min"
486            );
487            received = 0;
488            last_report = std::time::Instant::now();
489        }
490    }
491}

527/// Validate configuration, prepare the reporter and assemble components before account readiness.
528async fn pre_build_inner<
529    Chain,
530    Graph,
531    Net,
532    Ct,
533    #[cfg(feature = "state-dump")] Dump: StateDumpFactory<Chain, Graph, Net>,
534>(
535    configured: HoprBuilderConfigured<Chain, Graph, Net, Ct, Dump>,
536    session_tx: futures::channel::mpsc::Sender<IncomingSession>,
537    admission_tx: Option<hopr_transport::SessionAdmissionSink>,
538    mut processes: AbortableList<HoprLibProcess>,
539) -> Result<PreHopr<Chain, Graph, Net, Ct>, HoprLibError>
540where
541    Chain: HoprChainApi + Clone + Send + Sync + 'static,
542    Graph: HoprGraphApi<HoprNodeId = hopr_api::OffchainPublicKey> + Clone + Send + Sync + 'static,
543    <Graph as hopr_api::graph::NetworkGraphTraverse>::Observed:
544        hopr_api::graph::traits::EdgeObservableRead + Send + 'static,
545    <Graph as hopr_api::graph::NetworkGraphWrite>::Observed: hopr_api::graph::traits::EdgeObservableWrite + Send,
546    Net: NetworkView + NetworkStreamControl + Send + Sync + Clone + 'static,
547    Ct: ProbingTrafficGeneration + CoverTrafficGeneration + Send + Sync + 'static,
548{
549    let peer_discovery_tx = Some(configured.peer_discovery_tx);
550    let ctx = configured.ctx;
551    // First, so it is in the log even when the configuration or the chain fails below.
552    log_startup(&ctx);
553    ctx.cfg.validate()?;
554    #[cfg(feature = "state-dump")]
555    let mut state_dump_factory = configured.state_dump_factory;
556    #[cfg(feature = "state-dump")]
557    state_dump_factory.prepare()?;
558
559    let network_tuning = crate::helpers::NetworkTuningReport::collect();
560    tracing::info!(?network_tuning, "OS network tuning specs");
561    for recommendation in network_tuning.recommendations() {
562        tracing::warn!("{recommendation}");
563    }
564
565    let chain_api = (configured
566        .chain_factory
567        .ok_or(HoprLibError::BuilderError("missing chain factory"))?)(&ctx);
568    let graph = (configured
569        .graph_factory
570        .ok_or(HoprLibError::BuilderError("missing graph factory"))?)(&ctx);
571    let (network, network_process) =
572        (configured
573            .network_factory
574            .ok_or(HoprLibError::BuilderError("missing network factory"))?)(ctx.clone())
575        .await?;
576    let cover_traffic = (configured
577        .ct_factory
578        .ok_or(HoprLibError::BuilderError("missing cover traffic factory"))?)(&ctx);
579
580    let (chain_id, transport_id) = (ctx.chain_key.clone(), ctx.packet_key.clone());
581
582    #[cfg(feature = "state-dump")]
583    let state_dump = state_dump_factory.create(
584        chain_api.clone(),
585        graph.clone(),
586        network.clone(),
587        ctx.cfg.safe_module.safe_address,
588        ctx.state_dump_sections,
589    );
590    #[cfg(feature = "state-dump")]
591    let transport_observer = state_dump.as_ref().and_then(|reporter| reporter.observer.clone());
592
593    let transport_api = HoprTransport::new(
594        (&chain_id, &transport_id),
595        chain_api.clone(),
596        graph.clone(),
597        vec![(&ctx.cfg.host).try_into().map_err(HoprLibError::TransportError)?],
598        ctx.cfg.protocol.clone(),
599    )
600    .map_err(HoprLibError::TransportError)?;
601    #[cfg(feature = "state-dump")]
602    let transport_api = match transport_observer {
603        Some(observer) => transport_api.with_observer(observer),
604        None => transport_api,
605    };
606
607    #[cfg(all(feature = "telemetry", not(test)))]
608    {
609        use hopr_api::types::primitive::traits::AsUnixTimestamp;
610        METRIC_PROCESS_START_TIME.set(std::time::SystemTime::now().as_unix_timestamp().as_secs_f64());
611        METRIC_HOPR_LIB_VERSION.set(
612            &[const_format::formatcp!("{}", constants::APP_VERSION)],
613            const_format::formatcp!(
614                "{}.{}",
615                env!("CARGO_PKG_VERSION_MAJOR"),
616                env!("CARGO_PKG_VERSION_MINOR")
617            )
618            .parse()
619            .unwrap_or(0.0),
620        );
621    }
622
623    let (mut new_tickets_tx, new_tickets_rx) = async_broadcast::broadcast(2048);
624    new_tickets_tx.set_await_active(false);
625    new_tickets_tx.set_overflow(true);
626
627    let (mut ssa_tx, ssa_rx) = async_broadcast::broadcast(2048);
628    ssa_tx.set_await_active(false);
629    ssa_tx.set_overflow(true);
630
631    let me_onchain = chain_id.public().to_address();
632
633    #[cfg(feature = "testing")]
634    tracing::warn!("!! FOR TESTING ONLY !! Node is running with some safety checks disabled!");
635
636    tracing::info!(
637        address = %me_onchain,
638        minimum_balance = %*SUGGESTED_NATIVE_BALANCE,
639        "node is not started, please fund this node",
640    );
641
642    tracing::info!(
643        suggested_minimum_balance = %*SUGGESTED_NATIVE_BALANCE,
644        "node about to start, checking for funds",
645    );
646    let funding_timeout = Duration::from_secs(200);
647    crate::helpers::wait_for_balance(*MIN_NATIVE_BALANCE, funding_timeout, me_onchain, &chain_api)
648        .await
649        .map_err(|_| {
650            HoprLibError::InsufficientFunds(format!(
651                "failed to fund the node within {} seconds",
652                funding_timeout.as_secs()
653            ))
654        })?;
655
656    tracing::info!("starting HOPR node...");
657    let balance: hopr_api::types::primitive::prelude::XDaiBalance =
658        chain_api.balance(me_onchain).await.map_err(HoprLibError::chain)?;
659    let minimum_balance = *constants::MIN_NATIVE_BALANCE;
660
661    tracing::info!(address = %me_onchain, %balance, %minimum_balance, "node information");
662
663    if balance.le(&minimum_balance) {
664        return Err(HoprLibError::InsufficientFunds(
665            "cannot start the node without a sufficiently funded wallet".into(),
666        ));
667    }
668
669    #[cfg(debug_assertions)]
670    let skip_protocol_checks = ctx.cfg.disable_protocol_checks;
671    #[cfg(not(debug_assertions))]
672    let skip_protocol_checks = false;
673
674    let network_min_ticket_price = chain_api.minimum_ticket_price().await.map_err(HoprLibError::chain)?;
675    let configured_ticket_price = ctx.cfg.protocol.packet.codec.outgoing_ticket_price;
676    if !skip_protocol_checks && configured_ticket_price.is_some_and(|c| c < network_min_ticket_price) {
677        return Err(HoprLibError::GeneralError(format!(
678            "configured outgoing ticket price < network minimum: {configured_ticket_price:?} < \
679             {network_min_ticket_price}"
680        )));
681    }
682
683    let network_min_win_prob = chain_api
684        .minimum_incoming_ticket_win_prob()
685        .await
686        .map_err(HoprLibError::chain)?;
687    let configured_win_prob = ctx.cfg.protocol.packet.codec.outgoing_win_prob;
688
689    if !skip_protocol_checks && configured_win_prob.is_some_and(|c| c.approx_cmp(&network_min_win_prob).is_lt()) {
690        return Err(HoprLibError::GeneralError(format!(
691            "configured outgoing win probability < network minimum: {configured_win_prob:?} < {network_min_win_prob}"
692        )));
693    }
694
695    // The network the node joined, with the minimums it enforces: what a log reader needs to judge
696    // ticket problems. Best effort, because the checks above have already passed.
697    log_network_information(
698        chain_api.chain_info().await,
699        &network_min_ticket_price,
700        &network_min_win_prob,
701    );
702
703    let safe_addr = ctx.cfg.safe_module.safe_address;
704    if me_onchain == safe_addr {
705        return Err(HoprLibError::GeneralError(
706            "cannot use self as staking safe address".into(),
707        ));
708    }
709
710    tracing::info!(%safe_addr, "registering safe with this node");
711    match chain_api.register_safe(&safe_addr).await {
712        Ok(awaiter) => {
713            awaiter.await.map_err(|error| {
714                tracing::error!(%safe_addr, %error, "safe registration failed");
715                HoprLibError::chain(error)
716            })?;
717            tracing::info!(%safe_addr, "safe successfully registered with this node");
718        }
719        Err(SafeRegistrationError::AlreadyRegistered(registered_safe)) => {
720            if registered_safe == safe_addr {
721                tracing::info!(%safe_addr, "this safe is already registered with this node");
722            } else {
723                tracing::error!(%safe_addr, %registered_safe, "node registered with different safe");
724                return Err(HoprLibError::GeneralError("node registered with different safe".into()));
725            }
726        }
727        Err(error) => {
728            tracing::error!(%safe_addr, %error, "safe registration failed");
729            return Err(HoprLibError::chain(error));
730        }
731    }
732
733    let multiaddresses_to_announce = if ctx.cfg.publish {
734        transport_api.announceable_multiaddresses()
735    } else {
736        Vec::new()
737    };
738
739    multiaddresses_to_announce
740        .iter()
741        .filter(|a| !is_public_address(a))
742        .for_each(|multi_addr| tracing::warn!(?multi_addr, "announcing private multiaddress"));
743
744    // Preflight for the on-chain announcement
745    if ctx.cfg.publish {
746        let key_binding_fee = chain_api.key_binding_fee().await.map_err(HoprLibError::chain)?;
747        if !key_binding_fee.is_zero() {
748            let safe_hopr_balance: hopr_api::types::primitive::prelude::HoprBalance =
749                chain_api.balance(safe_addr).await.map_err(HoprLibError::chain)?;
750            // The key-binding is a persistent on-chain state: if this node was already bound in a
751            // previous run, the fee has already been paid and we can announce even if the safe has
752            // since been drained below the fee. Hence we only wait for funds when the safe is
753            // underfunded *and* the node is not yet key-bound.
754            if safe_hopr_balance < key_binding_fee
755                && chain_api
756                    .await_key_binding(transport_id.public(), Duration::from_secs(1))
757                    .await
758                    .is_err()
759            {
760                tracing::warn!(
761                    %safe_addr,
762                    safe_balance = %safe_hopr_balance,
763                    required = %key_binding_fee,
764                    "the safe does not hold enough wxHOPR to pay the node announcement (key-binding) fee, waiting for \
765                     it to be funded before announcing the node",
766                );
767                let announce_timeout = Duration::from_secs(200);
768                crate::helpers::wait_for_balance(key_binding_fee, announce_timeout, safe_addr, &chain_api)
769                    .await
770                    .map_err(|_| {
771                        HoprLibError::InsufficientFunds(format!(
772                            "the safe {safe_addr} does not hold enough wxHOPR (needs at least {key_binding_fee}) to \
773                             announce the node on chain; fund the safe with wxHOPR and restart the node"
774                        ))
775                    })?;
776            }
777        }
778    }
779
780    let chain_api_clone = chain_api.clone();
781    let me_offchain = *transport_id.public();
782    let node_ready = spawn(async move {
783        chain_api_clone
784            .await_key_binding(&me_offchain, NODE_READY_TIMEOUT)
785            .await
786    });
787
788    tracing::info!(?multiaddresses_to_announce, "announcing node on chain");
789    match chain_api.announce(&multiaddresses_to_announce, &transport_id).await {
790        Ok(awaiter) => {
791            awaiter.await.map_err(|error| {
792                tracing::error!(
793                    ?multiaddresses_to_announce,
794                    %error,
795                    "node announcement failed; this is commonly caused by the safe not holding enough wxHOPR to pay \
796                     the key-binding fee",
797                );
798                HoprLibError::chain(error)
799            })?;
800            tracing::info!(?multiaddresses_to_announce, "node announced successfully");
801        }
802        Err(AnnouncementError::AlreadyAnnounced) => {
803            tracing::info!("node already announced on chain");
804        }
805        Err(error) => {
806            tracing::error!(%error, "failed to transmit node announcement");
807            return Err(HoprLibError::chain(error));
808        }
809    }
810
811    let this_node_account = node_ready
812        .await
813        .map_err(HoprLibError::other)?
814        .map_err(HoprLibError::chain)?;
815    if this_node_account.chain_addr != me_onchain || this_node_account.safe_address.is_none_or(|a| a != safe_addr) {
816        tracing::error!(%this_node_account, "account key-binding mismatch");
817        return Err(HoprLibError::GeneralError("account key-binding mismatch".into()));
818    }
819
820    tracing::info!(%this_node_account, "node account is ready");
821
822    // Network → graph event wiring (subscribe before transport starts)
823    {
824        let network_events = network.subscribe_network_events();
825        let graph_updater = graph.clone();
826        let health = transport_api.relayer_health();
827        let net = network.clone();
828        spawn(async move {
829            network_events
830                .for_each(|event| {
831                    let graph_updater = graph_updater.clone();
832                    let health = health.clone();
833                    let net = net.clone();
834                    async move {
835                        let (peer_id, connected) = match event {
836                            hopr_api::network::NetworkEvent::PeerConnected(p) => (p, true),
837                            hopr_api::network::NetworkEvent::PeerDisconnected(p) => (p, false),
838                        };
839                        if let Ok(opk) = hopr_api::OffchainPublicKey::from_peerid(&peer_id) {
840                            graph_updater.record_edge(hopr_api::graph::MeasurableEdge::<
841                                hopr_transport::NeighborTelemetry,
842                                hopr_transport::PathTelemetry,
843                            >::ConnectionStatus {
844                                peer: opk,
845                                connected,
846                            });
847
848                            // Only the position next to us depends on our own connection; the next
849                            // telemetry flush re-plans around it.
850                            if connected || !net.is_connected(&peer_id) {
851                                health.set_neighbour_connected(opk, connected);
852                            }
853                        } else {
854                            tracing::error!(%peer_id, "failed to convert peer ID to public key for graph update");
855                        }
856                    }
857                })
858                .await;
859        });
860    }
861
862    // Chain → graph event wiring
863    {
864        let chain_events = chain_api
865            .subscribe_with_state_sync([StateSyncOptions::PublicAccounts, StateSyncOptions::OpenedChannels])
866            .map_err(HoprLibError::chain)?;
867
868        let graph_updater = graph.clone();
869        let chain_reader = chain_api.clone();
870
871        let own_chain_addr = me_onchain;
872        let own_packet_key = *transport_id.public();
873
874        let ticket_price = Arc::new(parking_lot::RwLock::new(
875            chain_reader.minimum_ticket_price().await.unwrap_or_default(),
876        ));
877        let win_probability = Arc::new(parking_lot::RwLock::new(
878            chain_reader
879                .minimum_incoming_ticket_win_prob()
880                .await
881                .unwrap_or_default(),
882        ));
883
884        let proc = chain_wiring::process_chain_events(
885            chain_reader,
886            graph_updater,
887            transport_api.surb_store().clone(),
888            chain_events,
889            own_chain_addr,
890            own_packet_key,
891            ticket_price,
892            win_probability,
893            peer_discovery_tx,
894        )
895        .inspect(|_| {
896            tracing::warn!(
897                task = "chain-to-graph event wiring",
898                "long-running background task finished"
899            )
900        });
901        processes.insert(HoprLibProcess::ChannelEvents, hopr_utils::spawn_as_abortable!(proc));
902    }
903
904    #[cfg(feature = "state-dump")]
905    if let Some(reporter) = state_dump {
906        processes.insert(
907            HoprLibProcess::StateDump,
908            hopr_utils::spawn_as_abortable!(reporter.process),
909        );
910    }
911
912    Ok(PreHopr {
913        chain_id,
914        transport_id,
915        cfg: ctx.cfg,
916        state: Arc::new(AtomicHoprState::new(HoprState::Uninitialized)),
917        transport_api,
918        chain_api,
919        ticket_event_subscribers: (new_tickets_tx, new_tickets_rx.deactivate()),
920        pix_event_subscribers: (ssa_tx, ssa_rx.deactivate()),
921        processes,
922        session_tx,
923        admission_tx,
924        cover_traffic,
925        network,
926        network_process,
927    })
928}

1112        /// Builds a full (relay) [`Hopr`] node.
1113        pub async fn build_full<TMgr, TFact>(
1114            self,
1115            ticket_manager: TMgr,
1116            ticket_factory: TFact,
1117        ) -> Result<Hopr<Chain, Graph, Net, TMgr>, HoprLibError>
1118        where
1119            TMgr: TicketManagement + Clone + Send + Sync + 'static,
1120            TFact: TicketFactory + Clone + Send + Sync + 'static,
1121        {
1122            let (configured, session_tx, admission_tx, processes) = self.into_parts();
1123            let pre = pre_build_inner(configured, session_tx, admission_tx, processes).await?;
1124            let mut processes = pre.processes;
1125
1126            tracing::info!("starting ticket events processor");
1127            let (tickets_tx, tickets_rx) = bounded_sink_channel::<TicketEvent>(8192);
1128
1129            // Need to use DropAbortable, so that the receiver is dropped when aborted and no new items can be sent by the senders.
1130            let (tickets_rx_stream, tickets_handle) = hopr_utils::runtime::DropAbortable::new(tickets_rx);
1131
1132            processes.insert(HoprLibProcess::TicketEvents, tickets_handle);
1133            let new_ticket_tx = pre.ticket_event_subscribers.0.clone();
1134            let tmgr_clone = ticket_manager.clone();
1135            spawn(
1136                hopr_utils::runtime::diagnostics::instrument(
1137                    tickets_rx_stream
1138                    .for_each(move |event| {
1139                        if let TicketEvent::WinningTicket(ticket) = &event
1140                            && let Err(error) = tmgr_clone.insert_incoming_ticket(**ticket)
1141                        {
1142                            tracing::error!(%error, "failed to insert incoming ticket");
1143                        }
1144                        match new_ticket_tx.try_broadcast(event) {
1145                            // No subscriber is listening right now, so there is nothing to deliver to
1146                            Ok(_) | Err(async_broadcast::TrySendError::Inactive(_)) => {}
1147                            Err(error) => tracing::error!(%error, "failed to broadcast ticket event"),
1148                        }
1149                        futures::future::ready(())
1150                    })
1151                    .inspect(|_| {
1152                        tracing::warn!(task = %HoprLibProcess::TicketEvents, "long-running background task finished")
1153                    }),
1154                    "hopr_lib_ticket_events",
1155                    module_path!(),
1156                    file!(),
1157                    line!(),
1158                ),
1159            );
1160
1161            {
1162                let chain_for_neglect = pre.chain_api.clone();
1163                let tmgr_for_neglect = ticket_manager.clone();
1164                let events = pre.chain_api.subscribe().map_err(HoprLibError::chain)?;
1165                let (neglect_handle, neglect_reg) = hopr_utils::runtime::AbortHandle::new_pair();
1166                let neglect_task = futures::stream::Abortable::new(
1167                    events.filter_map(move |event| {
1168                        futures::future::ready(match event {
1169                            ChainEvent::ChannelClosed(ch) => Some(ch),
1170                            _ => None,
1171                        })
1172                    }),
1173                    neglect_reg,
1174                )
1175                .for_each(move |closed_channel| {
1176                    let chain = chain_for_neglect.clone();
1177                    let tmgr = tmgr_for_neglect.clone();
1178                    async move {
1179                        match closed_channel.direction(chain.me()) {
1180                            Some(ChannelDirection::Incoming) => {
1181                                match tmgr.neglect_tickets(closed_channel.get_id(), None) {
1182                                    Ok(neglected) if !neglected.is_empty() => {
1183                                        tracing::warn!(
1184                                            num_neglected = neglected.len(),
1185                                            %closed_channel,
1186                                            "tickets on incoming closed channel were neglected"
1187                                        );
1188                                    }
1189                                    Ok(_) => {}
1190                                    Err(error) => {
1191                                        tracing::error!(
1192                                            %error, %closed_channel,
1193                                            "failed to neglect tickets on closed channel"
1194                                        );
1195                                    }
1196                                }
1197                            }
1198                            Some(ChannelDirection::Outgoing) => {}
1199                            _ => {}
1200                        }
1201                    }
1202                })
1203                .inspect(|_| {
1204                    tracing::warn!(
1205                        task = %HoprLibProcess::ChannelClosureNeglect,
1206                        "channel closure ticket neglect task finished"
1207                    )
1208                });
1209                spawn(hopr_utils::runtime::diagnostics::instrument(
1210                    neglect_task,
1211                    "hopr_lib_channel_closure_neglect",
1212                    module_path!(),
1213                    file!(),
1214                    line!(),
1215                ));
1216                processes.insert(HoprLibProcess::ChannelClosureNeglect, neglect_handle);
1217            }
1218
1219            tracing::info!("starting transport for full (relay) node");
1220            let (socket, transport_processes) = pre
1221                .transport_api
1222                .run_relay(
1223                    pre.cover_traffic,
1224                    pre.network,
1225                    pre.network_process,
1226                    tickets_tx,
1227                    ticket_factory,
1228                    Some(BroadcastSenderSink(pre.pix_event_subscribers.0.clone())),
1229                    pre.session_tx,
1230                    pre.admission_tx,
1231                )
1232                .await?;
1233            // Drain unrelated packets to avoid missing blackhole
1234            spawn(drain_incoming_data(socket.reader()));
1235            processes.flat_map_extend_from(transport_processes, HoprLibProcess::Transport);
1236
1237            let hopr = Hopr {
1238                chain_id: NodeOnchainIdentity {
1239                    node_address: pre.chain_id.public().to_address(),
1240                    safe_address: pre.cfg.safe_module.safe_address,
1241                    module_address: pre.cfg.safe_module.module_address,
1242                },
1243                cfg: pre.cfg,
1244                state: pre.state.clone(),
1245                ticket_event_subscribers: pre.ticket_event_subscribers,
1246                pix_event_subscribers: pre.pix_event_subscribers,
1247                transport_id: pre.transport_id,
1248                transport_api: pre.transport_api,
1249                chain_api: pre.chain_api,
1250                processes,
1251                ticket_manager,
1252            };
1253
1254            hopr.state.store(HoprState::Running, std::sync::atomic::Ordering::Relaxed);
1255
1256            tracing::info!(
1257                id = %hopr.transport_id.public().to_peerid_str(),
1258                hopr_lib_version = constants::APP_VERSION,
1259                "FULL NODE STARTED AND RUNNING"
1260            );
1261
1262            #[cfg(all(feature = "telemetry", not(test)))]
1263            METRIC_HOPR_NODE_INFO.set(
1264                &[
1265                    &hopr.transport_id.public().to_peerid_str(),
1266                    &hopr.chain_id.node_address.to_string(),
1267                    &hopr.chain_id.safe_address.to_string(),
1268                    &hopr.chain_id.module_address.to_string(),
1269                ],
1270                1.0,
1271            );
1272

1274        }
