24/// Compact retirement state for one Session.
25///
26/// `highest_retired` alone is not enough: a later member of a batch may retire while an earlier
27/// member is still live. The exception sets retain only indices that still own a registration or
28/// were released before going live and may be retried. Live registrations are bounded by the
29/// Session layer's generation limit; a failed pre-live request closes a production Session, whose
30/// scope releases its retriable entries. The number of completed cycles is represented by one
31/// watermark rather than one allocation each.
32#[derive(Default)]
33struct RetirementFrontier {
34    highest_retired: Option<SsaIndex>,
35    registered: HashSet<SsaIndex>,
36    retriable: HashSet<SsaIndex>,
37    /// The most recently recovered cycle's bounded liveness tail.
38    ///
39    /// One slot is sufficient and security-critical: the Entry emits cycles in index order, so
40    /// only the immediate recovered predecessor can legitimately remain in the Exit's FIFO. A
41    /// newer recovered cycle replaces it, preventing old shares from buying service indefinitely
42    /// while keeping retirement memory constant over the Session's lifetime.
43    recovered_tail: Option<RecoveredSsaTail>,
44}
45
46impl RetirementFrontier {
47    fn is_retired(&self, index: SsaIndex) -> bool {
48        self.highest_retired.is_some_and(|highest| index <= highest)
49            && !self.registered.contains(&index)
50            && !self.retriable.contains(&index)
51    }
52
53    fn register(&mut self, index: SsaIndex) -> bool {
54        if self.is_retired(index) {
55            return false;
56        }
57        self.retriable.remove(&index);
58        self.registered.insert(index);
59        true
60    }
61
62    /// Releases a registration that never went live while leaving its index reusable.
63    fn abandon(&mut self, index: SsaIndex) {
64        if self.registered.remove(&index) {
65            self.retriable.insert(index);
66        }
67    }
68
69    /// Retires a known registration and raises the compact watermark.
70    ///
71    /// Returning `false` for an unknown index is load-bearing: allowing an arbitrary `retire_ssa`
72    /// call to raise the watermark would condemn legitimate lower indices that are registered
73    /// later.
74    fn retire(&mut self, index: SsaIndex) -> bool {
75        let was_registered = self.registered.remove(&index);
76        let was_retriable = self.retriable.remove(&index);
77        if !was_registered && !was_retriable {
78            return false;
79        }
80
81        self.highest_retired = Some(match self.highest_retired {
82            Some(highest) if highest >= index => highest,
83            _ => index,
84        });
85        true
86    }
87
88    fn retire_recovered(&mut self, index: SsaIndex, tail: RecoveredSsaTail) -> bool {
89        if !self.retire(index) {
90            return false;
91        }
92        debug_assert_eq!(index, tail.ssa_index());
93        self.recovered_tail = Some(tail);
94        true
95    }
96}

141/// Rejects an early-recovery fraction that is not a finite value in `0.0..=1.0`.
142///
143/// A `range` validator does not cover this. Every IEEE comparison against `NaN` is false, so `NaN`
144/// satisfies `min = 0.0, max = 1.0` untouched — and it does not stay inert downstream:
145/// [`SsaBuilder::check_early_threshold`](utils::SsaBuilder::check_early_threshold) computes
146/// `(threshold * num_polys).ceil() as usize`,
147/// and casting a `NaN` float to an integer saturates to zero in Rust. The early-recovery signal then
148/// fires on the very first reconstructed polynomial rather than at 85 % of them, which is the
149/// earliest instant the Exit can possibly ask for its next batch — and the Entry's successor gate,
150/// which exists to refuse exactly that, drops the request and leaves the Session to die on its
151/// commitment deadline.
152fn validate_early_recovery_threshold(threshold: f64) -> Result<(), validator::ValidationError> {
153    if threshold.is_finite() && (0.0..=1.0).contains(&threshold) {
154        Ok(())
155    } else {
156        Err(validator::ValidationError::new(
157            "early_recovery_threshold must be a finite fraction between 0.0 and 1.0",
158        ))
159    }
160}

163#[derive(Debug, Clone, Copy, PartialEq, smart_default::SmartDefault, validator::Validate)]
164pub struct SsaReconstructorConfig {
165    /// Time until the complete commitment to an SSA must be received.
166    ///
167    /// Default is 2 minutes.
168    #[default(Self::DEFAULT_INCOMPLETE_COMMITMENT_LIFETIME)]
169    pub incomplete_commitment_lifetime: std::time::Duration,
170    /// Maximum time an SSA cycle can go without progress before it is discarded.
171    ///
172    /// Measured from the last acknowledged share *anywhere* in the cycle, not per polynomial — see
173    /// `SsaCycle` for why that distinction is load-bearing. A cycle that is still being served
174    /// therefore never expires, whatever the line rate.
175    ///
176    /// Default is 30 minutes.
177    #[default(Self::DEFAULT_UNUSED_VERIFIER_LIFETIME)]
178    pub unused_verifier_lifetime: std::time::Duration,
179    /// Maximum number of peers that can be tracked simultaneously with unacknowledged shares.
180    ///
181    /// Default is 2000, minimum is 10.
182    ///
183    /// This is a per-*peer* fan-out bound, and it guards the opposite concentration to
184    /// [`max_awaiting_acks`](Self::max_awaiting_acks): traffic spread thinly across many
185    /// first-relayers. The two cannot both be saturated at once, which is why their product is not
186    /// the reconstructor's memory bound — see `PixReconstructorConfig` in `hopr-transport` for the
187    /// bound that is.
188    #[validate(range(min = 10))]
189    #[default(Self::DEFAULT_MAX_TRACKED_PEERS)]
190    pub max_tracked_peers: usize,
191    /// Maximum number of awaited acknowledgements to extract a single share.
192    ///
193    /// This corresponds to the maximum number of unacknowledged HOPR packets awaiting acknowledgement.
194    ///
195    /// Default is 1 000 000, must be at least 10 000.
196    ///
197    /// Sizes one inner cache **per peer**, so it must cover the concentrated case: every Session on
198    /// the node returning through a single first-relayer. At the operating point
199    /// `tests/memory_profile.rs` models that is ~542 000 entries, which is what makes a cap of this
200    /// order the right one rather than an oversight.
201    #[default(Self::DEFAULT_MAX_AWAITING_ACKS)]
202    #[validate(range(min = 10000))]
203    pub max_awaiting_acks: usize,
204    /// Maximum time an acknowledgement can be awaited before it is discarded.
205    ///
206    /// Default is 30 seconds.
207    ///
208    /// Multiplies the whole awaiting-ack buffer: the reachable state is the Exit's share-emission
209    /// rate times this window, so it — not either cap above — is the dial that actually sizes the
210    /// buffer.
211    #[default(Self::DEFAULT_MAX_ACK_AWAIT_TIME)]
212    pub max_ack_await_time: std::time::Duration,
213    /// Indicates whether to use batch verification algorithm for acknowledgements.
214    ///
215    /// Default is false.
216    ///
217    /// Batching only covers the acknowledgement *signature* check. While each share also cost a
218    /// `threshold`-term multi-scalar multiplication, that MSM dominated and the choice was
219    /// immaterial — measured 2.46 MiB/s batched against 2.50 MiB/s unbatched. Committing to the
220    /// constant term alone removed the MSM, and the batching overhead is no longer hidden by it:
221    /// the same benchmark then measures 50.2 MiB/s batched against 92.8 MiB/s unbatched, so
222    /// batching costs 46 % of the sustained rate.
223    ///
224    /// **Settled against the concurrent pipeline shape**, which is what the Exit actually runs and
225    /// what the sequential figures above could not speak to. `concurrent_quota_rate` on 48 cores at
226    /// production width, aggregate MiB/s of Session quota:
227    ///
228    /// | callers | unbatched | batched  |                              |
229    /// | ------- | --------- | -------- | ---------------------------- |
230    /// | 1       | **90.9**  | 46.4     | unbatched 1.96×              |
231    /// | 10      | 130.2     | 126.2    | tie — confidence intervals overlap |
232    /// | 48      | 139.8     | **152.6**| batched 1.09×                |
233    ///
234    /// So batching does eventually pay, but only *above* the concurrency the pipeline is
235    /// configured for: `DEFAULT_ACK_INPUT_CONCURRENCY` is 10, which is precisely the row where the
236    /// two are indistinguishable. `false` stays the default because it is far better at low
237    /// concurrency and no worse at the configured one, making it the safer choice across the range
238    /// an operator can set — not because batching is slower everywhere.
239    ///
240    /// Kept configurable for the operator who raises `ack_input_concurrency` well past its default,
241    /// where the last row says the choice flips.
242    ///
243    /// **Headroom is narrower than it looks.** A deployed Exit serves 10–30 clients at 16–20 Mbps,
244    /// so it absorbs 19–72 MiB/s; against the 130 MiB/s above that is **1.8× at the top of the
245    /// range**, not the 7.3× an earlier reading of these figures claimed. That claim came from
246    /// comparing against the benchmark suite's own model of production — 100 Sessions at 1.5 Mbps,
247    /// 18.75 MiB/s — which understates real per-Session rate by 13×. The rates here are also
248    /// measured at 4096 polynomials rather than the 512 used previously, which costs about 5 %.
249    #[default(Self::DEFAULT_USE_BATCH_VERIFICATION)]
250    pub use_batch_verification: bool,
251    /// Fraction of reconstructed polynomials at which to emit an early recovery
252    /// notification, triggering pipelined SSA request preparation.
253    ///
254    /// Range: 0.0..1.0. Default: 0.85.
255    ///
256    /// A reconstructor driving a real Session is additionally held to
257    /// [`MIN_EARLY_RECOVERY_THRESHOLD`](crate::MIN_EARLY_RECOVERY_THRESHOLD), the floor an Entry
258    /// assumes when it decides whether a successor `SsaRequest` has been earned. Anything below it is
259    /// unusable rather than merely aggressive — the request goes out early and the Entry refuses it.
260    /// The bound is not in this validator because it belongs to the pairing rather than to the
261    /// reconstructor: `hopr-transport-session` checks it where both halves are known.
262    #[default(Self::DEFAULT_EARLY_RECOVERY_THRESHOLD)]
263    #[validate(custom(function = "validate_early_recovery_threshold"))]
264    pub early_recovery_threshold: f64,
265    /// Ceiling on the total live state held in the awaiting-acknowledgement buffer, across every
266    /// peer, in bytes.
267    ///
268    /// This is the *global* bound; [`max_tracked_peers`](Self::max_tracked_peers) and
269    /// [`max_awaiting_acks`](Self::max_awaiting_acks) are per-dimension backstops and their product
270    /// is not one. See [`SsaReconstructor::insert_encrypted_share`] for why the product overstates
271    /// by roughly three thousandfold and why bounding it instead would lose shares.
272    ///
273    /// Enforced at insertion time rather than by validating a workload model. A model has to assume
274    /// a Session count and a packet rate; the node enforces neither — `maximum_managed_sessions`

277    /// Counting what is actually held is indifferent to all of that.
278    ///
279    /// Default is 1 GiB, which at [`AWAITING_ACK_ENTRY_BYTES`] is ~2.68 M shares in flight.
280    ///
281    /// The minimum is 25 600 B — 64 entries. That is a sanity floor, not a sizing recommendation:
282    /// its only job is to stop the budget rounding down to a handful of shares. Whether a given
283    /// value is *adequate* depends on the node's traffic, which is exactly the thing this design
284    /// stopped trying to predict, so the floor deliberately does not pretend to encode it. A node
285    /// configured near it will drop shares and say so.
286    #[default(Self::DEFAULT_MAX_ACK_BUFFER_BYTES)]
287    #[validate(range(min = 25_600))]
288    pub max_ack_buffer_bytes: usize,
289}

305/// Per-polynomial live heap a cycle holds that `size_of` cannot account for, in bytes.
306///
307/// [`peak_cycle_bytes`] reads the structural per-polynomial terms — the part builder, the shared
308/// surplus counter, and the commitment map entry retained behind them — straight from the types,
309/// so they track the curve and the structs without anyone maintaining a number. This covers what
310/// is left:
311///
312/// * `SsaBuilder`'s received-index set, sized at `num_polys` on construction;
313/// * the `Box<[_]>` header the part builders live in, and allocator rounding on both collections;
314/// * the acknowledgement-path residue a cycle drags along while it fills. `awaiting_acks` is bounded globally by
315///   [`max_ack_buffer_bytes`](SsaReconstructorConfig::max_ack_buffer_bytes) rather than per cycle, but `moka` applies
316///   removals on a later maintenance pass, so a cycle being fed at line rate carries a backlog of redeemed-but-not-yet
317///   -reclaimed entries. Charging a slice of it twice is deliberate: it is the difference between a model that is a
318///   ceiling and one that is merely a good estimate.
319///
320/// **Measured, not derived**, like [`AWAITING_ACK_ENTRY_BYTES`]:
321/// `exit_reconstructor_worst_case_share_order` in `tests/memory_profile.rs` prints the modelled and
322/// measured per-polynomial figures side by side with the share buffers subtracted, which is exactly
323/// this number. Run it to re-derive every figure below.
324///
325/// The arithmetic, at the time of writing and at the deployed dimensions:
326///
327/// * measured per polynomial, share buffers excluded: **~880–905 B**. A range because it is a range: the
328///   acknowledgement residue is a scheduling artefact, and two runs of the same test on the same machine gave 876 B and
329///   904 B. That variance is the reason this constant is set with headroom rather than at the measurement;
330/// * of which `size_of` accounts for **458 B**, leaving a **~420–450 B** residue the types cannot see;
331/// * this constant is **704 B**, ~1.6× that residue, so the model charges 458 + 704 = **1162 B** per polynomial — ~30 %
332///   above the measurement;
333/// * across a whole cycle that dilutes to ~6 %, because the share buffers are modelled exactly and dominate: **41.1
334///   MiB** modelled against **38.8 MiB** measured, of which 32.0 MiB is share buffers.
335///
336/// Both headroom figures are worth keeping straight, because they answer different questions: ~30 %
337/// is the margin on the term this constant actually sets, and ~6 % is the margin on the number the
338/// Session layer's live-cycle budget is denominated in.
339///
340/// Headroom rather than the measured figure for the variance above, and because understating this
341/// would let that budget be exceeded.
342pub const PART_BUILDER_OVERHEAD_BYTES: usize = 704;

344/// The per-polynomial share buffer [`peak_cycle_bytes`] models, in bytes.
345///
346/// `Vec` doubles from a minimum of four elements, so a buffer holding `threshold - 1` shares has a
347/// capacity of `threshold.next_power_of_two()` — up to twice what is stored. Modelling the
348/// allocation rather than the occupancy is what makes [`peak_cycle_bytes`] a ceiling.
349///
350/// Public because it is the one term of that model a consumer has reason to subtract back out:
351/// `tests/memory_profile.rs` reports the per-polynomial figure both with and without it, since the
352/// buffers dominate at the deployed dimensions and the residue is what
353/// [`PART_BUILDER_OVERHEAD_BYTES`] is set against. Restating the expression at the call site is what
354/// this exists to prevent — an external caller cannot reproduce it even in principle, because
355/// `CompletedShare` is crate-private and they would have to assume it is two scalars. True of the
356/// shipped specs, and not what the model reads.
357pub fn peak_share_buffer_bytes<S: PixSpec>(params: &PixParams) -> u64 {
358    (params.shares_per_poly() as u64)
359        .next_power_of_two()
360        .max(4)
361        .saturating_mul(size_of::<CompletedShare<S>>() as u64)
362}

364/// Worst-case live heap one Exit-side SSA cycle can hold, in bytes.
365///
366/// This is the figure the Session layer's live-cycle budget is denominated in, and it is
367/// deliberately the **adversarial** peak rather than the conforming one. Nothing constrains the
368/// order in which an Entry emits shares within a cycle: the shipped generator walks polynomials in
369/// blocks of [`SHARE_EMISSION_WINDOW`](crate::SHARE_EMISSION_WINDOW), so only that window holds
370/// share buffers at once and a conforming cycle peaks five times below this — 7.8 MiB against
371/// 39.1 MiB measured at the deployed dimensions — but a peer running anything else can drive *every*
372/// polynomial to one share short of its threshold and hold the lot.
373/// `SsaPartBuilder::release_verification_state` frees a buffer when its polynomial reconstructs; a
374/// polynomial one share short never reconstructs.
375///
376/// Not free to the peer, which is why this is a capacity bound and not a security one: a share only
377/// reaches the Exit on a return SURB the Exit itself spends, so reaching this peak costs a full
378/// quota deposit per cycle. What it bounds is an Exit selling more service than its memory can hold.
379///
380/// Excludes the awaiting-acknowledgement buffer, which is bounded separately and globally by
381/// [`max_ack_buffer_bytes`](SsaReconstructorConfig::max_ack_buffer_bytes) — counting it here would
382/// charge the same bytes twice.
383pub fn peak_cycle_bytes<S: PixSpec>(params: &PixParams) -> u64 {
384    let share_buffer = peak_share_buffer_bytes::<S>(params);
385
386    // Read from the types rather than restated as a constant, so the model follows the curve this
387    // node was built for and cannot drift when a field is added to either struct.
388    //
389    // The commitment map term is the one that surprises: `SsaCommitmentBuilder` is *drained* when
390    // the part builders are handed out, but `drain` does not return the allocation and the builder
391    // lives until `remove_cycle` — so a completed cycle still holds a bucket per polynomial. The
392    // factor of two is hashbrown's power-of-two bucket count at its 87.5 % load factor, and the
393    // extra byte is the control byte beside each one.
394    let per_poly = size_of::<parking_lot::Mutex<utils::SsaPartBuilder<S>>>() as u64
395        // This allocation is shared with the compact recovered tail. It used to be the
396        // `surplus_seen` field inside each part builder; keeping it explicit here preserves the
397        // admission bound when the counter moves behind an Arc to span recovery.
398        + size_of::<std::sync::atomic::AtomicUsize>() as u64
399        + 2 * (size_of::<(PolynomialIndex, PixGroup<S>)>() as u64 + 1)
400        + PART_BUILDER_OVERHEAD_BYTES as u64;
401
402    (params.polys_per_ssa() as u64).saturating_mul(per_poly + share_buffer)
403}

439/// Deferred acknowledgements for one cycle, drained in one shot when its part builders install.
440///
441/// Plain `Vec`s under one mutex rather than nested caches: a bucket is only ever appended to and
442/// then drained whole, so per-entry cache bookkeeping (and its ~200 B overhead per entry) buys
443/// nothing. The mutex serialises deferrals *within one cycle* only, and deferral is O(1) work off
444/// the steady-state path.
445#[derive(Default)]
446struct DeferredAcks {
447    by_poly: std::collections::HashMap<PolynomialIndex, Vec<DeferredAck>>,
448    /// Running sum of the `by_poly` lengths, maintained so the per-cycle cap is an O(1) check.
449    ///
450    /// Recomputing it would walk every sub-bucket inside the mutex on every deferral, and there
451    /// can be one sub-bucket per entry: filling a bucket to
452    /// [`MAX_DEFERRED_ACKS_PER_CYCLE`] would then cost ~33M map-entry visits, on the path every
453    /// acknowledgement takes during the commitment window. The invariant
454    /// `total == by_poly.values().map(Vec::len).sum()` has one increment site and one reset site,
455    /// both under this mutex.
456    total: usize,
457    /// Set by the one drain this bucket will ever get, in the same critical section as the take.
458    ///
459    /// A bucket is reachable through two routes: the `pending_acks` key, and an `Arc` a
460    /// [`defer_ack`](SsaReconstructor::defer_ack) already holds. The drain removes the first but
461    /// cannot revoke the second, so an append that lands after it would sit in a bucket nothing
462    /// will ever read again. This flag is how such an append notices — see
463    /// [`Deferral::Orphaned`].
464    drained: bool,
465}

469/// What became of an acknowledgement handed to [`defer_ack`](SsaReconstructor::defer_ack).
470///
471/// Fieldless — the caller still owns the acknowledgement, since [`DeferredAck`] is `Copy` — so this
472/// stays a discriminant rather than carrying 264 bytes back out of the critical section.
473#[derive(Clone, Copy)]
474enum Deferral {
475    /// Appended to a live bucket. The drain that installs the cycle will redeem it.
476    Buffered,
477    /// The bucket had already been drained, so no drain will come for this one. The caller must
478    /// redeem it inline; parking it would be silent loss.
479    Orphaned,
480    /// Over one of the two caps. Already warned about, and deliberately discarded.
481    Dropped,
482}

484/// Cap on deferred acknowledgements held for a single polynomial.
485///
486/// A conforming Entry emits `threshold + surplus` shares per polynomial — 96 at the default
487/// dimensions — across all return paths combined, so at the defaults this cannot be reached without
488/// the peer exceeding its own share budget. Anything above the cap is dropped rather than buffered.
489///
490/// It is *not* unreachable in general: both halves are a byte wide, so a conforming Entry may
491/// legitimately announce up to `255 + 255` and have its excess deferrals silently discarded. Both
492/// values now travel in [`PixParams`], so an Exit that cares *can* compare
493/// `shares_per_poly + surplus_shares` against this cap when it accepts a Session.
494///
495/// Deliberately not refused on that comparison, though, and the distinction matters. Only shares
496/// arriving *before* the cycle's commitments install are deferred at all — a prefix, not the whole
497/// emission — so `shares_per_poly + surplus_shares <= MAX_DEFERRED_ACKS_PER_POLYNOMIAL` is a
498/// sufficient condition for "no acknowledgement is ever dropped for want of room", not a necessary
499/// one for the dimensions to work. Enforcing it would reject legitimate re-splits: `4096 x 128` holds
500/// the profiled product exactly and sums to 160. The shipping defaults sum to 96 and are asserted to
501/// stay under the cap in `hopr-transport`'s config tests; wider dimensions lose the guarantee and
502/// fall back on the deferral window being short.
503///
504/// Public because it is observable behaviour, not an implementation detail: past the cap an
505/// acknowledgement is discarded, so anything measuring or exercising the deferral path has to stay
506/// underneath it or it silently measures the discard instead.
507pub const MAX_DEFERRED_ACKS_PER_POLYNOMIAL: usize = 128;
508
509/// Cap on deferred acknowledgements held across all polynomials of one cycle.
510///
511/// The per-polynomial cap alone leaves the cycle total at `num_polys × 128` — a million entries at
512/// production dimensions, which is no bound at all. This one is derived rather than chosen: the
513/// drain discards any acknowledgement whose share has already left `awaiting_acks`, and that cache
514/// expires entries after
515/// `max_ack_await_time` (30 s by default). An older deferral is therefore provably dead, so the
516/// ceiling only has to cover the shares one cycle can receive inside that window: ~181 shares/s at
517/// the deployed 1.5 Mbps per-Session cap, so ~5 400. 8192 leaves ~1.5× headroom and costs at most
518/// ~786 KB per cycle.
519///
520/// Public for the same reason as [`MAX_DEFERRED_ACKS_PER_POLYNOMIAL`].
521pub const MAX_DEFERRED_ACKS_PER_CYCLE: usize = 8192;

523/// Allows server-side reconstruction of SSAs.
524///
525/// There are 3 inputs that reconstructor is dependent on (in order):
526/// 1. SSA commitments from the Client (delivered via
527///    [`insert_coefficient_commitments`](ExitAcknowledgementShareProcessor::insert_coefficient_commitments))
528/// 2. Extraction of pending encrypted shares (added via
529///    [`insert_encrypted_share`](ExitAcknowledgementShareProcessor::insert_encrypted_share)
530/// 3. Decryption of pending encrypted shares via [`Acknowledgement`]s (via
531///    [`acknowledge_shares`](ExitAcknowledgementShareProcessor::acknowledge_shares))
532///
533/// It is able to track SSA for multiple different pseudonyms (Sessions).
534pub struct SsaReconstructor<S: PixSpec> {
535    commitment_builder: moka::sync::Cache<SsaId<S::Pseudonym>, std::sync::Arc<RegisteredCommitment<S>>>,
536    /// Post-commitment state of every live cycle: the part accumulator and all part builders,
537    /// published and reclaimed as one unit. See [`SsaCycle`].
538    ssa_cycles: moka::sync::Cache<SsaId<S::Pseudonym>, std::sync::Arc<RegisteredCycle<S>>>,
539    /// One compact retirement frontier per active Session pseudonym.
540    ///
541    /// A plain `HashMap` under a short synchronous lock is deliberate. Session scopes remove their
542    /// entries immediately on teardown; a cache would defer physical removal and put the same
543    /// turnover-rate assumption back into the memory bound this state replaces.
544    retirement_frontiers: parking_lot::Mutex<HashMap<S::Pseudonym, SharedRetirementFrontier>>,
545    awaiting_acks: moka::sync::Cache<OffchainPublicKey, EncryptedShareCache<S>>,
546    /// Acknowledgements that arrived before their cycle's part builders were installed, bucketed by
547    /// cycle and then by polynomial.
548    ///
549    /// ## Why bucketed at all
550    ///
551    /// The bucket key is exactly the thing whose arrival unblocks the entries inside it, so a
552    /// bucket is drained once, by the installation of its own cycle, and never scanned
553    /// speculatively. That is what keeps [`Self::acknowledge_shares`] free of retry work: it only
554    /// ever *appends* to a bucket.
555    ///
556    /// The original per-peer stash had to be re-scanned in full on every `acknowledge_shares` call,
557    /// because a per-peer key says nothing about which entries have become viable. That is
558    /// quadratic in the number of acks received while a cycle's commitments are in flight, and the
559    /// per-peer key aggregates across every Session sharing a first-relayer.
560    ///
561    /// ## Why keyed by cycle, sub-bucketed by polynomial
562    ///
563    /// Part builders are installed for a whole cycle at once, so a per-polynomial *key* no longer
564    /// buys selective draining — the drain would just walk every polynomial of the cycle. Keying by
565    /// cycle makes it one lookup. The per-polynomial sub-bucket is kept because its cap is what
566    /// bounds a misbehaving peer (see [`MAX_DEFERRED_ACKS_PER_POLYNOMIAL`]).
567    ///
568    /// The capacity unit is cycles, not polynomials. Keyed per polynomial it was `2 *
569    /// MAX_POLYS_PER_SSA` entries — which one cycle can exhaust on its own, so past roughly four
570    /// concurrent cycles node-wide, moka began LRU-evicting buckets and silently dropping real
571    /// shares. A size eviction here is share loss, so the headroom is deliberate and the
572    /// `max_ack_await_time` TTL is the operative bound.
573    pending_acks: moka::sync::Cache<SsaId<S::Pseudonym>, DeferredAckBucket>,
574    /// Resolutions produced by draining deferred-ack buckets at verifier-installation time, waiting
575    /// for either the transport notification stream or an acknowledgement batch to collect them.
576    ///
577    /// Draining happens on the commitment path (`insert_coefficient_commitments`), which is where
578    /// the verifier that unblocks the acks is installed. That deliberately keeps the share
579    /// verification off the acknowledgement hot path, but it also means the resolutions surface
580    /// on a synchronous worker. The buffer bridges that worker to the transport's asynchronous
581    /// resolution stream without blocking it or making recovered deposit keys depend on later
582    /// acknowledgement traffic.
583    ready_resolutions: parking_lot::Mutex<Vec<ShareResolution<S::Pseudonym, S::AddressPrivateKey>>>,
584    /// Length of [`ready_resolutions`](Self::ready_resolutions), so the common case (nothing to pick
585    /// up) costs one relaxed load instead of a mutex acquisition on every ack batch.
586    ready_resolutions_len: std::sync::atomic::AtomicUsize,
587    /// Non-blocking wake-up for the asynchronous consumer of [`ready_resolutions`](Self::ready_resolutions).
588    ///
589    /// The callback carries no data: resolutions remain owned by `ready_resolutions` until a
590    /// consumer atomically takes them, so wake-ups may safely coalesce and channel backpressure
591    /// cannot drop a recovered deposit key. `OnceLock` prevents two transport streams from racing
592    /// as competing designated consumers; `acknowledge_shares` remains an opportunistic fallback.
593    ready_resolution_notifier: std::sync::OnceLock<ReadyResolutionNotifier>,
594    /// Running estimate of the entries live in [`awaiting_acks`](Self::awaiting_acks), summed over
595    /// every peer, so the global budget costs one relaxed load per insertion.
596    ///
597    /// An *estimate*, and knowingly so — see [`Self::resync_ack_buffer`] for the one drift source
598    /// that cannot be listened for and what keeps it from accumulating.
599    ///
600    /// Behind an `Arc` because each peer's inner cache decrements it from an eviction listener, and
601    /// moka requires those to be `'static` — they cannot borrow the reconstructor that owns them.
602    ack_buffer_entries: std::sync::Arc<std::sync::atomic::AtomicUsize>,
603    /// [`max_ack_buffer_bytes`](SsaReconstructorConfig::max_ack_buffer_bytes) in entries, divided
604    /// once here rather than on every insertion.
605    max_ack_buffer_entries: usize,
606    /// When [`resync_ack_buffer`](Self::resync_ack_buffer) last ran, and the lock that keeps two
607    /// from running at once.
608    ///
609    /// `None` until the first run, so a buffer that saturates immediately is not made to wait out
610    /// an interval before its first ground-truth reading.
611    ack_buffer_resync: parking_lot::Mutex<Option<std::time::Instant>>,
612    cfg: SsaReconstructorConfig,
613}

615impl<S: PixSpec> Drop for SsaRetirementScope<S> {
616    fn drop(&mut self) {
617        let Some(reconstructor) = self.reconstructor.upgrade() else {
618            return;
619        };
620
621        let mut frontiers = reconstructor.retirement_frontiers.lock();
622        let owns_entry = frontiers
623            .get(&self.pseudonym)
624            .is_some_and(|current| std::sync::Arc::ptr_eq(current, &self.retirement));
625        if owns_entry {
626            frontiers.remove(&self.pseudonym);
627        }
628    }
629}

631/// Result of processing a single verified acknowledgement in the SSA reconstructor.
632///
633/// The counters behind [`SsaRecoveryProgress`] are updated by `process_verified_ack` itself, on the
634/// cycle it already holds, so these variants only have to say *whether* a snapshot is worth emitting
635/// — not what changed. That is why a duplicate, a surplus share and an unmatched acknowledgement all
636/// collapse into [`NoProgress`](Self::NoProgress): none of them moves a counter, so none of them can
637/// make a snapshot differ from the last one sent.
638enum ProcessedAckResult<S: PixSpec> {
639    /// Nothing to report: the acknowledgement matched no pending share, or the share was a duplicate
640    /// or was absorbed by an already-failed polynomial.
641    ///
642    /// A *surplus* share is not in this set. It advances nothing, but it is evidence the Entry is
643    /// still serving, and a consumer needs to be able to tell that from silence — see
644    /// [`Progressed`](Self::Progressed).
645    NoProgress,
646    /// The share is valid but its polynomial's verifier is not installed yet, so it cannot be
647    /// checked. Deferral, not failure: the ack is bucketed under this
648    /// [`SsaPolynomialId`] and retried once the verifier arrives.
649    VerifierNotReady(SsaPolynomialId<<S as PixSpec>::Pseudonym>),
650    /// A share landed on this cycle without finishing it. The snapshot says whether it advanced
651    /// recovery: `useful_shares` moves for a useful share, only `shares_seen` for a surplus one.
652    Progressed(SsaRecoveryProgress<<S as PixSpec>::Pseudonym>),
653    /// The share failed verification. Carries the SSA's aggregate fault total across all peers.
654    InvalidShare(SsaId<<S as PixSpec>::Pseudonym>, u64),
655    /// The early recovery threshold was crossed.
656    EarlyRecovery(SsaRecoveryProgress<<S as PixSpec>::Pseudonym>),
657    /// Full SSA was recovered. Carries the progress at the recovery instant; later buffered shares
658    /// may advance only the compact tail's liveness counter.
659    FullRecovery(
660        RecoveredSsa<<S as PixSpec>::Pseudonym, <S as PixSpec>::AddressPrivateKey>,
661        SsaRecoveryProgress<<S as PixSpec>::Pseudonym>,
662    ),
663}

665/// Merges a snapshot into the batch's pending set, keeping the furthest-along one per SSA.
666///
667/// Concurrent batches share a cycle's counters, so snapshots taken microseconds apart can be
668/// unordered. Keeping the maximum means one batch never reports its own SSA going backwards.
669///
670/// Merged **componentwise** rather than by choosing one snapshot whole, because the counters are two
671/// independent axes and a batch can advance either alone. A surplus share moves only `shares_seen`,
672/// so selecting on `useful_shares` discarded exactly the snapshots that say an Entry is still
673/// serving during a surplus run — the run this crate emits `Progressed` for in the first place, and
674/// the axis the Session layer's egress gate and idle deadline both read. Selecting on `shares_seen`
675/// instead would only move the same defect onto the other axis, since the two orderings disagree
676/// precisely when a snapshot is worth keeping.
677///
678/// Every field merged here is monotone over a cycle's life, so the componentwise maximum is itself a
679/// reachable state rather than an artefact. `target_useful_shares` is fixed at construction and
680/// `ssa_id` is the key, so neither is merged.
681fn record_progress<P: PartialEq>(acc: &mut Vec<SsaRecoveryProgress<P>>, snapshot: SsaRecoveryProgress<P>) {
682    match acc.iter_mut().find(|p| p.ssa_id == snapshot.ssa_id) {
683        Some(existing) => {
684            existing.useful_shares = existing.useful_shares.max(snapshot.useful_shares);
685            existing.shares_seen = existing.shares_seen.max(snapshot.shares_seen);
686            existing.recovered_polynomials = existing.recovered_polynomials.max(snapshot.recovered_polynomials);
687        }
688        None => acc.push(snapshot),
689    }
690}

692/// One SSA's fault observation, with the relayer that carried the offending share.
693type FaultObservation<P> = (Box<OffchainPublicKey>, SsaId<P>, u64);
694
695/// Merges a fault observation into the batch's pending set, keeping the highest total per SSA.
696///
697/// The relayer travels with the total rather than being filled in at emission time: a batch can also
698/// carry faults redeemed from deferral, and those were relayed by whoever held the share at the time
699/// — not by the peer whose acknowledgements are being processed now.
700fn record_fault<P: PartialEq>(acc: &mut Vec<FaultObservation<P>>, observation: FaultObservation<P>) {
701    match acc.iter_mut().find(|(_, id, _)| *id == observation.1) {
702        Some(existing) if existing.2 >= observation.2 => {}
703        Some(existing) => *existing = observation,
704        None => acc.push(observation),
705    }
706}

711        acc.push(resolution);

761    /// Creates a new SSA reconstructor from the given configuration.
762    ///
763    /// Fails if the configuration does not validate. Prefer this over [`Self::new`] anywhere the
764    /// configuration is assembled at runtime — a config built programmatically or read from a file
765    /// is input, not a constant, and turning it into a panic makes it un-handleable by the caller.
766    pub fn try_new(cfg: SsaReconstructorConfig) -> Result<Self, PixError<S::Pseudonym>> {
767        cfg.validate()?;
768        Ok(Self {
769            commitment_builder: moka::sync::Cache::builder()
770                .time_to_idle(cfg.incomplete_commitment_lifetime)
771                .build(),
772            // Indispensable per-cycle state: never size-evicted. Built without a `max_capacity`,
773            // so only `time_to_idle` reclaims it. Active removal happens via `remove_cycle` on
774            // full recovery and `retire_ssa` on session teardown; the TTL is the backstop.
775            // A hard capacity would silently strand a live cycle.
776            //
777            // The idle timer is refreshed by an acknowledgement for *any* polynomial of the cycle,
778            // because the whole cycle is one entry. That is what makes reclamation correct at any
779            // line rate — see `SsaCycle`.
780            ssa_cycles: moka::sync::Cache::builder()
781                .time_to_idle(cfg.unused_verifier_lifetime)
782                .build(),
783            retirement_frontiers: parking_lot::Mutex::new(HashMap::new()),
784            awaiting_acks: moka::sync::CacheBuilder::new(cfg.max_tracked_peers as u64)
785                .time_to_idle(cfg.max_ack_await_time)
786                // Dropping a peer entry drops its whole inner cache, and dropping a moka handle
787                // does not run that cache's eviction listener — so without this, every entry the
788                // peer still held would stay counted against the global budget forever. Invalidating
789                // routes them through the inner listener instead.
790                //
791                // `run_pending_tasks()` is deliberately not called here: it is unbounded work on
792                // whichever thread happened to trigger maintenance. That leaves the invalidation
793                // best-effort, which is precisely why `resync_ack_buffer` exists — this narrows the
794                // drift, it does not close it.
795                .eviction_listener(|_, shares: EncryptedShareCache<S>, _| shares.invalidate_all())
796                .build(),
797            // One bucket per cycle, expiring on the same clock as the shares it belongs to: an ack
798            // whose share has left `awaiting_acks` can never be used again, so there is nothing to
799            // keep. `time_to_live`, not idle — appending to a bucket must not extend the life of
800            // entries already in it.
801            //
802            // The capacity is in cycles. `MAX_POLYS_PER_SSA` is reused only as a generous count of
803            // concurrently deferring cycles; a size eviction here is share loss, so it is
804            // deliberately far above the pipelining factor and the TTL is the operative bound.
805            pending_acks: moka::sync::CacheBuilder::new(MAX_POLYS_PER_SSA as u64)
806                .time_to_live(cfg.max_ack_await_time)
807                .build(),
808            ready_resolutions: parking_lot::Mutex::new(Vec::new()),
809            ready_resolutions_len: std::sync::atomic::AtomicUsize::new(0),
810            ready_resolution_notifier: std::sync::OnceLock::new(),
811            ack_buffer_entries: std::sync::Arc::new(std::sync::atomic::AtomicUsize::new(0)),
812            // At least one, so a budget rounded below one entry refuses everything rather than
813            // dividing to zero and admitting everything.
814            max_ack_buffer_entries: (cfg.max_ack_buffer_bytes / AWAITING_ACK_ENTRY_BYTES).max(1),
815            ack_buffer_resync: parking_lot::Mutex::new(None),
816            cfg,
817        })
818    }

868    /// Removes the heavyweight reconstructor state for a single SSA cycle.
869    ///
870    /// A successfully recovered cycle's compact accounting tail is owned by the Session retirement
871    /// frontier, not by these caches, and intentionally survives this operation while paid SURBs
872    /// drain.
873    ///
874    /// Idempotent: invalidating an absent key is a no-op.
875    fn remove_cycle(&self, ssa_id: SsaId<S::Pseudonym>) {
876        self.ssa_cycles.invalidate(&ssa_id);
877        // Deferred acks for a retired cycle can never be redeemed — their part builders will not
878        // come back and their shares are about to expire.
879        self.pending_acks.invalidate(&ssa_id);
880        self.commitment_builder.invalidate(&ssa_id);
881    }

883    /// Removes a cycle that has reached a terminal state and advances its Session frontier.
884    ///
885    /// Retirement is recorded *before* the removal, so a commitment completing concurrently sees
886    /// the retirement and undoes its own publication rather than resurrecting the cycle.
887    ///
888    /// This is the operation for a cycle that went **live** and failed terminally. Successful
889    /// recovery uses [`retire_recovered_cycle`](Self::retire_recovered_cycle), which performs the
890    /// same heavyweight teardown but installs a bounded liveness tail. It is deliberately *not* the
891    /// operation for a commitment released before going live: that index is retried at the same
892    /// value by design, and retirement would block the retry permanently.
893    /// [`release_abandoned_commitment`](Self::release_abandoned_commitment) draws exactly that line,
894    /// and escalates to this for the live case.
895    fn retire_cycle(&self, ssa_id: SsaId<S::Pseudonym>, retirement: &SharedRetirementFrontier) {
896        // Keep the frontier lock through cache invalidation. Registration takes the same lock, so a
897        // same-index retry cannot install itself between the retirement marker and the removal and
898        // then be mistaken for the state this call intended to tear down.
899        let mut frontier = retirement.lock();
900        frontier.retire(ssa_id.ssa_index());
901        self.remove_cycle(ssa_id);
902    }

904    /// Retires the heavyweight recovery state while retaining only the bounded allowance needed by
905    /// SURBs that were minted before recovery and are still queued in the Exit's FIFO.
906    fn retire_recovered_cycle(
907        &self,
908        ssa_id: SsaId<S::Pseudonym>,
909        retirement: &SharedRetirementFrontier,
910        tail: RecoveredSsaTail,
911    ) {
912        // Publication and retirement use this same lock. Installing the tail before invalidating
913        // the live cache leaves no miss window in which a legitimate trailing acknowledgement is
914        // mistaken for a not-yet-published cycle and deferred.
915        let mut frontier = retirement.lock();
916        frontier.retire_recovered(ssa_id.ssa_index(), tail);
917        self.remove_cycle(ssa_id);
918    }

920    fn process_verified_ack(
921        &self,
922        ack: HalfKey,
923        ack_challenge: HalfKeyChallenge,
924        awaiting_ack_from_peer: &moka::sync::Cache<HalfKeyChallenge, TaggedEncryptedPartialSsaShare<S>>,
925    ) -> Result<ProcessedAckResult<S>, PixError<S::Pseudonym>> {
926        let Some(share) = awaiting_ack_from_peer.get(&ack_challenge) else {
927            tracing::trace!(?ack_challenge, "received ack for unknown share");
928            return Ok(ProcessedAckResult::NoProgress);
929        };
930
931        let spi = share.ssa_polynomial_id().ok_or(PixError::ShareIsEmpty)?;
932
933        // One lookup for the whole cycle: the part builders and the accumulator are published and
934        // reclaimed together, so there is no state in which one is reachable and the other is not.
935        // The lookup also refreshes the cycle's idle timer, which is what keeps a cycle that is
936        // still being served from being reclaimed underneath itself.
937        let Some(cycle) = self.ssa_cycles.get(spi.as_ref()) else {
938            // Two states share this miss and only one of them is worth deferring for. A cycle that
939            // has not been published yet will arrive, so the share stays put and the caller buckets
940            // the ack against it. A cycle that has been retired — recovered, or failed terminally —
941            // never will, and deferring holds both the bucket and the share for `max_ack_await_time`
942            // to no end.
943            //
944            // Not a corner case at the deployed dimensions: a conforming Entry ends every emission
945            // window in a run of shares that advance nothing, so *every* recovered cycle is followed
946            // by acks landing here. Deferring them would hold up to `MAX_DEFERRED_ACKS_PER_CYCLE`
947            // dead entries per cycle, and — the part that actually costs something — keep their
948            // shares in `awaiting_acks`, whose budget is global. There they crowd out live cycles
949            // rather than merely wasting space.
950            if let Some(frontier) = self.existing_retirement_frontier(spi.pseudonym())
951                && frontier.lock().is_retired(spi.ssa_index())
952            {
953                // Frees the buffer entry: the per-peer cache's eviction listener decrements the
954                // global count on an explicit removal, which is what the redeeming path below
955                // relies on too.
956                awaiting_ack_from_peer.remove(&ack_challenge);
957
958                // Decrypt before taking the short Session frontier lock below. The tail itself is
959                // immutable apart from atomics, but matching and crediting happen under that lock
960                // so a newer recovered cycle cannot replace the one slot between those operations.
961                let partial_share = share.partial_share.decrypt(spi.pseudonym(), &ack)?;
962                let ssa_id = *spi.as_ref();
963                let frontier = frontier.lock();
964                let Some(tail) = frontier
965                    .recovered_tail
966                    .as_ref()
967                    .filter(|tail| tail.ssa_index() == spi.ssa_index())
968                else {
969                    tracing::trace!(%spi, "ack for a retired cycle outside the active recovered tail — dropped");
970                    return Ok(ProcessedAckResult::NoProgress);
971                };
972
973                return match tail.credit_share::<S>(ssa_id, spi.poly_index(), share.nonce, &partial_share) {
974                    Ok(RecoveredTailCredit::Credited(progress)) => {
975                        tracing::trace!(%spi, "credited buffered surplus against recovered cycle tail");
976                        Ok(ProcessedAckResult::Progressed(progress))
977                    }
978                    Ok(RecoveredTailCredit::OverBudget) => {
979                        tracing::debug!(%spi, "buffered share is past the recovered polynomial's negotiated surplus");
980                        Ok(ProcessedAckResult::NoProgress)
981                    }
982                    Ok(RecoveredTailCredit::InvalidPolynomial) => {
983                        tracing::error!(%spi, "buffered share names a polynomial outside its recovered cycle");
984                        Err(PixError::InvalidInput)
985                    }
986                    Ok(RecoveredTailCredit::InvalidShare(observed_total)) => {
987                        tracing::error!(%spi, observed_total, "invalid buffered share addressed to a recovered polynomial");
988                        Ok(ProcessedAckResult::InvalidShare(ssa_id, observed_total))
989                    }
990                    Err(error) => Err(error),
991                };
992            }
993            // Not an error: the constant-term set is still incomplete, so no part builder exists
994            // yet. Leave the share in `awaiting_acks` and hand the caller the key it needs to
995            // bucket the ack.
996            //
997            // A third state also lands here and is deliberately left to the TTL: a cycle that idled
998            // out of `ssa_cycles` through `unused_verifier_lifetime` is still registered, not
999            // retired, so its late acks are bucketed as though the cycle were still coming. Both
1000            // the bucket and retained share expire on `max_ack_await_time`; retirement paths above
1001            // are classified exactly without allocating one historical entry per cycle.
1002            return Ok(ProcessedAckResult::VerifierNotReady(spi));
1003        };
1004
1005        // The polynomial index comes from the peer's own share, so it is untrusted. Once the cycle
1006        // is known its dimensions are too, which makes an out-of-range index definitively invalid
1007        // rather than merely early — there is no later state in which it becomes meaningful.
1008        let Some(part) = cycle.part(spi.poly_index()) else {
1009            tracing::error!(%spi, num_polys = cycle.num_polys(), "share names a polynomial outside the cycle");
1010            return Err(PixError::InvalidInput);
1011        };
1012
1013        // Cycle confirmed — safe to consume the share.
1014        awaiting_ack_from_peer.remove(&ack_challenge);
1015
1016        // The share cannot be empty at this point because we prevent empty share insertions
1017        let partial_share = share.partial_share.decrypt(spi.pseudonym(), &ack)?;
1018
1019        let ssa_id = *spi.as_ref();
1020
1021        // The part lock is released before the accumulator is taken below. That order is the one
1022        // callers must keep, and neither lock is ever held across the other.
1023        let ssa_part = match part.lock().add_share(share.nonce, partial_share) {
1024            Ok(AddShareOutcome::Completed(share)) => {
1025                tracing::trace!(%spi, "ssa part complete");
1026                cycle.record_useful_share();
1027                cycle.record_completed_part();
1028                share
1029            }
1030            Ok(AddShareOutcome::Useful) => {
1031                tracing::trace!(%spi, "ssa part not yet complete, waiting for more shares");
1032                cycle.record_useful_share();
1033                // Snapshot here rather than making the caller look the cycle up again: this is the
1034                // steady-state outcome for all but one share in `threshold`, so a second cache get
1035                // would double the lookups on the hot path.
1036                return Ok(ProcessedAckResult::Progressed(cycle.progress()));
1037            }
1038            // Expected traffic, not a fault: a conforming Entry emits `threshold + surplus` shares
1039            // per polynomial, so every polynomial ends its life absorbing surplus.
1040            //
1041            // Reported rather than swallowed, even though it advances nothing. Polynomials in an
1042            // emission window run in lockstep, so they reach `threshold` together and then take their
1043            // surplus together — `surplus × window` consecutive shares, 4096 at the deployed
1044            // dimensions, none of them useful. A consumer that only ever hears about useful shares
1045            // reads that as the Entry having gone quiet. The snapshot says which it is: `shares_seen`
1046            // moves, `useful_shares` does not.
1047            Ok(AddShareOutcome::Surplus) => {
1048                match cycle.credit_surplus(spi.poly_index()) {
1049                    Some(true) => {
1050                        tracing::trace!(%spi, "share arrived after its polynomial was reconstructed");
1051                        cycle.record_surplus_share();
1052                        return Ok(ProcessedAckResult::Progressed(cycle.progress()));
1053                    }
1054                    Some(false) => {
1055                        // Past the negotiated surplus the peer is emitting more for this polynomial
1056                        // than it said it would, and reconstruction has already released the share
1057                        // set, so these cannot even be told apart as duplicates. Withholding the
1058                        // snapshot prevents one completed polynomial from refreshing the Exit's
1059                        // gate indefinitely while the rest of the cycle is untouched.
1060                        tracing::debug!(%spi, "share arrived past the peer's negotiated surplus for this polynomial");
1061                        return Ok(ProcessedAckResult::NoProgress);
1062                    }
1063                    None => {
1064                        // `cycle.part` performed the same checked lookup above, so this can only be
1065                        // an internal disagreement between the part and budget dimensions.
1066                        tracing::error!(%spi, "surplus budget does not contain the cycle's polynomial");
1067                        return Err(PixError::InvalidInput);
1068                    }
1069                }
1070            }
1071            Ok(AddShareOutcome::Duplicate) => {
1072                tracing::trace!(%spi, "duplicate evaluation identifier");
1073                return Ok(ProcessedAckResult::NoProgress);
1074            }
1075            Ok(AddShareOutcome::Absorbed) => {
1076                tracing::trace!(%spi, "share for a polynomial that already failed its commitment");
1077                return Ok(ProcessedAckResult::NoProgress);
1078            }
1079            Err(PixError::VsssError(vsss_rs::Error::InvalidShare)) => {
1080                // Counted rather than raised: the caller reports it as a resolution, and the count
1081                // has to be taken here because the cycle that holds it is in hand.
1082                //
1083                // Almost always this means the polynomial's reconstructed constant term did not
1084                // open its commitment, in which case the offending share is one of the `threshold`
1085                // that went into it and cannot be singled out. The whole cycle is lost either way,
1086                // since the SSA needs every polynomial.
1087                let observed_total = cycle.record_invalid_share();
1088                tracing::error!(%spi, observed_total, "ssa part failed to open its commitment");
1089                return Ok(ProcessedAckResult::InvalidShare(ssa_id, observed_total));
1090            }
1091            Err(e) => return Err(e),
1092        };
1093
1094        let mut builder_guard = cycle.builder().lock();
1095        let ssa = match builder_guard.add_recovered_ssa_part(spi.poly_index(), ssa_part) {
1096            Ok(ssa) => ssa,
1097            Err(error) => {
1098                // As terminal as `scalar_to_private_key` returning `None` below, and torn down the
1099                // same way. Propagating alone would leave the accumulator and every part builder in
1100                // place, and each further share for the cycle would refresh the idle timer that is
1101                // supposed to reclaim them — so a Session that keeps sending holds a cycle that can
1102                // never reconstruct for as long as it likes.
1103                //
1104                // The lock goes first: `retire_cycle` drops the last `Arc` to this very cycle.
1105                drop(builder_guard);
1106                tracing::error!(%spi, %error, "ssa part could not be added to its accumulator");
1107                self.retire_cycle(ssa_id, &cycle.registration.retirement);
1108                return Err(error);
1109            }
1110        };
1111        match ssa {
1112            Some(scalar) => {
1113                // Capture the recovery instant before retirement. The counters survive in the
1114                // compact tail, but any later movement is buffered liveness and must not be folded
1115                // backwards into the terminal recovery event.
1116                let progress = cycle.progress();
1117                // Release the accumulator lock before retiring, so `retire_cycle` — which drops
1118                // the last `Arc` to this very cycle — does not run while it is held.
1119                drop(builder_guard);
1120                let Some(ssa) = S::scalar_to_private_key(scalar) else {
1121                    tracing::error!(%spi, "ssa reconstruction failed");
1122                    self.retire_cycle(ssa_id, &cycle.registration.retirement);
1123                    return Err(PixError::InvalidSsa);
1124                };
1125                // Full recovery releases every heavyweight builder immediately. Only the shared
1126                // counters and per-polynomial remainder survive, so SURBs already queued with this
1127                // cycle's shares can drain as bounded liveness without retaining the recovered key
1128                // or giving the Entry a replayable cycle-wide allowance.
1129                let tail = cycle.recovered_tail();
1130                self.retire_recovered_cycle(ssa_id, &cycle.registration.retirement, tail);
1131                tracing::info!(%ssa_id, "ssa recovered");
1132                Ok(ProcessedAckResult::FullRecovery(RecoveredSsa { ssa_id, ssa }, progress))
1133            }
1134            None => {
1135                tracing::trace!(%spi, "ssa not yet complete, waiting for more ssa parts");
1136                // Check early threshold while we hold the lock
1137                let early = builder_guard.check_early_threshold(self.cfg.early_recovery_threshold);
1138                drop(builder_guard);
1139                let progress = cycle.progress();
1140                if early {
1141                    tracing::info!(%ssa_id, "early recovery threshold reached");
1142                    Ok(ProcessedAckResult::EarlyRecovery(progress))
1143                } else {
1144                    Ok(ProcessedAckResult::Progressed(progress))
1145                }
1146            }
1147        }
1148    }

1150    /// Buckets an acknowledgement whose cycle's part builders have not been installed yet.
1151    ///
1152    /// O(1) — this is the entire cost the acknowledgement path pays for a deferral.
1153    fn defer_ack(&self, spi: SsaPolynomialId<S::Pseudonym>, deferred: DeferredAck) {
1154        let bucket = self.pending_acks.get_with(*spi.as_ref(), || {
1155            std::sync::Arc::new(parking_lot::Mutex::new(Default::default()))
1156        });
1157        self.defer_ack_into(&bucket, spi, deferred);
1158    }

1160    /// The bucket half of [`defer_ack`](Self::defer_ack), taking the bucket rather than looking it
1161    /// up.
1162    ///
1163    /// Split out so a test can hold a handle across the drain that invalidates the cache key,
1164    /// which is the interleaving this guards against and the one thing a single thread cannot
1165    /// otherwise produce — after the invalidate, `get_with` hands out a *fresh* bucket.
1166    fn defer_ack_into(&self, bucket: &DeferredAckBucket, spi: SsaPolynomialId<S::Pseudonym>, deferred: DeferredAck) {
1167        let ssa_id = *spi.as_ref();
1168        let outcome = {
1169            let mut bucket = bucket.lock();
1170            if bucket.drained {
1171                Deferral::Orphaned
1172            } else if bucket.total >= MAX_DEFERRED_ACKS_PER_CYCLE {
1173                // The cycle as a whole is holding more than the shares it could plausibly have
1174                // received inside `max_ack_await_time`, so the excess cannot be redeemable.
1175                tracing::warn!(
1176                    %ssa_id,
1177                    cap = MAX_DEFERRED_ACKS_PER_CYCLE,
1178                    "dropping deferred acknowledgement: cycle bucket is full"
1179                );
1180                Deferral::Dropped
1181            } else {
1182                // Reborrowed off the guard so `by_poly` and `total` are disjoint field borrows.
1183                let bucket = &mut *bucket;
1184                let per_poly = bucket.by_poly.entry(spi.poly_index()).or_default();
1185                if per_poly.len() >= MAX_DEFERRED_ACKS_PER_POLYNOMIAL {
1186                    // Only reachable if the peer emits more shares for one polynomial than its own
1187                    // `threshold + surplus` budget allows, so the excess is almost certainly
1188                    // duplicate.
1189                    tracing::warn!(
1190                        %spi,
1191                        cap = MAX_DEFERRED_ACKS_PER_POLYNOMIAL,
1192                        "dropping deferred acknowledgement: polynomial bucket is full"
1193                    );
1194                    Deferral::Dropped
1195                } else {
1196                    per_poly.push(deferred);
1197                    bucket.total += 1;
1198                    Deferral::Buffered
1199                }
1200            }
1201        };
1202
1203        match outcome {
1204            // The drain took this bucket while we were on our way into it. Redeeming here is what
1205            // makes the mutex the whole synchronisation point: the drain's take and this append
1206            // are serialised by it, so exactly one of them owns the ack.
1207            Deferral::Orphaned => {
1208                tracing::trace!(%spi, "redeeming an acknowledgement deferred into a drained bucket");
1209                self.redeem_deferred_acks(&ssa_id, std::iter::once(deferred));
1210            }
1211            // Close the race against a concurrent installation. The decision to defer was made on
1212            // a cycle lookup that missed; if the cycle has appeared since, the drain that would
1213            // have redeemed this ack may already have run against a bucket we never saw.
1214            Deferral::Buffered => {
1215                if self.ssa_cycles.contains_key(&ssa_id) {
1216                    self.drain_deferred_acks(&ssa_id);
1217                }
1218            }
1219            Deferral::Dropped => {}
1220        }
1221    }

1223    /// Redeems the acknowledgements that were waiting for this cycle's part builders.
1224    ///
1225    /// Called from the commitment path immediately after the cycle is installed, so each bucket is
1226    /// processed exactly once and never speculatively re-scanned. Resolutions are parked in
1227    /// [`ready_resolutions`](Self::ready_resolutions) for the next `acknowledge_shares` call, since
1228    /// the commitment path has no route to the upper layer.
1229    fn drain_deferred_acks(&self, ssa_id: &SsaId<S::Pseudonym>) {
1230        let Some(bucket) = self.pending_acks.get(ssa_id) else {
1231            return;
1232        };
1233        self.pending_acks.invalidate(ssa_id);
1234
1235        // Take and tombstone in one critical section. A `defer_ack_into` that looked this bucket
1236        // up before the invalidate still holds an `Arc` to it; the flag is what stops its append
1237        // from disappearing into a bucket nothing will read again.
1238        let deferred = {
1239            let mut bucket = bucket.lock();
1240            bucket.drained = true;
1241            bucket.total = 0;
1242            std::mem::take(&mut bucket.by_poly)
1243        };
1244        if deferred.is_empty() {
1245            return;
1246        }
1247
1248        self.redeem_deferred_acks(ssa_id, deferred.into_values().flatten());
1249    }

1251    /// Processes acknowledgements whose verifier has since been installed, parking whatever they
1252    /// resolve to.
1253    ///
1254    /// Shared by the two routes that can redeem a deferral — the drain on the commitment path and
1255    /// an [`orphaned`](Deferral::Orphaned) append — so both produce the same resolutions in the
1256    /// same order.
1257    fn redeem_deferred_acks(&self, ssa_id: &SsaId<S::Pseudonym>, deferred: impl IntoIterator<Item = DeferredAck>) {
1258        let mut resolved = Vec::new();
1259        // The furthest-along snapshot any redeemed ack produced. Shares recovered here would
1260        // otherwise be invisible to the consumer until some later batch happened to touch this same
1261        // SSA, since only `acknowledge_shares` emits snapshots.
1262        let mut progress = Vec::new();
1263        for (peer, challenge, ack) in deferred {
1264            // The share lives in the peer's own awaiting-acks cache; if the peer entry is gone the
1265            // share has expired with it and the ack is dead.
1266            let Some(awaiting) = self.awaiting_acks.get(&peer) else {
1267                continue;
1268            };
1269            match self.process_verified_ack(ack, challenge, &awaiting) {
1270                Ok(ProcessedAckResult::FullRecovery(ssa, snapshot)) => {
1271                    record_progress(&mut progress, snapshot);
1272                    resolved.push(ShareResolution::RecoveredSsa(ssa));
1273                }
1274                Ok(ProcessedAckResult::EarlyRecovery(snapshot)) => {
1275                    let id = snapshot.ssa_id;
1276                    record_progress(&mut progress, snapshot);
1277                    resolved.push(ShareResolution::AlmostRecoveredSsa(id));
1278                }
1279                Ok(ProcessedAckResult::Progressed(snapshot)) => record_progress(&mut progress, snapshot),
1280                Ok(ProcessedAckResult::InvalidShare(id, observed_total)) => {
1281                    tracing::error!(%id, observed_total, "deferred share could not be verified");
1282                    resolved.push(ShareResolution::InvalidShares {
1283                        peer: peer.into(),
1284                        ssa_id: id,
1285                        observed_total,
1286                    });
1287                }
1288                Ok(ProcessedAckResult::NoProgress) => {}
1289                Ok(ProcessedAckResult::VerifierNotReady(_)) => {
1290                    // The cycle was installed and then immediately withdrawn, which only the
1291                    // retirement path does. Re-bucketing would leak, so drop.
1292                    tracing::trace!(%ssa_id, "cycle withdrawn while draining deferred acknowledgements");
1293                }
1294                Err(error) => tracing::debug!(%ssa_id, %error, "failed to process deferred acknowledgement"),
1295            }
1296        }
1297
1298        // Snapshots go in ahead of the terminal events they belong to, matching what
1299        // `acknowledge_shares` emits — the consumer sees one order regardless of which path resolved
1300        // a share.
1301        if !progress.is_empty() {
1302            let mut ordered = progress.into_iter().map(ShareResolution::Progress).collect::<Vec<_>>();
1303            ordered.append(&mut resolved);
1304            resolved = ordered;
1305        }
1306
1307        if !resolved.is_empty() {
1308            tracing::debug!(%ssa_id, num = resolved.len(), "redeemed deferred acknowledgements");
1309            let mut ready = self.ready_resolutions.lock();
1310            ready.extend(resolved);
1311            self.ready_resolutions_len
1312                .store(ready.len(), std::sync::atomic::Ordering::Release);
1313            drop(ready);
1314            self.notify_ready_resolutions();
1315        }
1316    }

1318    /// Takes any resolutions parked by `drain_deferred_acks`.
1319    ///
1320    /// The take is atomic with respect to producers and other consumers. This lets the transport's
1321    /// notification stream and the acknowledgement fallback race safely: exactly one receives each
1322    /// queued resolution.
1323    ///
1324    /// One atomic load is the entire common-case cost — the buckets are empty whenever the Entry
1325    /// finishes the constant-term pass before the shares that reference it arrive.
1326    pub fn take_ready_resolutions(&self) -> Vec<ShareResolution<S::Pseudonym, S::AddressPrivateKey>> {
1327        if self.ready_resolutions_len.load(std::sync::atomic::Ordering::Acquire) == 0 {
1328            return Vec::new();
1329        }
1330        let mut ready = self.ready_resolutions.lock();
1331        self.ready_resolutions_len
1332            .store(0, std::sync::atomic::Ordering::Release);
1333        std::mem::take(&mut *ready)
1334    }

1336    /// Minimum wall time between two [`resync_ack_buffer`](Self::resync_ack_buffer) passes.
1337    ///
1338    /// A saturated buffer would otherwise turn every rejected insertion into an
1339    /// `O(max_tracked_peers)` scan — the overload path amplifying its own cost, which is the shape
1340    /// of bug this budget exists to prevent.
1341    ///
1342    /// Derived from [`max_ack_await_time`](SsaReconstructorConfig::max_ack_await_time) rather than
1343    /// fixed, because what a resync reclaims is entries that have aged out of *that* window. A
1344    /// constant would be wrong at both ends: too slow for a short window, so a drained buffer keeps
1345    /// refusing shares long after it emptied, and needlessly eager for a long one.
1346    ///
1347    /// The resulting staleness — up to ~1.9 s at the 30 s default — only bites when the counter has
1348    /// drifted high *and* nothing is touching the caches, since redemption and any cache access
1349    /// drive moka's expiry maintenance and fire the listener directly. That is the traffic-stopped
1350    /// case, where refusing a share costs nothing.
1351    fn ack_buffer_resync_interval(&self) -> std::time::Duration {
1352        (self.cfg.max_ack_await_time / 16).max(std::time::Duration::from_millis(1))
1353    }

1355    /// Recomputes [`ack_buffer_entries`](Self::ack_buffer_entries) from what the caches actually
1356    /// hold.
1357    ///
1358    /// # Why a counter needs a backstop at all
1359    ///
1360    /// Entries leave the buffer four ways: redeemed by their acknowledgement, expired, size-evicted
1361    /// from their peer's cache, or dropped wholesale when the peer's entry leaves `awaiting_acks`.
1362    /// The inner eviction listener catches the first three exactly. The fourth cannot be caught:
1363    /// dropping a moka handle does not run its eviction listener, so the outer listener falls back
1364    /// to `invalidate_all`, which is best-effort and races an insertion landing on the very cache
1365    /// being discarded.
1366    ///
1367    /// Left alone, that residue only ever accumulates *upward*, and an over-count is far worse than
1368    /// an under-count: it would eventually refuse every share while the buffer sat empty, turning a
1369    /// memory ceiling into a permanent outage of the acknowledgement path. (The sibling
1370    /// `HoprUnacknowledgedTicketProcessor` in `hopr-protocol-hopr` has the same nesting and the same
1371    /// residue; there it only skews metrics.)
1372    ///
1373    /// So the counter is treated as a hint that is allowed to be wrong, and ground truth is
1374    /// consulted at the one moment being wrong would cost something — when it says the buffer is
1375    /// full. `try_lock` rather than `lock`: a caller that finds a resync already running should
1376    /// proceed on the current estimate, not queue up behind it.
1377    fn resync_ack_buffer(&self) {
1378        let Some(mut last_run) = self.ack_buffer_resync.try_lock() else {
1379            return;
1380        };
1381        if last_run.is_some_and(|at| at.elapsed() < self.ack_buffer_resync_interval()) {
1382            return;
1383        }
1384
1385        let held = self.count_ack_buffer_entries();
1386        let previous = self.ack_buffer_entries.swap(held, std::sync::atomic::Ordering::Relaxed);
1387        *last_run = Some(std::time::Instant::now());
1388
1389        if previous != held {
1390            tracing::debug!(
1391                previous,
1392                held,
1393                "resynchronised the awaiting-acknowledgement buffer count"
1394            );
1395        }
1396    }

1398    /// Ground truth: the entries actually held across every peer.
1399    ///
1400    /// Never reads [`ack_buffer_entries`](Self::ack_buffer_entries), so a test asserting the two
1401    /// agree is testing the counter rather than agreeing with it — the same reason
1402    /// `deferred_ack_count` recomputes from `by_poly` instead of reading `DeferredAcks::total`.
1403    ///
1404    /// `O(max_tracked_peers)`, and each `run_pending_tasks` is bounded by that cache's pending write
1405    /// queue rather than its size. Both callers keep it off the steady-state path.
1406    fn count_ack_buffer_entries(&self) -> usize {
1407        self.awaiting_acks.run_pending_tasks();
1408        self.awaiting_acks
1409            .iter()
1410            .map(|(_, shares)| {
1411                shares.run_pending_tasks();
1412                shares.entry_count() as usize
1413            })
1414            .sum()
1415    }

1431    ///
1432    /// The per-polynomial cache entry count used to express this. It has to be asked of the cycle

1434    /// `entry_count()` alone can no longer tell "every part installed" from "one part installed".
1435    #[cfg(test)]

1438    }

1441    ///
1442    /// Recomputed from `by_poly` rather than read off `DeferredAcks::total` deliberately: a

1448            .get(ssa_id)

1454/// Ownership of an Exit SSA commitment, released when dropped.
1455///
1456/// Registering an Exit commitment is the first fallible step of many: the request still has to be
1457/// encoded, sent, and answered. Every early return between here and the point where a permanent
1458/// owner takes over would otherwise strand the commitment in the reconstructor until its own
1459/// lifetime expired, and a stranded commitment is not inert — its `SsaId` is occupied, so a retry at
1460/// the same index is rejected as a duplicate.
1461///
1462/// Move-only by design: no `Clone`, no `Copy`, so there is exactly one release point. A success path
1463/// hands ownership on with [`disarm`](Self::disarm) rather than letting the guard fall out of scope.
1464///
1465/// Dropping releases the registration **without** retiring the SSA, so the same index can be
1466/// requested again — see `SsaReconstructor::release_abandoned_commitment` for why that
1467/// distinction is load-bearing.
1468#[must_use = "dropping the guard immediately releases the SSA it owns"]
1469pub struct SsaCommitmentGuard<S: PixSpec + Clone> {
1470    /// `None` once disarmed, which is the only state in which `Drop` does nothing.
1471    owned: Option<OwnedCommitment<S>>,
1472}
1473
1474/// What an [`SsaCommitmentGuard`] needs to release its SSA: where it is registered, which one, and
1475/// the Session frontier that must outlive a concurrent commitment worker.
1476type OwnedCommitment<S> = (
1477    std::sync::Arc<SsaReconstructor<S>>,
1478    SsaId<<S as PixSpec>::Pseudonym>,
1479    std::sync::Arc<RegisteredCommitment<S>>,
1480);
1481
1482/// A registered Exit commitment, paired with ownership of its lifetime.
1483type GuardedExitCommitment<S> = (PixGroup<S>, SsaCommitmentGuard<S>);

1515impl<S: PixSpec + Clone> SsaReconstructor<S> {
1516    fn register_exit_commitment(
1517        &self,
1518        id: SsaId<S::Pseudonym>,
1519        params: PixParams,
1520    ) -> Result<RegisteredExitCommitment<S>, PixError<S::Pseudonym>> {
1521        let exit_commitment_secret = PixScalar::<S>::random(&mut hopr_types::crypto_random::rng());
1522        let exit_commitment_public = PixGroup::<S>::mul_by_generator(&exit_commitment_secret);
1523        let retirement = self.retirement_frontier(*id.pseudonym());
1524        let registration = std::sync::Arc::new(RegisteredCommitment {
1525            builder: parking_lot::Mutex::new(SsaCommitmentBuilder::new(
1526                id,
1527                params,
1528                exit_commitment_secret,
1529                exit_commitment_public,
1530            )),
1531            retirement: retirement.clone(),
1532        });
1533
1534        // Registration and retirement serialize on this Session-local lock. The cache insertion is
1535        // inside the critical section so a retiring call cannot remove the old registration and
1536        // accidentally remove a same-index retry installed between its marker and invalidation.
1537        let mut frontier = retirement.lock();
1538        if frontier.is_retired(id.ssa_index()) || self.ssa_cycles.contains_key(&id) {
1539            return Err(PixError::DuplicateCommitment);
1540        }
1541
1542        self.commitment_builder
1543            .entry(id)
1544            .and_try_compute_with(|entry| match entry {
1545                Some(_) => Err(PixError::DuplicateCommitment),
1546                None => Ok(moka::ops::compute::Op::Put(registration.clone())),
1547            })?;
1548
1549        // The retired check above and this mutation share the same lock, so this cannot fail unless
1550        // the frontier's own invariants were broken locally.
1551        let newly_registered = frontier.register(id.ssa_index());
1552        debug_assert!(newly_registered, "registration raced the frontier under its own lock");
1553        drop(frontier);
1554
1555        Ok((exit_commitment_public, registration))
1556    }
1557
1558    /// Releases a commitment that was registered but never taken over by an owner.
1559    ///
1560    /// Deliberately **not** [`retire_ssa`](ExitAcknowledgementShareProcessor::retire_ssa), which
1561    /// permanently moves the index behind the Session's retirement frontier. A pre-live release
1562    /// instead marks the index retriable, so a replacement registration can use the same value.
1563    ///
1564    /// Same-index retry is not a corner case: the SSA index is advanced only after every fallible
1565    /// step of a request has succeeded, so a request that failed keeps its index by design and the
1566    /// next attempt reuses it.
1567    ///
1568    /// Escalates to a full retirement if a cycle did go live, which means the peer was asked and
1569    /// answered — and therefore that ownership should already have been transferred with
1570    /// [`disarm`](SsaCommitmentGuard::disarm). That branch is a caller error, and retiring is the
1571    /// safe response to it, because a live cycle is exactly what the retirement frontier protects.
1572    fn release_abandoned_commitment(
1573        &self,
1574        ssa_id: SsaId<S::Pseudonym>,
1575        registration: &std::sync::Arc<RegisteredCommitment<S>>,
1576    ) {
1577        // Hold the frontier lock through both classification and removal. A completion that already
1578        // published is retired and removed here; one that publishes later sees either retirement or
1579        // an abandoned registration and withdraws its own state.
1580        let mut frontier = registration.retirement.lock();
1581        let owns_live_cycle = self
1582            .ssa_cycles
1583            .get(&ssa_id)
1584            .is_some_and(|cycle| std::sync::Arc::ptr_eq(&cycle.registration, registration));
1585        let owns_pending_commitment = self
1586            .commitment_builder
1587            .get(&ssa_id)
1588            .is_some_and(|current| std::sync::Arc::ptr_eq(&current, registration));
1589
1590        if owns_live_cycle {
1591            tracing::warn!(%ssa_id, "abandoned ssa commitment was already live — retiring it");
1592            frontier.retire(ssa_id.ssa_index());
1593        } else if owns_pending_commitment {
1594            tracing::debug!(%ssa_id, "releasing ssa commitment abandoned by its owner");
1595            frontier.abandon(ssa_id.ssa_index());
1596        } else {
1597            // This guard belonged to an expired registration that has since been retried. Pointer
1598            // identity is what prevents the old owner from tearing down the replacement.
1599            return;
1600        }
1601        self.remove_cycle(ssa_id);
1602    }
1603
1604    /// [`new_exit_commitment`](ExitAcknowledgementShareProcessor::new_exit_commitment), with the
1605    /// registration owned by an [`SsaCommitmentGuard`].
1606    ///
1607    /// No guard is produced on failure, so a rejected duplicate never retires the registration that
1608    /// caused the rejection.
1609    pub fn new_guarded_exit_commitment(
1610        self: &std::sync::Arc<Self>,
1611        id: SsaId<S::Pseudonym>,
1612        params: PixParams,
1613    ) -> Result<GuardedExitCommitment<S>, PixError<S::Pseudonym>> {
1614        let (exit_commitment, registration) = self.register_exit_commitment(id, params)?;
1615        Ok((
1616            exit_commitment,
1617            SsaCommitmentGuard {
1618                owned: Some((self.clone(), id, registration)),
1619            },
1620        ))
1621    }
1622}

1662    fn retire_ssa(&self, ssa_id: SsaId<S::Pseudonym>) {
1663        // Every heavyweight key is the SsaId itself, so there is nothing to enumerate and no way for
1664        // one builder to survive the removal of the rest. The frontier-before-removal ordering the
1665        // commitment completion path depends on lives in `retire_cycle`, which the terminal
1666        // acknowledgement paths share with this one.
1667        //
1668        // A successfully recovered id is already retired, so `RetirementFrontier::retire` returns
1669        // false and deliberately leaves its compact tail intact. The Session tombstone is a short
1670        // diagnostic record; the FIFO can take longer to drain. The one-slot tail is replaced by
1671        // the next successful recovery and finally released with the Session retirement scope. An
1672        // entirely unknown id has no frontier to advance and remains the documented idempotent no-op.
1673        if let Some(retirement) = self.retirement_frontier_for(&ssa_id) {
1674            self.retire_cycle(ssa_id, &retirement);
1675        } else {
1676            self.remove_cycle(ssa_id);
1677        }

1683    ///
1684    /// "The two bounded fields" is `polys_per_ssa` and `shares_per_poly`, and the omission of
1685    /// `surplus_shares` is deliberate rather than an oversight — worth stating, because the surplus

1687    /// traffic one completed polynomial may credit as liveness. `try_new` accepts the whole `u8`
1688    /// range for it on purpose: the surplus is a *wire* value, `surplus_must_not_exceed_threshold`
1689    /// bounds only what this node will locally configure, and an extravagant surplus is caught where

1699    fn insert_coefficient_commitments(
1700        &self,
1701        ssa_id: SsaId<S::Pseudonym>,
1702        index: CoefficientIndex,
1703        proof: Option<SsaCommitmentProof<S>>,
1704        commitments: impl Iterator<Item = (PolynomialIndex, PixGroupRepr<S>)>,
1705    ) -> Result<SsaCommitmentState<S::Pseudonym, S::DepositAddress>, Self::Error> {
1706        let mut res = SsaCommitmentState::new(ssa_id);
1707
1708        // The Server commitment must be present first
1709        let Some(builder) = self.commitment_builder.get(&ssa_id) else {
1710            return Err(PixError::MissingSsaCommitment);
1711        };
1712        let retirement = builder.retirement.clone();
1713
1714        let progress = {
1715            let mut builder = builder.lock();
1716            res.is_first_encountered = builder.is_empty();
1717            res.ssa_deposit_address = builder.get_deposit_address().copied();
1718            builder.add_transposed(index, proof, commitments)?
1719        };
1720
1721        // `ssa_deposit_address` was read *before* the insertion, so it being absent here means the
1722        // address (if any) was discovered by this very call.
1723        res.deposit_address_first_encountered = res.ssa_deposit_address.is_none();
1724
1725        let Some(full_ssa_commitment) = progress.full_commitment else {
1726            res.deposit_address_first_encountered = false; // Not yet encountered
1727            tracing::trace!(%ssa_id, "ssa commitment not yet complete, waiting for more constant terms");
1728            return Ok(res);
1729        };
1730        res.ssa_deposit_address = Some(S::group_to_deposit_address(full_ssa_commitment).ok_or(PixError::InvalidSsa)?);
1731
1732        // The accumulator and every part builder go in as one entry. There is deliberately no
1733        // ordering to get right here: a share can never observe a cycle in which one is reachable
1734        // and the other is not, which used to be a permanent-drop hazard.
1735        let mut publication_rejected = false;
1736        let installed = if let Some(ssa_builder) = progress.ssa_builder {
1737            let num_polys = ssa_builder.num_polys();
1738            let cycle = SsaCycle::new(ssa_id, ssa_builder, progress.new_verifiers)?;
1739            let registered_cycle = std::sync::Arc::new(RegisteredCycle {
1740                cycle,
1741                registration: builder.clone(),
1742            });
1743
1744            // Registration, abandonment and retirement all take this same Session-local lock. A
1745            // stale worker therefore cannot overwrite a newer retry: it verifies pointer identity
1746            // and publishes while every state transition for this index is excluded.
1747            let frontier = retirement.lock();
1748            let registration_is_current = self
1749                .commitment_builder
1750                .get(&ssa_id)
1751                .is_some_and(|current| std::sync::Arc::ptr_eq(&current, &builder));
1752            let was_retired = frontier.is_retired(ssa_id.ssa_index());
1753            if !registration_is_current || was_retired {
1754                publication_rejected = true;
1755                false
1756            } else {
1757                self.ssa_cycles.insert(ssa_id, registered_cycle);
1758
1759                // Keep the check after publication as well. Cache expiry is independent of the
1760                // frontier lock; if it withdraws this registration during insertion, undo the cycle
1761                // before releasing the lock and allowing a retry.
1762                let registration_is_current = self
1763                    .commitment_builder
1764                    .get(&ssa_id)
1765                    .is_some_and(|current| std::sync::Arc::ptr_eq(&current, &builder));
1766                if !registration_is_current {
1767                    self.remove_cycle(ssa_id);
1768                    publication_rejected = true;
1769                    false
1770                } else {
1771                    tracing::debug!(%ssa_id, num_polys, "ssa commitment known — cycle is live");
1772                    true
1773                }
1774            }
1775        } else {
1776            false
1777        };
1778
1779        if publication_rejected {
1780            tracing::trace!(%ssa_id, "ssa commitment progressed after its registration ended — dropped published state");
1781            res.deposit_address_first_encountered = false;
1782            return Ok(res);
1783        }
1784
1785        // Installing the cycle unblocks every acknowledgement bucketed under it. Doing this here
1786        // rather than on the acknowledgement path is what keeps `acknowledge_shares` free of retry
1787        // scanning.
1788        if installed {
1789            self.drain_deferred_acks(&ssa_id);
1790        }
1791
1792        res.is_verifiable = progress.fully_committed;
1793        if progress.fully_committed {
1794            tracing::trace!(%ssa_id, "ssa commitment completed");
1795        }
1796
1797        Ok(res)
1798    }

1800    /// Buffers an encrypted share until its acknowledgement arrives, subject to the global byte
1801    /// budget.
1802    ///
1803    /// # The bound is global, and it is not the product of the two caps
1804    ///
1805    /// [`max_tracked_peers`](SsaReconstructorConfig::max_tracked_peers) and
1806    /// [`max_awaiting_acks`](SsaReconstructorConfig::max_awaiting_acks) bound one dimension each,
1807    /// and their product — 2 000 × 1 000 000 by default, some 800 GB — is neither reachable nor the
1808    /// right thing to bound. Not reachable, because an entry exists only for a share this node has
1809    /// already *sent*, so filling it would take 66 M packets/s of egress inside the default 30 s
1810    /// window. Not right, because the two guard **mutually exclusive** concentrations:
1811    /// `max_awaiting_acks` sizes one cache per peer and has to cover every Session returning through
1812    /// a single first-relayer, while `max_tracked_peers` covers traffic spread thin. Squeezing the
1813    /// product would push one of them below what its own case needs, and a `max_awaiting_acks` set
1814    /// too low does not save memory — it size-evicts shares before their acknowledgements arrive.
1815    ///
1816    /// So the real bound is [`max_ack_buffer_bytes`](SsaReconstructorConfig::max_ack_buffer_bytes),
1817    /// counted here across all peers at once. Validating a workload model instead would not do:
1818    /// a model has to assume a Session count and a packet rate, and this node enforces neither.
1819    ///
1820    /// # Behaviour at the ceiling
1821    ///
1822    /// The newest share is refused, rather than the oldest evicted: there is no cheap global
1823    /// "oldest" across per-peer caches, and the oldest is nearest its TTL anyway. Either way a full
1824    /// buffer means share loss — the packet is already on the wire and its acknowledgement will find
1825    /// nothing — which is the honest cost of a hard ceiling. [`PixError::AckBufferFull`] is
1826    /// deliberately not an expected error so the caller logs it.
1827    ///
1828    /// The check and the insertion are not atomic. Concurrent inserters can overshoot the ceiling by
1829    /// their own number, which is the right trade: a lock on this path would cost more than the few
1830    /// hundred kilobytes of overshoot it would prevent.
1831    fn insert_encrypted_share(
1832        &self,
1833        peer: &OffchainPublicKey,
1834        challenge: HalfKeyChallenge,
1835        tagged_enc_share: TaggedEncryptedPartialSsaShare<S>,
1836    ) -> Result<(), Self::Error> {
1837        if tagged_enc_share.partial_share.is_empty() {
1838            return Err(PixError::ShareIsEmpty);
1839        }
1840
1841        // One relaxed load in the steady state. Only a buffer that has actually reached its ceiling
1842        // pays for the ground-truth pass, and even then at most one caller at a time does.
1843        if self.ack_buffer_entries.load(std::sync::atomic::Ordering::Relaxed) >= self.max_ack_buffer_entries {
1844            self.resync_ack_buffer();
1845            if self.ack_buffer_entries.load(std::sync::atomic::Ordering::Relaxed) >= self.max_ack_buffer_entries {
1846                tracing::error!(
1847                    %peer,
1848                    budget_bytes = self.cfg.max_ack_buffer_bytes,
1849                    "awaiting-acknowledgement buffer is full — dropping share"
1850                );
1851                return Err(PixError::AckBufferFull);
1852            }
1853        }
1854
1855        // Incremented unconditionally, including when this replaces an entry under the same
1856        // challenge. The inner listener fires for `Replaced` too, so the pair nets to zero.
1857        self.ack_buffer_entries
1858            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
1859        self.awaiting_acks
1860            .get_with_by_ref(peer, || {
1861                // Inner cache keyed by HalfKeyChallenge — each entry gets its own TTL
1862                // so a late-arriving share gets the full max_ack_await_time window.
1863                let released = self.ack_buffer_entries.clone();
1864                moka::sync::CacheBuilder::new(self.cfg.max_awaiting_acks as u64)
1865                    .time_to_live(self.cfg.max_ack_await_time)
1866                    // Every way an entry can leave this cache — redeemed by its acknowledgement,
1867                    // expired, or size-evicted — arrives here, which is what keeps the global count
1868                    // falling without the removal sites having to know about it.
1869                    .eviction_listener(move |_, _, _| {
1870                        released.fetch_sub(1, std::sync::atomic::Ordering::Relaxed);
1871                    })
1872                    .build()
1873            })
1874            .insert(challenge, tagged_enc_share);
1875
1876        Ok(())
1877    }

1879    fn acknowledge_shares(
1880        &self,
1881        peer: OffchainPublicKey,
1882        acks: Vec<Acknowledgement>,
1883    ) -> Result<Vec<ShareResolution<S::Pseudonym, S::AddressPrivateKey>>, Self::Error> {
1884        let Some((awaiting_ack_from_peer, half_keys_challenges)) = crate::ack_verify::verify_expected_acknowledgements(
1885            peer,
1886            acks,
1887            &self.awaiting_acks,
1888            self.cfg.use_batch_verification,
1889        ) else {
1890            return Err(PixError::UnexpectedShare);
1891        };
1892
1893        // Three accumulators rather than one deduplicating set, because the emission contract is an
1894        // ordering as well as a multiplicity: every SSA's counters precede its terminal event.
1895        //
1896        // All three are `Vec`s deduplicated by linear scan. A per-peer batch resolves shares for the
1897        // cycle that peer is relaying for — one, or a small handful while a Session pipelines the
1898        // next — so the scan is over a one- or two-element list and a hash set would cost more than
1899        // it saves.
1900        let mut progress: Vec<SsaRecoveryProgress<S::Pseudonym>> = Vec::new();
1901        let mut faults: Vec<FaultObservation<S::Pseudonym>> = Vec::new();
1902        let mut terminal: Vec<ShareResolution<S::Pseudonym, S::AddressPrivateKey>> = Vec::new();
1903
1904        // Collect anything redeemed while verifiers were being installed. No retry scanning happens
1905        // here: a deferred ack is retried exactly once, by the installation of the verifier it was
1906        // waiting for (see `drain_deferred_acks`).
1907        for resolution in self.take_ready_resolutions() {
1908            match resolution {
1909                ShareResolution::Progress(snapshot) => record_progress(&mut progress, snapshot),
1910                ShareResolution::InvalidShares {
1911                    peer,
1912                    ssa_id,
1913                    observed_total,
1914                } => record_fault(&mut faults, (peer, ssa_id, observed_total)),
1915                other => push_unique(&mut terminal, other),
1916            }
1917        }
1918
1919        for (ack, ack_challenge) in half_keys_challenges {
1920            match self.process_verified_ack(ack, ack_challenge, &awaiting_ack_from_peer) {
1921                Ok(ProcessedAckResult::FullRecovery(ssa, snapshot)) => {
1922                    record_progress(&mut progress, snapshot);
1923                    push_unique(&mut terminal, ShareResolution::RecoveredSsa(ssa));
1924                }
1925                Ok(ProcessedAckResult::EarlyRecovery(snapshot)) => {
1926                    let ssa_id = snapshot.ssa_id;
1927                    record_progress(&mut progress, snapshot);
1928                    push_unique(&mut terminal, ShareResolution::AlmostRecoveredSsa(ssa_id));
1929                }
1930                Ok(ProcessedAckResult::Progressed(snapshot)) => record_progress(&mut progress, snapshot),
1931                Ok(ProcessedAckResult::InvalidShare(ssa_id, observed_total)) => {
1932                    tracing::error!(%ssa_id, observed_total, "encountered share that could not be verified");
1933                    record_fault(&mut faults, (Box::new(peer), ssa_id, observed_total));
1934                }
1935                Ok(ProcessedAckResult::NoProgress) => {}
1936                Ok(ProcessedAckResult::VerifierNotReady(spi)) => {
1937                    // The share stays in `awaiting_acks`; bucket the ack under the polynomial whose
1938                    // verifier it needs, so installing that verifier redeems it.
1939                    tracing::trace!(%peer, %spi, "verifier not yet installed, deferring acknowledgement");
1940                    self.defer_ack(spi, (peer, ack_challenge, ack));
1941                }
1942                Err(PixError::ShareIsEmpty) => tracing::trace!(%peer, "received empty share"),
1943                Err(error) => {
1944                    tracing::error!(%error, "failed to process acknowledgement");
1945                }
1946            }
1947        }
1948
1949        let mut res = Vec::with_capacity(progress.len() + faults.len() + terminal.len());
1950        res.extend(progress.into_iter().map(ShareResolution::Progress));
1951        res.extend(
1952            faults
1953                .into_iter()
1954                .map(|(peer, ssa_id, observed_total)| ShareResolution::InvalidShares {
1955                    peer,
1956                    ssa_id,
1957                    observed_total,
1958                }),
1959        );
1960        res.append(&mut terminal);
1961        Ok(res)
1962    }

4552    ///

4554    /// them apart from the cache miss alone: a cycle not yet published, whose acks are worth
4555    /// bucketing, or one already retired. Advancing the Session frontier at recovery separates
4556    /// them, and the one recovered-tail slot preserves only the latter cycle's paid remainder.

5148    /// **H8 regression.** Reclamation is scoped to the cycle, so a share for *any* polynomial keeps
5149    /// the whole cycle alive.
5150    ///
5151    /// This used to fail. The part builders were keyed per polynomial with an idle timer, so the
5152    /// clock measured "time since a share for *this* polynomial arrived". Commitments are a
5153    /// fraction of a percent of a cycle's bytes, so every builder was installed in the opening
5154    /// moments and then waited, while shares arrive polynomial-major and spread across the whole
5155    /// cycle. Any polynomial late in the emission order had its builder evicted before its first
5156    /// share landed — unrecoverably, since the commitment cannot be retransmitted and a deferred
5157    /// ack's only drain is an installation that had already happened. The SSA never completed and
5158    /// the deposit burned.
5159    ///
5160    /// The condition was `quota / line_rate > unused_verifier_lifetime`. At the deployed 1.5 Mbps
5161    /// per-Session cap a 519 MiB cycle runs 48.4 minutes against a 30-minute default, so every
5162    /// polynomial past ≈62 % of the cycle was lost.

5165    ///
5166    /// The essential geometry is that the cycle is *continuously* busy while any single polynomial
5167    /// is not. Shares are spaced at half the lifetime, so the cycle never goes idle, but the four

5673    /// A client commitment crafted so the Entry alone knows the *combined* deposit key must be
5674    /// refused, and must not produce a deposit address.
5675    ///
5676    /// The deposit key is `s + e`, where `s` is the sum of the Entry's polynomial constant terms and
5677    /// `e` is the Exit's commitment secret. Neither party is supposed to know the sum. But
5678    /// `SsaRequest` hands `e·G` to the Entry *before* it chooses its own constant terms
5679    /// (`protocols/start/src/lib.rs:305`, consumed at `transport/session/src/manager.rs:2879-2883`),
5680    /// so an Entry that is made to prove nothing can pick `w`, publish constant terms summing to
5681    /// `w·G − e·G`, and end up with a combined commitment of `w·G` — a key it can sweep alone.
5682    ///
5683    /// It cannot then produce shares for the polynomial whose constant term it does not know, so the
5684    /// Exit never recovers the SSA and is never paid. But the Entry controls emission order
5685    /// (`generator.rs` builds `poly_queue`, `next_share` drains `front_mut()`) and puts that
5686    /// polynomial last, so it is served nearly the whole cycle before the Exit notices — by which
5687    /// time it has already swept the deposit. Note this is distinct from the by-design burn
5688    /// semantics: there *neither* party can recover, whereas here the party that owes can.
5689    ///
5690    /// Before [`SsaCommitmentProof`] existed this construction was accepted, and the Exit published
5691    /// `addr(w·G)` as the address to watch — verified by asserting exactly that. The proof cannot be
5692    /// forged here because producing it would require `dlog(w·G − e·G)`, and knowing that together
5693    /// with `w` yields `e`. So the assertion is now inverted.
5694    #[test]
5695    fn exit_refuses_a_client_commitment_whose_deposit_key_the_entry_knows() -> anyhow::Result<()> {
5696        const POLYS: u16 = 3;
5697        const THRESHOLD: u8 = 2;
5698
5699        let mut rng = hopr_types::crypto_random::rng();
5700        let ssa_id = SsaId::new(SimplePseudonym::random(), SsaIndex::MIN);
5701        let reconstructor = SsaReconstructor::<TestSpec>::new(SsaReconstructorConfig::default());
5702
5703        // The Exit reveals its half. This is exactly what `SsaRequest` carries to the Entry.
5704        let exit_public = reconstructor.new_exit_commitment(ssa_id, test_params(POLYS, THRESHOLD))?;
5705
5706        // The attacker picks the deposit key it wants to end up holding.
5707        let w = PixScalar::<TestSpec>::random(&mut rng);
5708        let target = PixGroup::<TestSpec>::mul_by_generator(&w);
5709
5710        // Honest constant terms for every polynomial but the last.
5711        let honest: Vec<PixGroup<TestSpec>> = (0..POLYS - 1)
5712            .map(|_| PixGroup::<TestSpec>::mul_by_generator(&PixScalar::<TestSpec>::random(&mut rng)))
5713            .collect();
5714        let honest_sum: PixGroup<TestSpec> = honest.iter().copied().sum();
5715
5716        // The last one is obtained by group subtraction. The attacker never learns its discrete log
5717        // — that would require the Exit's secret — and does not need to.
5718        let rogue = target - exit_public - honest_sum;
5719
5720        let mut constant_terms: HashMap<PolynomialIndex, PixGroupRepr<TestSpec>> = honest
5721            .iter()
5722            .enumerate()
5723            .map(|(poly_index, c0)| (poly_index as PolynomialIndex, c0.to_bytes()))
5724            .collect();
5725        constant_terms.insert((POLYS - 1) as PolynomialIndex, rogue.to_bytes());
5726
5727        // The best the attacker can offer is a proof over the client commitment it actually
5728        // published, using the only scalar it knows — which is not that commitment's discrete log.
5729        let bogus_proof = SsaCommitmentProof::prove(&ssa_id, &w, &(target - exit_public))?;
5730
5731        let rejected =
5732            reconstructor.insert_coefficient_commitments(ssa_id, 0, Some(bogus_proof), constant_terms.into_iter());
5733        assert!(
5734            matches!(rejected, Err(PixError::UnprovenSsaCommitment)),
5735            "a commitment whose discrete logarithm the sender does not know must be refused, got {rejected:?}"
5736        );
5737
5738        // Nothing about the cycle may have been published: with no deposit address the strategy is
5739        // never asked to fund an SSA the Entry could reclaim.
5740        assert!(
5741            reconstructor
5742                .commitment_builder
5743                .get(&ssa_id)
5744                .is_some_and(|b| b.lock().get_deposit_address().is_none()),
5745            "no deposit address may be derived from an unproven commitment"
5746        );
5747        // The published cycle is what makes an SSA live and able to accept recovered shares.
5748        // `commitment_builder` legitimately still exists — `new_exit_commitment` created it.
5749        assert_eq!(
5750            0,
5751            reconstructor.live_cycles(),
5752            "no cycle must be published for an unproven commitment"
5753        );
5754
5755        Ok(())
5756    }
