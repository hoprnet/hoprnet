1//! Shared-pool virtual-clock timing-wheel mixer.
2//!
3//! The pool lives behind an `Arc<Mutex<_>>` and the sweep + wake timer run inline on the
4//! consumer's `poll_next` — no dedicated thread, no cross-thread channel hand-off. The pool is a
5//! flat `Vec` (see the internal `pool` module for why not a heap); the tradeoff for the dropped
6//! thread isolation is holding the mutex across each sweep and mixing on the consumer task rather
7//! than an isolated thread.
8
9use std::{
10    collections::VecDeque,
11    pin::Pin,

13        Arc,
14        atomic::{AtomicBool, AtomicUsize, Ordering},

16    task::{Context, Poll, Waker},
17    time::{Duration, Instant},
18};

30struct Pool<T> {
31    entries: Vec<Entry<T>>,
32    waker: Option<Waker>,
33    /// Wakers for senders parked on the capacity bound — signalled by the receiver whenever a
34    /// sweep releases items, when it observes closure, and on drop.
35    sender_wakers: Vec<Waker>,
36    clock: VirtualClock,
37}
38
39impl<T> Pool<T> {
40    /// Wake every sender parked on capacity and clear the list.
41    fn wake_sender_wakers(&mut self) {
42        for waker in &self.sender_wakers {
43            waker.wake_by_ref();
44        }
45        self.sender_wakers.clear();
46    }
47}
48
49struct Shared<T> {
50    pool: Arc<Mutex<Pool<T>>>,
51    sender_count: Arc<AtomicUsize>,
52    receiver_active: Arc<AtomicBool>,
53    params: PoissonParams,
54}
55
56impl<T> Clone for Shared<T> {
57    fn clone(&self) -> Self {
58        Self {
59            pool: self.pool.clone(),
60            sender_count: self.sender_count.clone(),
61            receiver_active: self.receiver_active.clone(),
62            params: self.params,
63        }
64    }
65}
66
67fn wake<T>(pool: &mut Pool<T>) {
68    if let Some(w) = pool.waker.take() {
69        w.wake();
70    }
71}

73/// Sender end of the shared-pool mixer.
74pub struct Sender<T> {
75    shared: Shared<T>,
76}
77
78impl<T> Clone for Sender<T> {
79    fn clone(&self) -> Self {
80        self.shared.sender_count.fetch_add(1, Ordering::Relaxed);
81        Self {
82            shared: self.shared.clone(),
83        }
84    }
85}
86
87impl<T> Drop for Sender<T> {
88    fn drop(&mut self) {
89        // `AcqRel` so the receiver's `Acquire` load of a zero count synchronizes-with this
90        // decrement (and thus with the push that preceded the sender's drop): without it, on a
91        // weak memory model the receiver could observe `0`, find the pool empty, and close one
92        // packet short.
93        if self.shared.sender_count.fetch_sub(1, Ordering::AcqRel) == 1 {
94            // Last sender gone: wake the receiver so it can observe closure and drain.
95            wake(&mut self.shared.pool.lock());
96        }
97    }
98}

100impl<T> Sender<T> {
101    /// Push one item into the shared pool (never blocks beyond the brief lock).
102    pub fn send(&self, item: T) -> Result<(), SenderError> {
103        let mut pool = self.shared.pool.lock();
104        // Check the flag *under the lock* so it orders against `Receiver::drop`, which sets it
105        // while holding the same lock. An unlocked check would race: a receiver dropping between
106        // the check and the push would leave the item in a pool nothing will ever sweep, yet
107        // `send` would still return `Ok` and the caller would record a lost packet as delivered.
108        if !self.shared.receiver_active.load(Ordering::Relaxed) {
109            return Err(SenderError::Closed);
110        }
111        let now = Instant::now();
112        // Reborrow through the guard's `DerefMut` once, up front: splitting `pool.entries` and
113        // `pool.clock` into two disjoint `&mut` requires a single concrete `&mut Pool<T>` — two
114        // separate `&mut pool.field` expressions each re-invoke `DerefMut` on the guard itself and
115        // the borrow checker can't see they're disjoint through that indirection.
116        let pool: &mut Pool<T> = &mut pool;
117        match pool::enqueue(&mut pool.entries, &mut pool.clock, &self.shared.params, now, now, item) {
118            Ok(()) => {
119                wake(pool);
120                Ok(())
121            }
122            // Nothing entered the pool and no clock state advanced, so there is nothing to wake
123            // the receiver for.
124            Err(_) => Err(SenderError::Full),
125        }
126    }
127}

129impl<T> futures::sink::Sink<T> for Sender<T> {
130    type Error = SenderError;
131
132    fn poll_ready(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
133        if !self.shared.receiver_active.load(Ordering::Relaxed) {
134            return Poll::Ready(Err(SenderError::Closed));
135        }
136
137        let mut pool = self.shared.pool.lock();
138        // Re-check under the lock: `Receiver::drop` clears the flag while holding it, so a
139        // receiver dropped between the fast-path load above and this acquire is caught here —
140        // `Closed` must win over `Pending`, or a sender would wait on a sweep that never runs.
141        if !self.shared.receiver_active.load(Ordering::Relaxed) {
142            return Poll::Ready(Err(SenderError::Closed));
143        }
144
145        if pool.entries.len() >= self.shared.params.capacity() {
146            pool.sender_wakers.push(cx.waker().clone());
147            return Poll::Pending;
148        }
149
150        Poll::Ready(Ok(()))
151    }
152
153    fn start_send(self: Pin<&mut Self>, item: T) -> Result<(), Self::Error> {
154        self.get_mut().send(item)
155    }
156
157    fn poll_flush(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
158        Poll::Ready(Ok(()))
159    }
160
161    fn poll_close(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
162        Poll::Ready(Ok(()))
163    }
164}

166/// Receiver end of the shared-pool mixer. Runs the sweep + adaptive timer inline.
167pub struct Receiver<T> {
168    shared: Shared<T>,
169    timer: Delay,
170    /// Items released by the last sweep, yielded one per `poll_next`.
171    ready: VecDeque<T>,
172    /// Reused scratch for `sweep` output.
173    scratch: Vec<(Duration, T)>,
174}
175
176// `T: Unpin` is required so `poll_next` can `get_mut()` the receiver: its buffered fields
177// (`VecDeque<T>`/`Vec<(_, T)>`) only implement `Unpin` when `T` does under the current toolchain.
178// Removing it would need `unsafe` pin-projection, not worth it here.
179impl<T: Unpin> Stream for Receiver<T> {
180    type Item = T;
181
182    fn poll_next(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
183        let this = self.get_mut();
184        loop {
185            if let Some(item) = this.ready.pop_front() {
186                return Poll::Ready(Some(item));
187            }
188
189            let now = Instant::now();
190            // `Acquire` pairs with the sender's `AcqRel` decrement so observing `0` guarantees
191            // every pushed entry is visible below.
192            let no_senders = this.shared.sender_count.load(Ordering::Acquire) == 0;
193
194            // Sweep outcome, decided under the lock and acted on after it drops.
195            enum Next {
196                Released,
197                Closed,
198                Sleep(Duration),
199            }
200
201            // Sweep under the lock, writing released items straight into the reused `scratch`
202            // (a field disjoint from the pool guard). Draining `scratch` into `ready` and parking
203            // the timer both happen *after* the guard drops, so senders never wait on that work.
204            let next = {
205                let mut pool = this.shared.pool.lock();
206                // See the matching comment in `Sender::send`: split-borrowing `entries` and
207                // `clock` needs one concrete `&mut Pool<T>`, not two separate derefs of the guard.
208                let pool: &mut Pool<T> = &mut pool;
209
210                let earliest_v_release = pool::sweep(
211                    &mut pool.entries,
212                    &mut pool.clock,
213                    &this.shared.params,
214                    now,
215                    &mut this.scratch,
216                );
217
218                #[cfg(all(feature = "telemetry", not(test)))]
219                {
220                    let len = pool.entries.len();
221                    crate::metrics::METRIC_QUEUE_SIZE.set(len as f64);
222                    crate::metrics::record_anonymity_set(len);
223                }
224
225                if !this.scratch.is_empty() {
226                    // A release batch is the only event that frees slots, so wake any sender parked
227                    // on the capacity bound while the lock is still held.
228                    pool.wake_sender_wakers();
229                    Next::Released
230                } else if pool.entries.is_empty() && no_senders {
231                    // Mark the channel closed under the lock, before waking, so a parked sender
232                    // re-polled by the wake below observes `Closed` rather than parking again on a
233                    // sweep that will never run.
234                    this.shared.receiver_active.store(false, Ordering::Relaxed);
235                    pool.wake_sender_wakers();
236                    Next::Closed
237                } else {
238                    match pool.waker.as_mut() {
239                        Some(w) => w.clone_from(cx.waker()),
240                        None => pool.waker = Some(cx.waker().clone()),
241                    }
242                    Next::Sleep(pool::next_wake(earliest_v_release, &pool.clock, &this.shared.params))
243                }
244            };
245
246            match next {
247                Next::Released => {
248                    for (delay, item) in this.scratch.drain(..) {
249                        tracing::trace!(delay_ms = delay.as_millis() as u64, "mixer released packet");
250                        #[cfg(all(feature = "telemetry", not(test)))]
251                        {
252                            crate::metrics::record_packet_delay(
253                                delay.as_millis() as f64,
254                                this.shared.params.metric_delay_window,
255                            );
256                            crate::metrics::record_window_miss(delay > this.shared.params.max_delay());
257                        }
258                        #[cfg(not(all(feature = "telemetry", not(test))))]
259                        let _ = delay;
260                        this.ready.push_back(item);
261                    }
262                    continue;
263                }
264                Next::Closed => {
265                    return Poll::Ready(None);
266                }
267                Next::Sleep(sleep_for) => {
268                    this.timer.reset(sleep_for);
269                    match this.timer.poll_unpin(cx) {
270                        Poll::Ready(()) => continue, // timer already elapsed; sweep again
271                        Poll::Pending => return Poll::Pending,
272                    }
273                }
274            }
275        }
276    }
277}

279impl<T: Unpin> Receiver<T> {
280    /// Receive a single mixed item.
281    pub async fn recv(&mut self) -> Option<T> {
282        self.next().await
283    }

286impl<T> Drop for Receiver<T> {
287    fn drop(&mut self) {
288        // Set the flag under the pool lock so it orders against `Sender::send`'s re-check: a
289        // sender either sees the pool still open (and its push will be swept before this drop
290        // completes) or sees `false` and returns `Closed` — never a silently orphaned push.
291        let mut pool = self.shared.pool.lock();
292        self.shared.receiver_active.store(false, Ordering::Relaxed);
293        // Wake senders parked on capacity so they re-poll and observe `Closed`.
294        pool.wake_sender_wakers();
295    }
296}

298/// Instantiate a shared-pool timing-wheel mixing channel. No engine thread is spawned.
299pub fn poisson_channel<T: Send>(cfg: MixerConfig) -> (Sender<T>, Receiver<T>) {
300    let params = PoissonParams::from_mixer(&cfg);
301    let shared = Shared {
302        pool: Arc::new(Mutex::new(Pool {
303            entries: Vec::with_capacity(params.capacity()),
304            waker: None,
305            sender_wakers: Vec::new(),
306            clock: VirtualClock::new(Instant::now()),
307        })),
308        sender_count: Arc::new(AtomicUsize::new(1)),
309        receiver_active: Arc::new(AtomicBool::new(true)),
310        params,
311    };
312    (
313        Sender { shared: shared.clone() },
314        Receiver {
315            shared,
316            timer: Delay::new(pool::IDLE_HEARTBEAT),
317            ready: VecDeque::new(),
318            scratch: Vec::new(),
319        },
320    )
321}
