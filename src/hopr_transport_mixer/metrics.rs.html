1//! Telemetry gauges and the packet-delay histogram shared by all mixer implementations.
2//!
3//! Only present with the `telemetry` feature (and never under `test`); callers guard their
4//! use sites with the same `cfg`.
5
6/// Packet count the window-miss ratio EMA is smoothed over. Independent of the configurable
7/// `metric_delay_window` (which tracks the active `max_delay` and would make the miss-ratio EMA's
8/// responsiveness swing with configuration); a fixed window keeps this gauge comparable across
9/// configs.
10#[cfg(all(feature = "telemetry", not(test)))]
11const WINDOW_MISS_EMA_PACKETS: u64 = 200;
12
13#[cfg(all(feature = "telemetry", not(test)))]
14lazy_static::lazy_static! {
15    pub static ref METRIC_QUEUE_SIZE: hopr_types::telemetry::SimpleGauge =
16        hopr_types::telemetry::SimpleGauge::new("hopr_mixer_queue_size", "Current mixer queue size").unwrap();
17    pub static ref METRIC_MIXER_AVERAGE_DELAY: hopr_types::telemetry::SimpleGauge =
18        hopr_types::telemetry::SimpleGauge::new(
19            "hopr_mixer_average_packet_delay",
20            "Average mixer packet delay averaged over a packet window"
21        )
22        .unwrap();
23    pub static ref METRIC_MIXER_PACKET_DELAY: hopr_types::telemetry::SimpleHistogram =
24        hopr_types::telemetry::SimpleHistogram::new(
25            "hopr_mixer_packet_delay_ms",
26            "Distribution of per-packet mixer output delay in milliseconds",
27            vec![1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0],
28        )
29        .unwrap();
30    /// `e * queue_size`, the live effective-anonymity-set estimate (`2^H`) for a memoryless-shaped
31    /// release distribution — the operational signal that the mixer has stopped buying anything
32    /// once traffic is too thin (see the PR description for the derivation).
33    pub static ref METRIC_MIXER_ANONYMITY_SET: hopr_types::telemetry::SimpleGauge =
34        hopr_types::telemetry::SimpleGauge::new(
35            "hopr_mixer_effective_anonymity_set",
36            "Estimated effective anonymity set (2^H) of the mixer's current occupancy"
37        )
38        .unwrap();
39    /// EMA of `P(delay > max_delay)`, the live check that the configured `miss_probability`
40    /// actually holds in production.
41    pub static ref METRIC_MIXER_WINDOW_MISS_RATIO: hopr_types::telemetry::SimpleGauge =
42        hopr_types::telemetry::SimpleGauge::new(
43            "hopr_mixer_window_miss_ratio",
44            "EMA fraction of packets whose realized delay exceeded the configured hard bound"
45        )
46        .unwrap();
47}

49/// Record one packet's output delay (ms): observe it into the `hopr_mixer_packet_delay_ms`
50/// distribution and feed the `hopr_mixer_average_packet_delay` EMA gauge (weight `1 / window`).
51/// `window` is clamped to at least 1 so a misconfigured zero can't divide-by-zero into NaN.
52#[cfg(all(feature = "telemetry", not(test)))]
53pub(crate) fn record_packet_delay(delay_ms: f64, window: u64) {
54    METRIC_MIXER_PACKET_DELAY.observe(delay_ms);
55    let weight = 1.0f64 / window.max(1) as f64;
56    METRIC_MIXER_AVERAGE_DELAY.set(weight * delay_ms + (1.0f64 - weight) * METRIC_MIXER_AVERAGE_DELAY.get());
57}
58
59/// Set `hopr_mixer_effective_anonymity_set` from the current queue size. Euler's number, not the
60/// queue-size-derived shape factor `k(x)`, is used as the multiplier: `e` is the memoryless-clock
61/// (`x -> infinity`) limit and a conservative (slight over-)estimate across the configured range.
62#[cfg(all(feature = "telemetry", not(test)))]
63pub(crate) fn record_anonymity_set(queue_size: usize) {
64    METRIC_MIXER_ANONYMITY_SET.set(std::f64::consts::E * queue_size as f64);
65}
66
67/// Feed one release's window-miss outcome into the `hopr_mixer_window_miss_ratio` EMA.
68#[cfg(all(feature = "telemetry", not(test)))]
69pub(crate) fn record_window_miss(exceeded_window: bool) {
70    let weight = 1.0f64 / WINDOW_MISS_EMA_PACKETS as f64;
71    let sample = if exceeded_window { 1.0 } else { 0.0 };
72    METRIC_MIXER_WINDOW_MISS_RATIO.set(weight * sample + (1.0f64 - weight) * METRIC_MIXER_WINDOW_MISS_RATIO.get());
73}
