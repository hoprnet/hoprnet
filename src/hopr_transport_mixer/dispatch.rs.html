1//! Runtime dispatch over the configured mixer engine.
2//!
3//! [`create`] instantiates the engine named by [`MixerConfig::mixer_type`] and wraps its sender
4//! and receiver in enums that forward `Sink`/`Stream` to the active variant, so callers stay
5//! agnostic to which engine runs. The `match` per poll is negligible against packet processing.
6
7use std::{
8    pin::Pin,
9    task::{Context, Poll},
10};
11
12use futures::{Sink, Stream};
13
14use crate::{
15    config::{MixerConfig, MixerType},
16    error::SenderError,
17};
18
19/// Match an `Any*` value and run `$call` against the inner engine handle, with the feature-gated
20/// arms declared once. The variant name never appears in the body, so all arms share `$call`.
21macro_rules! forward {
22    ($ty:ident, $val:expr, $inner:ident => $call:expr) => {
23        match $val {
24            #[cfg(feature = "uniform-channel")]
25            $ty::Uniform($inner) => $call,
26            #[cfg(feature = "poisson")]
27            $ty::Poisson($inner) => $call,
28        }
29    };
30}
31
32/// Sender over whichever engine [`create`] selected.
33pub enum AnySender<T> {
34    #[cfg(feature = "uniform-channel")]
35    Uniform(crate::channel::Sender<T>),
36    #[cfg(feature = "poisson")]
37    Poisson(crate::poisson::Sender<T>),
38}
39
40impl<T> Clone for AnySender<T> {
41    fn clone(&self) -> Self {
42        match self {
43            #[cfg(feature = "uniform-channel")]
44            AnySender::Uniform(s) => AnySender::Uniform(s.clone()),
45            #[cfg(feature = "poisson")]
46            AnySender::Poisson(s) => AnySender::Poisson(s.clone()),
47        }
48    }
49}
50
51impl<T> Sink<T> for AnySender<T> {
52    type Error = SenderError;
53
54    fn poll_ready(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
55        forward!(AnySender, self.get_mut(), s => Pin::new(s).poll_ready(cx))
56    }
57
58    fn start_send(self: Pin<&mut Self>, item: T) -> Result<(), Self::Error> {
59        forward!(AnySender, self.get_mut(), s => Pin::new(s).start_send(item))
60    }
61
62    fn poll_flush(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
63        forward!(AnySender, self.get_mut(), s => Pin::new(s).poll_flush(cx))
64    }
65
66    fn poll_close(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
67        forward!(AnySender, self.get_mut(), s => Pin::new(s).poll_close(cx))
68    }
69}

71/// Receiver over whichever engine [`create`] selected.
72pub enum AnyReceiver<T> {
73    #[cfg(feature = "uniform-channel")]
74    Uniform(crate::channel::Receiver<T>),
75    #[cfg(feature = "poisson")]
76    Poisson(crate::poisson::Receiver<T>),
77}
78
79impl<T: Unpin> Stream for AnyReceiver<T> {
80    type Item = T;
81
82    fn poll_next(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Option<Self::Item>> {
83        forward!(AnyReceiver, self.get_mut(), r => Pin::new(r).poll_next(cx))
84    }
85}

87/// Instantiate the mixer engine selected by `cfg.mixer_type`, returning an engine-agnostic
88/// sender/receiver pair.
89pub fn create<T: Send + Unpin + 'static>(cfg: MixerConfig) -> (AnySender<T>, AnyReceiver<T>) {
90    match cfg.mixer_type {
91        #[cfg(feature = "uniform-channel")]
92        MixerType::Uniform(_) => {
93            let (tx, rx) = crate::channel::channel(cfg);
94            (AnySender::Uniform(tx), AnyReceiver::Uniform(rx))
95        }
96        #[cfg(feature = "poisson")]
97        MixerType::Poisson(_) => {
98            let (tx, rx) = crate::poisson::poisson_channel(cfg);
99            (AnySender::Poisson(tx), AnyReceiver::Poisson(rx))
100        }
101        // `MixerType::Uniform` also exists under `uniform-adapter` alone (so `MixerSink` can be
102        // configured standalone), but that feature provides no `channel()` to dispatch to here.
103        #[cfg(all(feature = "uniform-adapter", not(feature = "uniform-channel")))]
104        MixerType::Uniform(_) => panic!(
105            "MixerType::Uniform selected but the `uniform-channel` feature is disabled; enable it to dispatch this \
106             engine via `create()` (`uniform-adapter` alone only provides the `MixerSink` adapter, not a channel() \
107             Sender/Receiver pair)"
108        ),
109    }
110}
