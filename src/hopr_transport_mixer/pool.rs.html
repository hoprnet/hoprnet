1//! Pure buffer-and-release logic for the virtual-clock timing-wheel mixer.
2//!
3//! This module is deliberately free of any async, threading or channel concerns so the release
4//! decision can be unit-tested in isolation. The engine in [`crate::poisson`] owns a
5//! `Vec<Entry<T>>` and calls [`enqueue`] on ingress and [`sweep`] on every wake.
6//!
7//! # Mechanism
8//!
9//! Every entry is tagged **once**, at enqueue, with a release threshold `v_release` in
10//! dimensionless virtual time: `v_release = V + g`, where `V` is the pool's [`VirtualClock`] and
11//! `g` is drawn from a truncated `Exp(1)` on `[0, g_max]` (`g_max = ln(1/miss_probability)`). The clock
12//! advances by `dt / mu_max` on every event (`dt` = elapsed wall-clock time since the last sync)
13//! plus, when `target_occupancy > 0`, one `1/target_occupancy` increment per arrival. An entry releases once
14//! `V >= v_release` — a plain comparison, no per-packet CSPRNG draw during a sweep (the one draw
15//! happens at enqueue).
16//!
17//! Three properties fall out of the clock update with no dedicated code (see the PR description
18//! for the derivation and validation numbers):
19//! - **Hard bound**: `V` advances by at least `dt/mu_max`, so an entry clears within `mu_max * g_max = max_delay` of
20//!   enqueue, regardless of load.
21//! - **Catch-up**: a delayed sweep has a larger `dt`, so `V` jumps further and more entries clear at once.
22//! - **Self-limiting overload**: every arrival advances `V`, so a burst of arrivals is itself the drain signal — no
23//!   separate overload valve is needed.
24

31/// Idle heartbeat when the pool is empty, so an engine periodically notices a dropped receiver
32/// and can shut down instead of parking forever. Both engines use it as the initial timer.
33pub(crate) const IDLE_HEARTBEAT: Duration = Duration::from_millis(200);
34/// Floor on the wake interval, so an engine never busy-spins even when the earliest pending
35/// entry's deadline is imminent. This is deliberately the *only* floor: [`next_wake`] computes an
36/// exact deadline from `v_release`, and a coarser, config-driven floor on top of that would delay
37/// waking past a known-soon deadline for no benefit — measured to cost a multi-x throughput
38/// regression under a tight `max_delay` (where `mu_max` ends up well under 1 ms) before this was
39/// caught, because the sweep would then wait a full extra tick before re-checking a backlog that
40/// was already known to be almost due.
41const MIN_WAKE: Duration = Duration::from_micros(100);
42
43/// The pool's release clock: a dimensionless virtual time `V` that advances from wall-clock time
44/// and, optionally, from arrivals. See the module docs for the update rule.
45#[derive(Clone, Copy)]
46pub(crate) struct VirtualClock {
47    v: f64,
48    synced_at: Instant,
49}
50
51impl VirtualClock {
52    pub(crate) fn new(now: Instant) -> Self {
53        Self { v: 0.0, synced_at: now }
54    }
55
56    /// Advance the time term to `now` and add `arrival_increment` (`0.0` for a pure time sync,
57    /// e.g. from [`sweep`]; `params.inv_target_occupancy` for one arrival, from [`enqueue`]). Returns the
58    /// resulting `V`. `now` going backwards relative to the last sync is a normal occurrence, not
59    /// an edge case: `Sender::send` and `Receiver::poll_next` each capture their own `Instant`
60    /// before finally acquiring the shared pool lock, so a caller holding an older timestamp can
61    /// still apply it here after a caller with a newer one already has. `saturating_duration_since`
62    /// keeps that call's own `dt` at zero rather than negative, and `synced_at` only ever moves
63    /// forward (`self.synced_at.max(now)`) so the stale call can't rewind it — otherwise the next
64    /// call's `dt` would double-count the interval already accounted for, over-advancing `V` and
65    /// releasing packets earlier than their tag intended.
66    fn advance(&mut self, now: Instant, mu_max_secs: f64, arrival_increment: f64) -> f64 {
67        debug_assert!(mu_max_secs > 0.0, "advance() must not be called in passthrough mode");
68        let dt = now.saturating_duration_since(self.synced_at).as_secs_f64();
69        self.v += dt / mu_max_secs + arrival_increment;
70        self.synced_at = self.synced_at.max(now);
71        self.v
72    }
73
74    /// Monotonically increasing key for passthrough mode (`max_delay == 0`), giving FIFO release
75    /// order with no virtual-time semantics.
76    fn next_sequence(&mut self) -> f64 {
77        self.v += 1.0;
78        self.v
79    }
80
81    /// Reset to a fresh origin at `now`. Only called from [`sweep`] at the moment the pool goes
82    /// empty, which is the one point a reset cannot affect any live entry's relative order —
83    /// bounding `V`'s growth over a long-lived, intermittently-idle engine instead of letting it
84    /// grow for the process lifetime. Also clamped to move `synced_at` only forward, for the same
85    /// reason as [`Self::advance`].
86    fn rebase(&mut self, now: Instant) {
87        self.v = 0.0;
88        self.synced_at = self.synced_at.max(now);
89    }
90}

92/// Draw `g ~ truncated Exp(1)` on `[0, g_max)` via inverse-CDF sampling: with `u ~ U[0,1)`,
93/// `g = -ln(1 - u*z)`, `z = 1 - e^-g_max`. `g -> 0` as `u -> 0`; `g -> g_max` as `u -> 1`.
94///
95/// Takes the precomputed `z` rather than `g_max` itself: `z` is constant for the engine's
96/// lifetime, so computing it here (an `exp()` this function would otherwise repeat on every
97/// single enqueue) would be pure waste on the hot path — [`PoissonParams`] computes it once.
98fn sample_g(z: f64) -> f64 {
99    let u = crypto_random::random_float();
100    -(1.0 - u * z).ln()
101}

103/// Timing-wheel tuning resolved from a [`MixerConfig`]/[`PoissonConfig`], holding everything
104/// [`enqueue`]/[`sweep`]/[`next_wake`] need without re-reading the public config each call.
105#[derive(Clone, Copy)]
106pub(crate) struct PoissonParams {
107    /// The configured hard bound. Zero selects passthrough (no delay, FIFO order). Read outside
108    /// tests only by the engines' telemetry path, so it is dead in non-telemetry, non-test builds.
109    #[allow(dead_code)]
110    max_delay: Duration,
111    /// `max_delay.as_secs_f64() / g_max`; the slowest the release clock ever holds an entry via the
112    /// time term alone. `0.0` in passthrough mode (never read there).
113    mu_max_secs: f64,
114    /// `1 - e^-g_max`, precomputed once so [`sample_g`] does no `exp()` per packet — `g_max`
115    /// itself (`ln(1/miss_probability)`) is only ever needed to derive this and `mu_max_secs`.
116    z: f64,
117    /// `1/target_occupancy`, or `0.0` when `target_occupancy == 0` (arrival term folds to a no-op).
118    inv_target_occupancy: f64,
119    capacity: usize,
120    // Read only by the engines' telemetry path, so it is dead in non-telemetry / test builds.
121    #[allow(dead_code)]
122    pub(crate) metric_delay_window: u64,
123}
124
125impl PoissonParams {
126    pub(crate) fn new(cfg: &MixerConfig, poisson: &PoissonConfig) -> Self {
127        let g_max = (1.0 / poisson.miss_probability).ln();
128        let mu_max_secs = if poisson.max_delay.is_zero() {
129            0.0
130        } else {
131            poisson.max_delay.as_secs_f64() / g_max
132        };
133        let inv_target_occupancy = if poisson.target_occupancy > 0 {
134            1.0 / poisson.target_occupancy as f64
135        } else {
136            0.0
137        };
138        Self {
139            max_delay: poisson.max_delay,
140            mu_max_secs,
141            z: 1.0 - (-g_max).exp(),
142            inv_target_occupancy,
143            capacity: cfg.capacity,
144            metric_delay_window: cfg.metric_delay_window(),
145        }
146    }
147
148    /// Resolve the params from a config, taking the [`PoissonConfig`] from its `mixer_type` (or
149    /// defaults when a non-Poisson variant is passed to a Poisson engine directly).
150    pub(crate) fn from_mixer(cfg: &MixerConfig) -> Self {
151        let poisson = match cfg.mixer_type {
152            crate::config::MixerType::Poisson(poisson) => poisson,
153            // `create` only routes the Poisson variant here, so this is unreachable in practice;
154            // assert in debug so a direct caller passing a non-Poisson config fails loudly.
155            #[allow(unreachable_patterns)]
156            _ => {
157                debug_assert!(false, "PoissonParams::from_mixer called with a non-Poisson mixer_type");
158                PoissonConfig::default()
159            }
160        };
161        Self::new(cfg, &poisson)
162    }
163
164    pub(crate) fn capacity(&self) -> usize {
165        self.capacity
166    }
167
168    fn is_passthrough(&self) -> bool {
169        self.mu_max_secs <= 0.0
170    }

172    /// The configured hard bound, read by the engines' telemetry path to classify a realized
173    /// delay as a window miss. Dead outside tests in a non-telemetry build, same as the field.
174    #[allow(dead_code)]
175    pub(crate) fn max_delay(&self) -> Duration {
176        self.max_delay
177    }

179    #[cfg(test)]
180    pub(crate) fn mu_max_secs(&self) -> f64 {
181        self.mu_max_secs
182    }

185/// A single buffered item awaiting release.
186///
187/// The pool is a flat `Vec`, not a heap ordered by `v_release`: under the workload this engine
188/// actually sees, most of a sweep's due entries become due **together** (a burst that arrived
189/// close in time shares a similar `v_release`), so a heap pays `O(log n)` on every insert to
190/// maintain a global order that a same-sweep batch release makes irrelevant a moment later. A
191/// flat scan is `O(n)` with no such tax, and — measured — clearly faster on the reused-channel
192/// throughput benchmark. `sweep` still tracks the minimum `v_release` among kept entries in the
193/// same pass, so [`next_wake`] loses no precision from dropping the heap.
194pub(crate) struct Entry<T> {
195    /// Virtual-time release threshold, fixed at enqueue: the entry releases once the pool's
196    /// [`VirtualClock`] reaches this value.
197    pub v_release: f64,
198    /// When the item entered the mixer (wall clock), used only to compute the realized delay at
199    /// release — never re-read for release timing itself.
200    pub enqueued_at: Instant,
201    pub item: T,
202}
203
204/// Enqueue one item at `now` (used to advance the clock's time term) with delay measured from
205/// `sent_at` (its own ingress instant — the same as `now` for a same-thread push; earlier than
206/// `now` when absorbed from a channel, so the realized delay reported at release reflects the
207/// packet's actual wait rather than the engine's batch-processing instant).
208///
209/// Draws this entry's release tag once, here — the sweep never draws randomness.
210pub(crate) fn enqueue<T>(
211    pool: &mut Vec<Entry<T>>,
212    clock: &mut VirtualClock,
213    params: &PoissonParams,
214    now: Instant,
215    sent_at: Instant,
216    item: T,
217) {
218    let v_release = if params.is_passthrough() {
219        clock.next_sequence()
220    } else {
221        let v = clock.advance(now, params.mu_max_secs, params.inv_target_occupancy);
222        v + sample_g(params.z)
223    };
224    pool.push(Entry {
225        v_release,
226        enqueued_at: sent_at,
227        item,
228    });
229}
230
231/// Evaluate the whole pool at `now`, moving due items into `out` (cleared first) paired with
232/// their realized delay (`now - enqueued_at`), in randomized order. Returns the minimum
233/// `v_release` still buffered afterwards (for [`next_wake`]), or `None` when the pool ends up
234/// empty.
235///
236/// - Empty pool: rebase the clock (see [`VirtualClock::rebase`]) and return — the free, safe point to bound `V`'s
237///   long-run growth.
238/// - Passthrough (`max_delay == 0`): drain everything in FIFO order, no shuffle.
239/// - Otherwise: advance the clock's time term (no arrival increment — a sweep is not an arrival), then scan once,
240///   moving every entry with `v_release <= V` into `out` (`swap_remove` — pool order carries no meaning) and tracking
241///   the minimum `v_release` among the rest, then shuffle the released batch.
242pub(crate) fn sweep<T>(
243    pool: &mut Vec<Entry<T>>,
244    clock: &mut VirtualClock,
245    params: &PoissonParams,
246    now: Instant,
247    out: &mut Vec<(Duration, T)>,
248) -> Option<f64> {
249    out.clear();
250
251    if pool.is_empty() {
252        clock.rebase(now);
253        return None;
254    }
255
256    if params.is_passthrough() {
257        for e in pool.drain(..) {
258            out.push((now.saturating_duration_since(e.enqueued_at), e.item));
259        }
260        return None;
261    }
262
263    let v = clock.advance(now, params.mu_max_secs, 0.0);
264    let mut earliest: Option<f64> = None;
265    let mut i = 0;
266    while i < pool.len() {
267        if pool[i].v_release <= v {
268            let e = pool.swap_remove(i);
269            out.push((now.saturating_duration_since(e.enqueued_at), e.item));
270        } else {
271            let v_release = pool[i].v_release;
272            earliest = Some(earliest.map_or(v_release, |e| e.min(v_release)));
273            i += 1;
274        }
275    }
276
277    shuffle(out);
278    earliest
279}
280
281/// Next wake: exactly when the earliest still-buffered `v_release` (from [`sweep`]'s return
282/// value, so no extra scan) becomes due, converted back to a wall-clock duration
283/// (`dt = mu_max * (v_release - V)`), floored only at [`MIN_WAKE`] — see its doc comment for why
284/// no coarser, config-driven floor belongs here. An idle heartbeat when the pool is empty, or in
285/// passthrough mode (which drains fully on every sweep, so there is nothing to wait for beyond
286/// noticing a dropped receiver).
287pub(crate) fn next_wake(earliest_v_release: Option<f64>, clock: &VirtualClock, params: &PoissonParams) -> Duration {
288    if params.is_passthrough() {
289        return IDLE_HEARTBEAT;
290    }
291    match earliest_v_release {
292        None => IDLE_HEARTBEAT,
293        Some(v_release) => {
294            let dv = (v_release - clock.v).max(0.0);
295            Duration::from_secs_f64(dv * params.mu_max_secs).max(MIN_WAKE)
296        }
297    }
298}

300/// In-place cryptographic Fisher–Yates shuffle, so the enqueue order of a released
301/// batch cannot be read off the output within a single wake.
302fn shuffle<T>(items: &mut [T]) {
303    let n = items.len();
304    if n <= 1 {
305        return;
306    }
307
308    for i in (1..n).rev() {
309        let j = crypto_random::random_integer(0, Some((i + 1) as u64)) as usize;
310        items.swap(i, j);
311    }
312}

350    /// Draw one `Exp(lambda_per_s)` inter-arrival gap, in seconds.
351    fn exp_gap_s(lambda_per_s: f64) -> f64 {
352        -(1.0 - crypto_random::random_float()).ln() / lambda_per_s
353    }
354
355    /// Outcome of a timer-driven simulation.
356    struct SimResult {
357        /// Realized delay (ms) of each released packet, in release order.
358        delays_ms: Vec<f64>,
359        /// Occupancy sampled once per tick, averaged over the run.
360        mean_occupancy: f64,
361        /// Peak occupancy observed over the run.
362        max_occupancy: usize,
363    }
364
365    /// Drive Poisson arrivals of rate `lambda_per_s` through the pool for `duration_s` of virtual
366    /// time, sweeping on a **fixed wall-clock tick** rather than only at arrivals.
367    ///
368    /// This distinction is load-bearing: checking releases only when a new arrival happens
369    /// inflates both the realized mean and max delay by up to one inter-arrival gap, because a
370    /// due entry sits unreleased until the next arrival triggers a check. That artifact produced
371    /// a spurious hard-bound violation during design (226 ms against a 200 ms bound at a low
372    /// arrival rate) and must not reappear here.
373    fn simulate(params: &PoissonParams, lambda_per_s: f64, duration_s: f64, tick: Duration) -> SimResult {
374        let t0 = Instant::now();
375        let mut pool: Vec<Entry<u64>> = Vec::new();
376        let mut clock = VirtualClock::new(t0);
377        let mut out = Vec::new();
378        let mut delays_ms = Vec::new();
379        let (mut occ_sum, mut occ_samples, mut max_occ) = (0u64, 0u64, 0usize);
380
381        let mut next_arrival_s = exp_gap_s(lambda_per_s);
382        let mut t_s = 0.0f64;
383        let mut seq = 0u64;
384        let tick_s = tick.as_secs_f64();
385
386        while t_s < duration_s {
387            let t_next_s = t_s + tick_s;
388            while next_arrival_s <= t_next_s {
389                let arrival_now = t0 + Duration::from_secs_f64(next_arrival_s);
390                enqueue(&mut pool, &mut clock, params, arrival_now, arrival_now, seq);
391                seq += 1;
392                next_arrival_s += exp_gap_s(lambda_per_s);
393            }
394            let now_tick = t0 + Duration::from_secs_f64(t_next_s);
395            sweep(&mut pool, &mut clock, params, now_tick, &mut out);
396            delays_ms.extend(out.drain(..).map(|(d, _)| d.as_secs_f64() * 1000.0));
397
398            t_s = t_next_s;
399            occ_sum += pool.len() as u64;
400            occ_samples += 1;
401            max_occ = max_occ.max(pool.len());
402        }
403
404        SimResult {
405            delays_ms,
406            mean_occupancy: occ_sum as f64 / occ_samples.max(1) as f64,
407            max_occupancy: max_occ,
408        }
409    }
410
411    fn mean_of(xs: &[f64]) -> f64 {
412        xs.iter().sum::<f64>() / xs.len() as f64
413    }
414
415    fn coefficient_of_variation(xs: &[f64]) -> f64 {
416        let m = mean_of(xs);
417        let var = xs.iter().map(|x| (x - m).powi(2)).sum::<f64>() / xs.len() as f64;
418        var.sqrt() / m
419    }

425    /// `Sender::send` and `Receiver::poll_next` both call into the shared `VirtualClock` while
426    /// holding the pool lock, but each captures its own `Instant`. If a caller that captured an
427    /// *older* `now` (e.g. the receiver, before it managed to acquire a lock a sender briefly
428    /// held) ends up applying it to the clock *after* a caller with a *newer* `now` already did,
429    /// `synced_at` must not regress — otherwise the next call's `dt` double-counts the interval
430    /// between the two, over-advancing `V` and releasing packets earlier than their tag intended.
431    #[test]
432    fn virtual_clock_should_not_rewind_when_advanced_out_of_order() {
433        let t0 = Instant::now();
434        let mut clock = VirtualClock::new(t0);
435
436        // A fresher `now` (e.g. a sender that just acquired the lock) advances the clock first.
437        clock.advance(t0 + Duration::from_millis(10), 1.0, 0.0);
438
439        // A stale `now` (e.g. a receiver that captured it before waiting on the lock) arrives
440        // second. It must contribute no elapsed time and must not move `synced_at` backwards.
441        clock.advance(t0 + Duration::from_millis(5), 1.0, 0.0);
442
443        // The true elapsed wall time from `t0` is 20ms; with `mu_max_secs = 1.0`, `V` should equal
444        // exactly that — not 25ms, which is what re-opening the [5ms, 10ms] interval would give.
445        let v = clock.advance(t0 + Duration::from_millis(20), 1.0, 0.0);
446        let expected = Duration::from_millis(20).as_secs_f64();
447        assert!(
448            (v - expected).abs() < 1e-9,
449            "V should equal total elapsed wall time ({expected}s), got {v}s — synced_at was rewound"
450        );
451    }

511    #[test]
512    fn a_delayed_sweep_should_release_a_larger_batch() {
513        // A gap in sweeps is a larger `dt` on the next one, so V jumps further and more entries
514        // clear at once — this should already hold with no dedicated code, since it falls
515        // directly out of the clock update. Two independent trials from an identical starting
516        // pool, differing only in the gap before the one sweep each performs — comparing batch
517        // sizes from *sequential* sweeps on one shared pool would confound "bigger gap" with
518        // "pool was fuller when this sweep ran first", which is a different effect entirely.
519        let params = bounded_params(20, 0.01);
520
521        let batch_after_gap = |gap: Duration| {
522            let t0 = Instant::now();
523            let mut pool: Vec<Entry<u32>> = Vec::new();
524            let mut clock = VirtualClock::new(t0);
525            let mut out = Vec::new();
526            for i in 0..2000u32 {
527                enqueue(&mut pool, &mut clock, &params, t0, t0, i);
528            }
529            sweep(&mut pool, &mut clock, &params, t0 + gap, &mut out);
530            out.len()
531        };
532
533        let normal_batch = batch_after_gap(Duration::from_millis(2));
534        let delayed_batch = batch_after_gap(Duration::from_millis(2) + Duration::from_millis(10));
535
536        assert!(
537            delayed_batch > normal_batch,
538            "a delayed sweep (batch {delayed_batch}) should release more than a normal one (batch {normal_batch})"
539        );
540    }

561    #[test]
562    fn holding_time_should_be_geometric_in_virtual_time() {
563        // Memorylessness of the truncated-Exp(1) tag: P(g > 2k | g > k) should be roughly
564        // constant in k (up to the truncation at g_max), the discrete/virtual-time analogue of
565        // the exponential survival property.
566        const N: usize = 20_000;
567        let g_max: f64 = 10.0; // relaxed so truncation is negligible near k=0.5..1.5
568        let z = 1.0 - (-g_max).exp();
569        let gs: Vec<f64> = (0..N).map(|_| sample_g(z)).collect();
570        let survival = |k: f64| gs.iter().filter(|g| **g > k).count() as f64 / N as f64;
571        let s1 = survival(0.5);
572        let s2 = survival(1.0);
573        let s3 = survival(1.5);
574        // exp(-0.5) ratio between consecutive steps of 0.5.
575        let expected_ratio = (-0.5f64).exp();
576        assert!(
577            ((s2 / s1) - expected_ratio).abs() < 0.05,
578            "survival ratio s(1.0)/s(0.5) = {:.3} should be near e^-0.5 = {expected_ratio:.3}",
579            s2 / s1
580        );
581        assert!(
582            ((s3 / s2) - expected_ratio).abs() < 0.05,
583            "survival ratio s(1.5)/s(1.0) = {:.3} should be near e^-0.5 = {expected_ratio:.3}",
584            s3 / s2
585        );
586    }

633    #[test]
634    fn overload_should_not_require_a_valve() {
635        // Baseline load, then a sudden burst well above it: with no high_watermark/valve present
636        // at all, occupancy should still grow sublinearly with the burst multiplier, because the
637        // arrival term (target_occupancy > 0) makes a burst its own drain signal.
638        let params = privacy_params(200, 0.01, 14);
639        let t0 = Instant::now();
640        let mut pool: Vec<Entry<u64>> = Vec::new();
641        let mut clock = VirtualClock::new(t0);
642        let mut out = Vec::new();
643        let mut seq = 0u64;
644        let mut t_s = 0.0f64;
645        let mut next_arrival_s = exp_gap_s(1000.0);
646        let tick_s = Duration::from_micros(200).as_secs_f64();
647
648        // Runs one phase at `lambda_per_s` for `duration_s` starting from the current `t_s`,
649        // continuing the same pool/clock across phases (so a burst can follow a warm-up without
650        // restarting state), and returns the peak occupancy observed during that phase.
651        let mut run_phase = |lambda_per_s: f64, duration_s: f64| -> usize {
652            let until_s = t_s + duration_s;
653            let mut max_occ = pool.len();
654            while t_s < until_s {
655                let t_next_s = t_s + tick_s;
656                while next_arrival_s <= t_next_s {
657                    let now = t0 + Duration::from_secs_f64(next_arrival_s);
658                    enqueue(&mut pool, &mut clock, &params, now, now, seq);
659                    seq += 1;
660                    next_arrival_s += exp_gap_s(lambda_per_s);
661                }
662                sweep(
663                    &mut pool,
664                    &mut clock,
665                    &params,
666                    t0 + Duration::from_secs_f64(t_next_s),
667                    &mut out,
668                );
669                t_s = t_next_s;
670                max_occ = max_occ.max(pool.len());
671            }
672            max_occ
673        };
674
675        run_phase(1000.0, 5.0); // warm up at baseline load
676        let max_occ_during_burst = run_phase(200_000.0, 0.05); // 200x burst for 50ms
677
678        assert!(
679            (max_occ_during_burst as f64) < 2.0 * 14.0,
680            "a 200x burst should not push occupancy past 2x target_occupancy with no valve present, got \
681             {max_occ_during_burst}"
682        );
683    }
