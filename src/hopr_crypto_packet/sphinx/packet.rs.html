24/// Holds data that are padded up to `P + 1`.
25///
26/// Data in this instance is guaranteed to be always `P + 1` bytes-long.
27// TODO: make P a typenum argument
28#[derive(Clone, Debug, PartialEq, Eq)]
29pub struct PaddedPayload<const P: usize>(Bytes);
30
31impl<const P: usize> PaddedPayload<P> {
32    /// Byte used to pad the data.
33    pub const PADDING: u8 = 0x00;
34    /// Tag used to separate padding from data
35    pub const PADDING_TAG: u8 = 0xaa;
36    /// Size of the padded data.
37    pub const SIZE: usize = P + size_of_val(&Self::PADDING_TAG);
38
39    /// Creates a new instance from the given message `msg` shorter than [`PaddedPayload::SIZE`] and pads it.
40    ///
41    /// The padding consists of prepending a [`PaddedPayload::PADDING_TAG`], preceded by as many zero bytes
42    /// to fill it up to [`PaddedPayload::SIZE`]. If data is `P` bytes-long, only the padding tag is prepended.
43    ///
44    /// If the argument's length is greater or equal to [`PaddedPayload::SIZE`], [`SphinxError::PaddingError`] is
45    /// returned.
46    pub fn new(msg: &[u8]) -> Result<Self, SphinxError> {
47        if msg.len() < Self::SIZE {
48            // Zeroes followed by the PADDING_TAG and then the message
49            let mut ret = vec![Self::PADDING; Self::SIZE];
50            ret[Self::SIZE - msg.len() - 1] = Self::PADDING_TAG;
51            ret[Self::SIZE - msg.len()..].copy_from_slice(msg);
52
53            Ok(Self(ret.into()))
54        } else {
55            Err(SphinxError::PaddingError)
56        }
57    }
58
59    /// Similar like [`PaddedPayload::new`], but creates a new instance from a vector,
60    /// reallocating only if the given vector has insufficient capacity.
61    pub fn new_from_vec(mut msg: Vec<u8>) -> Result<Self, SphinxError> {
62        let len = msg.len();
63        if len >= Self::SIZE {
64            return Err(SphinxError::PaddingError);
65        }
66
67        msg.resize(Self::SIZE, Self::PADDING); // Reallocates only if capacity is not enough
68        msg.copy_within(0..len, Self::SIZE - len);
69        msg[0..Self::SIZE - len].fill(Self::PADDING);
70        msg[Self::SIZE - len - 1] = Self::PADDING_TAG;
71
72        Ok(Self(msg.into()))
73    }
74
75    /// Creates a new instance from an already padded message `msg` and takes its ownership.
76    ///
77    /// This method only checks the length of the argument, it does not verify
78    /// the presence of the padding tag. If the padding tag is not present, an error
79    /// is later returned when [`PaddedPayload::into_unpadded`] is called.
80    ///
81    /// If the argument's length is not equal to [`PaddedPayload::SIZE`], [`SphinxError::PaddingError`] is returned.
82    pub fn from_padded_bytes(msg: Bytes) -> Result<Self, SphinxError> {
83        if msg.len() == Self::SIZE {
84            Ok(Self(msg))
85        } else {
86            Err(SphinxError::PaddingError)
87        }
88    }
89
90    /// Consumes the instance by removing the padding and returning the unpadded data
91    /// as a view into the same buffer. The original length of the data is restored.
92    ///
93    /// If the padding tag could not be found, [`SphinxError::PaddingError`] is returned.
94    /// This means this instance was created using [`PaddedPayload::from_padded_bytes`] with invalid data.
95    pub fn into_unpadded(self) -> Result<Bytes, SphinxError> {
96        let tag_pos = self
97            .0
98            .iter()
99            .position(|x| *x == Self::PADDING_TAG)
100            .ok_or(SphinxError::PaddingError)?;
101        let mut data = self.0;
102        data.advance(tag_pos + 1);
103        Ok(data)
104    }
105}

166/// Describes how a [`MetaPacket`] should be routed to the destination.
167pub enum MetaPacketRouting<'a, S: SphinxSuite, H: SphinxHeaderSpec> {
168    /// Uses an explicitly given path to deliver the packet.
169    ForwardPath {
170        /// Shared keys with individual hops
171        shared_keys: SharedKeys<S::E, S::G>,
172        /// Public keys on the path corresponding to the shared keys
173        forward_path: &'a [<S::P as Keypair>::Public],
174        /// Additional data for individual relayers
175        additional_data_relayer: &'a [H::RelayerData],
176        /// Additional data delivered to the packet's final recipient.
177        receiver_data: &'a H::PacketReceiverData,
178        /// Special flag used for acknowledgement signaling to the recipient
179        no_ack: bool,
180    },
181    /// Uses a SURB to deliver the packet and some additional data to the SURB's creator.
182    Surb(SURB<S, H>, &'a H::PacketReceiverData),
183}

185/// Represents a packet that is only partially instantiated,
186/// that is - it contains only the routing information and the Alpha value.
187///
188/// This object can be used to pre-compute a packet without a payload
189/// and possibly serialize it, and later to be
190/// deserialized and used to construct the final [`MetaPacket`] via
191/// a call to [`PartialPacket::into_meta_packet`].
192#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
193pub struct PartialPacket<S: SphinxSuite, H: SphinxHeaderSpec> {
194    alpha: Alpha<<S::G as GroupElement<S::E>>::AlphaLen>,
195    routing_info: RoutingInfo<H>,
196    prp_inits: Vec<IvKey<S::PRP>>,
197}
198
199impl<S: SphinxSuite, H: SphinxHeaderSpec> PartialPacket<S, H> {
200    /// Creates a new partial packet using the given routing information and
201    /// public key identifier mapper.
202    pub fn new<M: ProtocolKeyIdMapper<S, H>>(
203        routing: MetaPacketRouting<S, H>,
204        key_mapper: &M,
205    ) -> Result<Self, SphinxError> {
206        match routing {
207            MetaPacketRouting::ForwardPath {
208                shared_keys,
209                forward_path,
210                additional_data_relayer,
211                receiver_data,
212                no_ack,
213            } => {
214                let routing_info = RoutingInfo::<H>::new(
215                    &forward_path
216                        .iter()
217                        .map(|key| {
218                            key_mapper.map_key_to_id(key).ok_or_else(|| {
219                                SphinxError::PacketConstructionError(format!("key id not found for {}", key.to_hex()))
220                            })
221                        })
222                        .collect::<Result<Vec<_>, SphinxError>>()?,
223                    &shared_keys.secrets,
224                    additional_data_relayer,
225                    receiver_data,
226                    false,
227                    no_ack,
228                )?;
229
230                Ok(Self {
231                    alpha: shared_keys.alpha,
232                    routing_info,
233                    prp_inits: shared_keys
234                        .secrets
235                        .into_iter()
236                        .rev()
237                        .map(|key| S::new_prp_init(&key))
238                        .collect::<Result<Vec<_>, _>>()?,
239                })
240            }
241            MetaPacketRouting::Surb(surb, receiver_data) => Ok(Self {
242                alpha: surb.alpha,
243                routing_info: surb.header,
244                prp_inits: vec![S::new_reply_prp_init(&surb.sender_key, receiver_data.as_ref())?],
245            }),
246        }
247    }
248
249    /// Transform this partial packet into an actual [`MetaPacket`] using the given payload.
250    pub fn into_meta_packet<const P: usize>(self, payload: PaddedPayload<P>) -> MetaPacket<S, H, P> {
251        let mut packet = MetaPacket::new_from_parts(self.alpha, self.routing_info, &payload);
252        for iv_key in self.prp_inits {
253            let prp = iv_key.into_init::<S::PRP>();
254            // The following won't panic, because PaddedPayload<P> is guaranteed to be S::PRP::BlockSize bytes-long
255            // However, it would be nicer to make PaddedPayload take P as a typenum parameter
256            // and enforce this invariant at compile time.
257            let block =
258                <&mut crypto_traits::Block<S::PRP>>::try_from(packet.payload_mut()).expect("block size mismatch");
259            prp.forward(block);
260        }
261
262        packet
263    }
264}

294/// An encrypted packet with a payload of size `P`.
295/// The final packet size is given by [`MetaPacket::SIZE`].
296///
297/// A sender can create a new packet via [`MetaPacket::new`] and send it.
298/// Once received by the recipient, it is parsed first by calling [`MetaPacket::try_from`]
299/// and then it can be transformed into [`ForwardedMetaPacket`] by calling
300/// the [`MetaPacket::into_forwarded`] method. The [`ForwardedMetaPacket`] then contains the information
301/// about the next recipient of this packet or the payload for the final destination.
302///
303/// The packet format is directly dependent on the used [`SphinxSuite`].
304pub struct MetaPacket<S, H, const P: usize> {
305    packet: Vec<u8>,
306    _d: PhantomData<(S, H)>,
307}
308

310    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
311        write!(f, "{}", self.to_hex())
312    }
313}

325/// What [`MetaPacket::forward_secret`] derives for this hop, and what
326/// [`MetaPacket::into_forwarded_with_secret`] consumes.
327///
328/// The fields are private and the tag is derived only here, so a caller cannot pair a tag with a secret it does not
329/// belong to - the invariant is carried by the type rather than by a doc comment.
330pub struct ForwardSecret<S: SphinxSuite> {
331    alpha: Alpha<<S::G as GroupElement<S::E>>::AlphaLen>,
332    secret: SharedSecret,
333    tag: PacketTag,
334}
335
336/// The three fields of a [`ForwardSecret`], for the transform's internal destructuring.
337///
338/// The tag is here as well as in `ForwardSecret`, so the pairing is decided inside `forward_secret` rather than by a
339/// caller.
340type ForwardSecretParts<S> = (
341    Alpha<<<S as SphinxSuite>::G as GroupElement<<S as SphinxSuite>::E>>::AlphaLen>,
342    SharedSecret,
343    PacketTag,
344);
345
346impl<S: SphinxSuite> ForwardSecret<S> {
347    /// The packet's [`PacketTag`], derived from the secret above.
348    ///
349    /// This is what a replay filter keys on: a value only a party that derived this hop's secret can produce, and
350    /// which is stable across replays of the same packet.
351    pub fn tag(&self) -> &PacketTag {
352        &self.tag
353    }
354
355    /// Splits the value into the parts the transform needs.
356    fn into_parts(self) -> ForwardSecretParts<S> {
357        (self.alpha, self.secret, self.tag)
358    }
359}
360
361impl<S: SphinxSuite> std::fmt::Debug for ForwardSecret<S> {
362    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
363        // The secret is redacted, as everywhere else it appears.
364        f.debug_struct("ForwardSecret")
365            .field("tag", &self.tag)
366            .finish_non_exhaustive()
367    }
368}

370/// Represent a [`MetaPacket`] with one layer of encryption removed, exposing the details
371/// about the next hop.
372///
373/// There are two possible states - either the packet is intended for the recipient,
374/// and is thus [`ForwardedMetaPacket::Final`], or it is meant to be sent (relayed)
375/// to the next hop - thus it is [`ForwardedMetaPacket::Relayed`].
376#[allow(dead_code)]
377pub enum ForwardedMetaPacket<S: SphinxSuite, H: SphinxHeaderSpec, const P: usize> {
378    /// The content is another [`MetaPacket`] meant to be sent to the next hop.
379    Relayed {
380        /// Packet for the next hop.
381        packet: MetaPacket<S, H, P>,
382        /// Public key of the next hop.
383        next_node: <S::P as Keypair>::Public,
384        /// Position in the channel path of this packet.
385        path_pos: u8,
386        /// Additional data for the relayer.
387        ///
388        /// In HOPR protocol, this contains the PoR challenge that will be solved when we receive
389        /// the acknowledgement after we forward the inner packet to the next hop.
390        additional_info: H::RelayerData,
391        /// Shared secret that was used to encrypt the removed layer.
392        derived_secret: SharedSecret,
393        /// Packet checksum.
394        packet_tag: PacketTag,
395    },
396    /// The content is the actual payload for the packet's destination.
397    Final {
398        /// Decrypted payload
399        plain_text: PaddedPayload<P>,
400        /// Data for the packet receiver (containing the sender's pseudonym).
401        receiver_data: H::PacketReceiverData,
402        /// Shared secret that was used to encrypt the removed layer.
403        derived_secret: SharedSecret,
404        /// Packet checksum.
405        packet_tag: PacketTag,
406        /// Special flag used for acknowledgement signaling to the recipient
407        no_ack: bool,
408    },
409}

411impl<S: SphinxSuite, H: SphinxHeaderSpec, const P: usize> MetaPacket<S, H, P> {
412    /// The fixed length of the padded packet.
413    pub const PACKET_LEN: usize = <S::P as Keypair>::Public::SIZE + RoutingInfo::<H>::SIZE + PaddedPayload::<P>::SIZE;
414
415    /// Creates a new outgoing packet with the given payload `msg` and `routing`.
416    ///
417    /// The size of the `msg` must be less or equal `P`, otherwise the
418    /// constructor will return an error.
419    pub fn new<M: ProtocolKeyIdMapper<S, H>>(
420        payload: PaddedPayload<P>,
421        routing: MetaPacketRouting<S, H>,
422        key_mapper: &M,
423    ) -> Result<Self, SphinxError> {
424        Ok(PartialPacket::new(routing, key_mapper)?.into_meta_packet(payload))
425    }
426
427    fn new_from_parts(
428        alpha: Alpha<<S::G as GroupElement<S::E>>::AlphaLen>,
429        routing_info: RoutingInfo<H>,
430        payload: &[u8],
431    ) -> Self {
432        let mut packet = Vec::with_capacity(Self::SIZE);
433        packet.extend_from_slice(&alpha);
434        packet.extend_from_slice(routing_info.as_ref());
435        packet.extend_from_slice(&payload[0..PaddedPayload::<P>::SIZE]);
436
437        Self {
438            packet,
439            _d: PhantomData,
440        }
441    }
442
443    /// Consumes the packet and returns its buffer, including any spare capacity.
444    pub fn into_vec(self) -> Vec<u8> {
445        self.packet
446    }
447
448    /// Returns the Alpha value subslice from the packet data.
449    fn alpha(&self) -> &[u8] {
450        let len = <S::G as GroupElement<S::E>>::AlphaLen::USIZE;
451        &self.packet[..len]
452    }
453
454    /// Returns the Alpha value from the packet data as a mutable slice.
455    fn alpha_mut(&mut self) -> &mut [u8] {
456        let len = <S::G as GroupElement<S::E>>::AlphaLen::USIZE;
457        &mut self.packet[..len]
458    }
459
460    /// Returns the routing information from the packet data as a mutable slice.
461    fn routing_info_mut(&mut self) -> &mut [u8] {
462        let base = <S::G as GroupElement<S::E>>::AlphaLen::USIZE;
463        &mut self.packet[base..base + RoutingInfo::<H>::SIZE]
464    }
465
466    /// Returns the payload subslice from the packet data.
467    ///
468    /// This data is guaranteed to be `PaddedPayload::<P>::SIZE` bytes-long, which currently
469    /// is `P + 1` bytes.
470    fn payload_mut(&mut self) -> &mut [u8] {
471        let base = <S::G as GroupElement<S::E>>::AlphaLen::USIZE + RoutingInfo::<H>::SIZE;
472        &mut self.packet[base..base + PaddedPayload::<P>::SIZE]
473    }
474
475    /// Derives the shared secret this packet was built for `node_keypair`, together with the blinded alpha the
476    /// next hop will see.
477    ///
478    /// This is the first and most expensive step of [`MetaPacket::into_forwarded`] - an elliptic-curve scalar
479    /// multiplication - and it is also the only step needed to derive the packet's [`PacketTag`]. Splitting it out
480    /// lets a caller that wants to look at the tag before paying for the rest of the decode (a replay filter, for
481    /// one) do so without repeating the multiplication: pass the secret to
482    /// [`MetaPacket::into_forwarded_with_secret`].
483    pub fn forward_secret<'a>(&self, node_keypair: &'a S::P) -> Result<ForwardSecret<S>, SphinxError>
484    where
485        &'a Alpha<<S::G as GroupElement<S::E>>::AlphaLen>: From<&'a <S::P as Keypair>::Public>,
486    {
487        let alpha_arr = Array::<u8, <S::G as GroupElement<S::E>>::AlphaLen>::try_from(self.alpha())
488            .map_err(|_| SphinxError::PacketDecodingError("invalid alpha".into()))?;
489        let (alpha, secret) = SharedKeys::<S::E, S::G>::forward_transform(
490            &alpha_arr,
491            &(node_keypair.into()),
492            node_keypair.public().into(),
493        )
494        .map_err(SphinxError::from)?;
495
496        let tag = derive_packet_tag(&secret)?;
497        Ok(ForwardSecret { alpha, secret, tag })
498    }
499
500    /// Attempts to remove the layer of encryption in this packet by using the given `node_keypair`.
501    /// This will transform this packet into the [`ForwardedMetaPacket`].
502    pub fn into_forwarded<'a, K, F>(
503        self,
504        node_keypair: &'a S::P,
505        key_mapper: &K,
506        reply_openers: F,
507    ) -> Result<ForwardedMetaPacket<S, H, P>, SphinxError>
508    where
509        K: ProtocolKeyIdMapper<S, H>,
510        F: FnMut(&H::PacketReceiverData) -> Option<ReplyOpener>,
511        &'a Alpha<<S::G as GroupElement<S::E>>::AlphaLen>: From<&'a <S::P as Keypair>::Public>,
512    {
513        let state = self.forward_secret(node_keypair)?;
514        self.into_forwarded_with_secret(state, key_mapper, reply_openers, || {})
515    }
516
517    /// As [`MetaPacket::into_forwarded`], but for a caller that has already derived the shared secret with
518    /// [`MetaPacket::forward_secret`].
519    /// `state` is what [`MetaPacket::forward_secret`] returned, so the tag is the one belonging to this secret.
520    ///
521    /// `on_authenticated` runs once the header MAC has verified, i.e. once the sender has proven knowledge of the
522    /// shared secret, and before anything on this hop's side is consumed. A caller that records the tag can use it
523    /// to distinguish "this packet was well-formed and then failed on our side" from "this packet never
524    /// authenticated".
525    pub fn into_forwarded_with_secret<K, F, A>(
526        mut self,
527        state: ForwardSecret<S>,
528        key_mapper: &K,
529        mut reply_openers: F,
530        on_authenticated: A,
531    ) -> Result<ForwardedMetaPacket<S, H, P>, SphinxError>
532    where
533        K: ProtocolKeyIdMapper<S, H>,
534        F: FnMut(&H::PacketReceiverData) -> Option<ReplyOpener>,
535        A: FnOnce(),
536    {
537        let (alpha, secret, packet_tag) = state.into_parts();
538        // Forward the packet header
539        let fwd_header = forward_header::<H>(&secret, self.routing_info_mut())?;
540        on_authenticated();
541
542        // Perform initial decryption over the payload
543        let decrypted = self.payload_mut();
544        let prp = S::new_prp_init(&secret)?.into_init::<S::PRP>();
545        prp.inverse(<&mut crypto_traits::Block<S::PRP>>::try_from(&mut *decrypted).expect("block size mismatch"));
546
547        Ok(match fwd_header {
548            ForwardedHeader::Relayed {
549                next_header,
550                path_pos,
551                next_node,
552                additional_info,
553            } => {
554                // The payload is already decrypted in place; write back the blinded alpha and the next header.
555                self.alpha_mut().copy_from_slice(&alpha);
556                self.routing_info_mut().copy_from_slice(next_header.as_ref());
557
558                ForwardedMetaPacket::Relayed {
559                    packet: self,
560                    packet_tag,
561                    derived_secret: secret,
562                    next_node: key_mapper.map_id_to_public(&next_node).ok_or_else(|| {
563                        SphinxError::PacketDecodingError(format!(
564                            "couldn't map id to public key: {}",
565                            next_node.to_hex()
566                        ))
567                    })?,
568                    path_pos,
569                    additional_info,
570                }
571            }
572            ForwardedHeader::Final {
573                receiver_data,
574                is_reply,
575                no_ack,
576            } => {
577                // If the received packet contains a reply message for a pseudonym,
578                // we must perform additional steps to decrypt it
579                if is_reply {
580                    let local_surb = reply_openers(&receiver_data)
581                        .ok_or_else(|| SphinxError::MissingReplyOpener(receiver_data.to_hex()))?;
582
583                    // Encrypt the packet payload using the derived shared secrets
584                    // to reverse the decryption done by individual hops
585                    for secret in local_surb.shared_secrets.into_iter().rev() {
586                        let prp = S::new_prp_init(&secret)?.into_init::<S::PRP>();
587                        prp.forward(
588                            <&mut crypto_traits::Block<S::PRP>>::try_from(&mut *decrypted)
589                                .expect("block size mismatch"),
590                        );
591                    }
592
593                    // Invert the initial encryption using the sender key
594                    let prp =
595                        S::new_reply_prp_init(&local_surb.sender_key, receiver_data.as_ref())?.into_init::<S::PRP>();
596                    prp.inverse(
597                        <&mut crypto_traits::Block<S::PRP>>::try_from(&mut *decrypted).expect("block size mismatch"),
598                    );
599                }
600
601                // The decrypted payload is a view into the packet buffer.
602                let mut payload = Bytes::from(self.packet);
603                payload.advance(<S::G as GroupElement<S::E>>::AlphaLen::USIZE + RoutingInfo::<H>::SIZE);
604
605                ForwardedMetaPacket::Final {
606                    packet_tag,
607                    derived_secret: secret,
608                    plain_text: PaddedPayload::from_padded_bytes(payload)?,
609                    receiver_data,
610                    no_ack,
611                }
612            }
613        })
614    }
615}

624    type Error = GeneralError;

626    fn try_from(value: &[u8]) -> Result<Self, Self::Error> {
627        if value.len() == Self::SIZE {
628            Ok(Self {
629                packet: value.to_vec(),
630                _d: PhantomData,
631            })
632        } else {
633            Err(GeneralError::ParseError("MetaPacket".into()))
634        }
635    }
