10/// Represents a shared secret with a remote peer.
11pub type SharedSecret = SecretValue<hopr_types::primitive::typenum::U32>;

13/// Types representing a scalar of an additive abelian group.
14pub trait Scalar: Mul<Output = Self> + Sized {
15    /// Generates a random scalar using a cryptographically secure RNG.
16    fn random() -> Self;
17
18    /// Create scalar from bytes
19    fn from_bytes(bytes: &[u8]) -> hopr_types::crypto::errors::Result<Self>;
20
21    /// Whether this scalar is the additive identity. `from_bytes` does not always exclude it:
22    /// secp256k1 accepts an all-zero encoding, and zero blinds the alpha to the identity.
23    ///
24    /// Curve25519 clamping cannot produce it: clamping sets bit 254, so the value is a positive
25    /// multiple of eight in `[2^254, 2^255)`, and the order `L` is odd, so its first multiple
26    /// divisible by eight is `8L > 2^255`. Reduction modulo `L` is therefore never zero.
27    fn is_zero(&self) -> bool;
28}

30/// Represents the Alpha value of a certain length in the Sphinx protocol
31/// The length of the alpha value is directly dependent on the group element.
32pub type Alpha<A> = Array<u8, A>;
33
34/// Generic additive abelian group element with an associated scalar type.
35/// It also comes with the associated Alpha value size.
36/// A group element is considered valid if it is not neutral or a torsion element of small order.
37pub trait GroupElement<E: Scalar>: Clone + for<'a> Mul<&'a E, Output = Self> {
38    /// Length of the Alpha value - a binary representation of the group element.
39    type AlphaLen: ArraySize;
40
41    /// Converts the group element to a binary format suitable for representing the Alpha value.
42    fn to_alpha(&self) -> Alpha<Self::AlphaLen>;
43
44    /// Converts the group element from the binary format representing an Alpha value.
45    fn from_alpha(alpha: Alpha<Self::AlphaLen>) -> hopr_types::crypto::errors::Result<Self>;
46
47    /// Create a group element using the group generator and the given scalar
48    fn generate(scalar: &E) -> Self;
49
50    /// Whether a group element this crate *derived itself* from a validated scalar is usable.
51    ///
52    /// Distinct from [`GroupElement::is_valid`], which validates untrusted wire input: a point
53    /// produced as `s * G` for an already-parsed scalar is in the prime-order subgroup by
54    /// construction, so only the identity (a scalar that is a multiple of the group order) can be
55    /// wrong. Suites whose `is_valid` is more than an identity test can override this to skip that
56    /// extra work on the per-packet path.
57    fn is_valid_derived(&self) -> bool {
58        self.is_valid()
59    }
60
61    /// Group element is considered valid if it is not a neutral element and also not a torsion element of small order.
62    fn is_valid(&self) -> bool;
63
64    /// Generates a random pair of group element and secret scalar.
65    /// This is a convenience method that internally calls the `random` method of the associated Scalar
66    /// and constructs the group element using `generate`.
67    fn random_pair() -> (Self, E) {
68        let scalar = E::random();
69        (Self::generate(&scalar), scalar)
70    }
71
72    /// Extract a keying material from a group element using a KDF
73    fn extract_key(&self, context: &str, salt: &[u8]) -> SharedSecret {
74        Self::extract_key_from_alpha(&self.to_alpha(), context, salt)
75    }
76
77    /// As [`GroupElement::extract_key`], but from an already-computed Alpha encoding.
78    ///
79    /// Deriving two keys from the same point (the secret and the blinding factor) would otherwise
80    /// call [`GroupElement::to_alpha`] twice, and on Edwards that encoding costs a field inversion.
81    /// Both the sender (once per hop of every path) and the relayer (once per packet) do exactly
82    /// that pair, so they encode the point once and reuse it.
83    fn extract_key_from_alpha(alpha: &Alpha<Self::AlphaLen>, context: &str, salt: &[u8]) -> SharedSecret {
84        let mut output = create_kdf_instance(alpha, context, Some(salt)).expect("invalid sphinx key length");
85        let mut out = SharedSecret::default();
86        output.fill(out.as_mut());
87        out
88    }
89}

91/// Structure containing shared keys for peers using the Sphinx algorithm.
92pub struct SharedKeys<E: Scalar, G: GroupElement<E>> {
93    pub alpha: Alpha<G::AlphaLen>,
94    pub secrets: Vec<SharedSecret>,
95    _d: PhantomData<(E, G)>,
96}
97
98const HASH_KEY_SPHINX_SECRET: &str = "HASH_KEY_SPHINX_SECRET";
99const HASH_KEY_SPHINX_BLINDING: &str = "HASH_KEY_SPHINX_BLINDING";
100
101impl<E: Scalar, G: GroupElement<E>> SharedKeys<E, G> {
102    /// Generates shared secrets given the group element of the peers and their Alpha value encodings.
103    ///
104    /// The order of the peer group elements is preserved for resulting shared keys.
105    pub(crate) fn generate(
106        peer_group_elements: Vec<(G, &Alpha<G::AlphaLen>)>,
107    ) -> hopr_types::crypto::errors::Result<SharedKeys<E, G>> {
108        let mut shared_keys = Vec::with_capacity(peer_group_elements.len());
109
110        // coeff_prev becomes: x * b_0 * b_1 * b_2 * ...
111        // alpha_prev becomes: x * b_0 * b_1 * b_2 * ... * G
112        let (mut alpha_prev, mut coeff_prev) = G::random_pair();
113
114        // Save the part of the result
115        let alpha = alpha_prev.to_alpha();
116
117        // Iterate through all the given peer public keys
118        let keys_len = peer_group_elements.len();
119        for (i, (group_element, salt)) in peer_group_elements.into_iter().enumerate() {
120            // Multiply the public key by the current coefficient
121            let shared_secret = group_element.mul(&coeff_prev);
122
123            // Encoded once and reused for the blinding factor: `to_alpha` is a field inversion on
124            // Edwards.
125            let shared_secret_alpha = shared_secret.to_alpha();
126            shared_keys.push(G::extract_key_from_alpha(
127                &shared_secret_alpha,
128                HASH_KEY_SPHINX_SECRET,
129                salt.as_ref(),
130            ));
131
132            // Stop here, we don't need to compute anything more
133            if i == keys_len - 1 {
134                break;
135            }
136
137            // Compute the new blinding factor b_k (alpha needs compressing first)
138            let b_k = G::extract_key_from_alpha(&shared_secret_alpha, HASH_KEY_SPHINX_BLINDING, &alpha_prev.to_alpha());
139            let b_k_checked = Self::parse_blinding_scalar(&b_k)?;
140
141            // Same point as `alpha_prev * b_k` by associativity, but as a fixed-base multiply
142            // `(coeff * b) * G`, which uses the precomputed base-point table. Once per hop, per packet.
143            coeff_prev = coeff_prev.mul(b_k_checked);
144            alpha_prev = G::generate(&coeff_prev);
145
146            // A multiple of the generator, so only the identity can be wrong - and `is_valid` would
147            // not even establish that much, since it admits mixed-order points.
148            if !alpha_prev.is_valid_derived() {
149                return Err(CryptoError::CalculationError);
150            }
151        }
152
153        Ok(SharedKeys {
154            alpha,
155            secrets: shared_keys,
156            _d: PhantomData,
157        })
158    }
159
160    /// The first half of the forward transform: everything but the blinded alpha.
161    ///
162    /// Blinding the alpha is a second variable-base scalar multiplication, and a packet's *last* hop
163    /// never forwards, so that point is never read there. Deferring it lets a decode pay only the
164    /// multiplication that derives the shared secret when the packet is addressed to this node,
165    /// which is the common case for an Exit and for any node's own traffic.
166    ///
167    /// The caller that does forward must call [`DeferredAlpha::finalize`] to finish the transform.
168    pub(crate) fn forward_transform_deferred(
169        alpha: &Alpha<G::AlphaLen>,
170        private_scalar: &E,
171        public_element_alpha: &Alpha<G::AlphaLen>,
172    ) -> hopr_types::crypto::errors::Result<(DeferredAlpha<E, G>, SharedSecret)> {
173        let alpha_point = G::from_alpha(alpha.clone())?;
174
175        // Reject a degenerate alpha *before* the private-scalar multiplication, on every hop.
176        //
177        // For alpha `P` of order `d`, `P * sk` takes at most `d` values (and for `d = 2`, `-P = P`,
178        // so just `O` or `P`), however the key was chosen. A third party can compute those candidates
179        // without the node's key, so the hop secret and header MAC stop being secret.
180        //
181        // The base validated the *blinded output* instead, which caught only the identity; it accepted
182        // all 200 sampled keys for an order-2 input. Rejecting the whole class on the input is new.
183        if !alpha_point.is_valid() {
184            return Err(CryptoError::CalculationError);
185        }
186
187        let s_k = alpha_point.clone().mul(private_scalar);
188
189        // Encode the shared point once: both keys below come from it, and `to_alpha` costs a field
190        // inversion on Edwards. This is the per-packet cost of relaying, so the second encoding is
191        // worth avoiding.
192        let s_k_alpha = s_k.to_alpha();
193        let secret = G::extract_key_from_alpha(&s_k_alpha, HASH_KEY_SPHINX_SECRET, public_element_alpha);
194
195        let b_k = G::extract_key_from_alpha(&s_k_alpha, HASH_KEY_SPHINX_BLINDING, alpha);
196
197        Self::parse_blinding_scalar(&b_k)?;
198
199        Ok((
200            DeferredAlpha {
201                alpha_point,
202                b_k,
203                _d: PhantomData,
204            },
205            secret,
206        ))
207    }
208
209    /// Validates a KDF output as a blinding scalar, in the first half of the transform on every hop.
210    ///
211    /// It cannot live in [`DeferredAlpha::finalize`]: a final hop never calls that, so a rejected KDF
212    /// output would be accepted there. secp256k1 both rejects values at or above the group order and
213    /// *accepts* all-zero, and zero blinds the alpha to the identity - which the eager transform
214    /// refused on every hop, so the explicit zero test preserves that. (Curve25519 clamping cannot
215    /// produce zero; see [`Scalar::is_zero`].)
216    fn parse_blinding_scalar(b_k: &SharedSecret) -> hopr_types::crypto::errors::Result<E> {
217        let b_k = E::from_bytes(b_k.as_ref())?;
218        if b_k.is_zero() {
219            return Err(CryptoError::CalculationError);
220        }
221        Ok(b_k)
222    }
223}

225/// The blinded alpha a [`SharedKeys::forward_transform_deferred`] has yet to compute.
226///
227/// A final hop discards it, so it never pays the multiplication.
228pub(crate) struct DeferredAlpha<E: Scalar, G: GroupElement<E>> {
229    alpha_point: G,
230    /// Blinding factor `b_k` in raw KDF form, so it is zeroized on drop (`E` has no `ZeroizeOnDrop`).
231    /// Already parsed and checked nonzero in the first half.
232    b_k: SharedSecret,
233    _d: PhantomData<E>,
234}
235
236impl<E: Scalar, G: GroupElement<E>> DeferredAlpha<E, G> {
237    /// Computes the blinded alpha - the multiplication the first half leaves out.
238    pub(crate) fn finalize(self) -> hopr_types::crypto::errors::Result<Alpha<G::AlphaLen>> {
239        let b_k_checked = SharedKeys::<E, G>::parse_blinding_scalar(&self.b_k)?;
240        let alpha_new = self.alpha_point.mul(&b_k_checked);
241
242        // Not the prime-order subgroup: `is_valid` admits mixed-order points. But for `P = T + Q`
243        // with `T` small order and `Q` of prime order, `b * P` keeps a nonzero `b * Q` for any
244        // nonzero `b mod L`, so this can never be the identity. The check is defensive.
245        if !alpha_new.is_valid_derived() {
246            return Err(hopr_types::crypto::errors::CryptoError::CalculationError);
247        }
248
249        Ok(alpha_new.to_alpha())
250    }
251
252    /// Lets a test build the state that makes `finalize` fail (an identity point).
253    #[cfg(test)]
254    pub(crate) fn for_test(alpha_point: G, b_k: SharedSecret) -> Self {
255        Self {
256            alpha_point,
257            b_k,
258            _d: PhantomData,
259        }
260    }
261}

267/// Represents an instantiation of the Spinx protocol using the given EC group and corresponding public key object.
268pub trait SphinxSuite {
269    /// Keypair corresponding to the EC group
270    type P: Keypair;
271
272    /// Scalar type supported by the EC group
273    type E: Scalar + for<'a> From<&'a Self::P>;
274
275    /// Expanded form of [`Self::P`]'s public key, suitable for EC computations.
276    ///
277    /// Deriving it from the compact [`Keypair::Public`] can be CPU-intensive, which is why it is
278    /// a type of its own: a path can be expanded once and reused across several packets.
279    type ExpandedPublic;
280
281    /// EC group element
282    type G: GroupElement<Self::E> + for<'a> From<&'a Self::ExpandedPublic>;
283
284    /// Pseudo-Random Permutation used to encrypt and decrypt packet payload
285    type PRP: crypto_traits::PRP + crypto_traits::KeyIvInit;
286
287    /// Expands a compact public key into [`Self::ExpandedPublic`].
288    ///
289    /// Can be CPU-intensive. Fails if `public_key` is not a valid point on the curve.
290    fn expand_public(
291        public_key: &<Self::P as Keypair>::Public,
292    ) -> hopr_types::crypto::errors::Result<Self::ExpandedPublic>;
293
294    /// Expands an entire path. See [`SphinxSuite::expand_public`].
295    fn expand_path(
296        public_keys: &[<Self::P as Keypair>::Public],
297    ) -> hopr_types::crypto::errors::Result<Vec<Self::ExpandedPublic>> {
298        public_keys.iter().map(|pk| Self::expand_public(pk)).collect()
299    }
300
301    /// Convenience function to generate shared keys from the path of public keys.
302    ///
303    /// Expands every key on the path; use [`SphinxSuite::new_shared_keys_expanded`] when the path
304    /// has already been expanded, so that the cost is not paid once per packet.
305    fn new_shared_keys<'a>(
306        public_keys: &'a [<Self::P as Keypair>::Public],
307    ) -> hopr_types::crypto::errors::Result<SharedKeys<Self::E, Self::G>>
308    where
309        &'a Alpha<<Self::G as GroupElement<Self::E>>::AlphaLen>: From<&'a <Self::P as Keypair>::Public>,
310    {
311        let expanded = Self::expand_path(public_keys)?;
312
313        SharedKeys::generate(
314            expanded
315                .iter()
316                .zip(public_keys)
317                .map(|(expanded, compact)| (expanded.into(), compact.into()))
318                .collect(),
319        )
320    }
321
322    /// As [`SphinxSuite::new_shared_keys`], but for a path that is already expanded.
323    ///
324    /// Kept next to [`SphinxSuite::new_shared_keys`] so the two cannot diverge.
325    fn new_shared_keys_expanded<'a>(
326        public_keys: &'a [Self::ExpandedPublic],
327    ) -> hopr_types::crypto::errors::Result<SharedKeys<Self::E, Self::G>>
328    where
329        &'a Alpha<<Self::G as GroupElement<Self::E>>::AlphaLen>: From<&'a Self::ExpandedPublic>,
330    {
331        SharedKeys::generate(public_keys.iter().map(|pk| (pk.into(), pk.into())).collect())
332    }
333
334    /// Instantiates a new Pseudo-Random Permutation IV and key for general packet data.
335    fn new_prp_init(secret: &SecretKey) -> hopr_types::crypto::errors::Result<IvKey<Self::PRP>> {
336        generate_key_iv(secret, HASH_KEY_PRP, None)
337    }
338
339    /// Instantiates a new Pseudo-Random Permutation IV and key for reply data.
340    fn new_reply_prp_init(secret: &SecretKey16, salt: &[u8]) -> hopr_types::crypto::errors::Result<IvKey<Self::PRP>> {
341        generate_key_iv(secret, HASH_KEY_REPLY_PRP, Some(salt))
342    }
343}

352    #[allow(clippy::type_complexity)]
353    pub fn generic_sphinx_suite_test<S: SphinxSuite>(node_count: usize) -> anyhow::Result<()> {
354        let (pub_keys, priv_keys): (Vec<(S::G, Alpha<<S::G as GroupElement<S::E>>::AlphaLen>)>, Vec<S::E>) = (0
355            ..node_count)
356            .map(|_| {
357                let pair = S::G::random_pair();
358                ((pair.0.clone(), pair.0.to_alpha()), pair.1)
359            })
360            .unzip();
361
362        let pub_keys_alpha = pub_keys
363            .iter()
364            .map(|(pk, alpha)| (pk.clone(), alpha))
365            .collect::<Vec<_>>();
366
367        // Now generate the key shares for the public keys
368        let generated_shares =
369            SharedKeys::<S::E, S::G>::generate(pub_keys_alpha.clone()).context("sender: generate the shared keys")?;
370        assert_eq!(
371            node_count,
372            generated_shares.secrets.len(),
373            "number of generated keys should be equal to the number of nodes"
374        );
375
376        let mut alpha_cpy = generated_shares.alpha.clone();
377        for (i, priv_key) in priv_keys.into_iter().enumerate() {
378            // Both halves of the split transform are exercised here, against a *reference*
379            // implementation of the pre-deferral eager transform written out below. A comparison
380            // against a helper that merely calls `forward_transform_deferred` + `finalize` would not
381            // do: a change to the blinding salt or the alpha encoding would move both sides and leave
382            // the assertion passing.
383            let (expected_alpha, expected_secret) =
384                reference_forward_transform::<S>(&alpha_cpy, &priv_key, &pub_keys[i].1)
385                    .with_context(|| format!("hop {i}: reference transform"))?;
386
387            let (deferred, secret) =
388                SharedKeys::<S::E, S::G>::forward_transform_deferred(&alpha_cpy, &priv_key, &pub_keys[i].1)
389                    .with_context(|| format!("hop {i}: deferred transform"))?;
390
391            assert_eq!(
392                secret.ct_eq(&generated_shares.secrets[i]).unwrap_u8(),
393                1,
394                "hop {i}: forward transform should yield the same shared secret"
395            );
396            assert_eq!(
397                secret.ct_eq(&expected_secret).unwrap_u8(),
398                1,
399                "hop {i}: the deferred secret must match the reference transform"
400            );
401
402            let alpha = deferred.finalize().with_context(|| format!("hop {i}: finalize"))?;
403            assert_eq!(
404                alpha, expected_alpha,
405                "hop {i}: the finalized alpha must match the reference transform"
406            );
407
408            alpha_cpy = alpha;
409        }
410        Ok(())
411    }

469    /// The pre-deferral eager transform, re-derived rather than calling the code under test.
470    #[allow(clippy::type_complexity)] // the alpha type is spelled out by the trait, not chosen here
471    fn reference_forward_transform<S: SphinxSuite>(
472        alpha: &Alpha<<S::G as GroupElement<S::E>>::AlphaLen>,
473        private_scalar: &S::E,
474        public_element_alpha: &Alpha<<S::G as GroupElement<S::E>>::AlphaLen>,
475    ) -> anyhow::Result<(Alpha<<S::G as GroupElement<S::E>>::AlphaLen>, SharedSecret)> {
476        let alpha_point = S::G::from_alpha(alpha.clone()).context("reference: decode the alpha")?;
477        let s_k = alpha_point.clone().mul(private_scalar);
478        let secret = s_k.extract_key(HASH_KEY_SPHINX_SECRET, public_element_alpha);
479        let b_k = s_k.extract_key(HASH_KEY_SPHINX_BLINDING, alpha);
480        let b_k = S::E::from_bytes(b_k.as_ref()).context("reference: parse the blinding scalar")?;
481        let alpha_new = alpha_point.mul(&b_k);
482        anyhow::ensure!(alpha_new.is_valid(), "reference transform produced an invalid alpha");
483        Ok((alpha_new.to_alpha(), secret))
484    }

603    /// A zero blinding scalar must be rejected on every hop, including the final one, where
604    /// `finalize` no longer runs to catch the identity it produces.
605    #[cfg(feature = "secp256k1")]
606    #[test]
607    fn a_zero_blinding_scalar_is_rejected() -> anyhow::Result<()> {
608        use crate::sphinx::ec_groups::{Secp256k1Point, Secp256k1Scalar};
609
610        // The premise: `from_bytes` really does accept an all-zero encoding, so the value is
611        // reachable from a KDF output rather than only from an uninitialised scalar.
612        let zero_bytes = SharedSecret::default();
613        let parsed = <Secp256k1Scalar as Scalar>::from_bytes(zero_bytes.as_ref())
614            .expect("secp256k1 accepts an all-zero scalar encoding");
615        assert!(parsed.is_zero(), "an all-zero encoding must parse to zero");
616        assert!(Secp256k1Scalar::default().is_zero(), "the default scalar must be zero");
617
618        // The check under test must reject it. This is the assertion the fix exists for: without the
619        // `is_zero` term the parse alone succeeds and a final hop accepts the packet.
620        assert!(
621            SharedKeys::<Secp256k1Scalar, Secp256k1Point>::parse_blinding_scalar(&zero_bytes).is_err(),
622            "a zero blinding scalar must be rejected on every hop"
623        );
624
625        // And a nonzero scalar must still be accepted, so the guard does not reject everything.
626        // k256 scalars are big-endian, so the scalar one is a trailing 1.
627        let mut one_bytes = [0u8; 32];
628        one_bytes[31] = 1;
629        let one = <Secp256k1Scalar as Scalar>::from_bytes(&one_bytes).context("one must parse")?;
630        assert!(!one.is_zero(), "a nonzero scalar must not report as zero");
631        let one_secret = SharedSecret::try_from(one_bytes.as_slice()).context("one is 32 bytes")?;
632        assert!(
633            SharedKeys::<Secp256k1Scalar, Secp256k1Point>::parse_blinding_scalar(&one_secret).is_ok(),
634            "a nonzero blinding scalar must be accepted"
635        );
636
637        // Zero really does blind the alpha away to the identity, which is what makes it unacceptable:
638        // `P * 0 = O` for every alpha.
639        use hopr_types::primitive::typenum::U33;
640
641        let mut alpha_bytes = Alpha::<U33>::default();
642        alpha_bytes.copy_from_slice(
643            const_hex::decode("0279be667ef9dcbbac55a06295ce870b07029bfcdb2dce28d959f2815b16f81798")
644                .context("secp256k1 generator in compressed SEC1 form")?
645                .as_slice(),
646        );
647        let generator = <Secp256k1Point as GroupElement<Secp256k1Scalar>>::from_alpha(alpha_bytes)
648            .context("the generator must decode")?;
649        let blinded = generator.mul(&parsed);
650        assert!(
651            !blinded.is_valid_derived(),
652            "multiplying by a zero scalar must give the identity, which the derived check rejects"
653        );
654        Ok(())
655    }
