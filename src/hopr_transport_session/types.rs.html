109/// Turns the deposit data a pool produced for a batch into the per-SSA map an `SsaRequest` carries.
110///
111/// Every SSA in the batch must come out of this with an entry, and the absence of one is fatal — see
112/// [`MissingDepositData`](crate::errors::SessionManagerError::MissingDepositData) for why the batch
113/// cannot simply travel short. An *empty* entry is not an absent one: a pool with nothing to attach
114/// says so by answering with empty data, which is a value
115/// ([`PixDepositData::is_empty`]) and travels fine.
116///
117/// Entries the batch has no place for — another Session's, an SSA outside this batch, a second answer
118/// for an index already answered — are dropped and reported. They are not themselves fatal: what they
119/// are evidence of is, and the gap they leave behind is what fails.
120pub(crate) fn deposit_data_for_batch(
121    session_id: &SessionId,
122    batch: &[hopr_protocol_pix::SsaIndex],
123    deposit_data: HoprPixDepositData,
124) -> Result<std::collections::HashMap<hopr_protocol_pix::SsaIndex, HoprPixDepositPayload>, SessionManagerError> {
125    let mut out = std::collections::HashMap::with_capacity(batch.len());
126    let (mut foreign, mut duplicate) = (0usize, 0usize);
127
128    for entry in deposit_data {
129        let ssa_index = entry.id.ssa_index();
130        if &entry.id.session_id() != session_id || !batch.contains(&ssa_index) {
131            foreign += 1;
132            continue;
133        }
134        if out.insert(ssa_index, HoprPixDepositPayload(entry.data)).is_some() {
135            duplicate += 1;
136        }
137    }
138
139    if foreign > 0 || duplicate > 0 {
140        warn!(
141            %session_id,
142            requested = batch.len(),
143            usable = out.len(),
144            foreign,
145            duplicate,
146            "deposit pool answered with entries this batch has no place for"
147        );
148    }
149
150    let missing = batch.iter().filter(|i| !out.contains_key(i)).collect::<Vec<_>>();
151    if !missing.is_empty() {
152        return Err(SessionManagerError::MissingDepositData(format!(
153            "session {session_id} is missing deposit data for {} of {} SSAs in the batch: {missing:?} ({foreign} \
154             entries were for other SSAs, {duplicate} were repeats)",
155            missing.len(),
156            batch.len(),
157        )));
158    }
159
160    Ok(out)
161}

163/// Quota per single SSA in bytes.
164///
165/// The quota in bytes has only informative value for the user - the volume of Exit -> Entry data
166/// one SSA cycle carries, which is what a single deposit pays for.
167///
168/// Not the volume at which the Exit *recovers* the SSA private key: recovery needs
169/// `polys × threshold` useful shares, and so happens before the cycle drains. The quota covers the
170/// whole cycle, surplus included — see [`pix_params_to_quota`].
171///
172/// The SessionManager always counts in packets, not in bytes, when it comes to quota management.
173pub type SsaQuota = u64;
174
175/// What a single SSA deposit buys: every Exit → Entry payload byte the cycle carries.
176///
177/// Counts [`PixParams::emitted_shares_per_poly`], i.e. `threshold + surplus`, and not the threshold
178/// alone. A polynomial leaves the generator's queue only once it has emitted `threshold + surplus`
179/// shares (`SsaShareGenerator::next_share`), whether or not any were lost, and each share rides back
180/// on one Exit → Entry data packet — so this is exactly the traffic the Exit serves per cycle.
181/// Charging the threshold alone left the surplus unpaid, which at the deployed 1.25× factor is a
182/// fifth of all Exit → Entry traffic.
183///
184/// The surplus is insurance the Entry buys against share loss, and it is billed like insurance: on
185/// purchase, not on claim. One deposit per cycle at this rate is what makes paid-for equal served.
186///
187/// The other half of that equality lives in the generator's own tests
188/// (`drain_shares_by_polynomial` in `hopr-protocol-pix`), which pin a cycle's emission at
189/// `polys × (threshold + surplus)`. Change either expression and the other has to move with it.
190pub(crate) const fn pix_params_to_quota(params: &PixParams) -> SsaQuota {
191    params.polys_per_ssa() as SsaQuota
192        * params.emitted_shares_per_poly() as SsaQuota
193        * HoprPacket::PAYLOAD_SIZE as SsaQuota
194}
195
196/// Default number of polynomials ("SSA parts") a single SSA is split into.
197///
198/// This is the single source of truth for the Entry-side generator dimension
199/// (`PixGlobalConfig::num_ssa_parts`) and for the Exit-side acceptance policy
200/// ([`IncomingSessionPixConfig::quota_range`](crate::IncomingSessionPixConfig::quota_range)).
201/// Both must be derived from it so the two cannot drift apart: the Exit computes the
202/// offered quota as `polys × (shares + surplus) × PAYLOAD_SIZE` and rejects the Session if neither
203/// it nor an allowed dynamic batch brings the total inside the configured range. At the default
204/// batch ceiling of one, a hard-coded range that no longer matches the dimension defaults makes
205/// every PIX Session fail to establish.
206///
207/// ## Choosing the split
208///
209/// For a fixed useful-share count `U = polys × threshold` the *product* is pinned, so the split
210/// between the two is free — but the costs scale differently, and dropping the non-constant
211/// coefficient commitments (see [`hopr_protocol_pix::SsaPartCommitment`]) changed which way they
212/// pull:
213///
214/// * Commitment wire volume and Exit ingest are one commitment per polynomial — **linear in `polys`**, and formerly
215///   `polys × threshold`. Ingest is dominated by point decompression plus the cofactor-8 subgroup check.
216/// * Reconstructor commitment memory is likewise `polys`, no longer `U`.
217/// * Share verification is one scalar multiplication per *polynomial*, not `O(threshold)` per share. It used to be `U ×
218///   threshold` and is now simply `polys`.
219/// * Interpolating a polynomial is `O(threshold²)` field operations, and there are `polys` of them — `U × threshold`,
220///   **linear in `threshold`**, and it is field arithmetic rather than curve arithmetic.
221/// * Detection of a dishonest Entry takes `threshold` return packets, since a share set is only checked once it
222///   interpolates.
223/// * On the **Entry**, `SsaShareGenerator::next_share` evaluates a `threshold`-wide polynomial by Horner for every
224///   share it emits — `U × threshold` again. This is much the smaller of the two per-share terms, but it is not zero,
225///   and describing the Entry as threshold-free (as this list once did) is wrong.
226///
227/// So raising `threshold` (and lowering `polys`) buys a proportionally smaller commitment phase at
228/// the cost of more interpolation, later fault detection and more Entry evaluation.
229///
230/// Both sides have since been measured, and `8192 × 64` stands — see
231/// [`hopr_protocol_pix::DEFAULT_POLY_THRESHOLD`] for the tables. The objective is **Exit
232/// reconstruction capacity**, because the Exit serves 10–30 clients while an Entry serves only
233/// itself: on that measure the deployed threshold is within 0.4 % of the optimum, and the fixed
234/// per-polynomial cost the Exit amortises over `threshold` shares means a *lower* threshold is
235/// worse, not better. Summing Entry and Exit per-share cost instead would favour 48 by about 3 %;
236/// that reading is recorded there and deliberately not acted on.
237///
238/// The quota is fixed by `polys × (threshold + surplus)`, so the split can be re-tuned without
239/// touching session negotiation as long as that product holds.
240///
241/// ## Why this is an alias
242///
243/// The generator that produces the shares lives in `hopr-protocol-pix` and carries its own
244/// defaults, so the split existed as two independent literals — and they drifted: this side was
245/// re-tuned to `8192 × 64` while the pix crate stayed at a threshold of 128, which made
246/// [`hopr_protocol_pix::SsaGeneratorConfig::default`] imply a 1.01 GiB quota, outside the very
247/// range derived below. Four benchmarks had grown comments explaining which of the two to
248/// believe. Aliasing removes the choice.
249pub const DEFAULT_PIX_POLYS_PER_SSA: u16 = hopr_protocol_pix::DEFAULT_POLYS_PER_SSA;
250
251/// Default number of shares required to reconstruct a single SSA part.
252///
253/// See [`DEFAULT_PIX_POLYS_PER_SSA`] for why this is shared between both sides, why it is kept
254/// small relative to the polynomial count, and why it is an alias rather than a literal.
255pub const DEFAULT_PIX_SHARES_PER_POLY: u8 = hopr_protocol_pix::DEFAULT_POLY_THRESHOLD;

346/// Events received by the [`crate::manager::SessionManager`] in reaction to received shares from the packet pipeline.
347#[derive(Debug, Clone)]
348pub enum HoprSessionInPixEvent {
349    /// Informs the [`crate::manager::SessionManager`] that an SSA was fully recovered.
350    SsaRecovered(SsaId<HoprPseudonym>),
351    /// Informs the [`crate::manager::SessionManager`] that the early recovery threshold was reached
352    /// for an SSA — the next SSA request can be made.
353    SsaAlmostRecovered(SsaId<HoprPseudonym>),
354    /// Informs the [`crate::manager::SessionManager`] that unverifiable shares were encountered.
355    ///
356    /// Carries the reconstructor's running total for the SSA rather than a delta: shares for one
357    /// cycle reach the Exit over whichever return path is available, so no single relayer's count is
358    /// the whole story, and an absolute total makes a duplicated or reordered report harmless.
359    UnverifiableShares {
360        ssa_id: SsaId<HoprPseudonym>,
361        observed_total: u64,
362    },
363    /// Reports how far an SSA has got towards recovery.
364    ///
365    /// Unlike the others this is paced by traffic rather than by lifecycle transitions, and it is
366    /// what keeps a funded Session's egress flowing — the gate stops serving a cycle that shows no
367    /// progress. Delivered without backpressure for that reason; see
368    /// `SessionPixSupervisorHandle::try_send_progress`.
369    RecoveryProgress(SsaRecoveryProgress<HoprPseudonym>),
370}

372impl HoprSessionInPixEvent {
373    /// Extracts the pseudonym of the SSA that might map to an existing Session.

378            HoprSessionInPixEvent::UnverifiableShares { ssa_id, .. } => ssa_id.pseudonym(),

407/// Indicates the closure reason of a [`HoprSession`].
408///
409/// Delivered to the `on_close` callback a Session is constructed with — see
410/// [`HoprSession::new`] — and to the `SessionManager`'s closure notifier, so a caller has to be able
411/// to name this type to write either.
412#[derive(Clone, Copy, Debug, PartialEq, Eq, strum::Display)]
413pub enum ClosureReason {
414    /// Write-half of the Session has been closed.
415    WriteClosed,
416    /// Read-part of the Session has been closed (encountered empty read).
417    EmptyRead,
418    /// Session has been evicted from the cache due to inactivity or capacity reasons.
419    Eviction,
420    /// Deposit to an SSA has not been made on-time on a PIX-enabled Session.
421    UnrealizedDeposit,
422    /// The local deposit pool did not supply the deposit data an SSA batch needs.
423    ///
424    /// Exit-side and locally caused, unlike [`UnrealizedDeposit`](Self::UnrealizedDeposit) which is
425    /// the Entry failing to pay. Kept apart from it for exactly that reason: the two look alike from
426    /// the outside — a PIX Session that stopped — and point at opposite nodes.
427    MissingDepositData,
428    /// The PIX supervisor ended the Session — a recovery deadline elapsed, too many shares failed
429    /// verification, or the supervisor itself became unavailable.
430    ///
431    /// Distinct from [`UnrealizedDeposit`](Self::UnrealizedDeposit) and
432    /// [`MissingDepositData`](Self::MissingDepositData), which name the two PIX failures an operator
433    /// can act on directly; the rest are collapsed here and distinguished in telemetry by the
434    /// supervisor's own close reason.
435    PixFailure,
436}

521    /// based on the given `capabilities`.
522    ///
523    /// The `on_close` closure can be optionally called when the Session has been closed via `poll_close`.
524    #[tracing::instrument(skip_all, fields(id, routing, cfg, session_id = %id))]
525    pub fn new<Tx, Rx>(
526        id: SessionId,
527        routing: DestinationRouting,
528        cfg: HoprSessionConfig,
529        hopr: (Tx, Rx),
530        on_close: Option<Box<dyn FnOnce(SessionId, ClosureReason) + Send + Sync>>,
531    ) -> Result<Self, TransportSessionError>
532    where
533        Tx: futures::Sink<(DestinationRouting, ApplicationDataOut)> + Send + Unpin + 'static,
534        Rx: futures::Stream<Item = ApplicationDataIn> + Send + Unpin + 'static,
535        Tx::Error: std::error::Error + Send + Sync,
536    {
537        Self::new_with_surb_state(id, routing, cfg, hopr, on_close, None, None)
538    }
539
540    /// Like [`new`](Self::new) but threads the SURB balancer state and the opt-in client-side
541    /// flow-control config. `flow_control` = the client's [`FlowControlConfig`] for this session
542    /// (`None` leaves it unpaced); `surb_mgmt` gives the window its anti-grief down-only SURB ceiling.
543    /// The entry (sending) side passes `Some(..)`; sites without them pass `None`.
544    #[allow(clippy::too_many_arguments)]
545    #[tracing::instrument(skip_all, fields(id, routing, cfg, session_id = %id))]
546    pub fn new_with_surb_state<Tx, Rx>(
547        id: SessionId,
548        routing: DestinationRouting,
549        cfg: HoprSessionConfig,
550        hopr: (Tx, Rx),
551        on_close: Option<Box<dyn FnOnce(SessionId, ClosureReason) + Send + Sync>>,
552        surb_mgmt: Option<Arc<BalancerStateValues>>,
553        flow_control: Option<FlowControlConfig>,
554    ) -> Result<Self, TransportSessionError>
555    where
556        Tx: futures::Sink<(DestinationRouting, ApplicationDataOut)> + Send + Unpin + 'static,
557        Rx: futures::Stream<Item = ApplicationDataIn> + Send + Unpin + 'static,
558        Tx::Error: std::error::Error + Send + Sync,
559    {
560        let routing_clone = routing.clone();
561
562        #[cfg(feature = "telemetry")]
563        let (session_id_write, session_id_read) = (id, id);
564
565        // Wrap the HOPR transport so that it appears as regular transport to the SessionSocket
566        let transport = DuplexIO(
567            AsyncWriteSink::<{ ApplicationData::PAYLOAD_SIZE }, _>(hopr.0.sink_map_err(std::io::Error::other).with(
568                move |buf: Box<[u8]>| {
569                    #[cfg(feature = "telemetry")]
570                    crate::telemetry::record_session_write(&session_id_write, buf.len());
571                    // The Session protocol does not set any packet info on outgoing packets.
572                    // However, the SessionManager on top usually overrides this.
573                    futures::future::ready(
574                        ApplicationData::new(SESSION_APPLICATION_TAG, buf.into_vec())
575                            .map(|data| (routing_clone.clone(), ApplicationDataOut::with_no_packet_info(data)))
576                            .map_err(std::io::Error::other),
577                    )
578                },
579            )),
580            // The Session protocol ignores the packet info on incoming packets.
581            // It is typically SessionManager's job to interpret those.
582            hopr.1
583                .map(move |data| {
584                    #[cfg(feature = "telemetry")]
585                    crate::telemetry::record_session_read(&session_id_read, data.data.plain_text.len());
586                    Ok::<_, std::io::Error>(data.data.plain_text)
587                })
588                .into_async_read(),
589        );
590
591        // Based on the requested capabilities, see if we should use the Session protocol
592        let inner: Box<dyn AsyncReadWrite> = if cfg.capabilities.contains(Capability::Segmentation) {
593            let socket_cfg = SessionSocketConfig {
594                frame_size: cfg.frame_mtu,
595                frame_timeout: cfg.frame_timeout,
596                capacity: SESSION_SOCKET_CAPACITY,
597                flush_immediately: cfg.capabilities.contains(Capability::NoDelay),
598                max_buffered_segments: cfg.max_buffered_segments,
599                // Anti-bufferbloat bound; only meaningful when flow control is enabled, which is
600                // also where the honest clock that observes the resulting loss lives.
601                max_frame_age: flow_control.and_then(|c| c.max_frame_age),
602                // Head-of-line bound, and deliberately *not* gated on flow control the way
603                // `max_frame_age` is. The reasoning there runs backwards for this one: a session
604                // that can retransmit may still recover a missing frame, so waiting is
605                // productive, while a session without retransmission is waiting for something
606                // that is never coming and holds everything already received behind it. The
607                // sessions that need this bound most are exactly the ones flow control excludes.
608                max_frames_behind_gap: cfg.max_frames_behind_gap,
609                ..Default::default()
610            };
611
612            // Need to test the capabilities separately, because any Retransmission capability
613            // implies Segmentation, and therefore `is_disjoint` would fail
614            if cfg.capabilities.contains(Capability::RetransmissionAck)
615                || cfg.capabilities.contains(Capability::RetransmissionNack)
616            {
617                let fc = flow_control;
618
619                // TODO: update config values
620                let ack_cfg = AcknowledgementStateConfig {
621                    // This is a very coarse assumption, that a single 3-hop packet
622                    // takes on average 200 ms to deliver.
623                    // We can no longer base this timeout on the number of hops because
624                    // it is not known for SURB-based routing.
625                    expected_packet_latency: Duration::from_millis(200),
626                    mode: caps_to_ack_mode(cfg.capabilities),
627                    backoff_base: 0.2,
628                    max_incoming_frame_retries: 1,
629                    // Under flow control the sender is paced to the SURB drain rate, so an un-acked
630                    // frame is usually just a *delayed* ack on a temporarily-starved return path, not
631                    // a genuine loss. The retry budget is a flow-control config knob (`frame_retries`,
632                    // default 2 = original); a robust profile raises it so delayed frames recover
633                    // instead of being abandoned (an abandoned frame leaves a gap → stream corruption).
634                    // `.max(1)`: never drop the retry budget to 0 — an abandoned frame under
635                    // reliable-mode flow control leaves a gap and corrupts the stream.
636                    max_outgoing_frame_retries: fc.map(|c| c.frame_retries.max(1) as usize).unwrap_or(2),
637                    // Retire an outgoing frame that is already too stale to be worth delivering,
638                    // rather than spending the remaining retry budget on it.
639                    max_frame_age: fc.and_then(|c| c.max_frame_age),
640                    ..Default::default()
641                };
642
643                debug!(
644                    ?socket_cfg,
645                    ?ack_cfg,
646                    flow_control = fc.is_some(),
647                    "opening new stateful session socket"
648                );
649
650                // Opt-in client-side flow control: when enabled, install the honest-clock tap on the
651                // ack state and keep the paired clock to drive the paced writer.
652                let (ack_state, flow_control) = match fc {
653                    Some(fc_cfg) => {
654                        let meter = DeliveryMeter::default();
655                        let ack_state = AcknowledgementState::<{ ApplicationData::PAYLOAD_SIZE }>::new(id, ack_cfg)
656                            .with_delivery_tap(DeliveryTap::new(meter.clone(), cfg.frame_mtu));
657                        let clock = DeliveryClock::new(meter, Some(ack_cfg.expected_packet_latency));
658                        (ack_state, Some((fc_cfg, clock)))
659                    }
660                    None => (
661                        AcknowledgementState::<{ ApplicationData::PAYLOAD_SIZE }>::new(id, ack_cfg),
662                        None,
663                    ),
664                };
665
666                let socket = ReliableSocket::new(
667                    transport,
668                    ack_state,
669                    socket_cfg,
670                    #[cfg(feature = "telemetry")]
671                    NoopTracker,
672                )?;
673
674                match flow_control {
675                    Some((fc_cfg, clock)) => {
676                        let surb_state = surb_mgmt
677                            .clone()
678                            .unwrap_or_else(|| Arc::new(BalancerStateValues::default()));
679                        let supply = SurbSupply::new(surb_state, cfg.frame_mtu);
680                        debug!(?fc_cfg, "wrapping session socket with paced flow-control writer");
681                        Box::new(PacedWriter::new(socket, fc_cfg, clock, supply))
682                    }
683                    None => Box::new(socket),
684                }
685            } else {
686                debug!(?socket_cfg, "opening new stateless session socket");
687
688                Box::new(UnreliableSocket::<{ ApplicationData::PAYLOAD_SIZE }>::new_stateless(
689                    id,
690                    transport,
691                    socket_cfg,
692                    #[cfg(feature = "telemetry")]
693                    NoopTracker,
694                )?)
695            }
696        } else {
697            debug!("opening raw session socket");
698            Box::new(transport)
699        };
700
701        Ok(Self {
702            id,
703            inner,
704            routing,
705            cfg,
706            on_close,
707        })
708    }

735impl futures::AsyncRead for HoprSession {
736    #[instrument(name = "Session::poll_read", level = "trace", skip_all, fields(session_id = %self.id), ret)]
737    fn poll_read(self: Pin<&mut Self>, cx: &mut Context<'_>, buf: &mut [u8]) -> Poll<std::io::Result<usize>> {
738        let this = self.project();
739        let read = futures::ready!(this.inner.poll_read(cx, buf))?;
740        if read == 0 {
741            tracing::trace!("hopr session empty read");
742            // Empty read signals end of the socket, notify if needed
743            if let Some(notifier) = this.on_close.take() {
744                tracing::trace!("notifying read half closure of session");
745                notifier(*this.id, ClosureReason::EmptyRead);
746            }
747        }
748        Poll::Ready(Ok(read))
749    }
750}

773            tracing::trace!("notifying write half closure of session");
774            notifier(*this.id, ClosureReason::WriteClosed);
775        }
776
777        Poll::Ready(Ok(()))
778    }

1074        // The array above is hand-maintained, and a snapshot that silently stops covering a variant
1075        // is worse than no snapshot: it reads as a guarantee it no longer gives. `PixFailure` was
1076        // missing for exactly that reason. This match is exhaustive and wildcard-free, so adding a
1077        // variant to `ClosureReason` fails to compile *here*, which is where the array is.
1078        for reason in reasons {
1079            match reason {
1080                ClosureReason::WriteClosed
1081                | ClosureReason::EmptyRead
1082                | ClosureReason::Eviction
1083                | ClosureReason::UnrealizedDeposit
1084                | ClosureReason::MissingDepositData
1085                | ClosureReason::PixFailure => {}
1086            }
1087        }
1088
1089        insta::assert_debug_snapshot!(reasons);
1090    }

1171                    .map(|(_, data)| ApplicationDataIn {
1172                        data: data.data,
1173                        packet_info: Default::default(),
1174                    })
1175                    .inspect(|d| debug!("bob rcvd: {}", d.data.total_len())),
1176            ),
1177            None,
1178        )?;
1179

1181        let bob_sent = hopr_api::types::crypto_random::random_bytes::<DATA_LEN>();
