22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23enum SsaPhase {
24    /// Request has been sent; waiting for complete verifiable commitment.
25    AwaitingCommitment,
26    /// Commitment is verifiable; waiting for a sufficient deposit.
27    AwaitingDeposit,
28    /// Deposit confirmed; recovering shares.
29    Recovering,
30    /// SSA fully recovered; diagnostic tombstone until `tombstone_until`.
31    /// Any paid FIFO remainder lives separately in [`PaidRecoveryTail`].
32    Recovered { tombstone_until: Instant },
33    /// Phase that will produce a close action on next deadline check.
34    Closing,
35}

41/// Internal state for one supervised SSA.
42struct PerSsaState {
43    ssa_id: SsaId<HoprPseudonym>,
44    /// First SSA index allocated in this request batch.
45    batch_id: u32,
46    phase: SsaPhase,
47
48    // Deadlines (None means not set for this phase).
49    commitment_deadline: Option<Instant>,
50    /// When to re-request the parts of this commitment that have not arrived, or `None` while there
51    /// is nothing to ask for.
52    ///
53    /// Set only once *some* of the commitment has arrived, and pushed out by every further part, so
54    /// it measures silence in the delivery rather than elapsed time. A cycle for which nothing was
55    /// ever delivered leaves it unset and dies on `commitment_deadline`, which is all the Exit can
56    /// do about one: it has no scope to name.
57    recommit_deadline: Option<Instant>,
58    /// Re-requests already made for this cycle, capped at
59    /// [`MAX_COMMITMENT_RETRANSMISSIONS`](hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS).
60    ///
61    /// The cap is a count and not a duration on purpose. `commitment_deadline` bounds the asks in
62    /// *time*, which is what protects the Entry, but the packets they cost are this node's own: at a
63    /// misconfigured millisecond interval inside a 20-second deadline the time bound alone permits
64    /// twenty thousand of them. Matching the figure the Entry will answer means none of the asks is
65    /// wasted either.
66    recommit_attempts: u8,
67    deposit_deadline: Option<Instant>,
68    recovery_idle_deadline: Option<Instant>,
69    recovery_hard_deadline: Option<Instant>,
70
71    // Progress tracking.
72    largest_useful_shares: u64,
73    /// Largest `shares_seen` observed for this cycle — the liveness counter, which moves for surplus
74    /// shares too. Also the staleness discriminator for snapshots, since it moves for every share
75    /// while `largest_useful_shares` does not.
76    largest_shares_seen: u64,
77    target_useful_shares: u64,
78    recovered_polynomials: u16,
79
80    // Overlap / deferred-request state.
81    next_request_pending_deposit: bool,
82    next_requested: bool,
83    /// Set on the last cycle allocated by [`SessionPixSupervisor::emit_request_next_ssa`], and only
84    /// on that one.
85    ///
86    /// The successor gate: a batch asks for its replacement once, when its *last* cycle is nearly
87    /// recovered. Every other flag here is per-cycle, so without this one each member of a batch
88    /// would answer its own `AlmostRecovered` with a whole batch of its own — `ssas_per_request`
89    /// batches per batch, compounding, each cycle a separate on-chain deposit for the Entry.
90    ///
91    /// The last cycle rather than the first because the batch is served in index order: gating on
92    /// the first would ask for the successor a whole batch too early, which is the exposure
93    /// `ssas_per_request` deliberately bounds.
94    ///
95    /// If the two-generation admission bound defers that request, `next_requested` remains set and
96    /// the supervisor-level `successor_request_deferred` flag owns the retry; the gate must not be
97    /// handed to another cycle and spend the request twice.
98    is_batch_last: bool,
99    /// Set when `Recovered` arrives before deposit confirms (i.e. during
100    /// `AwaitingCommitment` or `AwaitingDeposit`). Once deposit is confirmed
101    /// and the SSA enters `Recovering`, this flag triggers an immediate
102    /// tombstone transition — replaying the deferred `Recovered` event.
103    recovered_pending: bool,
104
105    // Service gating.
106    served_total_at_last_progress: u64,
107}

139/// Bounded handoff for SURBs minted against the cycle that has just recovered.
140///
141/// Cryptographic recovery and transport-buffer exhaustion are different instants. The former frees
142/// the reconstructor's heavy state; this small receipt keeps the paid service front on the recovered
143/// predecessor until either its negotiated tail is exhausted or any progress proves the successor
144/// has reached the head of the FIFO. The latter need not be funded: observing an unfunded successor
145/// is precisely when the predecessor's receipt must stop authorizing service.
146struct PaidRecoveryTail {
147    ssa_id: SsaId<HoprPseudonym>,
148    largest_shares_seen: u64,
149    served_total_at_last_progress: u64,
150    idle_deadline: Option<Instant>,
151    hard_deadline: Option<Instant>,
152}

158/// Deterministic core of the PIX session supervisor.
159pub struct SessionPixSupervisor {
160    pub(crate) cfg: SupervisorConfig,
161    pub(crate) dims: PixParams,
162    pub(crate) pseudonym: HoprPseudonym,
163    pub(crate) closed: bool,
164    next_ssa_index: u32,
165    /// The cycle the two share-order counters below are measured against — the earliest unrecovered
166    /// cycle, i.e. the one the Entry should currently be serving. `None` before the first batch exists.
167    share_order_front: Option<SsaIndex>,
168    /// Useful shares booked on [`Self::share_order_front`] since it took the front.
169    front_useful: u64,
170    /// Useful shares booked on any *other* live cycle over that same span.
171    ///
172    /// `off_front_useful / (front_useful + off_front_useful)` is the share-order ratio — see
173    /// [`SupervisorConfig::max_off_front_share_fraction`] for what it detects and why it is a fraction
174    /// rather than a count. Both counters reset when the front changes, which is what keeps the window
175    /// bounded to a single cycle's service.
176    off_front_useful: u64,
177    /// Whether the service gate currently follows a funded front cycle.
178    service_open: bool,
179    /// Front cycle for which the current gate mode was emitted.
180    ///
181    /// Tracking the identity as well as the mode lets a funded-to-funded handoff rebaseline the
182    /// served-without-progress ceiling for the newly promoted cycle.
183    service_front: Option<SsaIndex>,
184    /// A front cycle became paid and terminal in one event, so an unfunded successor still earns a
185    /// freshly restored predeposit allowance even though the gate never entered funded mode.
186    paid_front_handoff: bool,
187    /// The sole recovered predecessor authorized to report buffered liveness.
188    ///
189    /// This never becomes a set: accepting arbitrary recovered tombstones would let the Entry replay
190    /// old cycle tags to keep service open. Replacement on each in-order recovery is the supervisor's
191    /// independent bound behind the reconstructor's per-polynomial surplus counters.
192    paid_recovery_tail: Option<PaidRecoveryTail>,
193    /// A successor batch was earned but could not yet fit the two-generation reservation.
194    successor_request_deferred: bool,
195    /// Ordered SSAs (oldest first, newest last), including short-lived recovered tombstones.
196    ssas: Vec<PerSsaState>,
197    /// Tracks the first failure reason when multiple SSAs fail, so the
198    /// earliest cause is used for the final `Close` action rather than the last.
199    first_failure_reason: Option<SessionPixCloseReason>,
200    /// Cycles this Session has lost without recovering them, over its whole life.
201    ///
202    /// Cumulative, and that is the whole point: retiring a failed member leaves no trace on its
203    /// siblings, so without a counter that survives the retirement an Entry can lose one cycle per
204    /// batch indefinitely while a single funded sibling holds the Session open. Bounded by
205    /// [`SupervisorConfig::max_failed_cycles`].
206    failed_cycles: usize,
207    /// Plans the rate at which this Exit originates its own fill keep-alives.
208    ///
209    /// Kept here rather than in the worker because the rate is a function of the lifecycle state
210    /// this struct owns — which cycle is at the paid front and how far it has got — and because
211    /// every close path has to be able to stop it in the same action vector that carries the close.
212    fill: FillPlanner,
213    /// Greatest SSA index that has been retired (closed and removed).
214    ///
215    /// Prevents a stale `SsaRequestSent` from resurrecting a closed SSA. A high-watermark rather
216    /// than the set of every retired index: the set grew for the whole life of the Session and was
217    /// scanned linearly on every request, one entry per cycle, without bound.
218    ///
219    /// A watermark alone would be wrong if it were consulted first — retirement is not monotone,
220    /// since a later batch member can lose its deposit while an earlier one is still recovering, and
221    /// "at or below the greatest retired index" would then condemn that live earlier cycle. The
222    /// guard is therefore placed *after* the live-record lookup, so a cycle that still exists always
223    /// wins and the watermark only ever answers for indices that have none.
224    highest_retired_ssa_index: Option<SsaIndex>,
225}

261    /// Handle a lifecycle event.
262    pub fn handle_event(&mut self, ev: &SessionPixEvent, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
263        if self.closed {
264            return Vec::new();
265        }
266
267        let actions = match ev {
268            SessionPixEvent::SsaRequestSent(ssa_id) => self.on_ssa_request_sent(ssa_id, now),
269            SessionPixEvent::CommitmentProgress(ssa_id) => self.on_commitment_progress(ssa_id, now),
270            SessionPixEvent::CommitmentVerified(ssa_id) => self.on_commitment_verified(ssa_id, now),
271            SessionPixEvent::DepositConfirmed { ssa_id, amount } => {
272                self.on_deposit_confirmed(ssa_id, *amount, now, served_total)
273            }
274            SessionPixEvent::DepositObserverClosed(ssa_id) => self.on_deposit_observer_closed(ssa_id, now),
275            SessionPixEvent::RecoveryProgress(progress) => self.on_recovery_progress(progress, now, served_total),
276            SessionPixEvent::AlmostRecovered(ssa_id) => self.on_almost_recovered(ssa_id, now),
277            SessionPixEvent::Recovered(ssa_id) => self.on_recovered(ssa_id, now, served_total),
278            SessionPixEvent::UnverifiableShares { ssa_id, observed_total } => {
279                self.on_unverifiable_shares(ssa_id, *observed_total, now)
280            }
281        };
282
283        let mut lifecycle_actions = actions;
284        lifecycle_actions.extend(self.retry_deferred_successor_request(now));
285        self.consume_paid_tail_at_observed_boundary();
286        self.arm_recovery_clocks_for_earliest(now, served_total);
287        let mut actions = if lifecycle_actions
288            .iter()
289            .any(|action| matches!(action, SessionPixAction::Close(_)))
290        {
291            // A closing Session has no cycle left to fill for, and the fill stream outlives this
292            // state machine by however long it takes the driver to tear the Session down. Silencing
293            // it here rather than only there is what keeps a close from being followed by a burst of
294            // keep-alives for a cycle that no longer exists.
295            self.fill
296                .stop()
297                .map(SessionPixAction::SetFillRate)
298                .into_iter()
299                .collect()
300        } else {
301            self.sync_service_gate()
302        };
303        actions.extend(lifecycle_actions);
304        self.refresh_share_order_front();
305        actions
306    }

308    /// Raises the retired-index watermark.
309    ///
310    /// Takes the maximum rather than assuming the caller retires in order, because it does not: a
311    /// later batch member losing its deposit is retired before an earlier one that is still
312    /// recovering.
313    fn note_retired_index(&mut self, index: SsaIndex) {
314        self.highest_retired_ssa_index = Some(match self.highest_retired_ssa_index {
315            Some(highest) if highest >= index => highest,
316            _ => index,
317        });
318    }

320    /// Index of the earliest cycle that has not finished recovering — the one the Entry should be
321    /// serving, given that emission is clamped to a single cycle.
322    fn earliest_live_idx(&self) -> Option<usize> {
323        self.ssas
324            .iter()
325            .enumerate()
326            .filter(|(_, s)| !s.is_terminal())
327            .min_by_key(|(_, s)| s.ssa_id.ssa_index())
328            .map(|(i, _)| i)
329    }

331    /// Consumes the predecessor receipt once the current live front has reported any share.
332    ///
333    /// This check is centralized rather than tied only to `RecoveryProgress`: a previously queued
334    /// cycle can become the front because a sibling is retired or because its deposit arrives after
335    /// its first snapshot. In every case the observation proves the transport crossed the cycle
336    /// boundary. Keeping the old funded receipt beyond it would let an unfunded successor spend the
337    /// predecessor's remaining allowance.
338    fn consume_paid_tail_at_observed_boundary(&mut self) {
339        let Some(front_idx) = self.earliest_live_idx() else {
340            return;
341        };
342        let front_id = self.ssas[front_idx].ssa_id;
343        if self.ssas[front_idx].largest_shares_seen == 0 {
344            return;
345        }
346
347        if let Some(predecessor) = self
348            .paid_recovery_tail
349            .as_ref()
350            .map(|tail| tail.ssa_id)
351            .filter(|predecessor| predecessor.ssa_index() < front_id.ssa_index())
352        {
353            tracing::debug!(
354                %predecessor,
355                successor = %front_id,
356                "successor progress consumed the paid FIFO-tail handoff"
357            );
358            self.paid_recovery_tail = None;
359        }
360    }

362    /// Keeps the service gate aligned with the funding state of the earliest live cycle.
363    ///
364    /// A funded-to-unfunded handoff restores the configured predeposit allowance. A
365    /// funded-to-funded handoff emits `ReleaseService` too: the mode does not change, but the new
366    /// front must receive its own served-without-progress ceiling rather than inheriting the tail of
367    /// its predecessor's. An unfunded cycle that fails does not earn another allowance.
368    fn sync_service_gate(&mut self) -> Vec<SessionPixAction> {
369        if self.closed {
370            return Vec::new();
371        }
372
373        let front = self.earliest_live_idx();
374        let front_id = self
375            .paid_recovery_tail
376            .as_ref()
377            .map(|tail| tail.ssa_id.ssa_index())
378            .or_else(|| front.map(|idx| self.ssas[idx].ssa_id.ssa_index()));
379        let front_changed = front_id != self.service_front;
380        let front_funded =
381            self.paid_recovery_tail.is_some() || front.is_some_and(|idx| self.ssas[idx].phase == SsaPhase::Recovering);
382        let paid_front_handoff = std::mem::take(&mut self.paid_front_handoff);
383        self.service_front = front_id;
384
385        match (front_funded, self.service_open, front_changed) {
386            (true, false, _) | (true, true, true) => {
387                self.service_open = true;
388                tracing::debug!(?front_id, "aligning service gate with funded front cycle");
389                vec![SessionPixAction::ReleaseService]
390            }
391            (false, true, _) => {
392                self.service_open = false;
393                tracing::debug!(?front_id, "withholding service for unfunded front cycle");
394                vec![SessionPixAction::WithholdService]
395            }
396            (false, false, _) if paid_front_handoff => {
397                tracing::debug!(?front_id, "restoring predeposit service after paid front handoff");
398                vec![SessionPixAction::WithholdService]
399            }
400            _ => Vec::new(),
401        }
402    }

434    /// Re-points the share-order accounting at the current front of the batch, clearing it if the front
435    /// moved.
436    ///
437    /// Clearing on a change of front is what bounds the measurement window to one cycle's service, and
438    /// it is why an Entry cannot launder a spree by eventually finishing a cycle: the reset is only
439    /// reached by the front cycle *completing*, which is the thing the Exit wanted all along.
440    ///
441    /// Called after every event and every deadline sweep, so it covers both ways the front can move —
442    /// the front cycle recovering, or being retired.
443    fn refresh_share_order_front(&mut self) {
444        let front = self.earliest_live_idx().map(|i| self.ssas[i].ssa_id.ssa_index());
445        if front != self.share_order_front {
446            self.share_order_front = front;
447            self.front_useful = 0;
448            self.off_front_useful = 0;
449        }
450    }

452    /// Starts the recovery clocks of the earliest unrecovered cycle, if they are not running yet.
453    ///
454    /// A batch's cycles are served strictly in index order — the Entry's emission window is clamped to
455    /// one cycle (see [`hopr_protocol_pix::SHARE_EMISSION_WINDOW`]) — so a cycle behind the front of
456    /// the batch is *queued*, not stalled. Starting its clocks when its deposit confirmed, as this used
457    /// to, measured the queue wait rather than the recovery: at deployed dimensions one cycle takes
458    /// ~61 min of emission to exhaust, so every cycle after the first was retired by
459    /// `max_recovery_time` — or, worse, by `max_recovery_idle` inside a minute, since the idle gate
460    /// compares against *session-wide* service and a queued cycle sees plenty of that without any of it
461    /// being its own.
462    ///
463    /// So the clock starts when the cycle reaches the front, which is when it can actually make
464    /// progress, and [`SupervisorConfig::max_recovery_time`] then means what its name says: the ceiling
465    /// on recovering a single cycle. `served_total_at_last_progress` is re-baselined at the same moment
466    /// for the same reason — otherwise the idle gate would charge the queue wait against it.
467    ///
468    /// Called after every event and every deadline sweep, so it covers each way a cycle can reach the
469    /// front: its predecessor recovering, or its predecessor being retired.
470    fn arm_recovery_clocks_for_earliest(&mut self, now: Instant, served_total: u64) {
471        if self.closed {
472            return;
473        }
474        // The packets being served still carry the recovered predecessor's shares. Starting the
475        // successor now would charge that pipeline delay against both its idle and hard recovery
476        // budgets even though no successor share can possibly arrive yet.
477        if self.paid_recovery_tail.is_some() {
478            return;
479        }
480        let Some(idx) = self.earliest_live_idx() else {
481            return;
482        };
483
484        let ssa = &mut self.ssas[idx];
485        if ssa.phase != SsaPhase::Recovering || ssa.recovery_hard_deadline.is_some() {
486            return;
487        }
488
489        ssa.recovery_idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
490        ssa.recovery_hard_deadline = now.checked_add(self.cfg.max_recovery_time);
491        ssa.served_total_at_last_progress = served_total;
492        tracing::debug!(ssa_id = %ssa.ssa_id, "recovery clocks started — cycle is at the front of the batch");
493    }

495    /// Check all deadlines and emit actions for any that have expired.
496    pub fn handle_deadline(&mut self, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
497        if self.closed {
498            return Vec::new();
499        }
500
501        let mut actions = Vec::new();
502        let max_recovery_idle = self.cfg.max_recovery_idle;
503
504        // While the recovered predecessor still owns the service front, its original recovery
505        // clocks remain the resource bound. Progress refreshes only the idle clock; the hard clock
506        // is never extended. Thus a large but paid surplus may drain, while an Entry that declares
507        // one and then feeds neither tail nor successor cannot hold the Session forever.
508        if let Some(tail) = self.paid_recovery_tail.as_mut() {
509            let expired = if tail.hard_deadline.is_some_and(|deadline| now >= deadline) {
510                Some(SessionPixCloseReason::RecoveryDeadline)
511            } else if tail.idle_deadline.is_some_and(|deadline| now >= deadline) {
512                Some(SessionPixCloseReason::RecoveryIdle)
513            } else {
514                None
515            };
516
517            if let Some(reason) = expired {
518                if reason == SessionPixCloseReason::RecoveryIdle && served_total <= tail.served_total_at_last_progress {
519                    tail.idle_deadline = now.checked_add(max_recovery_idle);
520                } else {
521                    tracing::warn!(
522                        ssa_id = %tail.ssa_id,
523                        ?reason,
524                        largest_shares_seen = tail.largest_shares_seen,
525                        served_total_at_last_progress = tail.served_total_at_last_progress,
526                        "closing PIX Session while its paid recovery tail is stalled"
527                    );
528                    self.closed = true;
529                    return self.with_fill_stopped(vec![SessionPixAction::Close(reason)]);
530                }
531            }
532        }
533
534        let mut i = 0;
535        while i < self.ssas.len() {
536            if self.closed {
537                break;
538            }
539
540            let expired = {
541                let ssa = &self.ssas[i];
542                if ssa.is_terminal() {
543                    i += 1;
544                    continue;
545                }
546                ssa.check_deadlines(now)
547            };
548
549            // A stalled commitment is re-requested rather than given up on, and the timer re-arms so
550            // a repair that is itself lost is asked for again. This yields no close reason and is
551            // bounded by nothing of its own: the absolute `commitment_deadline` still ends the cycle,
552            // which is why it can repeat freely — and why it is skipped once that deadline has gone,
553            // rather than putting an ask on the wire for a Session about to close.
554            if expired.is_none()
555                && self.ssas[i].phase == SsaPhase::AwaitingCommitment
556                && self.ssas[i].recommit_deadline.is_some_and(|deadline| now >= deadline)
557            {
558                let ssa = &mut self.ssas[i];
559                ssa.recommit_attempts += 1;
560                // Disarmed rather than re-armed at the cap, so the worker stops waking for a timer
561                // that would produce nothing. Further arriving parts cannot re-arm it either.
562                ssa.recommit_deadline = (ssa.recommit_attempts < hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS)
563                    .then(|| now.checked_add(self.cfg.commitment_recommit_interval))
564                    .flatten();
565                tracing::debug!(
566                    ssa_id = %ssa.ssa_id, attempt = ssa.recommit_attempts,
567                    "commitment delivery stalled — re-requesting the parts that did not arrive"
568                );
569                actions.push(SessionPixAction::RequestCommitmentRetransmission {
570                    ssa_id: ssa.ssa_id,
571                    params: self.dims,
572                });
573            }
574
575            if let Some(reason) = expired {
576                // Service-gated idle: if no service consumed since last progress,
577                // re-arm instead of closing.
578                if reason == SessionPixCloseReason::RecoveryIdle
579                    && served_total <= self.ssas[i].served_total_at_last_progress
580                {
581                    self.ssas[i].recovery_idle_deadline = now.checked_add(max_recovery_idle);
582                    i += 1;
583                    continue;
584                }
585
586                actions.extend(self.close_ssa_and_collect(i, reason));
587                continue;
588            }
589            i += 1;
590        }
591
592        // If the session is already closing, skip tombstone retirement.
593        // Whole-session teardown retires everything via the retirement guard.
594        if self.closed {
595            return self.with_fill_stopped(actions);
596        }
597
598        // Remove tombstones that have expired and emit RetireSsa so the
599        // reconstructor and observer state is released mid-session.
600        let retired_ids: Vec<SsaId<_>> = self
601            .ssas
602            .iter()
603            .filter(|ssa| matches!(ssa.phase, SsaPhase::Recovered { tombstone_until } if now >= tombstone_until))
604            .map(|ssa| ssa.ssa_id)
605            .collect();
606        self.ssas
607            .retain(|ssa| !matches!(ssa.phase, SsaPhase::Recovered { tombstone_until } if now >= tombstone_until));
608        for id in retired_ids {
609            self.note_retired_index(id.ssa_index());
610            actions.push(SessionPixAction::RetireSsa(id));
611        }
612
613        // A terminal cycle may have released the older of two live batches. Keep retirement before
614        // the replacement request in the action stream so the reconstructor drops the old guard
615        // before allocating the successor.
616        actions.extend(self.retry_deferred_successor_request(now));
617        self.consume_paid_tail_at_observed_boundary();
618
619        // If no SSAs remain, close.
620        // Note: a successor request may still be in flight here (RequestSsa
621        // emitted, SsaRequestSent not yet observed) — intentionally not
622        // tracked. The tombstone retention window is the backstop; a request
623        // delayed beyond it (>30 s transport stall on an otherwise healthy
624        // session) is theoretical and does not warrant blocking normal
625        // tombstone expiry.
626        if self.ssas.is_empty() && !self.closed {
627            actions.push(SessionPixAction::Close(
628                self.first_failure_reason
629                    .unwrap_or(SessionPixCloseReason::NoSsaRemaining),
630            ));
631            self.closed = true;
632        }
633
634        // Retiring or tombstoning a cycle can promote the next one to the front of the batch — and if
635        // that one is already funded, promotion is the only moment left at which service can be
636        // released, since its deposit event is long past.
637        self.arm_recovery_clocks_for_earliest(now, served_total);
638        let mut gate_actions = self.sync_service_gate();
639        gate_actions.extend(actions);
640        self.refresh_share_order_front();
641
642        if self.closed {
643            return self.with_fill_stopped(gate_actions);
644        }
645        gate_actions
646    }

648    /// Runs the deadline sweep and, if one is due, the fill tick — in that order.
649    ///
650    /// The order is the point. The sweep is what arms a promoted cycle's recovery clocks, consumes a
651    /// spent FIFO tail and re-aligns the service gate, so running it first means the tick plans
652    /// against the lifecycle as it is *now* rather than as it was a second ago. A tick that ran first
653    /// would spend one whole sampling interval filling for a cycle that had already recovered.
654    ///
655    /// This is the entry point the worker uses; [`handle_deadline`](Self::handle_deadline) remains
656    /// separate because the deadline logic is meaningful — and tested — on its own.
657    pub fn handle_timers(&mut self, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
658        let mut actions = self.handle_deadline(now, served_total);
659        if self.closed {
660            return actions;
661        }
662        if self.next_fill_tick().is_some_and(|tick| now >= tick) {
663            actions.extend(self.handle_fill_tick(now, served_total));
664        }
665        actions
666    }

668    /// Re-plans the fill rate, emitting an action only when the rate actually changes.
669    pub fn handle_fill_tick(&mut self, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
670        if self.closed {
671            return Vec::new();
672        }
673        let target = self.fill_target();
674        self.fill
675            .plan(now, served_total, target)
676            .map(|rate| vec![SessionPixAction::SetFillRate(rate)])
677            .into_iter()
678            .flatten()
679            .collect()
680    }

690    /// When the fill planner next wants to run, or `None` if there is nothing for it to do.
691    ///
692    /// `None` while no cycle is being filled for *and* the stream is already silent, which is what
693    /// keeps an unfunded or non-PIX-idle Session from waking its worker once a second for nothing. A
694    /// running stream always yields a tick even without a target, because it is owed the zero that
695    /// stops it.
696    pub fn next_fill_tick(&self) -> Option<Instant> {
697        if self.closed || !self.fill.is_enabled() {
698            return None;
699        }
700        if self.fill_target().is_none() && self.fill.is_idle() {
701            return None;
702        }
703        Some(self.fill.next_tick())
704    }

706    /// The cycle the Exit should currently be filling for, if any.
707    ///
708    /// The recovered predecessor's FIFO tail wins whenever it exists, for two reasons that point the
709    /// same way: its SURBs sit ahead of the successor's in the Exit's own buffer, so packets sent now
710    /// carry *its* shares; and it holds the binding hard deadline, since the successor's clocks are
711    /// deliberately not armed until the tail is exhausted (see
712    /// [`arm_recovery_clocks_for_earliest`](Self::arm_recovery_clocks_for_earliest)).
713    ///
714    /// Otherwise it is the earliest live cycle, and only once it is both funded and clocked. An
715    /// unfunded cycle has nothing at stake to be filled for, and a funded one whose clocks have not
716    /// armed is queued behind a predecessor rather than being served — filling for it would spend
717    /// SURBs on shares the Entry cannot yet be emitting.
718    fn fill_target(&self) -> Option<FillTarget> {
719        if let Some(tail) = self.paid_recovery_tail.as_ref() {
720            return tail.hard_deadline.map(|hard_deadline| FillTarget {
721                ssa_id: tail.ssa_id,
722                largest_shares_seen: tail.largest_shares_seen,
723                hard_deadline,
724            });
725        }
726
727        let ssa = &self.ssas[self.earliest_live_idx()?];
728        (ssa.phase == SsaPhase::Recovering)
729            .then_some(ssa)
730            .and_then(|ssa| ssa.recovery_hard_deadline.map(|hard_deadline| (ssa, hard_deadline)))
731            .map(|(ssa, hard_deadline)| FillTarget {
732                ssa_id: ssa.ssa_id,
733                largest_shares_seen: ssa.largest_shares_seen,
734                hard_deadline,
735            })
736    }

738    /// Prepends the action that silences the fill stream, when there is one to silence.
739    ///
740    /// Returns `actions` untouched for a Session that never filled — a disabled one, or one that
741    /// closed before its first cycle funded — so the action stream of every non-filling Session is
742    /// exactly what it was before this existed.
743    fn with_fill_stopped(&mut self, actions: Vec<SessionPixAction>) -> Vec<SessionPixAction> {
744        match self.fill.stop() {
745            Some(rate) => {
746                let mut stopped = Vec::with_capacity(actions.len() + 1);
747                stopped.push(SessionPixAction::SetFillRate(rate));
748                stopped.extend(actions);
749                stopped
750            }
751            None => actions,
752        }
753    }

755    /// Returns the earliest deadline across all live SSAs, or `None`.
756    pub fn next_deadline(&self) -> Option<Instant> {
757        if self.closed || (self.ssas.is_empty() && self.paid_recovery_tail.is_none()) {
758            return None;
759        }
760
761        let ssa_deadline = self
762            .ssas
763            .iter()
764            .filter_map(|ssa| {
765                if ssa.is_terminal() {
766                    if let SsaPhase::Recovered { tombstone_until } = ssa.phase {
767                        return Some(tombstone_until);
768                    }
769                    return None;
770                }
771                // Return the earliest set deadline. The re-request timer belongs here like the rest:
772                // it is shorter than every other deadline of the phase it runs in, so leaving it out
773                // would mean the worker slept past it and only re-requested when something else woke
774                // it up.
775                ssa.commitment_deadline
776                    .into_iter()
777                    .chain(ssa.recommit_deadline)
778                    .chain(ssa.deposit_deadline)
779                    .chain(ssa.recovery_idle_deadline)
780                    .chain(ssa.recovery_hard_deadline)
781                    .min()
782            })
783            .min();
784        let tail_deadline = self
785            .paid_recovery_tail
786            .as_ref()
787            .and_then(|tail| tail.idle_deadline.into_iter().chain(tail.hard_deadline).min());
788
789        ssa_deadline.into_iter().chain(tail_deadline).min()
790    }

817    /// Arms the commitment deadline once the request this SSA was registered for has gone out.
818    ///
819    /// The record itself is created by [`emit_request_next_ssa`](Self::emit_request_next_ssa); this
820    /// only starts the clock, because the Entry cannot be late answering a request it has not been
821    /// sent yet.
822    fn on_ssa_request_sent(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
823        // Validate pseudonym. Checked before the lookup: a foreign pseudonym never matches a record
824        // of ours, so looking first would answer a cross-session confusion with silence.
825        if ssa_id.pseudonym() != &self.pseudonym {
826            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
827        }
828
829        // The live-record lookup comes first, and the staleness guard only answers for indices that
830        // have no record. Retirement is not monotone — a later batch member can lose its deposit while
831        // an earlier one is still recovering — so a watermark consulted first would silently ignore a
832        // legitimate request for a cycle that is very much alive. Asking the records first removes the
833        // assumption instead of relying on it.
834        let Some(idx) = self.find_ssa_idx(ssa_id) else {
835            // No live record. Either this index was retired and the event is a straggler, which is
836            // benign and ignored, or we were never asked for it at all, which is a protocol fault.
837            if self
838                .highest_retired_ssa_index
839                .is_some_and(|highest| ssa_id.ssa_index() <= highest)
840            {
841                return Vec::new();
842            }
843            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
844        };
845
846        // Idempotent: a repeated confirmation must not extend the deadline, and one arriving after
847        // the SSA has moved on must not resurrect a phase it has left.
848        if self.ssas[idx].phase == SsaPhase::AwaitingCommitment && self.ssas[idx].commitment_deadline.is_none() {
849            self.ssas[idx].commitment_deadline = now.checked_add(crate::supervision::scaled_deadline(
850                self.cfg.max_ssa_delivery_time,
851                self.cfg.ssas_per_request,
852            ));
853        }
854
855        Vec::new()
856    }

858    /// Arms — or re-arms — the re-request timer on a commitment that is under way but not complete.
859    ///
860    /// This is where the issue's rule lives: a cycle that received *nothing* never reaches here, so
861    /// it never asks, which is correct because there is no scope for it to name. And because every
862    /// arriving part pushes the timer out, the first ask cannot land in the middle of the burst —
863    /// what the timer measures is a stall in delivery.
864    ///
865    /// Unscaled by the batch size, unlike the two absolute deadlines: this is a per-cycle stall
866    /// detector, and a batch's commitment sets do not queue behind one another in a way that would
867    /// make one cycle's silence expected.
868    fn on_commitment_progress(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
869        if let Some(idx) = self.find_ssa_idx(ssa_id)
870            && self.ssas[idx].phase == SsaPhase::AwaitingCommitment
871            && self.ssas[idx].recommit_attempts < hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS
872        {
873            self.ssas[idx].recommit_deadline = now.checked_add(self.cfg.commitment_recommit_interval);
874        }
875
876        Vec::new()
877    }

879    fn on_commitment_verified(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
880        let idx = match self.find_ssa_idx(ssa_id) {
881            Some(i) => i,
882            None => return Vec::new(),
883        };
884        let ssa = &mut self.ssas[idx];
885        if ssa.phase != SsaPhase::AwaitingCommitment {
886            return Vec::new();
887        }
888
889        let deposit_deadline = now.checked_add(crate::supervision::scaled_deadline(
890            self.cfg.max_deposit_wait,
891            self.cfg.ssas_per_request,
892        ));
893        ssa.phase = SsaPhase::AwaitingDeposit;
894        ssa.deposit_deadline = deposit_deadline;
895        ssa.commitment_deadline = None;
896        // Nothing is missing any more, so nothing is to be asked for.
897        ssa.recommit_deadline = None;
898
899        Vec::new()
900    }

902    /// Funds the cycle on the deposit pool's verdict. The supervisor does not second-guess the amount.
903    ///
904    /// Sufficiency is priced and judged outside this workspace. The Exit's pool is handed the
905    /// [`DepositUpdated`](hopr_api::node::DepositUpdated) sender together with the byte quota when the
906    /// deposit address arrives, and it sends on that channel once the deposit clears the price it
907    /// charges for that quota — so by the time this runs, the question has been answered by the only
908    /// component that can answer it. Nothing in the Session layer prices a byte.
909    ///
910    /// This used to compare the running total against `max(expected_deposit, min_deposit)`. Both are
911    /// gone, and not because they were merely unused: a configured floor here is not a backstop behind
912    /// the pool but a second authority alongside it, and one set above the pool's price fails a cycle
913    /// that has already been paid for — silently, on `max_deposit_wait`, with neither side able to
914    /// break the tie. `expected_deposit` was the same idea arriving over the wire, and the commitment
915    /// never carried an amount to fill it.
916    ///
917    /// What remains is the one check that needs no price: a zero balance asserts that nothing was
918    /// deposited, so it is not a verdict and cannot release service. The deadline keeps running and a
919    /// later confirmation can still fund the cycle.
920    fn on_deposit_confirmed(
921        &mut self,
922        ssa_id: &SsaId<HoprPseudonym>,
923        amount: HoprBalance,
924        now: Instant,
925        served_total: u64,
926    ) -> Vec<SessionPixAction> {
927        let idx = match self.find_ssa_idx(ssa_id) {
928            Some(i) => i,
929            None => return Vec::new(),
930        };
931        let was_front = self.earliest_live_idx() == Some(idx);
932
933        let ssa = &mut self.ssas[idx];
934        if ssa.phase != SsaPhase::AwaitingDeposit {
935            return Vec::new();
936        }
937
938        if amount == HoprBalance::zero() {
939            return Vec::new();
940        }
941
942        // Transition to Recovering. The two recovery clocks are *not* started here — see
943        // `arm_recovery_clocks_for_earliest`, which starts them when this cycle's turn comes.
944        ssa.phase = SsaPhase::Recovering;
945        ssa.deposit_deadline = None;
946        ssa.served_total_at_last_progress = served_total;
947
948        // If recovery completed before the deposit arrived, immediately
949        // tombstone the SSA — the Recovered event was deferred.
950        let recovered_pending = ssa.recovered_pending;
951
952        let pending = ssa.next_request_pending_deposit;
953        if pending {
954            ssa.next_request_pending_deposit = false;
955        }
956        // End the mutable borrow on ssas[idx].
957        let _ = ssa;
958
959        let mut actions = Vec::new();
960
961        if recovered_pending {
962            if was_front {
963                self.paid_front_handoff = true;
964            }
965            actions.extend(self.perform_recovered_transition(idx, now, served_total));
966        }
967
968        if pending {
969            // When recovered_pending is also set, perform_recovered_transition
970            // won't emit RequestSsa (next_requested is already set by
971            // on_recovered), so emit it here.
972            actions.extend(self.emit_request_next_ssa(now));
973        }
974
975        actions
976    }

991    fn on_recovery_progress(
992        &mut self,
993        progress: &SsaRecoveryProgress<HoprPseudonym>,
994        now: Instant,
995        served_total: u64,
996    ) -> Vec<SessionPixAction> {
997        if self
998            .paid_recovery_tail
999            .as_ref()
1000            .is_some_and(|tail| tail.ssa_id == progress.ssa_id)
1001        {
1002            return self.on_recovered_tail_progress(progress, now, served_total);
1003        }
1004
1005        let idx = match self.find_ssa_idx(&progress.ssa_id) {
1006            Some(i) => i,
1007            None => return Vec::new(),
1008        };
1009
1010        // Absorb late progress on tombstones — the SSA is already fully
1011        // recovered and should not reset the session-wide gate watermark.
1012        if self.ssas[idx].is_terminal() {
1013            return Vec::new();
1014        }
1015
1016        // Validate every dimension-derived bound before mutating. Progress is produced locally, but
1017        // all of its source fields ultimately came from peer-tagged shares. Keeping the supervisor
1018        // as an independent backstop means a reconstructor regression cannot turn an oversized
1019        // counter into unbounded service.
1020        let target = self.dims.target_useful_shares();
1021        let max_shares_seen = self.dims.polys_per_ssa() as u64 * self.dims.emitted_shares_per_poly() as u64;
1022        if progress.target_useful_shares != target
1023            || progress.useful_shares > target
1024            || progress.shares_seen < progress.useful_shares
1025            || progress.shares_seen > max_shares_seen
1026            || progress.recovered_polynomials > self.dims.polys_per_ssa()
1027        {
1028            return vec![SessionPixAction::Close(SessionPixCloseReason::CounterRegression)];
1029        }
1030
1031        let ssa = &mut self.ssas[idx];
1032        let new_useful = progress.useful_shares;
1033        let new_seen = progress.shares_seen;
1034
1035        // Counter regression check, on the liveness counter because it is the one that moves for every
1036        // share — a snapshot can be newer than the last one while carrying the same `useful_shares`.
1037        //
1038        // The relay-as-Exit pipeline processes acknowledgement batches with
1039        // for_each_concurrent, so absolute progress snapshots from different
1040        // batches can arrive out of order. Treat a stale snapshot as benign
1041        // noise rather than a protocol violation.
1042        if new_seen <= ssa.largest_shares_seen {
1043            return Vec::new();
1044        }
1045        ssa.largest_shares_seen = new_seen;
1046
1047        // Liveness tier. A share arrived, useful or not, so the Entry is serving this cycle — which is
1048        // the question the egress gate and the recovery-idle deadline are asking. Keying either of them
1049        // on `useful_shares` instead makes a conforming Entry's surplus run look like silence, and for
1050        // the gate that is self-inflicted: withholding service removes the only thing that could
1051        // produce the next useful share. See `SsaRecoveryProgress::shares_seen`.
1052        ssa.served_total_at_last_progress = served_total;
1053        // Refreshing is not arming: only `arm_recovery_clocks_for_earliest` starts a cycle's clocks,
1054        // and `recovery_hard_deadline` is how a cycle records that it has reached the front. Without
1055        // that second condition a stray reordered share *starts* the idle clock on a cycle still
1056        // queued behind the front — which the Entry cannot then feed, because emission is clamped to
1057        // one cycle — and 60 s later the idle gate sees session-wide service climbing and retires it.
1058        // That is the queue-wait failure `arm_recovery_clocks_for_earliest` documents, reached
1059        // through the refresh path instead of the deposit one. The clocks arm together or not at all.
1060        if ssa.phase == SsaPhase::Recovering && ssa.recovery_hard_deadline.is_some() {
1061            ssa.recovery_idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
1062        }
1063
1064        // Payment tier. Only shares that advanced reconstruction are worth anything, so only they move
1065        // the recovery counters or count towards the share-order ratio.
1066        if new_useful > ssa.largest_useful_shares {
1067            let delta = new_useful - ssa.largest_useful_shares;
1068            ssa.largest_useful_shares = new_useful;
1069            ssa.recovered_polynomials = progress.recovered_polynomials;
1070
1071            if let Some(reason) = self.book_share_order_progress(&progress.ssa_id, delta) {
1072                return vec![SessionPixAction::Close(reason)];
1073            }
1074        }
1075
1076        let is_front = self.earliest_live_idx() == Some(idx);
1077
1078        // Only funded progress on the cycle service is currently charged against may reopen the
1079        // Session-wide ceiling. Unfunded progress is the H11 bypass; off-front progress belongs to a
1080        // queued cycle and must not buy service for the front either. The outer event transition
1081        // consumes any predecessor tail before synchronizing the gate, even for unfunded progress.
1082        if self.ssas[idx].phase == SsaPhase::Recovering && is_front {
1083            vec![SessionPixAction::ProgressNotification]
1084        } else {
1085            Vec::new()
1086        }
1087    }
1088
1089    /// Applies a progress snapshot from the sole recovered predecessor authorized to drain.
1090    ///
1091    /// The reconstructor enforces the strong per-polynomial budget. These aggregate checks are a
1092    /// second security boundary: even if that component regresses, the Entry cannot make a recovered
1093    /// cycle report more than `polys × (threshold + surplus)` packets, change its terminal useful
1094    /// count, or revive any older tombstone.
1095    fn on_recovered_tail_progress(
1096        &mut self,
1097        progress: &SsaRecoveryProgress<HoprPseudonym>,
1098        now: Instant,
1099        served_total: u64,
1100    ) -> Vec<SessionPixAction> {
1101        let target = self.dims.target_useful_shares();
1102        let expected_polynomials = self.dims.polys_per_ssa();
1103        let max_shares_seen = expected_polynomials as u64 * self.dims.emitted_shares_per_poly() as u64;
1104
1105        if progress.target_useful_shares != target
1106            || progress.useful_shares > target
1107            || progress.recovered_polynomials > expected_polynomials
1108            || progress.shares_seen < progress.useful_shares
1109            || progress.shares_seen > max_shares_seen
1110        {
1111            tracing::warn!(
1112                ssa_id = %progress.ssa_id,
1113                useful_shares = progress.useful_shares,
1114                shares_seen = progress.shares_seen,
1115                recovered_polynomials = progress.recovered_polynomials,
1116                "rejected impossible progress from a recovered PIX tail"
1117            );
1118            return vec![SessionPixAction::Close(SessionPixCloseReason::CounterRegression)];
1119        }
1120
1121        // Absolute snapshots from concurrent acknowledgement batches may arrive after the
1122        // `Recovered` event that overtook them. A non-terminal snapshot cannot buy liveness here,
1123        // but it is not evidence of peer misconduct either. Ignore it exactly as the live path
1124        // ignores an older absolute counter. Only a terminal snapshot can touch the paid receipt.
1125        if progress.useful_shares != target || progress.recovered_polynomials != expected_polynomials {
1126            return Vec::new();
1127        }
1128
1129        let exhausted = {
1130            let tail = self
1131                .paid_recovery_tail
1132                .as_mut()
1133                .expect("caller matched the paid recovery tail");
1134            if progress.shares_seen <= tail.largest_shares_seen {
1135                return Vec::new();
1136            }
1137
1138            tail.largest_shares_seen = progress.shares_seen;
1139            tail.served_total_at_last_progress = served_total;
1140            tail.idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
1141            progress.shares_seen == max_shares_seen
1142        };
1143
1144        // Keep the tombstone's diagnostics monotone while it is retained, but do not require the
1145        // record: tombstone cleanup is intentionally shorter than a Session's possible FIFO drain.
1146        if let Some(idx) = self.find_ssa_idx(&progress.ssa_id) {
1147            self.ssas[idx].largest_useful_shares = target;
1148            self.ssas[idx].largest_shares_seen = progress.shares_seen;
1149            self.ssas[idx].recovered_polynomials = expected_polynomials;
1150            self.ssas[idx].served_total_at_last_progress = served_total;
1151        }
1152
1153        if exhausted {
1154            self.paid_recovery_tail = None;
1155        }
1156        vec![SessionPixAction::ProgressNotification]
1157    }

1159    /// Attributes `delta` useful shares to the front of the batch or behind it, and judges the ratio.
1160    ///
1161    /// The Entry serves a batch strictly in index order, so progress on anything but the front cycle is
1162    /// service the Exit cannot yet be paid for. See
1163    /// [`SupervisorConfig::max_off_front_share_fraction`] for the threat and for why this is measured
1164    /// as a fraction; the judgement waits for
1165    /// [`SupervisorConfig::min_share_order_sample`] shares of evidence.
1166    fn book_share_order_progress(
1167        &mut self,
1168        ssa_id: &SsaId<HoprPseudonym>,
1169        delta: u64,
1170    ) -> Option<SessionPixCloseReason> {
1171        let front = self.share_order_front?;
1172        if ssa_id.ssa_index() == front {
1173            self.front_useful = self.front_useful.saturating_add(delta);
1174        } else {
1175            self.off_front_useful = self.off_front_useful.saturating_add(delta);
1176        }
1177
1178        // Judged on every event, not only off-front ones: the sample can cross the floor on a share
1179        // that lands *on* the front cycle, and the ratio is already whatever it is by then. Waiting for
1180        // the next off-front share would only delay the same verdict.
1181        let total = self.front_useful.saturating_add(self.off_front_useful);
1182        if total < self.cfg.min_share_order_sample || total == 0 {
1183            return None;
1184        }
1185
1186        let fraction = self.off_front_useful as f64 / total as f64;
1187        if fraction > self.cfg.max_off_front_share_fraction {
1188            tracing::error!(
1189                %ssa_id, %front, fraction,
1190                limit = self.cfg.max_off_front_share_fraction,
1191                front_useful = self.front_useful,
1192                off_front_useful = self.off_front_useful,
1193                "entry is serving the batch out of order"
1194            );
1195            return Some(SessionPixCloseReason::BatchServedOutOfOrder);
1196        }
1197
1198        tracing::trace!(%ssa_id, %front, fraction, "progress behind the front of the batch");
1199        None
1200    }

1202    fn on_almost_recovered(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
1203        let idx = match self.find_ssa_idx(ssa_id) {
1204            Some(i) => i,
1205            None => return Vec::new(),
1206        };
1207
1208        // If already recovered (e.g. a concurrent batch delivered Recovered
1209        // before this AlmostRecovered), the next SSA is already requested —
1210        // no-op.
1211        if self.ssas[idx].is_terminal() {
1212            return Vec::new();
1213        }
1214
1215        // The successor gate. A cycle in the middle of a batch reaching its early-recovery threshold
1216        // says nothing about the batch as a whole: the ones behind it still have their full quota to
1217        // serve. Only the last cycle standing in for the batch may ask for a replacement.
1218        if !self.ssas[idx].is_batch_last {
1219            return Vec::new();
1220        }
1221
1222        let next_requested = self.ssas[idx].next_requested;
1223        let phase = self.ssas[idx].phase;
1224
1225        if next_requested {
1226            return Vec::new();
1227        }
1228
1229        match phase {
1230            SsaPhase::Recovering => {
1231                self.ssas[idx].next_requested = true;
1232                self.emit_request_next_ssa(now)
1233            }
1234            SsaPhase::AwaitingDeposit => {
1235                self.ssas[idx].next_requested = true;
1236                self.ssas[idx].next_request_pending_deposit = true;
1237                Vec::new()
1238            }
1239            // AwaitingCommitment or unknown phase: the next SSA request will be
1240            // triggered when the recovered transition fires (normal Recovering
1241            // path or deferred replay via recovered_pending). Set the deferred
1242            // flag so that commitment_verified can propagate it forward.
1243            _ => {
1244                self.ssas[idx].next_request_pending_deposit = true;
1245                self.ssas[idx].next_requested = true;
1246                Vec::new()
1247            }
1248        }
1249    }

1251    fn on_recovered(
1252        &mut self,
1253        ssa_id: &SsaId<HoprPseudonym>,
1254        now: Instant,
1255        served_total: u64,
1256    ) -> Vec<SessionPixAction> {
1257        let idx = match self.find_ssa_idx(ssa_id) {
1258            Some(i) => i,
1259            None => return Vec::new(),
1260        };
1261
1262        // Guard against duplicate recovery events (possible with concurrent
1263        // batch processing). If already recovered, this is a no-op.
1264        if self.ssas[idx].is_terminal() {
1265            return Vec::new();
1266        }
1267
1268        match self.ssas[idx].phase {
1269            SsaPhase::Recovering => {
1270                // Normal path — transition to tombstone directly.
1271                self.perform_recovered_transition(idx, now, served_total)
1272            }
1273            SsaPhase::AwaitingDeposit => {
1274                // Recovery completed before deposit confirmed. Record the
1275                // pending flag so that when the deposit arrives we replay
1276                // the transition immediately after entering Recovering.
1277                self.ssas[idx].recovered_pending = true;
1278                // Also ensure the next SSA request is deferred — the
1279                // eventual tombstone will emit it. Subject to the same successor gate as
1280                // `on_almost_recovered`: a non-last cycle of a batch never asks, deferred or not.
1281                if !self.ssas[idx].next_requested && self.ssas[idx].is_batch_last {
1282                    self.ssas[idx].next_request_pending_deposit = true;
1283                    self.ssas[idx].next_requested = true;
1284                }
1285                Vec::new()
1286            }
1287            SsaPhase::AwaitingCommitment => {
1288                // Recovery outpaced commitment verification. Same semantics
1289                // as AwaitingDeposit: set pending flag so the transition
1290                // replays once commitment and deposit have both arrived.
1291                self.ssas[idx].recovered_pending = true;
1292                Vec::new()
1293            }
1294            // Closing / already Recovered: handled by the terminal guard above.
1295            _ => Vec::new(),
1296        }
1297    }

1299    /// Perform the terminal tombstone transition for a fully-recovered SSA.
1300    /// Called from `on_recovered` (normal Recovering path) and replayed from
1301    /// `on_deposit_confirmed`/`on_commitment_verified` when `recovered_pending`
1302    /// was set earlier.
1303    fn perform_recovered_transition(&mut self, idx: usize, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
1304        let next_requested = self.ssas[idx].next_requested;
1305        let was_front = self.earliest_live_idx() == Some(idx);
1306
1307        // `Recovered` is a local cryptographic fact, so it supplies the lower bound even if its
1308        // final absolute Progress snapshot is delivered just afterwards by a concurrent worker.
1309        // Every later tail snapshot must keep these terminal fields exact.
1310        let target = self.dims.target_useful_shares();
1311        let max_shares_seen = self.dims.polys_per_ssa() as u64 * self.dims.emitted_shares_per_poly() as u64;
1312        self.ssas[idx].largest_useful_shares = target;
1313        self.ssas[idx].largest_shares_seen = self.ssas[idx].largest_shares_seen.max(target);
1314        self.ssas[idx].recovered_polynomials = self.dims.polys_per_ssa();
1315
1316        if was_front {
1317            // Replacing rather than accumulating is intentional. Only the immediate recovered
1318            // predecessor can occupy a FIFO immediately ahead of the live front; accepting an
1319            // arbitrary tombstone here would turn old-cycle replay into gate liveness.
1320            self.paid_recovery_tail =
1321                (self.ssas[idx].largest_shares_seen < max_shares_seen).then(|| PaidRecoveryTail {
1322                    ssa_id: self.ssas[idx].ssa_id,
1323                    largest_shares_seen: self.ssas[idx].largest_shares_seen,
1324                    served_total_at_last_progress: served_total,
1325                    // Recovery itself is fresh, cryptographic progress. Reusing the prior idle
1326                    // instant would let delayed/dropped progress snapshots make an honest tail
1327                    // expire immediately after recovery. Refresh this clock once; unlike the hard
1328                    // deadline below, it is the clock progress is meant to extend.
1329                    idle_deadline: now.checked_add(self.cfg.max_recovery_idle),
1330                    hard_deadline: self.ssas[idx]
1331                        .recovery_hard_deadline
1332                        .or_else(|| now.checked_add(self.cfg.max_recovery_time)),
1333                });
1334        }
1335
1336        // Transition to tombstone.
1337        self.ssas[idx].phase = SsaPhase::Recovered {
1338            tombstone_until: now
1339                .checked_add(self.cfg.tombstone_retention_window)
1340                .unwrap_or_else(|| now + Duration::from_secs(86400 * 365)),
1341        };
1342        self.ssas[idx].commitment_deadline = None;
1343        self.ssas[idx].deposit_deadline = None;
1344        self.ssas[idx].recovery_idle_deadline = None;
1345        self.ssas[idx].recovery_hard_deadline = None;
1346        self.ssas[idx].recovered_pending = false;
1347
1348        let mut actions = Vec::new();
1349        // A cycle that recovered with nothing buffered behind it leaves the Exit with nothing to fill
1350        // for until its successor reaches the front and arms its own clocks — which is at least a
1351        // deposit away. Stopping now rather than waiting for the successor keeps those SURBs for the
1352        // cycle that will actually be paid for; the next tick starts the ramp again from the
1353        // successor's true remainder.
1354        if was_front && self.paid_recovery_tail.is_none() {
1355            actions.extend(self.fill.stop().map(SessionPixAction::SetFillRate));
1356        }
1357        if was_front {
1358            // `Recovered` is stronger evidence than an ordinary progress snapshot. Emit this even
1359            // though the reconstructor normally sends the final snapshot first: progress delivery
1360            // is deliberately lossy under command-channel pressure, and without this reset a gate
1361            // already at its served-without-progress ceiling cannot spend the first buffered SURB
1362            // needed to produce tail liveness. If the successor is unfunded, `sync_service_gate`
1363            // prepends `WithholdService`; notifying afterwards changes only the dormant funded
1364            // watermark and cannot restore service.
1365            actions.push(SessionPixAction::ProgressNotification);
1366        }
1367        // Full recovery is the fallback trigger for the early signal, so it carries the same
1368        // successor gate: a non-last cycle of a batch is tombstoned without asking for anything.
1369        if !next_requested && self.ssas[idx].is_batch_last {
1370            self.ssas[idx].next_requested = true;
1371            actions.extend(self.emit_request_next_ssa(now));
1372        }
1373
1374        actions
1375    }

1377    /// Closes the Session on the first unverifiable-share report. There is no tolerance to configure.
1378    ///
1379    /// Shares are not checked on arrival, so a report is not "a bad share" — it means a whole
1380    /// polynomial's share set failed to open its commitment, which surfaces only once `threshold` of
1381    /// them have been interpolated. `SsaPart::add_share` then marks that part failed and releases its
1382    /// shares, and nothing ever clears the flag; since the SSA is the sum of *every* polynomial's
1383    /// constant term, the cycle can no longer be reconstructed by any means and will never pay.
1384    ///
1385    /// So serving on is not tolerance, it is donation: every packet past this point is unpaid with no
1386    /// recovery to preserve at the end of it. That holds however the failure arose, which is why this
1387    /// is not a knob — a false positive from a verification bug leaves the part just as permanently
1388    /// failed, so a tolerance would buy unpaid service rather than the cycle it was raised to save.
1389    /// Closing here caps the exposure at the `threshold` packets already served rather than a multiple
1390    /// of it.
1391    ///
1392    /// `observed_total` is the reconstructor's cross-peer aggregate, and with no limit to compare it
1393    /// against it is logged rather than accumulated: it says how many polynomials the batch that
1394    /// triggered this took down, which is the difference between one bad relayer and a peer sending
1395    /// garbage wholesale.
1396    ///
1397    /// Note this closes the Session directly rather than through
1398    /// [`close_ssa_and_collect`](Self::close_ssa_and_collect), so a batch's surviving siblings do not
1399    /// keep it alive: unlike a lost deposit, this fault is evidence about the peer rather than about
1400    /// one cycle.
1401    fn on_unverifiable_shares(
1402        &mut self,
1403        ssa_id: &SsaId<HoprPseudonym>,
1404        observed_total: u64,
1405        _now: Instant,
1406    ) -> Vec<SessionPixAction> {
1407        let idx = match self.find_ssa_idx(ssa_id) {
1408            Some(i) => i,
1409            None => return Vec::new(),
1410        };
1411
1412        // Absorb late reports on a tombstoned or already-closing SSA: there is no service left to
1413        // stop, and the cycle they condemn has already left. Concurrent ack batches make these
1414        // ordinary rather than exceptional.
1415        if self.ssas[idx].is_terminal() {
1416            return Vec::new();
1417        }
1418
1419        tracing::warn!(
1420            %ssa_id,
1421            observed_total,
1422            phase = ?self.ssas[idx].phase,
1423            "closing PIX session: a polynomial's share set failed to open its commitment"
1424        );
1425
1426        self.with_fill_stopped(vec![SessionPixAction::Close(SessionPixCloseReason::UnverifiableShares)])
1427    }

1433    /// Close the SSA at `idx` and return close actions.
1434    fn close_ssa_and_collect(&mut self, idx: usize, reason: SessionPixCloseReason) -> Vec<SessionPixAction> {
1435        if matches!(self.ssas[idx].phase, SsaPhase::Closing) {
1436            return Vec::new();
1437        }
1438
1439        // Track the first close reason so it isn't lost when multiple SSAs close.
1440        self.first_failure_reason.get_or_insert(reason);
1441        let close_reason = self.first_failure_reason.unwrap_or(reason);
1442
1443        // Charged before the branches below, because the whole point is a count that outlives the
1444        // record being retired. Every path into this function is a cycle lost without recovering —
1445        // a recovered cycle leaves through tombstone expiry in `handle_deadline`, not through here.
1446        self.failed_cycles += 1;
1447
1448        // Warn-level diagnostic with full SSA state before closing.
1449        let ssa = &self.ssas[idx];
1450        tracing::warn!(
1451            ssa_id = %ssa.ssa_id,
1452            ?reason,
1453            phase = ?ssa.phase,
1454            largest_useful_shares = ssa.largest_useful_shares,
1455            target_useful_shares = ssa.target_useful_shares,
1456            recovered_polynomials = ssa.recovered_polynomials,
1457            served_total_at_last_progress = ssa.served_total_at_last_progress,
1458            ?ssa.commitment_deadline,
1459            ?ssa.deposit_deadline,
1460            ?ssa.recovery_idle_deadline,
1461            ?ssa.recovery_hard_deadline,
1462            "closing PIX SSA"
1463        );
1464
1465        self.ssas[idx].phase = SsaPhase::Closing;
1466
1467        if self.ssas.len() == 1 {
1468            self.closed = true;
1469            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1470        }
1471
1472        // A batch has siblings to fall back on, which is what makes retiring one member survivable —
1473        // but only up to a point. Past the limit the Session goes, with the *first* failure's reason
1474        // rather than this one's, so the report names the cause instead of the last symptom.
1475        //
1476        // Strictly greater: the field is how many failures are *tolerated*, so the shipping value of
1477        // one keeps the existing "retire the member and carry on" behaviour for a batch's first loss
1478        // — including handing on the successor gate below — and closes on the second.
1479        if self.failed_cycles > self.cfg.max_failed_cycles.max(1) {
1480            tracing::warn!(
1481                failed_cycles = self.failed_cycles,
1482                max_failed_cycles = self.cfg.max_failed_cycles,
1483                ?close_reason,
1484                "closing PIX session: too many cycles lost without recovering"
1485            );
1486            self.closed = true;
1487            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1488        }
1489
1490        // Clear deadlines on this SSA.
1491        self.ssas[idx].commitment_deadline = None;
1492        self.ssas[idx].deposit_deadline = None;
1493        self.ssas[idx].recovery_idle_deadline = None;
1494        self.ssas[idx].recovery_hard_deadline = None;
1495
1496        // If all SSAs are terminal, close the session.
1497        if self
1498            .ssas
1499            .iter()
1500            .all(|s| matches!(s.phase, SsaPhase::Closing | SsaPhase::Recovered { .. }))
1501        {
1502            self.closed = true;
1503            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1504        }
1505
1506        // Remove this closing SSA and emit RetireSsa so the reconstructor
1507        // releases its builder/verifier/counter state mid-session.
1508        // Record the retired index to prevent stale SsaRequestSent events
1509        // from resurrecting it.
1510        let retired = self.ssas[idx].ssa_id;
1511        self.note_retired_index(retired.ssa_index());
1512        let carried_successor_gate = self.ssas[idx].is_batch_last && !self.ssas[idx].next_requested;
1513        self.ssas.remove(idx);
1514
1515        // Retiring the cycle that carried the successor gate would strand the batch: its surviving
1516        // siblings are barred from asking, so nothing would ever request a replacement and the
1517        // Session would die on a recovery timeout that names the timer rather than the cause. Hand
1518        // the gate to the newest cycle still standing, which is now the last of the batch that will
1519        // actually serve its quota — so the request still cannot come early. `next_requested` guards
1520        // it: a gate already spent must not be handed on and fire a second time.
1521        if carried_successor_gate
1522            && let Some(newest) = self
1523                .ssas
1524                .iter_mut()
1525                .filter(|s| !s.is_terminal())
1526                .max_by_key(|s| s.ssa_id.ssa_index())
1527        {
1528            tracing::debug!(
1529                %retired, promoted = %newest.ssa_id,
1530                "successor gate handed on from a retired cycle"
1531            );
1532            newest.is_batch_last = true;
1533        }
1534        vec![SessionPixAction::RetireSsa(retired)]
1535    }

1537    /// Allocates the next batch of SSA indices and asks for all of them in one action.
1538    ///
1539    /// The batch size is [`SupervisorConfig::ssas_per_request`], clamped rather than trusted since a
1540    /// supervisor can be built from a config that never went through `validate_pix_supervision`.
1541    ///
1542    /// Exactly one cycle of the batch — the last one allocated — carries
1543    /// [`PerSsaState::is_batch_last`], and only that cycle may ask for the successor batch. Without
1544    /// it every member would ask for one of its own, since the request flags are per-cycle; see that
1545    /// field for why the last rather than the first.
1546    ///
1547    /// Before allocating, this enforces both halves of the admission reservation: no more than
1548    /// [`crate::MAX_OVERLAPPING_BATCHES`] live generations and no more than that many full batches'
1549    /// live cycles. A request earned while the reservation is full is retried after a generation
1550    /// releases its reconstructor state.
1551    ///
1552    /// What batching does change is the exposure *within* a batch: every cycle in it is unfunded at
1553    /// once, so the ceiling is `ssas_per_request` SSA quotas rather than one. That is the trade the
1554    /// knob exists to make, and it is why both deadlines are scaled by the same factor.
1555    fn emit_request_next_ssa(&mut self, now: Instant) -> Vec<SessionPixAction> {
1556        let batch = self.cfg.ssas_per_request.clamp(1, crate::MAX_SSA_BATCH_SIZE);
1557        let live_cycles = self.live_cycle_count();
1558        let live_batches = self.live_batch_count();
1559        if live_batches >= crate::MAX_OVERLAPPING_BATCHES as usize
1560            || live_cycles.saturating_add(batch) > self.reserved_cycle_slots()
1561        {
1562            self.successor_request_deferred = true;
1563            tracing::debug!(
1564                live_cycles,
1565                live_batches,
1566                requested = batch,
1567                reserved = self.reserved_cycle_slots(),
1568                "deferring successor SSA batch until an older batch is released"
1569            );
1570            return Vec::new();
1571        }
1572
1573        self.successor_request_deferred = false;
1574        let batch_id = self.next_ssa_index;
1575        let mut ssa_ids = Vec::with_capacity(batch);
1576
1577        for _ in 0..batch {
1578            let index = self.next_ssa_index;
1579
1580            let Ok(ssa_index) = SsaIndex::try_from(index) else {
1581                break;
1582            };
1583            let Some(next) = index.checked_add(1) else {
1584                // Index space exhausted mid-batch. Stop here rather than wrapping: a reused index
1585                // would collide with a live cycle. Whatever was allocated still goes out.
1586                break;
1587            };
1588            self.next_ssa_index = next;
1589
1590            let ssa_id = SsaId::new(self.pseudonym, ssa_index);
1591
1592            // Register each SSA now rather than on `SsaRequestSent`. Carrying out the action registers
1593            // an Exit commitment, which is what makes shares for that SSA processable — so
1594            // observations about it can reach us before the confirmation that we asked for it does.
1595            // Every handler ignores an SSA it has no record of, and for `UnverifiableShares` that
1596            // would mean failing open on exactly the signal that must fail closed.
1597            //
1598            // No deadline yet: the request has not gone out, so there is nothing to be late for. That
1599            // is what `SsaRequestSent` adds, per index.
1600            self.ssas.push(PerSsaState::new(
1601                ssa_id,
1602                batch_id,
1603                self.dims.target_useful_shares(),
1604                now,
1605            ));
1606            ssa_ids.push(ssa_id);
1607        }
1608
1609        // Nothing allocated means the index space is spent, and no further cycle can ever be funded.
1610        if ssa_ids.is_empty() {
1611            self.closed = true;
1612            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
1613        }
1614
1615        // The last cycle *actually* allocated carries the successor gate, which is what makes a
1616        // batch truncated by index exhaustion still able to ask for the next one.
1617        self.ssas
1618            .last_mut()
1619            .expect("a non-empty batch has just been pushed")
1620            .is_batch_last = true;
1621
1622        vec![SessionPixAction::RequestSsa {
1623            ssa_ids,
1624            params: self.dims,
1625        }]
1626    }
1627
1628    fn find_ssa_idx(&self, ssa_id: &SsaId<HoprPseudonym>) -> Option<usize> {
1629        self.ssas.iter().position(|s| s.ssa_id == *ssa_id)
1630    }
1631}

1635// ---------------------------------------------------------------------------

1637impl PerSsaState {
1638    /// Check which deadline expired, if any.
1639    fn check_deadlines(&self, now: Instant) -> Option<SessionPixCloseReason> {
1640        match self.phase {
1641            SsaPhase::AwaitingCommitment => self
1642                .commitment_deadline
1643                .filter(|d| now >= *d)
1644                .map(|_| SessionPixCloseReason::CommitmentTimeout),
1645            SsaPhase::AwaitingDeposit => self
1646                .deposit_deadline
1647                .filter(|d| now >= *d)
1648                .map(|_| SessionPixCloseReason::DepositTimeout),
1649            SsaPhase::Recovering => {
1650                // Hard deadline is immutable.
1651                if let Some(d) = self.recovery_hard_deadline
1652                    && now >= d
1653                {
1654                    return Some(SessionPixCloseReason::RecoveryDeadline);
1655                }
1656                // Idle deadline — service gating happens in handle_deadline.
1657                if let Some(d) = self.recovery_idle_deadline
1658                    && now >= d
1659                {
1660                    return Some(SessionPixCloseReason::RecoveryIdle);
1661                }
1662                None
1663            }
1664            SsaPhase::Recovered { .. } | SsaPhase::Closing => None,
1665        }
1666    }
1667}

4994    /// The whole point: an idle funded cycle is carried to completion inside its own deadline.
4995    ///
4996    /// The Session sends not one application packet — `served_total` never moves — so every share
4997    /// that comes back was earned by a fill keep-alive. The simulation closes the loop the deployment
4998    /// does: whatever fill emits in a second arrives as liveness in the next one, and the planner
4999    /// re-derives the rate from the remainder that leaves.
5000    ///
5001    /// Two properties, and the second is why `max_rate` is a configuration item rather than a
5002    /// constant: the cycle must finish *before the aim point*, and it must do so without the planner
5003    /// ever asking for more than the ceiling it was given.
5004    #[test]
5005    fn an_idle_funded_cycle_is_filled_to_completion_before_its_aim_point() {
5006        let cfg = default_cfg();
5007        let t0 = Instant::now();
5008        let (mut sup, p) = funded_at_front(cfg.clone(), fill_dims(), t0);
5009        let id = ssa_id(p, 1);
5010
5011        let cycle = max_cycle_shares(&sup);
5012        let aim_point = cfg.max_recovery_time.mul_f64(cfg.fill.finish_fraction);
5013
5014        let mut rate = FillRate::ZERO;
5015        let mut sent = 0.0_f64;
5016        let mut completed_at = None;
5017        for second in 1..=aim_point.as_secs() {
5018            let now = t0 + Duration::from_secs(second);
5019            if let Some(planned) = planned_rate(&sup.handle_timers(now, 0)) {
5020                assert!(
5021                    planned.as_packets_per_sec() <= cfg.fill.max_rate as f64,
5022                    "fill asked for {} packets/s against a ceiling of {}",
5023                    planned.as_packets_per_sec(),
5024                    cfg.fill.max_rate
5025                );
5026                rate = planned;
5027            }
5028
5029            // A lossless return path, so the packets sent in this second are the shares seen in it.
5030            sent += rate.as_packets_per_sec();
5031            if sent as u64 >= cycle {
5032                completed_at = Some(second);
5033                break;
5034            }
5035            feed_seen(&mut sup, id, sent as u64, now, 0);
5036        }
5037
5038        let completed_at = completed_at.unwrap_or_else(|| {
5039            panic!("an idle cycle of {cycle} packets was not filled to completion within {aim_point:?}")
5040        });
5041        assert!(
5042            Duration::from_secs(completed_at) < aim_point,
5043            "the cycle completed at {completed_at}s, which is not inside the {aim_point:?} aim point"
5044        );
