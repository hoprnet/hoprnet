23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24enum SsaPhase {
25    /// Request has been sent; waiting for complete verifiable commitment.
26    AwaitingCommitment,
27    /// Commitment is verifiable; waiting for a sufficient deposit.
28    AwaitingDeposit,
29    /// Deposit confirmed; recovering shares.
30    Recovering,
31    /// SSA fully recovered; diagnostic tombstone until `tombstone_until`.
32    /// Any paid FIFO remainder lives separately in [`PaidRecoveryTail`].
33    Recovered { tombstone_until: Instant },
34    /// Phase that will produce a close action on next deadline check.
35    Closing,
36}
37
38/// Wei in one µHOPR, the unit `hopr_pix_deposits_*_uhopr_total` counts.
39const WEI_PER_UHOPR: u128 = 1_000_000_000_000;
40
41/// A deposit in µHOPR, saturating rather than wrapping.
42///
43/// Integer because the counters it feeds are, and µHOPR because wei overflows a `u64` at 18 HOPR
44/// while whole HOPR would round every realistic deposit to nothing.
45fn deposit_uhopr(amount: HoprBalance) -> u64 {
46    let wei = amount.amount();
47    let wei = if wei.bits() > 128 { u128::MAX } else { wei.low_u128() };
48    (wei / WEI_PER_UHOPR).min(u64::MAX as u128) as u64
49}

55/// Internal state for one supervised SSA.
56struct PerSsaState {
57    ssa_id: SsaId<HoprPseudonym>,
58    /// First SSA index allocated in this request batch.
59    batch_id: u32,
60    phase: SsaPhase,
61
62    // Deadlines (None means not set for this phase).
63    commitment_deadline: Option<Instant>,
64    /// When to re-request the parts of this commitment that have not arrived, or `None` while there
65    /// is nothing to ask for.
66    ///
67    /// Set only once *some* of the commitment has arrived, and pushed out by every further part, so
68    /// it measures silence in the delivery rather than elapsed time. A cycle for which nothing was
69    /// ever delivered leaves it unset and dies on `commitment_deadline`, which is all the Exit can
70    /// do about one: it has no scope to name.
71    recommit_deadline: Option<Instant>,
72    /// Re-requests already made for this cycle, capped at
73    /// [`MAX_COMMITMENT_RETRANSMISSIONS`](hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS).
74    ///
75    /// The cap is a count and not a duration on purpose. `commitment_deadline` bounds the asks in
76    /// *time*, which is what protects the Entry, but the packets they cost are this node's own: at a
77    /// misconfigured millisecond interval inside a 20-second deadline the time bound alone permits
78    /// twenty thousand of them. Matching the figure the Entry will answer means none of the asks is
79    /// wasted either.
80    recommit_attempts: u8,
81    deposit_deadline: Option<Instant>,
82    recovery_idle_deadline: Option<Instant>,
83    recovery_hard_deadline: Option<Instant>,
84
85    /// What the Entry deposited for this cycle, in µHOPR, or zero until the deposit confirms.
86    deposit_uhopr: u64,
87    /// When [`phase`](Self::phase) was entered, for `hopr_pix_cycle_phase_seconds`.
88    phase_entered_at: Instant,
89
90    // Progress tracking.
91    largest_useful_shares: u64,
92    /// Largest `shares_seen` observed for this cycle — the liveness counter, which moves for surplus
93    /// shares too. Also the staleness discriminator for snapshots, since it moves for every share
94    /// while `largest_useful_shares` does not.
95    largest_shares_seen: u64,
96    target_useful_shares: u64,
97    recovered_polynomials: u16,
98
99    // Overlap / deferred-request state.
100    next_request_pending_deposit: bool,
101    next_requested: bool,
102    /// Set on the last cycle allocated by [`SessionPixSupervisor::emit_request_next_ssa`], and only
103    /// on that one.
104    ///
105    /// The successor gate: a batch asks for its replacement once, when its *last* cycle is nearly
106    /// recovered. Every other flag here is per-cycle, so without this one each member of a batch
107    /// would answer its own `AlmostRecovered` with a whole batch of its own — `ssas_per_request`
108    /// batches per batch, compounding, each cycle a separate on-chain deposit for the Entry.
109    ///
110    /// The last cycle rather than the first because the batch is served in index order: gating on
111    /// the first would ask for the successor a whole batch too early, which is the exposure
112    /// `ssas_per_request` deliberately bounds.
113    ///
114    /// If the two-generation admission bound defers that request, `next_requested` remains set and
115    /// the supervisor-level `successor_request_deferred` flag owns the retry; the gate must not be
116    /// handed to another cycle and spend the request twice.
117    is_batch_last: bool,
118    /// Set when `Recovered` arrives before deposit confirms (i.e. during
119    /// `AwaitingCommitment` or `AwaitingDeposit`). Once deposit is confirmed
120    /// and the SSA enters `Recovering`, this flag triggers an immediate
121    /// tombstone transition — replaying the deferred `Recovered` event.
122    recovered_pending: bool,
123
124    // Service gating.
125    served_total_at_last_progress: u64,
126    /// The Session's served count when this cycle reached the front of the batch, or `None` while
127    /// it is still queued behind one.
128    ///
129    /// Subtracted from the served count at finalization to give the cycle's own egress. Baselined
130    /// at the front rather than at the request, because a queued cycle is not being served: the
131    /// Entry's emission window is clamped to one cycle, so charging a batch member for everything
132    /// its predecessors consumed would report the last cycle of a batch as having cost the whole
133    /// batch. `None` on a cycle that never reached the front is therefore the honest answer, and it
134    /// is reported as zero egress rather than as the Session's whole history.
135    served_total_at_front: Option<u64>,
136}

139    fn new(ssa_id: SsaId<HoprPseudonym>, batch_id: u32, target_useful_shares: u64, now: Instant) -> Self {
140        Self {
141            ssa_id,
142            batch_id,
143            phase: SsaPhase::AwaitingCommitment,
144            phase_entered_at: now,
145            commitment_deadline: None,
146            recommit_deadline: None,
147            recommit_attempts: 0,
148            deposit_uhopr: 0,
149            deposit_deadline: None,
150            recovery_idle_deadline: None,
151            recovery_hard_deadline: None,
152            largest_useful_shares: 0,
153            largest_shares_seen: 0,
154            target_useful_shares,
155            recovered_polynomials: 0,
156            next_request_pending_deposit: false,
157            next_requested: false,
158            is_batch_last: false,
159            recovered_pending: false,
160            served_total_at_last_progress: 0,
161            served_total_at_front: None,
162        }
163    }
164
165    /// Moves to `phase`, returning how long the outgoing one lasted in milliseconds.
166    ///
167    /// `None` for an outgoing `Recovered` or `Closing`, which have no [`PixCyclePhase`] label — both
168    /// are retained state rather than live state, so neither is measured.
169    fn enter_phase(&mut self, phase: SsaPhase, now: Instant) -> Option<(PixCyclePhase, u64)> {
170        let left = match self.phase {
171            SsaPhase::AwaitingCommitment => Some(PixCyclePhase::AwaitingCommitment),
172            SsaPhase::AwaitingDeposit => Some(PixCyclePhase::AwaitingDeposit),
173            SsaPhase::Recovering => Some(PixCyclePhase::Recovering),
174            SsaPhase::Recovered { .. } | SsaPhase::Closing => None,
175        };
176        let elapsed = now.saturating_duration_since(self.phase_entered_at).as_millis() as u64;
177        self.phase = phase;
178        self.phase_entered_at = now;
179        left.map(|phase| (phase, elapsed))
180    }
181
182    /// This cycle's coverage, for the histograms observed once when it finalizes.
183    fn telemetry_summary(&self, outcome: PixCycleOutcome, served_total: u64) -> PixCycleSummary {
184        PixCycleSummary {
185            outcome,
186            // Saturating rather than assuming monotonicity: the baseline is taken from a
187            // `served_total` sample passed in by the worker, and a cycle retired on a deadline
188            // sweep can be summarized from an older sample than the one that baselined it.
189            egress_packets: self
190                .served_total_at_front
191                .map(|base| served_total.saturating_sub(base))
192                .unwrap_or(0),
193            accepted_shares: self.largest_shares_seen,
194            useful_shares: self.largest_useful_shares,
195            target_useful_shares: self.target_useful_shares,
196            deposit_uhopr: self.deposit_uhopr,
197        }
198    }
199
200    /// True if this SSA is past the recovery phase.
201    fn is_terminal(&self) -> bool {
202        matches!(self.phase, SsaPhase::Recovered { .. } | SsaPhase::Closing)
203    }
204}

206/// Bounded handoff for SURBs minted against the cycle that has just recovered.
207///
208/// Cryptographic recovery and transport-buffer exhaustion are different instants. The former frees
209/// the reconstructor's heavy state; this small receipt keeps the paid service front on the recovered
210/// predecessor until either its negotiated tail is exhausted or any progress proves the successor
211/// has reached the head of the FIFO. The latter need not be funded: observing an unfunded successor
212/// is precisely when the predecessor's receipt must stop authorizing service.
213struct PaidRecoveryTail {
214    ssa_id: SsaId<HoprPseudonym>,
215    /// When the tail took the paid front, for `hopr_pix_cycle_phase_seconds{phase="paid_tail"}`.
216    entered_at: Instant,
217    largest_shares_seen: u64,
218    served_total_at_last_progress: u64,
219    idle_deadline: Option<Instant>,
220    hard_deadline: Option<Instant>,
221}

227/// Deterministic core of the PIX session supervisor.
228pub struct SessionPixSupervisor {
229    pub(crate) cfg: SupervisorConfig,
230    pub(crate) dims: PixParams,
231    pub(crate) pseudonym: HoprPseudonym,
232    pub(crate) closed: bool,
233    /// The Session is gone and only its funded front cycle is still being worked for.
234    ///
235    /// Set by [`on_session_closed`](Self::on_session_closed) and never cleared: a drain ends by
236    /// closing, one way or another. While it is set the supervisor asks for no successor, plans fill
237    /// at the ceiling, and stops service-gating the recovery idle deadline — see "Draining a closed
238    /// Session" in this module's documentation for why each of those follows from the Session having
239    /// left rather than being separate policy.
240    pub(crate) draining: bool,
241    next_ssa_index: u32,
242    /// The cycle the two share-order counters below are measured against — the earliest unrecovered
243    /// cycle, i.e. the one the Entry should currently be serving. `None` before the first batch exists.
244    share_order_front: Option<SsaIndex>,
245    /// Useful shares booked on [`Self::share_order_front`] since it took the front.
246    front_useful: u64,
247    /// Useful shares booked on any *other* live cycle over that same span.
248    ///
249    /// `off_front_useful / (front_useful + off_front_useful)` is the share-order ratio — see
250    /// [`SupervisorConfig::max_off_front_share_fraction`] for what it detects and why it is a fraction
251    /// rather than a count. Both counters reset when the front changes, which is what keeps the window
252    /// bounded to a single cycle's service.
253    off_front_useful: u64,
254    /// Whether the service gate currently follows a funded front cycle.
255    service_open: bool,
256    /// Front cycle for which the current gate mode was emitted.
257    ///
258    /// Tracking the identity as well as the mode lets a funded-to-funded handoff rebaseline the
259    /// served-without-progress ceiling for the newly promoted cycle.
260    service_front: Option<SsaIndex>,
261    /// A front cycle became paid and terminal in one event, so an unfunded successor still earns a
262    /// freshly restored predeposit allowance even though the gate never entered funded mode.
263    paid_front_handoff: bool,
264    /// The sole recovered predecessor authorized to report buffered liveness.
265    ///
266    /// This never becomes a set: accepting arbitrary recovered tombstones would let the Entry replay
267    /// old cycle tags to keep service open. Replacement on each in-order recovery is the supervisor's
268    /// independent bound behind the reconstructor's per-polynomial surplus counters.
269    paid_recovery_tail: Option<PaidRecoveryTail>,
270    /// A successor batch was earned but could not yet fit the two-generation reservation.
271    successor_request_deferred: bool,
272    /// Ordered SSAs (oldest first, newest last), including short-lived recovered tombstones.
273    ssas: Vec<PerSsaState>,
274    /// Tracks the first failure reason when multiple SSAs fail, so the
275    /// earliest cause is used for the final `Close` action rather than the last.
276    first_failure_reason: Option<SessionPixCloseReason>,
277    /// Cycles this Session has lost without recovering them, over its whole life.
278    ///
279    /// Cumulative, and that is the whole point: retiring a failed member leaves no trace on its
280    /// siblings, so without a counter that survives the retirement an Entry can lose one cycle per
281    /// batch indefinitely while a single funded sibling holds the Session open. Bounded by
282    /// [`SupervisorConfig::max_failed_cycles`].
283    failed_cycles: usize,
284    /// Plans the rate at which this Exit originates its own fill keep-alives.
285    ///
286    /// Kept here rather than in the worker because the rate is a function of the lifecycle state
287    /// this struct owns — which cycle is at the paid front and how far it has got — and because
288    /// every close path has to be able to stop it in the same action vector that carries the close.
289    fill: FillPlanner,
290    /// Greatest SSA index that has been retired (closed and removed).
291    ///
292    /// Prevents a stale `SsaRequestSent` from resurrecting a closed SSA. A high-watermark rather
293    /// than the set of every retired index: the set grew for the whole life of the Session and was
294    /// scanned linearly on every request, one entry per cycle, without bound.
295    ///
296    /// A watermark alone would be wrong if it were consulted first — retirement is not monotone,
297    /// since a later batch member can lose its deposit while an earlier one is still recovering, and
298    /// "at or below the greatest retired index" would then condemn that live earlier cycle. The
299    /// guard is therefore placed *after* the live-record lookup, so a cycle that still exists always
300    /// wins and the watermark only ever answers for indices that have none.
301    highest_retired_ssa_index: Option<SsaIndex>,
302    /// Cycle lifecycle transitions since the worker last drained this.
303    ///
304    /// Latched rather than emitted because this type is a pure function of its inputs and a metric
305    /// is not — the same arrangement as [`FillPlanner`]'s stall, which is latched here and drained
306    /// in the worker. Always compiled, including without the `telemetry` feature, so the default
307    /// test configuration exercises the counting rather than only the feature-enabled one.
308    telemetry: PixTurnEvents,
309}

347    /// Handle a lifecycle event.
348    pub fn handle_event(&mut self, ev: &SessionPixEvent, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
349        if self.closed {
350            return Vec::new();
351        }
352
353        let actions = match ev {
354            SessionPixEvent::SsaRequestSent(ssa_id) => self.on_ssa_request_sent(ssa_id, now),
355            SessionPixEvent::CommitmentProgress(ssa_id) => self.on_commitment_progress(ssa_id, now),
356            SessionPixEvent::CommitmentVerified(ssa_id) => self.on_commitment_verified(ssa_id, now),
357            SessionPixEvent::DepositConfirmed { ssa_id, amount } => {
358                self.on_deposit_confirmed(ssa_id, *amount, now, served_total)
359            }
360            SessionPixEvent::DepositObserverClosed(ssa_id) => {
361                self.on_deposit_observer_closed(ssa_id, now, served_total)
362            }
363            SessionPixEvent::RecoveryProgress(progress) => self.on_recovery_progress(progress, now, served_total),
364            SessionPixEvent::AlmostRecovered(ssa_id) => self.on_almost_recovered(ssa_id, now),
365            SessionPixEvent::Recovered(ssa_id) => self.on_recovered(ssa_id, now, served_total),
366            SessionPixEvent::UnverifiableShares {
367                ssa_id,
368                observed_total,
369                peer,
370            } => self.on_unverifiable_shares(ssa_id, *observed_total, peer, now),
371            SessionPixEvent::SessionClosed { drainable_surbs } => self.on_session_closed(*drainable_surbs, now),
372        };
373
374        let mut lifecycle_actions = actions;
375        lifecycle_actions.extend(self.retry_deferred_successor_request(now));
376        self.consume_paid_tail_at_observed_boundary(now);
377        self.arm_recovery_clocks_for_earliest(now, served_total);
378        let mut actions = if lifecycle_actions
379            .iter()
380            .any(|action| matches!(action, SessionPixAction::Close(_)))
381        {
382            // A closing Session has no cycle left to fill for, and the fill stream outlives this
383            // state machine by however long it takes the driver to tear the Session down. Silencing
384            // it here rather than only there is what keeps a close from being followed by a burst of
385            // keep-alives for a cycle that no longer exists.
386            self.fill
387                .stop()
388                .map(SessionPixAction::SetFillRate)
389                .into_iter()
390                .collect()
391        } else {
392            self.sync_service_gate()
393        };
394        actions.extend(lifecycle_actions);
395        self.refresh_share_order_front();
396        actions
397    }

399    /// Raises the retired-index watermark.
400    ///
401    /// Takes the maximum rather than assuming the caller retires in order, because it does not: a
402    /// later batch member losing its deposit is retired before an earlier one that is still
403    /// recovering.
404    fn note_retired_index(&mut self, index: SsaIndex) {
405        self.highest_retired_ssa_index = Some(match self.highest_retired_ssa_index {
406            Some(highest) if highest >= index => highest,
407            _ => index,
408        });
409    }
410
411    /// Index of the earliest cycle that has not finished recovering — the one the Entry should be
412    /// serving, given that emission is clamped to a single cycle.
413    fn earliest_live_idx(&self) -> Option<usize> {
414        self.ssas
415            .iter()
416            .enumerate()
417            .filter(|(_, s)| !s.is_terminal())
418            .min_by_key(|(_, s)| s.ssa_id.ssa_index())
419            .map(|(i, _)| i)
420    }
421
422    /// Consumes the predecessor receipt once the current live front has reported any share.
423    ///
424    /// This check is centralized rather than tied only to `RecoveryProgress`: a previously queued
425    /// cycle can become the front because a sibling is retired or because its deposit arrives after
426    /// its first snapshot. In every case the observation proves the transport crossed the cycle
427    /// boundary. Keeping the old funded receipt beyond it would let an unfunded successor spend the
428    /// predecessor's remaining allowance.
429    /// Retires the paid tail, latching how long it held the front.
430    fn release_paid_tail(&mut self, now: Instant) {
431        if let Some(tail) = self.paid_recovery_tail.take() {
432            let elapsed = now.saturating_duration_since(tail.entered_at).as_millis() as u64;
433            self.telemetry.phase_durations.push((PixCyclePhase::PaidTail, elapsed));
434        }
435    }
436
437    fn consume_paid_tail_at_observed_boundary(&mut self, now: Instant) {
438        let Some(front_idx) = self.earliest_live_idx() else {
439            return;
440        };
441        let front_id = self.ssas[front_idx].ssa_id;
442        if self.ssas[front_idx].largest_shares_seen == 0 {
443            return;
444        }
445
446        if let Some(predecessor) = self
447            .paid_recovery_tail
448            .as_ref()
449            .map(|tail| tail.ssa_id)
450            .filter(|predecessor| predecessor.ssa_index() < front_id.ssa_index())
451        {
452            tracing::debug!(
453                %predecessor,
454                successor = %front_id,
455                "successor progress consumed the paid FIFO-tail handoff"
456            );
457            self.release_paid_tail(now);
458        }
459    }

461    /// Keeps the service gate aligned with the funding state of the earliest live cycle.
462    ///
463    /// A funded-to-unfunded handoff restores the configured predeposit allowance. A
464    /// funded-to-funded handoff emits `ReleaseService` too: the mode does not change, but the new
465    /// front must receive its own served-without-progress ceiling rather than inheriting the tail of
466    /// its predecessor's. An unfunded cycle that fails does not earn another allowance.
467    fn sync_service_gate(&mut self) -> Vec<SessionPixAction> {
468        if self.closed {
469            return Vec::new();
470        }
471
472        let front = self.earliest_live_idx();
473        let front_id = self
474            .paid_recovery_tail
475            .as_ref()
476            .map(|tail| tail.ssa_id.ssa_index())
477            .or_else(|| front.map(|idx| self.ssas[idx].ssa_id.ssa_index()));
478        let front_changed = front_id != self.service_front;
479        let front_funded =
480            self.paid_recovery_tail.is_some() || front.is_some_and(|idx| self.ssas[idx].phase == SsaPhase::Recovering);
481        let paid_front_handoff = std::mem::take(&mut self.paid_front_handoff);
482        self.service_front = front_id;
483
484        match (front_funded, self.service_open, front_changed) {
485            (true, false, _) | (true, true, true) => {
486                self.service_open = true;
487                tracing::debug!(?front_id, "aligning service gate with funded front cycle");
488                vec![SessionPixAction::ReleaseService]
489            }
490            (false, true, _) => {
491                self.service_open = false;
492                tracing::debug!(?front_id, "withholding service for unfunded front cycle");
493                vec![SessionPixAction::WithholdService]
494            }
495            (false, false, _) if paid_front_handoff => {
496                tracing::debug!(?front_id, "restoring predeposit service after paid front handoff");
497                vec![SessionPixAction::WithholdService]
498            }
499            _ => Vec::new(),
500        }
501    }
502
503    /// This Session's current contribution to the node-level live-set gauges.
504    ///
505    /// Recomputed from `ssas` and `paid_recovery_tail` rather than maintained incrementally, which
506    /// is what makes the gauges drain on every terminal path without any of those paths having to
507    /// know they exist. A Session holds at most `ssas_per_request × MAX_OVERLAPPING_BATCHES` cycles,
508    /// so the scan is over a handful of entries once per turn.
509    ///
510    /// `predeposit_exposure` comes from the gate, which the caller reads — this type deliberately
511    /// knows nothing about the gate's counters.
512    pub fn telemetry_snapshot(&self, predeposit_exposure: u64) -> PixSessionSnapshot {
513        let mut snapshot = PixSessionSnapshot {
514            // The mode the gate is actually in, which is what an operator reading this gauge next to
515            // `hopr_pix_predeposit_exposure_packets` is asking about. A paid recovered predecessor
516            // still holding the front counts as funded, because service continues to be authorized.
517            gate_mode: if self.service_open {
518                PixGateMode::Funded
519            } else {
520                PixGateMode::Predeposit
521            },
522            paid_tail: u32::from(self.paid_recovery_tail.is_some()),
523            predeposit_exposure_packets: predeposit_exposure,
524            stalled_share: u32::from(self.fill.is_stalled()),
525            ..Default::default()
526        };
527
528        for ssa in self.ssas.iter() {
529            // Recovered tombstones and `Closing` records are retained diagnostics, not live cycles.
530            // Counting them would leave `hopr_pix_cycles_active` unable to drain, since a tombstone
531            // outlives the cycle by `tombstone_retention_window`.
532            let bucket = match ssa.phase {
533                SsaPhase::AwaitingCommitment => &mut snapshot.awaiting_commitment,
534                SsaPhase::AwaitingDeposit => &mut snapshot.awaiting_deposit,
535                SsaPhase::Recovering => &mut snapshot.recovering,
536                SsaPhase::Recovered { .. } | SsaPhase::Closing => continue,
537            };
538            *bucket = bucket.saturating_add(1);
539        }
540
541        snapshot
542    }

579    /// Re-points the share-order accounting at the current front of the batch, clearing it if the front
580    /// moved.
581    ///
582    /// Clearing on a change of front is what bounds the measurement window to one cycle's service, and
583    /// it is why an Entry cannot launder a spree by eventually finishing a cycle: the reset is only
584    /// reached by the front cycle *completing*, which is the thing the Exit wanted all along.
585    ///
586    /// Called after every event and every deadline sweep, so it covers both ways the front can move —
587    /// the front cycle recovering, or being retired.
588    fn refresh_share_order_front(&mut self) {
589        let front = self.earliest_live_idx().map(|i| self.ssas[i].ssa_id.ssa_index());
590        if front != self.share_order_front {
591            self.share_order_front = front;
592            self.front_useful = 0;
593            self.off_front_useful = 0;
594        }
595    }

597    /// Starts the recovery clocks of the earliest unrecovered cycle, if they are not running yet.
598    ///
599    /// A batch's cycles are served strictly in index order — the Entry's emission window is clamped to
600    /// one cycle (see [`hopr_protocol_pix::SHARE_EMISSION_WINDOW`]) — so a cycle behind the front of
601    /// the batch is *queued*, not stalled. Starting its clocks when its deposit confirmed, as this used
602    /// to, measured the queue wait rather than the recovery: at deployed dimensions one cycle takes
603    /// ~61 min of emission to exhaust, so every cycle after the first was retired by
604    /// `max_recovery_time` — or, worse, by `max_recovery_idle` inside a minute, since the idle gate
605    /// compares against *session-wide* service and a queued cycle sees plenty of that without any of it
606    /// being its own.
607    ///
608    /// So the clock starts when the cycle reaches the front, which is when it can actually make
609    /// progress, and [`SupervisorConfig::max_recovery_time`] then means what its name says: the ceiling
610    /// on recovering a single cycle. `served_total_at_last_progress` is re-baselined at the same moment
611    /// for the same reason — otherwise the idle gate would charge the queue wait against it.
612    ///
613    /// Called after every event and every deadline sweep, so it covers each way a cycle can reach the
614    /// front: its predecessor recovering, or its predecessor being retired.
615    fn arm_recovery_clocks_for_earliest(&mut self, now: Instant, served_total: u64) {
616        if self.closed {
617            return;
618        }
619        // The packets being served still carry the recovered predecessor's shares. Starting the
620        // successor now would charge that pipeline delay against both its idle and hard recovery
621        // budgets even though no successor share can possibly arrive yet.
622        if self.paid_recovery_tail.is_some() {
623            return;
624        }
625        let Some(idx) = self.earliest_live_idx() else {
626            return;
627        };
628
629        let ssa = &mut self.ssas[idx];
630        if ssa.phase != SsaPhase::Recovering || ssa.recovery_hard_deadline.is_some() {
631            return;
632        }
633
634        ssa.recovery_idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
635        ssa.recovery_hard_deadline = now.checked_add(self.cfg.max_recovery_time);
636        ssa.served_total_at_last_progress = served_total;
637        // The same instant, for the same reason: this is when the cycle starts being served, so it
638        // is when its own egress begins. `get_or_insert` rather than a plain assignment because a
639        // cycle reaches the front once — the guard above returns early for one that already armed —
640        // but stating it here means a future caller cannot rebaseline a cycle mid-flight and make
641        // its recorded egress smaller than what it actually consumed.
642        ssa.served_total_at_front.get_or_insert(served_total);
643        tracing::debug!(ssa_id = %ssa.ssa_id, "recovery clocks started — cycle is at the front of the batch");
644    }

646    /// Check all deadlines and emit actions for any that have expired.
647    pub fn handle_deadline(&mut self, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
648        if self.closed {
649            return Vec::new();
650        }
651
652        let mut actions = Vec::new();
653        let max_recovery_idle = self.cfg.max_recovery_idle;
654
655        // While the recovered predecessor still owns the service front, its original recovery
656        // clocks remain the resource bound. Progress refreshes only the idle clock; the hard clock
657        // is never extended. Thus a large but paid surplus may drain, while an Entry that declares
658        // one and then feeds neither tail nor successor cannot hold the Session forever.
659        if let Some(tail) = self.paid_recovery_tail.as_mut() {
660            let expired = if tail.hard_deadline.is_some_and(|deadline| now >= deadline) {
661                Some(SessionPixCloseReason::RecoveryDeadline)
662            } else if tail.idle_deadline.is_some_and(|deadline| now >= deadline) {
663                Some(SessionPixCloseReason::RecoveryIdle)
664            } else {
665                None
666            };
667
668            if let Some(reason) = expired {
669                if reason == SessionPixCloseReason::RecoveryIdle && served_total <= tail.served_total_at_last_progress {
670                    tail.idle_deadline = now.checked_add(max_recovery_idle);
671                } else {
672                    tracing::warn!(
673                        ssa_id = %tail.ssa_id,
674                        ?reason,
675                        largest_shares_seen = tail.largest_shares_seen,
676                        served_total_at_last_progress = tail.served_total_at_last_progress,
677                        "closing PIX Session while its paid recovery tail is stalled"
678                    );
679                    // The tail is ending here, not merely being abandoned with the Session, so it
680                    // owes its duration like every other phase that reaches a terminal edge.
681                    self.release_paid_tail(now);
682                    self.closed = true;
683                    return self.with_fill_stopped(vec![SessionPixAction::Close(reason)]);
684                }
685            }
686        }
687
688        let mut i = 0;
689        while i < self.ssas.len() {
690            if self.closed {
691                break;
692            }
693
694            let expired = {
695                let ssa = &self.ssas[i];
696                if ssa.is_terminal() {
697                    i += 1;
698                    continue;
699                }
700                ssa.check_deadlines(now)
701            };
702
703            // A stalled commitment is re-requested rather than given up on, and the timer re-arms so
704            // a repair that is itself lost is asked for again. This yields no close reason and is
705            // bounded by nothing of its own: the absolute `commitment_deadline` still ends the cycle,
706            // which is why it can repeat freely — and why it is skipped once that deadline has gone,
707            // rather than putting an ask on the wire for a Session about to close.
708            if expired.is_none()
709                && self.ssas[i].phase == SsaPhase::AwaitingCommitment
710                && self.ssas[i].recommit_deadline.is_some_and(|deadline| now >= deadline)
711            {
712                let ssa = &mut self.ssas[i];
713                ssa.recommit_attempts += 1;
714                // Disarmed rather than re-armed at the cap, so the worker stops waking for a timer
715                // that would produce nothing. Further arriving parts cannot re-arm it either.
716                ssa.recommit_deadline = (ssa.recommit_attempts < hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS)
717                    .then(|| now.checked_add(self.cfg.commitment_recommit_interval))
718                    .flatten();
719                tracing::debug!(
720                    ssa_id = %ssa.ssa_id, attempt = ssa.recommit_attempts,
721                    "commitment delivery stalled — re-requesting the parts that did not arrive"
722                );
723                actions.push(SessionPixAction::RequestCommitmentRetransmission {
724                    ssa_id: ssa.ssa_id,
725                    params: self.dims,
726                });
727            }
728
729            if let Some(reason) = expired {
730                // Service-gated idle: if no service consumed since last progress,
731                // re-arm instead of closing.
732                //
733                // Not while draining, though. The re-arm asks "was any *gated* service consumed",
734                // and a drained Session has no application left to consume any — `served_total` is
735                // frozen for good, so the re-arm would fire for ever and the one bound on a drain
736                // admitted against an over-estimated buffer would be gone with it. A *productive*
737                // drain is unaffected: its own progress still refreshes `recovery_idle_deadline`
738                // through `on_recovery_progress`.
739                if reason == SessionPixCloseReason::RecoveryIdle
740                    && !self.draining
741                    && served_total <= self.ssas[i].served_total_at_last_progress
742                {
743                    self.ssas[i].recovery_idle_deadline = now.checked_add(max_recovery_idle);
744                    i += 1;
745                    continue;
746                }
747
748                actions.extend(self.close_ssa_and_collect(i, reason, served_total, now));
749                continue;
750            }
751            i += 1;
752        }
753
754        // If the session is already closing, skip tombstone retirement.
755        // Whole-session teardown retires everything via the retirement guard.
756        if self.closed {
757            return self.with_fill_stopped(actions);
758        }
759
760        // Remove tombstones that have expired and emit RetireSsa so the
761        // reconstructor and observer state is released mid-session.
762        let retired_ids: Vec<SsaId<_>> = self
763            .ssas
764            .iter()
765            .filter(|ssa| matches!(ssa.phase, SsaPhase::Recovered { tombstone_until } if now >= tombstone_until))
766            .map(|ssa| ssa.ssa_id)
767            .collect();
768        self.ssas
769            .retain(|ssa| !matches!(ssa.phase, SsaPhase::Recovered { tombstone_until } if now >= tombstone_until));
770        for id in retired_ids {
771            self.note_retired_index(id.ssa_index());
772            actions.push(SessionPixAction::RetireSsa(id));
773        }
774
775        // A terminal cycle may have released the older of two live batches. Keep retirement before
776        // the replacement request in the action stream so the reconstructor drops the old guard
777        // before allocating the successor.
778        actions.extend(self.retry_deferred_successor_request(now));
779        self.consume_paid_tail_at_observed_boundary(now);
780
781        // If no SSAs remain, close.
782        // Note: a successor request may still be in flight here (RequestSsa
783        // emitted, SsaRequestSent not yet observed) — intentionally not
784        // tracked. The tombstone retention window is the backstop; a request
785        // delayed beyond it (>30 s transport stall on an otherwise healthy
786        // session) is theoretical and does not warrant blocking normal
787        // tombstone expiry.
788        if self.ssas.is_empty() && !self.closed {
789            actions.push(SessionPixAction::Close(
790                self.first_failure_reason
791                    .unwrap_or(SessionPixCloseReason::NoSsaRemaining),
792            ));
793            self.closed = true;
794        }
795
796        // Retiring or tombstoning a cycle can promote the next one to the front of the batch — and if
797        // that one is already funded, promotion is the only moment left at which service can be
798        // released, since its deposit event is long past.
799        self.arm_recovery_clocks_for_earliest(now, served_total);
800        let mut gate_actions = self.sync_service_gate();
801        gate_actions.extend(actions);
802        self.refresh_share_order_front();
803
804        if self.closed {
805            return self.with_fill_stopped(gate_actions);
806        }
807        gate_actions
808    }

810    /// Runs the deadline sweep and, if one is due, the fill tick — in that order.
811    ///
812    /// The order is the point. The sweep is what arms a promoted cycle's recovery clocks, consumes a
813    /// spent FIFO tail and re-aligns the service gate, so running it first means the tick plans
814    /// against the lifecycle as it is *now* rather than as it was a second ago. A tick that ran first
815    /// would spend one whole sampling interval filling for a cycle that had already recovered.
816    ///
817    /// This is the entry point the worker uses; [`handle_deadline`](Self::handle_deadline) remains
818    /// separate because the deadline logic is meaningful — and tested — on its own.
819    pub fn handle_timers(&mut self, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
820        let mut actions = self.handle_deadline(now, served_total);
821        if self.closed {
822            return actions;
823        }
824        if self.next_fill_tick().is_some_and(|tick| now >= tick) {
825            actions.extend(self.handle_fill_tick(now, served_total));
826        }
827        actions
828    }

852    /// When the fill planner next wants to run, or `None` if there is nothing for it to do.
853    ///
854    /// `None` while no cycle is being filled for *and* the stream is already silent, which is what
855    /// keeps an unfunded or non-PIX-idle Session from waking its worker once a second for nothing. A
856    /// running stream always yields a tick even without a target, because it is owed the zero that
857    /// stops it.
858    pub fn next_fill_tick(&self) -> Option<Instant> {
859        if self.closed || !self.fill.is_enabled() {
860            return None;
861        }
862        if self.fill_target().is_none() && self.fill.is_idle() {
863            return None;
864        }
865        Some(self.fill.next_tick())

868    /// The cycle the Exit should currently be filling for, if any.
869    ///
870    /// The recovered predecessor's FIFO tail wins whenever it exists, for two reasons that point the
871    /// same way: its SURBs sit ahead of the successor's in the Exit's own buffer, so packets sent now
872    /// carry *its* shares; and it holds the binding hard deadline, since the successor's clocks are
873    /// deliberately not armed until the tail is exhausted (see
874    /// [`arm_recovery_clocks_for_earliest`](Self::arm_recovery_clocks_for_earliest)).
875    ///
876    /// Otherwise it is the earliest live cycle, and only once it is both funded and clocked. An
877    /// unfunded cycle has nothing at stake to be filled for, and a funded one whose clocks have not
878    /// armed is queued behind a predecessor rather than being served — filling for it would spend
879    /// SURBs on shares the Entry cannot yet be emitting.
880    fn fill_target(&self) -> Option<FillTarget> {
881        if let Some(tail) = self.paid_recovery_tail.as_ref() {
882            return tail.hard_deadline.map(|hard_deadline| FillTarget {
883                ssa_id: tail.ssa_id,
884                largest_shares_seen: tail.largest_shares_seen,
885                hard_deadline,
886                drain: self.draining,
887            });
888        }
889
890        let ssa = &self.ssas[self.earliest_live_idx()?];
891        (ssa.phase == SsaPhase::Recovering)
892            .then_some(ssa)
893            .and_then(|ssa| ssa.recovery_hard_deadline.map(|hard_deadline| (ssa, hard_deadline)))
894            .map(|(ssa, hard_deadline)| FillTarget {
895                ssa_id: ssa.ssa_id,
896                largest_shares_seen: ssa.largest_shares_seen,
897                hard_deadline,
898                drain: self.draining,
899            })
900    }

919    /// Returns the earliest deadline across all live SSAs, or `None`.
920    pub fn next_deadline(&self) -> Option<Instant> {
921        if self.closed || (self.ssas.is_empty() && self.paid_recovery_tail.is_none()) {
922            return None;
923        }
924
925        let ssa_deadline = self
926            .ssas
927            .iter()
928            .filter_map(|ssa| {
929                if ssa.is_terminal() {
930                    if let SsaPhase::Recovered { tombstone_until } = ssa.phase {
931                        return Some(tombstone_until);
932                    }
933                    return None;
934                }
935                // Return the earliest set deadline. The re-request timer belongs here like the rest:
936                // it is shorter than every other deadline of the phase it runs in, so leaving it out
937                // would mean the worker slept past it and only re-requested when something else woke
938                // it up.
939                ssa.commitment_deadline
940                    .into_iter()
941                    .chain(ssa.recommit_deadline)
942                    .chain(ssa.deposit_deadline)
943                    .chain(ssa.recovery_idle_deadline)
944                    .chain(ssa.recovery_hard_deadline)
945                    .min()
946            })
947            .min();
948        let tail_deadline = self
949            .paid_recovery_tail
950            .as_ref()
951            .and_then(|tail| tail.idle_deadline.into_iter().chain(tail.hard_deadline).min());
952
953        ssa_deadline.into_iter().chain(tail_deadline).min()
954    }

981    /// Arms the commitment deadline once the request this SSA was registered for has gone out.
982    ///
983    /// The record itself is created by [`emit_request_next_ssa`](Self::emit_request_next_ssa); this
984    /// only starts the clock, because the Entry cannot be late answering a request it has not been
985    /// sent yet.
986    fn on_ssa_request_sent(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
987        // Validate pseudonym. Checked before the lookup: a foreign pseudonym never matches a record
988        // of ours, so looking first would answer a cross-session confusion with silence.
989        if ssa_id.pseudonym() != &self.pseudonym {
990            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
991        }
992
993        // The live-record lookup comes first, and the staleness guard only answers for indices that
994        // have no record. Retirement is not monotone — a later batch member can lose its deposit while
995        // an earlier one is still recovering — so a watermark consulted first would silently ignore a
996        // legitimate request for a cycle that is very much alive. Asking the records first removes the
997        // assumption instead of relying on it.
998        let Some(idx) = self.find_ssa_idx(ssa_id) else {
999            // No live record. Either this index was retired and the event is a straggler, which is
1000            // benign and ignored, or we were never asked for it at all, which is a protocol fault.
1001            if self
1002                .highest_retired_ssa_index
1003                .is_some_and(|highest| ssa_id.ssa_index() <= highest)
1004            {
1005                return Vec::new();
1006            }
1007            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
1008        };
1009
1010        // Idempotent: a repeated confirmation must not extend the deadline, and one arriving after
1011        // the SSA has moved on must not resurrect a phase it has left.
1012        if self.ssas[idx].phase == SsaPhase::AwaitingCommitment && self.ssas[idx].commitment_deadline.is_none() {
1013            self.ssas[idx].commitment_deadline = now.checked_add(crate::supervision::scaled_deadline(
1014                self.cfg.max_ssa_delivery_time,
1015                self.cfg.ssas_per_request,
1016            ));
1017        }
1018
1019        Vec::new()
1020    }

1022    /// Arms — or re-arms — the re-request timer on a commitment that is under way but not complete.
1023    ///
1024    /// This is where the issue's rule lives: a cycle that received *nothing* never reaches here, so
1025    /// it never asks, which is correct because there is no scope for it to name. And because every
1026    /// arriving part pushes the timer out, the first ask cannot land in the middle of the burst —
1027    /// what the timer measures is a stall in delivery.
1028    ///
1029    /// Unscaled by the batch size, unlike the two absolute deadlines: this is a per-cycle stall
1030    /// detector, and a batch's commitment sets do not queue behind one another in a way that would
1031    /// make one cycle's silence expected.
1032    fn on_commitment_progress(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
1033        if let Some(idx) = self.find_ssa_idx(ssa_id)
1034            && self.ssas[idx].phase == SsaPhase::AwaitingCommitment
1035            && self.ssas[idx].recommit_attempts < hopr_protocol_pix::MAX_COMMITMENT_RETRANSMISSIONS
1036        {
1037            self.ssas[idx].recommit_deadline = now.checked_add(self.cfg.commitment_recommit_interval);
1038        }
1039
1040        Vec::new()
1041    }

1043    fn on_commitment_verified(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
1044        let idx = match self.find_ssa_idx(ssa_id) {
1045            Some(i) => i,
1046            None => return Vec::new(),
1047        };
1048        let ssa = &mut self.ssas[idx];
1049        if ssa.phase != SsaPhase::AwaitingCommitment {
1050            return Vec::new();
1051        }
1052
1053        let deposit_deadline = now.checked_add(crate::supervision::scaled_deadline(
1054            self.cfg.max_deposit_wait,
1055            self.cfg.ssas_per_request,
1056        ));
1057        let phase_sample = ssa.enter_phase(SsaPhase::AwaitingDeposit, now);
1058        ssa.deposit_deadline = deposit_deadline;
1059        ssa.commitment_deadline = None;
1060        // Nothing is missing any more, so nothing is to be asked for.
1061        ssa.recommit_deadline = None;
1062        self.telemetry.phase_durations.extend(phase_sample);
1063
1064        // Behind the phase guard above, so a repeated `CommitmentVerified` for a cycle that has
1065        // already moved on counts nothing.
1066        self.telemetry.committed = self.telemetry.committed.saturating_add(1);
1067
1068        Vec::new()
1069    }

1071    /// Funds the cycle on the deposit pool's verdict. The supervisor does not second-guess the amount.
1072    ///
1073    /// Sufficiency is priced and judged outside this workspace. The Exit's pool is handed the
1074    /// [`DepositUpdated`](hopr_api::node::DepositUpdated) sender together with the byte quota when the
1075    /// deposit address arrives, and it sends on that channel once the deposit clears the price it
1076    /// charges for that quota — so by the time this runs, the question has been answered by the only
1077    /// component that can answer it. Nothing in the Session layer prices a byte.
1078    ///
1079    /// This used to compare the running total against `max(expected_deposit, min_deposit)`. Both are
1080    /// gone, and not because they were merely unused: a configured floor here is not a backstop behind
1081    /// the pool but a second authority alongside it, and one set above the pool's price fails a cycle
1082    /// that has already been paid for — silently, on `max_deposit_wait`, with neither side able to
1083    /// break the tie. `expected_deposit` was the same idea arriving over the wire, and the commitment
1084    /// never carried an amount to fill it.
1085    ///
1086    /// What remains is the one check that needs no price: a zero balance asserts that nothing was
1087    /// deposited, so it is not a verdict and cannot release service. The deadline keeps running and a
1088    /// later confirmation can still fund the cycle.
1089    fn on_deposit_confirmed(
1090        &mut self,
1091        ssa_id: &SsaId<HoprPseudonym>,
1092        amount: HoprBalance,
1093        now: Instant,
1094        served_total: u64,
1095    ) -> Vec<SessionPixAction> {
1096        let idx = match self.find_ssa_idx(ssa_id) {
1097            Some(i) => i,
1098            None => return Vec::new(),
1099        };
1100        let was_front = self.earliest_live_idx() == Some(idx);
1101
1102        let ssa = &mut self.ssas[idx];
1103        if ssa.phase != SsaPhase::AwaitingDeposit {
1104            return Vec::new();
1105        }
1106
1107        if amount == HoprBalance::zero() {
1108            return Vec::new();
1109        }
1110
1111        // Transition to Recovering. The two recovery clocks are *not* started here — see
1112        // `arm_recovery_clocks_for_earliest`, which starts them when this cycle's turn comes.
1113        let phase_sample = ssa.enter_phase(SsaPhase::Recovering, now);
1114        ssa.deposit_deadline = None;
1115        ssa.served_total_at_last_progress = served_total;
1116        self.telemetry.phase_durations.extend(phase_sample);
1117
1118        // Behind both the phase guard and the zero-amount guard above, so a duplicate confirmation
1119        // and a zero balance — which is not a verdict — both count nothing.
1120        ssa.deposit_uhopr = deposit_uhopr(amount);
1121        self.telemetry.funded = self.telemetry.funded.saturating_add(1);
1122        self.telemetry.deposit_confirmed_uhopr =
1123            self.telemetry.deposit_confirmed_uhopr.saturating_add(ssa.deposit_uhopr);
1124
1125        // If recovery completed before the deposit arrived, immediately
1126        // tombstone the SSA — the Recovered event was deferred.
1127        let recovered_pending = ssa.recovered_pending;
1128
1129        let pending = ssa.next_request_pending_deposit;
1130        if pending {
1131            ssa.next_request_pending_deposit = false;
1132        }
1133        // End the mutable borrow on ssas[idx].
1134        let _ = ssa;
1135
1136        let mut actions = Vec::new();
1137
1138        if recovered_pending {
1139            if was_front {
1140                self.paid_front_handoff = true;
1141            }
1142            actions.extend(self.perform_recovered_transition(idx, now, served_total));
1143        }
1144
1145        if pending {
1146            // When recovered_pending is also set, perform_recovered_transition
1147            // won't emit RequestSsa (next_requested is already set by
1148            // on_recovered), so emit it here.
1149            actions.extend(self.emit_request_next_ssa(now));
1150        }
1151
1152        actions
1153    }

1173    fn on_recovery_progress(
1174        &mut self,
1175        progress: &SsaRecoveryProgress<HoprPseudonym>,
1176        now: Instant,
1177        served_total: u64,
1178    ) -> Vec<SessionPixAction> {
1179        if self
1180            .paid_recovery_tail
1181            .as_ref()
1182            .is_some_and(|tail| tail.ssa_id == progress.ssa_id)
1183        {
1184            return self.on_recovered_tail_progress(progress, now, served_total);
1185        }
1186
1187        let idx = match self.find_ssa_idx(&progress.ssa_id) {
1188            Some(i) => i,
1189            None => return Vec::new(),
1190        };
1191
1192        // Absorb late progress on tombstones — the SSA is already fully
1193        // recovered and should not reset the session-wide gate watermark.
1194        if self.ssas[idx].is_terminal() {
1195            return Vec::new();
1196        }
1197
1198        // Validate every dimension-derived bound before mutating. Progress is produced locally, but
1199        // all of its source fields ultimately came from peer-tagged shares. Keeping the supervisor
1200        // as an independent backstop means a reconstructor regression cannot turn an oversized
1201        // counter into unbounded service.
1202        let target = self.dims.target_useful_shares();
1203        let max_shares_seen = self.dims.polys_per_ssa() as u64 * self.dims.emitted_shares_per_poly() as u64;
1204        if progress.target_useful_shares != target
1205            || progress.useful_shares > target
1206            || progress.shares_seen < progress.useful_shares
1207            || progress.shares_seen > max_shares_seen
1208            || progress.recovered_polynomials > self.dims.polys_per_ssa()
1209        {
1210            return vec![SessionPixAction::Close(SessionPixCloseReason::CounterRegression)];
1211        }
1212
1213        let ssa = &mut self.ssas[idx];
1214        let new_useful = progress.useful_shares;
1215        let new_seen = progress.shares_seen;
1216
1217        // Counter regression check, on the liveness counter because it is the one that moves for every
1218        // share — a snapshot can be newer than the last one while carrying the same `useful_shares`.
1219        //
1220        // The relay-as-Exit pipeline processes acknowledgement batches with
1221        // for_each_concurrent, so absolute progress snapshots from different
1222        // batches can arrive out of order. Treat a stale snapshot as benign
1223        // noise rather than a protocol violation.
1224        if new_seen <= ssa.largest_shares_seen {
1225            return Vec::new();
1226        }
1227
1228        // The accepted delta, taken here because this is the one point at which a snapshot is known
1229        // to be both validated and newer than everything seen before. The absolute/reordered
1230        // semantics the issue asks for fall out of the two high-water marks rather than needing
1231        // their own machinery: a repeated or stale snapshot returned at the guard above and adds
1232        // zero, and a snapshot arriving after a dropped one carries the whole missing delta,
1233        // because a mark is a maximum and never a running sum.
1234        let accepted_delta = new_seen - ssa.largest_shares_seen;
1235        // Clamped to the accepted delta, which is not defensive clutter. The two marks are
1236        // independent maxima over possibly different snapshots, so a reconstructor regression could
1237        // present `(seen 10, useful 2)` followed by `(seen 11, useful 11)` — a useful delta of nine
1238        // against an accepted delta of one — and mint a negative surplus. This file is an
1239        // independent backstop against exactly that class of fault everywhere else; it is one here
1240        // too.
1241        let useful_delta = new_useful.saturating_sub(ssa.largest_useful_shares).min(accepted_delta);
1242        self.telemetry.useful_shares = self.telemetry.useful_shares.saturating_add(useful_delta);
1243        self.telemetry.surplus_shares = self
1244            .telemetry
1245            .surplus_shares
1246            .saturating_add(accepted_delta - useful_delta);
1247
1248        ssa.largest_shares_seen = new_seen;
1249
1250        // Liveness tier. A share arrived, useful or not, so the Entry is serving this cycle — which is
1251        // the question the egress gate and the recovery-idle deadline are asking. Keying either of them
1252        // on `useful_shares` instead makes a conforming Entry's surplus run look like silence, and for
1253        // the gate that is self-inflicted: withholding service removes the only thing that could
1254        // produce the next useful share. See `SsaRecoveryProgress::shares_seen`.
1255        ssa.served_total_at_last_progress = served_total;
1256        // Refreshing is not arming: only `arm_recovery_clocks_for_earliest` starts a cycle's clocks,
1257        // and `recovery_hard_deadline` is how a cycle records that it has reached the front. Without
1258        // that second condition a stray reordered share *starts* the idle clock on a cycle still
1259        // queued behind the front — which the Entry cannot then feed, because emission is clamped to
1260        // one cycle — and 60 s later the idle gate sees session-wide service climbing and retires it.
1261        // That is the queue-wait failure `arm_recovery_clocks_for_earliest` documents, reached
1262        // through the refresh path instead of the deposit one. The clocks arm together or not at all.
1263        if ssa.phase == SsaPhase::Recovering && ssa.recovery_hard_deadline.is_some() {
1264            ssa.recovery_idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
1265        }
1266
1267        // Payment tier. Only shares that advanced reconstruction are worth anything, so only they move
1268        // the recovery counters or count towards the share-order ratio.
1269        if new_useful > ssa.largest_useful_shares {
1270            let delta = new_useful - ssa.largest_useful_shares;
1271            ssa.largest_useful_shares = new_useful;
1272            ssa.recovered_polynomials = progress.recovered_polynomials;
1273
1274            if let Some(reason) = self.book_share_order_progress(&progress.ssa_id, delta) {
1275                return vec![SessionPixAction::Close(reason)];
1276            }
1277        }
1278
1279        let is_front = self.earliest_live_idx() == Some(idx);
1280
1281        // Only funded progress on the cycle service is currently charged against may reopen the
1282        // Session-wide ceiling. Unfunded progress is the H11 bypass; off-front progress belongs to a
1283        // queued cycle and must not buy service for the front either. The outer event transition
1284        // consumes any predecessor tail before synchronizing the gate, even for unfunded progress.
1285        if self.ssas[idx].phase == SsaPhase::Recovering && is_front {
1286            vec![SessionPixAction::ProgressNotification]
1287        } else {
1288            Vec::new()
1289        }
1290    }
1291
1292    /// Applies a progress snapshot from the sole recovered predecessor authorized to drain.
1293    ///
1294    /// The reconstructor enforces the strong per-polynomial budget. These aggregate checks are a
1295    /// second security boundary: even if that component regresses, the Entry cannot make a recovered
1296    /// cycle report more than `polys × (threshold + surplus)` packets, change its terminal useful
1297    /// count, or revive any older tombstone.
1298    fn on_recovered_tail_progress(
1299        &mut self,
1300        progress: &SsaRecoveryProgress<HoprPseudonym>,
1301        now: Instant,
1302        served_total: u64,
1303    ) -> Vec<SessionPixAction> {
1304        let target = self.dims.target_useful_shares();
1305        let expected_polynomials = self.dims.polys_per_ssa();
1306        let max_shares_seen = expected_polynomials as u64 * self.dims.emitted_shares_per_poly() as u64;
1307
1308        if progress.target_useful_shares != target
1309            || progress.useful_shares > target
1310            || progress.recovered_polynomials > expected_polynomials
1311            || progress.shares_seen < progress.useful_shares
1312            || progress.shares_seen > max_shares_seen
1313        {
1314            tracing::warn!(
1315                ssa_id = %progress.ssa_id,
1316                useful_shares = progress.useful_shares,
1317                shares_seen = progress.shares_seen,
1318                recovered_polynomials = progress.recovered_polynomials,
1319                "rejected impossible progress from a recovered PIX tail"
1320            );
1321            return vec![SessionPixAction::Close(SessionPixCloseReason::CounterRegression)];
1322        }
1323
1324        // Absolute snapshots from concurrent acknowledgement batches may arrive after the
1325        // `Recovered` event that overtook them. A non-terminal snapshot cannot buy liveness here,
1326        // but it is not evidence of peer misconduct either. Ignore it exactly as the live path
1327        // ignores an older absolute counter. Only a terminal snapshot can touch the paid receipt.
1328        if progress.useful_shares != target || progress.recovered_polynomials != expected_polynomials {
1329            return Vec::new();
1330        }
1331
1332        let (exhausted, accepted_delta) = {
1333            let tail = self
1334                .paid_recovery_tail
1335                .as_mut()
1336                .expect("caller matched the paid recovery tail");
1337            if progress.shares_seen <= tail.largest_shares_seen {
1338                return Vec::new();
1339            }
1340
1341            let accepted_delta = progress.shares_seen - tail.largest_shares_seen;
1342            tail.largest_shares_seen = progress.shares_seen;
1343            tail.served_total_at_last_progress = served_total;
1344            tail.idle_deadline = now.checked_add(self.cfg.max_recovery_idle);
1345            (progress.shares_seen == max_shares_seen, accepted_delta)
1346        };
1347
1348        // All surplus, by definition: this cycle has already reached its useful-share target, which
1349        // is what the guard above requires before a snapshot may touch the receipt at all. These
1350        // are the negotiated tail a conforming Entry keeps emitting while the FIFO drains — real
1351        // accepted traffic, and the evidence that keeps the paid front open, but no longer worth
1352        // anything towards recovery.
1353        self.telemetry.surplus_shares = self.telemetry.surplus_shares.saturating_add(accepted_delta);
1354
1355        // Keep the tombstone's diagnostics monotone while it is retained, but do not require the
1356        // record: tombstone cleanup is intentionally shorter than a Session's possible FIFO drain.
1357        if let Some(idx) = self.find_ssa_idx(&progress.ssa_id) {
1358            self.ssas[idx].largest_useful_shares = target;
1359            self.ssas[idx].largest_shares_seen = progress.shares_seen;
1360            self.ssas[idx].recovered_polynomials = expected_polynomials;
1361            self.ssas[idx].served_total_at_last_progress = served_total;
1362        }
1363
1364        if exhausted {
1365            self.release_paid_tail(now);
1366        }
1367        vec![SessionPixAction::ProgressNotification]
1368    }
1369
1370    /// Attributes `delta` useful shares to the front of the batch or behind it, and judges the ratio.
1371    ///
1372    /// The Entry serves a batch strictly in index order, so progress on anything but the front cycle is
1373    /// service the Exit cannot yet be paid for. See
1374    /// [`SupervisorConfig::max_off_front_share_fraction`] for the threat and for why this is measured
1375    /// as a fraction; the judgement waits for
1376    /// [`SupervisorConfig::min_share_order_sample`] shares of evidence.
1377    fn book_share_order_progress(
1378        &mut self,
1379        ssa_id: &SsaId<HoprPseudonym>,
1380        delta: u64,
1381    ) -> Option<SessionPixCloseReason> {
1382        let front = self.share_order_front?;
1383        if ssa_id.ssa_index() == front {
1384            self.front_useful = self.front_useful.saturating_add(delta);
1385        } else {
1386            self.off_front_useful = self.off_front_useful.saturating_add(delta);
1387        }
1388
1389        // Judged on every event, not only off-front ones: the sample can cross the floor on a share
1390        // that lands *on* the front cycle, and the ratio is already whatever it is by then. Waiting for
1391        // the next off-front share would only delay the same verdict.
1392        let total = self.front_useful.saturating_add(self.off_front_useful);
1393        if total < self.cfg.min_share_order_sample || total == 0 {
1394            return None;
1395        }
1396
1397        let fraction = self.off_front_useful as f64 / total as f64;
1398        if fraction > self.cfg.max_off_front_share_fraction {
1399            tracing::error!(
1400                %ssa_id, %front, fraction,
1401                limit = self.cfg.max_off_front_share_fraction,
1402                front_useful = self.front_useful,
1403                off_front_useful = self.off_front_useful,
1404                "entry is serving the batch out of order"
1405            );
1406            return Some(SessionPixCloseReason::BatchServedOutOfOrder);
1407        }
1408
1409        tracing::trace!(%ssa_id, %front, fraction, "progress behind the front of the batch");
1410        None
1411    }

1413    fn on_almost_recovered(&mut self, ssa_id: &SsaId<HoprPseudonym>, now: Instant) -> Vec<SessionPixAction> {
1414        let idx = match self.find_ssa_idx(ssa_id) {
1415            Some(i) => i,
1416            None => return Vec::new(),
1417        };
1418
1419        // If already recovered (e.g. a concurrent batch delivered Recovered
1420        // before this AlmostRecovered), the next SSA is already requested —
1421        // no-op.
1422        if self.ssas[idx].is_terminal() {
1423            return Vec::new();
1424        }
1425
1426        // The successor gate. A cycle in the middle of a batch reaching its early-recovery threshold
1427        // says nothing about the batch as a whole: the ones behind it still have their full quota to
1428        // serve. Only the last cycle standing in for the batch may ask for a replacement.
1429        if !self.ssas[idx].is_batch_last {
1430            return Vec::new();
1431        }
1432
1433        let next_requested = self.ssas[idx].next_requested;
1434        let phase = self.ssas[idx].phase;
1435
1436        if next_requested {
1437            return Vec::new();
1438        }
1439
1440        match phase {
1441            SsaPhase::Recovering => {
1442                self.ssas[idx].next_requested = true;
1443                self.emit_request_next_ssa(now)
1444            }
1445            SsaPhase::AwaitingDeposit => {
1446                self.ssas[idx].next_requested = true;
1447                self.ssas[idx].next_request_pending_deposit = true;
1448                Vec::new()
1449            }
1450            // AwaitingCommitment or unknown phase: the next SSA request will be
1451            // triggered when the recovered transition fires (normal Recovering
1452            // path or deferred replay via recovered_pending). Set the deferred
1453            // flag so that commitment_verified can propagate it forward.
1454            _ => {
1455                self.ssas[idx].next_request_pending_deposit = true;
1456                self.ssas[idx].next_requested = true;
1457                Vec::new()
1458            }
1459        }
1460    }

1462    fn on_recovered(
1463        &mut self,
1464        ssa_id: &SsaId<HoprPseudonym>,
1465        now: Instant,
1466        served_total: u64,
1467    ) -> Vec<SessionPixAction> {
1468        let idx = match self.find_ssa_idx(ssa_id) {
1469            Some(i) => i,
1470            None => return Vec::new(),
1471        };
1472
1473        // Guard against duplicate recovery events (possible with concurrent
1474        // batch processing). If already recovered, this is a no-op.
1475        if self.ssas[idx].is_terminal() {
1476            return Vec::new();
1477        }
1478
1479        match self.ssas[idx].phase {
1480            SsaPhase::Recovering => {
1481                // Normal path — transition to tombstone directly.
1482                self.perform_recovered_transition(idx, now, served_total)
1483            }
1484            SsaPhase::AwaitingDeposit => {
1485                // Recovery completed before deposit confirmed. Record the
1486                // pending flag so that when the deposit arrives we replay
1487                // the transition immediately after entering Recovering.
1488                self.ssas[idx].recovered_pending = true;
1489                // Also ensure the next SSA request is deferred — the
1490                // eventual tombstone will emit it. Subject to the same successor gate as
1491                // `on_almost_recovered`: a non-last cycle of a batch never asks, deferred or not.
1492                if !self.ssas[idx].next_requested && self.ssas[idx].is_batch_last {
1493                    self.ssas[idx].next_request_pending_deposit = true;
1494                    self.ssas[idx].next_requested = true;
1495                }
1496                Vec::new()
1497            }
1498            SsaPhase::AwaitingCommitment => {
1499                // Recovery outpaced commitment verification. Same semantics
1500                // as AwaitingDeposit: set pending flag so the transition
1501                // replays once commitment and deposit have both arrived.
1502                self.ssas[idx].recovered_pending = true;
1503                Vec::new()
1504            }
1505            // Closing / already Recovered: handled by the terminal guard above.
1506            _ => Vec::new(),
1507        }
1508    }

1510    /// Books what a recovered cycle owes the counters and retires it into its tombstone.
1511    ///
1512    /// Split out because a drain returns before the rest of the transition, and would otherwise
1513    /// never book the deposit it just unlocked — making the one outcome the drain exists for read as
1514    /// stranded value in `hopr_pix_deposits_recovered_uhopr_total`.
1515    fn book_recovered_cycle(&mut self, idx: usize, now: Instant, served_total: u64) {
1516        // `Recovered` is a local cryptographic fact, so it supplies the lower bound even if its
1517        // final absolute Progress snapshot is delivered just afterwards by a concurrent worker.
1518        // Every later tail snapshot must keep these terminal fields exact.
1519        let target = self.dims.target_useful_shares();
1520        self.ssas[idx].largest_useful_shares = target;
1521        self.ssas[idx].largest_shares_seen = self.ssas[idx].largest_shares_seen.max(target);
1522        self.ssas[idx].recovered_polynomials = self.dims.polys_per_ssa();
1523
1524        // Counted here rather than on the `Recovered` event, because that event is not the
1525        // transition: `on_recovered` defers it when the cycle is still awaiting its commitment or
1526        // deposit, and replays it through this function once both arrive. This is the one place a
1527        // cycle becomes terminal-by-recovery, and its callers all guard against reaching it twice.
1528        self.telemetry.recovered = self.telemetry.recovered.saturating_add(1);
1529        // Summarized before the tombstone transition below, and before `paid_recovery_tail` starts
1530        // collecting the drain, which belongs to the FIFO rather than to this cycle's recovery. The
1531        // terminal fields were just pinned to the target above, so a recovered cycle's useful
1532        // fraction is exactly one by construction.
1533        let summary = self.ssas[idx].telemetry_summary(PixCycleOutcome::Recovered, served_total);
1534        self.telemetry.finalized.push(summary);
1535
1536        let phase_sample = self.ssas[idx].enter_phase(
1537            SsaPhase::Recovered {
1538                tombstone_until: now
1539                    .checked_add(self.cfg.tombstone_retention_window)
1540                    .unwrap_or_else(|| now + Duration::from_secs(86400 * 365)),
1541            },
1542            now,
1543        );
1544        self.telemetry.phase_durations.extend(phase_sample);
1545    }
1546
1547    /// Perform the terminal tombstone transition for a fully-recovered SSA.
1548    /// Called from `on_recovered` (normal Recovering path) and replayed from
1549    /// `on_deposit_confirmed`/`on_commitment_verified` when `recovered_pending`
1550    /// was set earlier.
1551    fn perform_recovered_transition(&mut self, idx: usize, now: Instant, served_total: u64) -> Vec<SessionPixAction> {
1552        let next_requested = self.ssas[idx].next_requested;
1553        // Read before the cycle is retired below, which would make it terminal and change the answer.
1554        let was_front = self.earliest_live_idx() == Some(idx);
1555        let max_shares_seen = self.dims.polys_per_ssa() as u64 * self.dims.emitted_shares_per_poly() as u64;
1556
1557        self.book_recovered_cycle(idx, now, served_total);
1558
1559        // The end of a drain, and the outcome it exists for: the deposit this cycle was paid is
1560        // released rather than stranded. None of the bookkeeping below is worth doing — the tail
1561        // receipt, the gate notification and the successor request all serve a Session that is about
1562        // to be torn down, and `closed` gates every entry point from here on. Returning before the
1563        // tail is created also guarantees a drain can never leave one behind.
1564        if self.draining {
1565            tracing::info!(
1566                ssa_id = %self.ssas[idx].ssa_id,
1567                "the funded cycle of a closed session recovered — the SURB drain is complete"
1568            );
1569            self.closed = true;
1570            return self.with_fill_stopped(vec![SessionPixAction::Close(SessionPixCloseReason::Drained)]);
1571        }
1572
1573        if was_front {
1574            // Replacing rather than accumulating is intentional. Only the immediate recovered
1575            // predecessor can occupy a FIFO immediately ahead of the live front; accepting an
1576            // arbitrary tombstone here would turn old-cycle replay into gate liveness.
1577            self.paid_recovery_tail =
1578                (self.ssas[idx].largest_shares_seen < max_shares_seen).then(|| PaidRecoveryTail {
1579                    ssa_id: self.ssas[idx].ssa_id,
1580                    entered_at: now,
1581                    largest_shares_seen: self.ssas[idx].largest_shares_seen,
1582                    served_total_at_last_progress: served_total,
1583                    // Recovery itself is fresh, cryptographic progress. Reusing the prior idle
1584                    // instant would let delayed/dropped progress snapshots make an honest tail
1585                    // expire immediately after recovery. Refresh this clock once; unlike the hard
1586                    // deadline below, it is the clock progress is meant to extend.
1587                    idle_deadline: now.checked_add(self.cfg.max_recovery_idle),
1588                    hard_deadline: self.ssas[idx]
1589                        .recovery_hard_deadline
1590                        .or_else(|| now.checked_add(self.cfg.max_recovery_time)),
1591                });
1592        }
1593
1594        self.ssas[idx].commitment_deadline = None;
1595        self.ssas[idx].deposit_deadline = None;
1596        self.ssas[idx].recovery_idle_deadline = None;
1597        self.ssas[idx].recovery_hard_deadline = None;
1598        self.ssas[idx].recovered_pending = false;
1599
1600        let mut actions = Vec::new();
1601        // A cycle that recovered with nothing buffered behind it leaves the Exit with nothing to fill
1602        // for until its successor reaches the front and arms its own clocks — which is at least a
1603        // deposit away. Stopping now rather than waiting for the successor keeps those SURBs for the
1604        // cycle that will actually be paid for; the next tick starts the ramp again from the
1605        // successor's true remainder.
1606        if was_front && self.paid_recovery_tail.is_none() {
1607            actions.extend(self.fill.stop().map(SessionPixAction::SetFillRate));
1608        }
1609        if was_front {
1610            // `Recovered` is stronger evidence than an ordinary progress snapshot. Emit this even
1611            // though the reconstructor normally sends the final snapshot first: progress delivery
1612            // is deliberately lossy under command-channel pressure, and without this reset a gate
1613            // already at its served-without-progress ceiling cannot spend the first buffered SURB
1614            // needed to produce tail liveness. If the successor is unfunded, `sync_service_gate`
1615            // prepends `WithholdService`; notifying afterwards changes only the dormant funded
1616            // watermark and cannot restore service.
1617            actions.push(SessionPixAction::ProgressNotification);
1618        }
1619        // Full recovery is the fallback trigger for the early signal, so it carries the same
1620        // successor gate: a non-last cycle of a batch is tombstoned without asking for anything.
1621        if !next_requested && self.ssas[idx].is_batch_last {
1622            self.ssas[idx].next_requested = true;
1623            actions.extend(self.emit_request_next_ssa(now));
1624        }
1625
1626        actions
1627    }

1629    /// Closes the Session on the first unverifiable-share report. There is no tolerance to configure.
1630    ///
1631    /// Shares are not checked on arrival, so a report is not "a bad share" — it means a whole
1632    /// polynomial's share set failed to open its commitment, which surfaces only once `threshold` of
1633    /// them have been interpolated. `SsaPart::add_share` then marks that part failed and releases its
1634    /// shares, and nothing ever clears the flag; since the SSA is the sum of *every* polynomial's
1635    /// constant term, the cycle can no longer be reconstructed by any means and will never pay.
1636    ///
1637    /// So serving on is not tolerance, it is donation: every packet past this point is unpaid with no
1638    /// recovery to preserve at the end of it. That holds however the failure arose, which is why this
1639    /// is not a knob — a false positive from a verification bug leaves the part just as permanently
1640    /// failed, so a tolerance would buy unpaid service rather than the cycle it was raised to save.
1641    /// Closing here caps the exposure at the `threshold` packets already served rather than a multiple
1642    /// of it.
1643    ///
1644    /// `observed_total` is the reconstructor's cross-peer aggregate, and with no limit to compare it
1645    /// against it is logged rather than accumulated: it says how many polynomials the batch that
1646    /// triggered this took down, which is the difference between one bad relayer and a peer sending
1647    /// garbage wholesale.
1648    ///
1649    /// Note this closes the Session directly rather than through
1650    /// [`close_ssa_and_collect`](Self::close_ssa_and_collect), so a batch's surviving siblings do not
1651    /// keep it alive: unlike a lost deposit, this fault is evidence about the peer rather than about
1652    /// one cycle.
1653    fn on_unverifiable_shares(
1654        &mut self,
1655        ssa_id: &SsaId<HoprPseudonym>,
1656        observed_total: u64,
1657        peer: &OffchainPublicKey,
1658        _now: Instant,
1659    ) -> Vec<SessionPixAction> {
1660        let idx = match self.find_ssa_idx(ssa_id) {
1661            Some(i) => i,
1662            None => return Vec::new(),
1663        };
1664
1665        // Absorb late reports on a tombstoned or already-closing SSA: there is no service left to
1666        // stop, and the cycle they condemn has already left. Concurrent ack batches make these
1667        // ordinary rather than exceptional.
1668        if self.ssas[idx].is_terminal() {
1669            return Vec::new();
1670        }
1671
1672        tracing::warn!(
1673            %ssa_id,
1674            %peer,
1675            observed_total,
1676            phase = ?self.ssas[idx].phase,
1677            "closing PIX session: a polynomial's share set failed to open its commitment"
1678        );
1679
1680        self.with_fill_stopped(vec![SessionPixAction::Close(SessionPixCloseReason::UnverifiableShares)])
1681    }

1683    /// Decides whether a Session closed at the Exit is worth draining its buffered SURBs into.
1684    ///
1685    /// `available` is the manager's estimate of the SURBs still spendable, already net of the fill
1686    /// reserve. The cycle it is measured against is the earliest live one, and only when it is both
1687    /// funded and clocked — i.e. exactly the cycle fill would have been working for. Two shapes have
1688    /// nothing a drain could buy and close at once: an unfunded front, where no deposit is at stake;
1689    /// and a recovered predecessor's paid FIFO tail, whose key is already recovered and behind which
1690    /// the successor is queued without clocks of its own.
1691    ///
1692    /// The comparison is against the *whole* remaining emission, with no loss margin. That is
1693    /// conservative in both directions on purpose: the estimate is an upper bound (see
1694    /// [`SessionPixEvent::SessionClosed`]), while in-flight packets are already subtracted from it
1695    /// and full recovery in practice needs fewer than `E` shares. A drain that runs out partway has
1696    /// spent the Session's SURBs down to the reserve and recovered nothing, which is strictly worse
1697    /// than never starting.
1698    ///
1699    /// Siblings are retired directly rather than through
1700    /// [`close_ssa_and_collect`](Self::close_ssa_and_collect), which would be wrong twice over: it
1701    /// charges [`failed_cycles`](Self::failed_cycles), and these cycles were retired by an operator
1702    /// closing a Session rather than lost by an Entry; and its `ssas.len() == 1` branch would close
1703    /// the Session outright. Recovered tombstones are left alone — they hold no reconstructor state
1704    /// and expire on their own timer with an idempotent `RetireSsa`.
1705    ///
1706    /// `_now` is taken for symmetry with every other handler; nothing here is timed. A drain inherits
1707    /// the target's existing clocks rather than starting any of its own, which is the point — the
1708    /// resource bound on the cycle must not be extended by the Session ending.
1709    fn on_session_closed(&mut self, available: u64, _now: Instant) -> Vec<SessionPixAction> {
1710        let target = self
1711            .paid_recovery_tail
1712            .is_none()
1713            .then(|| self.earliest_live_idx())
1714            .flatten()
1715            .filter(|&idx| {
1716                self.ssas[idx].phase == SsaPhase::Recovering && self.ssas[idx].recovery_hard_deadline.is_some()
1717            });
1718
1719        let cycle_shares = self.dims.polys_per_ssa() as u64 * self.dims.emitted_shares_per_poly() as u64;
1720        let remaining = target.map(|idx| cycle_shares.saturating_sub(self.ssas[idx].largest_shares_seen));
1721
1722        let Some((idx, remaining)) = target.zip(remaining).filter(|&(_, remaining)| available >= remaining) else {
1723            tracing::info!(
1724                available,
1725                ?remaining,
1726                target = ?target.map(|idx| self.ssas[idx].ssa_id),
1727                "closing PIX session: the exit-side close left nothing worth draining for"
1728            );
1729            self.closed = true;
1730            return self.with_fill_stopped(vec![SessionPixAction::Close(SessionPixCloseReason::SessionClosed)]);
1731        };
1732
1733        let target_id = self.ssas[idx].ssa_id;
1734        tracing::info!(
1735            %target_id,
1736            available,
1737            remaining,
1738            "draining buffered SURBs into the funded cycle of a session closed at the exit"
1739        );
1740
1741        self.draining = true;
1742        // A deferred request can never be retried now, and leaving the flag set would have
1743        // `retry_deferred_successor_request` call into the (now silent) request path every event.
1744        self.successor_request_deferred = false;
1745
1746        // Every sibling is behind the target in index order — the target *is* `earliest_live_idx` —
1747        // so removing them cannot move the front, and the tail of `handle_event` re-derives the same
1748        // gate state and share-order front it already had.
1749        let mut retired = Vec::new();
1750        self.ssas.retain(|ssa| {
1751            if ssa.ssa_id == target_id || ssa.is_terminal() {
1752                true
1753            } else {
1754                retired.push(ssa.ssa_id);
1755                false
1756            }
1757        });
1758
1759        retired
1760            .into_iter()
1761            .map(|ssa_id| {
1762                self.note_retired_index(ssa_id.ssa_index());
1763                SessionPixAction::RetireSsa(ssa_id)
1764            })
1765            .collect()
1766    }

1772    /// Close the SSA at `idx` and return close actions.
1773    fn close_ssa_and_collect(
1774        &mut self,
1775        idx: usize,
1776        reason: SessionPixCloseReason,
1777        served_total: u64,
1778        now: Instant,
1779    ) -> Vec<SessionPixAction> {
1780        if matches!(self.ssas[idx].phase, SsaPhase::Closing) {
1781            return Vec::new();
1782        }
1783
1784        // Track the first close reason so it isn't lost when multiple SSAs close.
1785        self.first_failure_reason.get_or_insert(reason);
1786        let close_reason = self.first_failure_reason.unwrap_or(reason);
1787
1788        // Charged before the branches below, because the whole point is a count that outlives the
1789        // record being retired. Every path into this function is a cycle lost without recovering —
1790        // a recovered cycle leaves through tombstone expiry in `handle_deadline`, not through here.
1791        self.failed_cycles += 1;
1792        // Same transition, so the same guard covers both: the `Closing` check at the top of this
1793        // function is what makes a repeated close of one cycle count once. The summary rides along
1794        // for the same reason — this is the one place a cycle is retired without recovering, and
1795        // the counters are still intact because nothing below has cleared them yet.
1796        self.telemetry.failed = self.telemetry.failed.saturating_add(1);
1797        let summary = self.ssas[idx].telemetry_summary(PixCycleOutcome::Failed, served_total);
1798        self.telemetry.finalized.push(summary);
1799
1800        // Warn-level diagnostic with full SSA state before closing.
1801        let ssa = &self.ssas[idx];
1802        tracing::warn!(
1803            ssa_id = %ssa.ssa_id,
1804            ?reason,
1805            phase = ?ssa.phase,
1806            largest_useful_shares = ssa.largest_useful_shares,
1807            target_useful_shares = ssa.target_useful_shares,
1808            recovered_polynomials = ssa.recovered_polynomials,
1809            served_total_at_last_progress = ssa.served_total_at_last_progress,
1810            ?ssa.commitment_deadline,
1811            ?ssa.deposit_deadline,
1812            ?ssa.recovery_idle_deadline,
1813            ?ssa.recovery_hard_deadline,
1814            "closing PIX SSA"
1815        );
1816
1817        let phase_sample = self.ssas[idx].enter_phase(SsaPhase::Closing, now);
1818        self.telemetry.phase_durations.extend(phase_sample);
1819
1820        if self.ssas.len() == 1 {
1821            self.closed = true;
1822            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1823        }
1824
1825        // A batch has siblings to fall back on, which is what makes retiring one member survivable —
1826        // but only up to a point. Past the limit the Session goes, with the *first* failure's reason
1827        // rather than this one's, so the report names the cause instead of the last symptom.
1828        //
1829        // Strictly greater: the field is how many failures are *tolerated*, so the shipping value of
1830        // one keeps the existing "retire the member and carry on" behaviour for a batch's first loss
1831        // — including handing on the successor gate below — and closes on the second.
1832        if self.failed_cycles > self.cfg.max_failed_cycles.max(1) {
1833            tracing::warn!(
1834                failed_cycles = self.failed_cycles,
1835                max_failed_cycles = self.cfg.max_failed_cycles,
1836                ?close_reason,
1837                "closing PIX session: too many cycles lost without recovering"
1838            );
1839            self.closed = true;
1840            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1841        }
1842
1843        // Clear deadlines on this SSA.
1844        self.ssas[idx].commitment_deadline = None;
1845        self.ssas[idx].deposit_deadline = None;
1846        self.ssas[idx].recovery_idle_deadline = None;
1847        self.ssas[idx].recovery_hard_deadline = None;
1848
1849        // If all SSAs are terminal, close the session.
1850        if self
1851            .ssas
1852            .iter()
1853            .all(|s| matches!(s.phase, SsaPhase::Closing | SsaPhase::Recovered { .. }))
1854        {
1855            self.closed = true;
1856            return self.with_fill_stopped(vec![SessionPixAction::Close(close_reason)]);
1857        }
1858
1859        // Remove this closing SSA and emit RetireSsa so the reconstructor
1860        // releases its builder/verifier/counter state mid-session.
1861        // Record the retired index to prevent stale SsaRequestSent events
1862        // from resurrecting it.
1863        let retired = self.ssas[idx].ssa_id;
1864        self.note_retired_index(retired.ssa_index());
1865        let carried_successor_gate = self.ssas[idx].is_batch_last && !self.ssas[idx].next_requested;
1866        self.ssas.remove(idx);
1867
1868        // Retiring the cycle that carried the successor gate would strand the batch: its surviving
1869        // siblings are barred from asking, so nothing would ever request a replacement and the
1870        // Session would die on a recovery timeout that names the timer rather than the cause. Hand
1871        // the gate to the newest cycle still standing, which is now the last of the batch that will
1872        // actually serve its quota — so the request still cannot come early. `next_requested` guards
1873        // it: a gate already spent must not be handed on and fire a second time.
1874        if carried_successor_gate
1875            && let Some(newest) = self
1876                .ssas
1877                .iter_mut()
1878                .filter(|s| !s.is_terminal())
1879                .max_by_key(|s| s.ssa_id.ssa_index())
1880        {
1881            tracing::debug!(
1882                %retired, promoted = %newest.ssa_id,
1883                "successor gate handed on from a retired cycle"
1884            );
1885            newest.is_batch_last = true;
1886        }
1887        vec![SessionPixAction::RetireSsa(retired)]
1888    }
1889
1890    /// Allocates the next batch of SSA indices and asks for all of them in one action.
1891    ///
1892    /// The batch size is [`SupervisorConfig::ssas_per_request`], clamped rather than trusted since a
1893    /// supervisor can be built from a config that never went through `validate_pix_supervision`.
1894    ///
1895    /// Exactly one cycle of the batch — the last one allocated — carries
1896    /// [`PerSsaState::is_batch_last`], and only that cycle may ask for the successor batch. Without
1897    /// it every member would ask for one of its own, since the request flags are per-cycle; see that
1898    /// field for why the last rather than the first.
1899    ///
1900    /// Before allocating, this enforces both halves of the admission reservation: no more than
1901    /// [`crate::MAX_OVERLAPPING_BATCHES`] live generations and no more than that many full batches'
1902    /// live cycles. A request earned while the reservation is full is retried after a generation
1903    /// releases its reconstructor state.
1904    ///
1905    /// What batching does change is the exposure *within* a batch: every cycle in it is unfunded at
1906    /// once, so the ceiling is `ssas_per_request` SSA quotas rather than one. That is the trade the
1907    /// knob exists to make, and it is why both deadlines are scaled by the same factor.
1908    fn emit_request_next_ssa(&mut self, now: Instant) -> Vec<SessionPixAction> {
1909        // A drain has no Session left to serve a successor's quota, so asking for one would commit
1910        // the Entry to a deposit against a cycle that could never be served. Guarded here rather than
1911        // at each of the four call sites — the early signal, the recovered transition, the deferred
1912        // retry and the deposit replay — so no route into a request can be missed.
1913        if self.draining {
1914            return Vec::new();
1915        }
1916
1917        let batch = self.cfg.ssas_per_request.clamp(1, crate::MAX_SSA_BATCH_SIZE);
1918        let live_cycles = self.live_cycle_count();
1919        let live_batches = self.live_batch_count();
1920        if live_batches >= crate::MAX_OVERLAPPING_BATCHES as usize
1921            || live_cycles.saturating_add(batch) > self.reserved_cycle_slots()
1922        {
1923            self.successor_request_deferred = true;
1924            tracing::debug!(
1925                live_cycles,
1926                live_batches,
1927                requested = batch,
1928                reserved = self.reserved_cycle_slots(),
1929                "deferring successor SSA batch until an older batch is released"
1930            );
1931            return Vec::new();
1932        }
1933
1934        self.successor_request_deferred = false;
1935        let batch_id = self.next_ssa_index;
1936        let mut ssa_ids = Vec::with_capacity(batch);
1937
1938        for _ in 0..batch {
1939            let index = self.next_ssa_index;
1940
1941            let Ok(ssa_index) = SsaIndex::try_from(index) else {
1942                break;
1943            };
1944            let Some(next) = index.checked_add(1) else {
1945                // Index space exhausted mid-batch. Stop here rather than wrapping: a reused index
1946                // would collide with a live cycle. Whatever was allocated still goes out.
1947                break;
1948            };
1949            self.next_ssa_index = next;
1950
1951            let ssa_id = SsaId::new(self.pseudonym, ssa_index);
1952
1953            // Register each SSA now rather than on `SsaRequestSent`. Carrying out the action registers
1954            // an Exit commitment, which is what makes shares for that SSA processable — so
1955            // observations about it can reach us before the confirmation that we asked for it does.
1956            // Every handler ignores an SSA it has no record of, and for `UnverifiableShares` that
1957            // would mean failing open on exactly the signal that must fail closed.
1958            //
1959            // No deadline yet: the request has not gone out, so there is nothing to be late for. That
1960            // is what `SsaRequestSent` adds, per index.
1961            self.ssas.push(PerSsaState::new(
1962                ssa_id,
1963                batch_id,
1964                self.dims.target_useful_shares(),
1965                now,
1966            ));
1967            // Counted at allocation rather than at `SsaRequestSent`, for the same reason the record
1968            // is created here: a cycle is registered before its request goes out, and observations
1969            // about it can arrive before the confirmation that it was asked for. A batch truncated
1970            // by index exhaustion counts only what it allocated.
1971            self.telemetry.requested = self.telemetry.requested.saturating_add(1);
1972            ssa_ids.push(ssa_id);
1973        }
1974
1975        // Nothing allocated means the index space is spent, and no further cycle can ever be funded.
1976        if ssa_ids.is_empty() {
1977            self.closed = true;
1978            return vec![SessionPixAction::Close(SessionPixCloseReason::InvalidTransition)];
1979        }
1980
1981        // The last cycle *actually* allocated carries the successor gate, which is what makes a
1982        // batch truncated by index exhaustion still able to ask for the next one.
1983        self.ssas
1984            .last_mut()
1985            .expect("a non-empty batch has just been pushed")
1986            .is_batch_last = true;
1987
1988        vec![SessionPixAction::RequestSsa {
1989            ssa_ids,
1990            params: self.dims,
1991        }]
1992    }

1994    fn find_ssa_idx(&self, ssa_id: &SsaId<HoprPseudonym>) -> Option<usize> {
1995        self.ssas.iter().position(|s| s.ssa_id == *ssa_id)
1996    }

2004    /// Check which deadline expired, if any.
2005    fn check_deadlines(&self, now: Instant) -> Option<SessionPixCloseReason> {
2006        match self.phase {
2007            SsaPhase::AwaitingCommitment => self
2008                .commitment_deadline
2009                .filter(|d| now >= *d)
2010                .map(|_| SessionPixCloseReason::CommitmentTimeout),
2011            SsaPhase::AwaitingDeposit => self
2012                .deposit_deadline
2013                .filter(|d| now >= *d)
2014                .map(|_| SessionPixCloseReason::DepositTimeout),
2015            SsaPhase::Recovering => {
2016                // Hard deadline is immutable.
2017                if let Some(d) = self.recovery_hard_deadline
2018                    && now >= d
2019                {
2020                    return Some(SessionPixCloseReason::RecoveryDeadline);
2021                }
2022                // Idle deadline — service gating happens in handle_deadline.
2023                if let Some(d) = self.recovery_idle_deadline
2024                    && now >= d
2025                {
2026                    return Some(SessionPixCloseReason::RecoveryIdle);
2027                }
2028                None
2029            }
2030            SsaPhase::Recovered { .. } | SsaPhase::Closing => None,
2031        }
2032    }

4386    ///
4387    /// Retiring a member and carrying on is what keeps one bad cycle from costing a whole Session,
4388    /// but it leaves no trace on the survivors — so without a count that outlives the retirement an
4389    /// Entry can lose one cycle per batch forever, paying for a fraction of what it is served. The
4390    /// counter is cumulative for exactly that reason: here the two losses are in *different* batches,
4391    /// and nothing in a per-batch view would connect them.
4392    ///
4393    /// The close carries the *first* failure's reason, so the report names the cause rather than the
4394    /// last symptom.

4396    fn the_second_lost_cycle_closes_a_batched_session() {

4398
4399        let p = pseudonym();

4401            ssas_per_request: BATCH,
4402            max_failed_cycles: 1,
4403            ..default_cfg()
4404        };
4405        let (mut sup, _) = SessionPixSupervisor::new(cfg, dims(10, 5), p, Instant::now());
4406        let now = Instant::now();
4407
4408        // Cycles 1 and 2 are funded and recovering; 3 and 4 are stuck awaiting their deposits, which
4409        // is the state an unpaid cycle is lost from.
4410        fund_batch(&mut sup, p, 2, now);
4411        commit_unfunded(&mut sup, p, 3..=BATCH as u32, now);

6556    /// Feeds one snapshot and returns the `(useful, surplus)` it booked.
6557    fn feed(sup: &mut SessionPixSupervisor, id: SsaId<HoprPseudonym>, useful: u64, seen: u64) -> (u64, u64) {
6558        let target = sup.dims.target_useful_shares();
6559        let polys = sup.dims.polys_per_ssa();
6560        // `recovered_polynomials` is scaled with `useful` so the snapshot stays internally
6561        // consistent; the validator rejects one that is not, and this test is about deltas.
6562        let recovered = ((useful * polys as u64) / target.max(1)) as u16;
6563        sup.handle_event(
6564            &SessionPixEvent::RecoveryProgress(make_progress_seen(id, useful, seen, target, recovered)),
6565            Instant::now(),
6566            0,
6567        );
6568        let events = sup.take_telemetry_events();
6569        (events.useful_shares, events.surplus_shares)
6570    }

6572    /// Absolute snapshots become positive deltas, and every degenerate ordering adds zero.
6573    ///
6574    /// The property the issue names: `shares_seen` is absolute and acknowledgement batches arrive
6575    /// out of order or coalesced, so adding it to a counter would over-count wildly. What is added
6576    /// is the difference against a high-water mark.
6577    #[test]
6578    fn absolute_snapshots_become_positive_deltas() {
6579        let t0 = Instant::now();
6580        let p = pseudonym();
6581        let mut sup = recovering_session(p, t0);
6582        let id = ssa_id(p, 1);
6583
6584        assert_eq!((10, 0), feed(&mut sup, id, 10, 10), "the first snapshot is all delta");
6585        assert_eq!((5, 0), feed(&mut sup, id, 15, 15), "and the next is the difference");
6586
6587        // Repeated: the same snapshot again.
6588        assert_eq!((0, 0), feed(&mut sup, id, 15, 15), "a repeated snapshot adds nothing");
6589        // Stale: reordered behind one already seen.
6590        assert_eq!((0, 0), feed(&mut sup, id, 12, 12), "a stale snapshot adds nothing");
6591        // Still stale on the liveness counter even though useful is unchanged.
6592        assert_eq!(
6593            (0, 0),
6594            feed(&mut sup, id, 15, 14),
6595            "an older liveness count adds nothing"
6596        );
6597
6598        // Coalesced: a jump standing in for snapshots that were dropped. The whole missing delta
6599        // arrives with it, because a high-water mark is a maximum and not a running sum.
6600        assert_eq!(
6601            (85, 0),
6602            feed(&mut sup, id, 100, 100),
6603            "a jump carries the missing delta"
6604        );
6605    }

6630    /// Useful and surplus always partition the accepted delta, over an arbitrary sequence.
6631    ///
6632    /// Stated as an invariant over a generated sequence rather than as another worked example,
6633    /// because the failure it guards against is an accounting slip in one of the branches rather
6634    /// than a wrong answer to one case.
6635    #[test]
6636    fn useful_and_surplus_always_partition_the_accepted_delta() {
6637        let t0 = Instant::now();
6638        let p = pseudonym();
6639        let mut sup = recovering_session(p, t0);
6640        let id = ssa_id(p, 1);
6641        let target = sup.dims.target_useful_shares();
6642        // The greatest liveness count these dimensions can produce: anything above it is a protocol
6643        // violation the validator closes on, so a walk must stay inside it to be booking deltas at
6644        // all rather than testing the rejection path.
6645        let max_seen = max_cycle_shares(&sup);
6646
6647        let mut booked_useful = 0;
6648        let mut booked_surplus = 0;
6649        let mut highest_seen = 0;
6650
6651        // A deterministic walk that revisits, regresses and jumps, which is what real delivery does.
6652        for step in [
6653            (5u64, 5u64),
6654            (5, 9),
6655            (3, 4),
6656            (9, 9),
6657            (9, 20),
6658            (9, 20),
6659            (30, 41),
6660            (12, 12),
6661            (target, max_seen),
6662            (target, max_seen),
6663        ] {
6664            let (useful, seen) = step;
6665            let (u, s) = feed(&mut sup, id, useful, seen);
6666            assert_eq!(
6667                seen.saturating_sub(highest_seen),
6668                u + s,
6669                "snapshot {step:?} must book exactly its accepted delta"
6670            );
6671            highest_seen = highest_seen.max(seen);
6672            booked_useful += u;
6673            booked_surplus += s;
6674        }
6675
6676        assert_eq!(
6677            highest_seen,
6678            booked_useful + booked_surplus,
6679            "over the whole sequence the two must sum to the greatest liveness count seen"
6680        );
6681        assert_eq!(
6682            target, booked_useful,
6683            "and the useful half must sum to the greatest useful count seen"
6684        );
6685    }
