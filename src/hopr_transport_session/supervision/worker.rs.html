1//! Per-session actor for the [`SessionPixSupervisor`].
2//!
3//! The worker serializes lifecycle events through the deterministic core,
4//! manages the deadline timer, applies safety-critical gate actions locally,
5//! and forwards actions to an external I/O driver.
6//!
7//! Runtime-agnostic: uses crossfire channels and the runtime prelude from
8//! `hopr_utils` so no direct tokio dependency (tests use tokio freely).
9

39/// Cloneable handle to a running `SessionPixWorker`.
40#[derive(Clone)]
41pub struct SessionPixSupervisorHandle {
42    cmd_tx: CmdTx,
43    pub(crate) gate: Arc<ServiceGate>,
44    /// This Session's share of the node-level PIX aggregates.
45    ///
46    /// Carried here rather than on `SessionSlot` because the slot already holds this handle, and
47    /// both parties that need it can reach it: the worker publishes through its own clone, and
48    /// `close_session` releases through the slot's.
49    pub(crate) telemetry: Arc<PixSessionTelemetry>,
50}
51
52impl SessionPixSupervisorHandle {
53    /// Send a PIX event to the supervisor, awaiting capacity if the channel is
54    /// full.
55    ///
56    /// Returns `Err` if the worker is no longer running. Backpressures instead
57    /// of dropping events, so overflow cannot occur by construction.
58    pub async fn send_event(&self, ev: SessionPixEvent) -> Result<(), ()> {
59        match self.cmd_tx.send(WorkerCommand::Event(ev)).await {
60            Ok(()) => Ok(()),
61            Err(SendError(_)) => {
62                tracing::warn!("PIX supervisor command channel closed");
63                Err(())
64            }
65        }
66    }
67
68    /// Deliver a PIX event without blocking, dropping it if the channel is full.
69    ///
70    /// Returns `true` if the event was queued.
71    ///
72    /// The synchronous counterpart to [`send_event`](Self::send_event), and it exists because one
73    /// caller cannot await: `close_session_with_reason` is a plain `fn`, reached from the manager's
74    /// closure-notification task — which `HoprSession::poll_close` and the empty-read path feed
75    /// through a non-blocking channel — from the public synchronous `SessionManager::close_session`,
76    /// and from `handle_session_error` on a peer `SessionError`. None of them may park on a worker
77    /// that happens to be mid-tick.
78    ///
79    /// A refusal is not lost work, which is what makes dropping acceptable here where it would not be
80    /// for a lifecycle transition. The only event sent this way is
81    /// [`SessionPixEvent::SessionClosed`], and its caller treats `false` as "this close was not
82    /// answered by the supervisor" and falls back to the immediate teardown that has always been the
83    /// behaviour on that path.
84    pub fn try_send_event(&self, ev: SessionPixEvent) -> bool {
85        self.cmd_tx
86            .try_send(WorkerCommand::Event(ev))
87            .inspect_err(|e| match e {
88                TrySendError::Full(_) => tracing::debug!("supervisor command channel full — dropping event"),
89                TrySendError::Disconnected(_) => tracing::warn!("PIX supervisor command channel closed"),
90            })
91            .is_ok()
92    }
93
94    /// Send an action result feedback to the supervisor, awaiting capacity if
95    /// the channel is full.
96    pub async fn send_action_result(&self, action: SessionPixAction, ok: bool) -> Result<(), ()> {
97        match self.cmd_tx.send(WorkerCommand::ActionResult { action, ok }).await {
98            Ok(()) => Ok(()),
99            Err(SendError(_)) => {
100                tracing::warn!("PIX supervisor result channel closed");
101                Err(())
102            }
103        }
104    }
105
106    /// Deliver a recovery-progress snapshot without blocking, dropping it if the channel is full.
107    ///
108    /// Returns `true` if the snapshot was queued.
109    ///
110    /// Unlike every other event, progress arrives once per acknowledgement batch rather than once
111    /// per SSA lifecycle transition, so it is the one input whose rate is set by traffic. Awaiting
112    /// capacity for it would put the supervisor's scheduling latency on the acknowledgement path —
113    /// where a stall costs share verification throughput for the whole Session.
114    ///
115    /// Dropping is safe by construction rather than by tolerance: snapshots carry absolute counters,
116    /// [`on_recovery_progress`](super::supervisor::SessionPixSupervisor) keeps the maximum it has
117    /// seen, and a lower or repeated `useful_shares` is already treated as benign — concurrent
118    /// acknowledgement batches make out-of-order snapshots ordinary. A dropped snapshot is therefore
119    /// indistinguishable from one that arrived out of order, and the next batch supersedes it.
120    ///
121    /// The one thing a drop delays is the [`ProgressNotification`](SessionPixAction::ProgressNotification)
122    /// that resets the gate's served-without-progress ceiling, so `max_served_without_progress` has
123    /// to leave room for a few missed snapshots. Since the command channel is drained by a state
124    /// machine that performs no I/O, drops require a genuinely wedged worker.
125    pub fn try_send_progress(&self, progress: SsaRecoveryProgress<HoprPseudonym>) -> bool {
126        self.cmd_tx
127            .try_send(WorkerCommand::Event(SessionPixEvent::RecoveryProgress(progress)))
128            .inspect_err(|e| match e {
129                TrySendError::Full(_) => tracing::debug!(
130                    ssa_id = %progress.ssa_id,
131                    "supervisor command channel full — dropping progress snapshot"
132                ),
133                TrySendError::Disconnected(_) => tracing::warn!("PIX supervisor command channel closed"),
134            })
135            .is_ok()
136    }
137}

152/// Spawn a supervisor worker and return its handle and action driver receiver.
153pub fn spawn_supervisor_worker(
154    cfg: SupervisorConfig,
155    dims: PixParams,
156    pseudonym: hopr_api::types::internal::prelude::HoprPseudonym,
157    now: Instant,
158) -> (SessionPixSupervisorHandle, ActionRx) {
159    let (cmd_tx, cmd_rx) = crossfire::mpsc::bounded_async::<WorkerCommand>(64);
160    let (action_tx, action_rx) = crossfire::mpsc::bounded_async::<SessionPixAction>(64);
161
162    let predeposit_budget = std::cmp::min(
163        dims.target_useful_shares().saturating_sub(1),
164        cfg.max_predeposit_packets,
165    );
166    let gate = ServiceGate::new(predeposit_budget, cfg.max_served_without_progress);
167    let telemetry = Arc::new(PixSessionTelemetry::new(gate.clone()));
168
169    let handle = SessionPixSupervisorHandle {
170        cmd_tx,
171        gate: gate.clone(),
172        telemetry: telemetry.clone(),
173    };
174
175    let (mut supervisor, initial_actions) = SessionPixSupervisor::new(cfg, dims, pseudonym, now);
176
177    // Published here rather than left to the worker's first turn, because the worker is spawned and
178    // this function returns before it is scheduled. A Session torn down inside that window —
179    // establishment failing after the supervisor is stood up is the realistic way — would have its
180    // telemetry released before anything was ever published, and the first batch's `requested`
181    // count would be lost along with the census it belonged to.
182    //
183    // Correct to do before `dispatch`: a fresh supervisor has no funded front, so the initial
184    // actions change neither the gate mode nor its exposure, and the worker's own publish after
185    // dispatch then finds nothing to move.
186    publish_telemetry(&mut supervisor, &gate, &telemetry);
187
188    spawn(worker_loop(
189        supervisor,
190        cmd_rx,
191        action_tx,
192        gate,
193        telemetry,
194        initial_actions,
195    ));
196
197    (handle, action_rx)
198}

200// ---------------------------------------------------------------------------
201// Worker loop
202// ---------------------------------------------------------------------------
203
204async fn worker_loop(
205    mut supervisor: SessionPixSupervisor,
206    cmd_rx: CmdRx,
207    action_tx: ActionTx,
208    gate: Arc<ServiceGate>,
209    telemetry: Arc<PixSessionTelemetry>,
210    initial_actions: Vec<SessionPixAction>,
211) {
212    // Emit initial actions. `false` says only that a freshly built supervisor has not flagged
213    // itself, which it cannot have: `new` does not close. It is not a claim that these actions are
214    // non-terminal — `dispatch` reads the payload for that, so a construction-time `Close` would
215    // still fail closed here rather than being waved through on the strength of the flag.
216    let keep_running = dispatch(&initial_actions, false, &action_tx, &gate);
217    publish_telemetry(&mut supervisor, &gate, &telemetry);
218    if !keep_running {
219        return;
220    }
221
222    loop {
223        // Two independent timers, one wake-up. The deadline sweep is the lifecycle's, the fill tick
224        // is the rate planner's, and neither may starve the other — so the loop always sleeps until
225        // the earlier of the two and lets `handle_timers` decide which of them was actually due.
226        // `next_fill_tick` is `None` unless there is something to plan or something to stop, so a
227        // Session with no funded cycle waits on its command channel exactly as it did before.
228        let deadline = match (supervisor.next_deadline(), supervisor.next_fill_tick()) {
229            (Some(dl), Some(tick)) => Some(dl.min(tick)),
230            (dl, tick) => dl.or(tick),
231        };
232
233        if let Some(dl) = deadline {
234            let now = Instant::now();
235            if now >= dl {
236                let actions = supervisor.handle_timers(now, gate.served_total());
237                report_fill_stall(&mut supervisor);
238                let keep_running = dispatch(&actions, supervisor.closed, &action_tx, &gate);
239                publish_telemetry(&mut supervisor, &gate, &telemetry);
240                if !keep_running {
241                    return;
242                }
243                continue;
244            }
245
246            let duration = dl.saturating_duration_since(Instant::now());
247
248            match cmd_rx
249                .recv()
250                .timeout(futures_time::time::Duration::from(duration))
251                .await
252            {
253                Ok(result) => {
254                    if !process_cmd(result.ok(), &mut supervisor, &action_tx, &gate, &telemetry).await {
255                        return;
256                    }
257                }
258                Err(_) => {
259                    let now = Instant::now();
260                    let actions = supervisor.handle_timers(now, gate.served_total());
261                    report_fill_stall(&mut supervisor);
262                    let keep_running = dispatch(&actions, supervisor.closed, &action_tx, &gate);
263                    publish_telemetry(&mut supervisor, &gate, &telemetry);
264                    if !keep_running {
265                        return;
266                    }
267                }
268            }
269        } else {
270            let cmd = cmd_rx.recv().await.ok();
271            if !process_cmd(cmd, &mut supervisor, &action_tx, &gate, &telemetry).await {
272                return;
273            }
274        }
275    }
276}

278/// Counts a fill stall, if one just began.
279///
280/// The planner cannot do this itself — it is a pure function of its inputs, and a metric is neither —
281/// so the edge is latched there and drained here, in the actor that is allowed to have effects. Not
282/// folded into `dispatch`, because the stall is not an action: it produces no work for the driver,
283/// only a number for the operator.
284fn report_fill_stall(supervisor: &mut SessionPixSupervisor) {
285    if supervisor.take_fill_stall() {
286        #[cfg(feature = "telemetry")]
287        crate::telemetry::pix::record_pix_fill_backoff(crate::telemetry::pix::PixFillBackoff::Stalled);
288    }
289}

291/// Publishes this Session's share of the node-level PIX aggregates.
292///
293/// Called once per turn, after [`dispatch`] rather than before it, and on the terminal path as well
294/// as the ordinary one. Both halves of that matter:
295///
296/// * *after*, because `dispatch` applies the gate transitions locally before sending anything, and the gate is where
297///   the predeposit exposure is read from. Publishing first would report a rotating Session's exhausted predecessor
298///   allowance for one more turn.
299/// * *on the terminal path too*, because that turn is the one that carries the last `failed`/`recovered` counts, and
300///   the worker is about to stop. Returning the gauges to zero is not this function's job —
301///   [`PixSessionTelemetry::release`] does that from `close_session`, which knows the Session is gone rather than
302///   merely that this worker has stopped.
303///
304/// Kept out of `dispatch` for the reason `report_fill_stall` is: this produces no work for the
305/// action driver, only numbers for the operator, and `dispatch`'s callers hold the supervisor by
306/// different borrows.
307fn publish_telemetry(
308    supervisor: &mut SessionPixSupervisor,
309    gate: &Arc<ServiceGate>,
310    telemetry: &Arc<PixSessionTelemetry>,
311) {
312    let snapshot = supervisor.telemetry_snapshot(gate.predeposit_exposure());
313    telemetry.publish(snapshot, supervisor.take_telemetry_events());
314}

316/// Handle a received command from the handle.
317///
318/// Returns `false` to signal the worker loop to stop.
319async fn process_cmd(
320    cmd: Option<WorkerCommand>,
321    supervisor: &mut SessionPixSupervisor,
322    action_tx: &ActionTx,
323    gate: &Arc<ServiceGate>,
324    telemetry: &Arc<PixSessionTelemetry>,
325) -> bool {
326    let cmd = match cmd {
327        Some(c) => c,
328        None => {
329            // All senders dropped — close. Terminal by construction, so `dispatch` is told the
330            // supervisor is closed and its `false` is the verdict.
331            //
332            // Nothing is published: this close is synthesized here rather than by the state machine,
333            // so no transition was latched and the census is whatever the previous turn already
334            // reported. `release` charges its live cycles when the Session is torn down.
335            let actions = vec![SessionPixAction::Close(SessionPixCloseReason::SupervisorUnavailable)];
336            return dispatch(&actions, true, action_tx, gate);
337        }
338    };
339
340    let keep_running = match cmd {
341        WorkerCommand::Event(ev) => {
342            let now = Instant::now();
343            let actions = supervisor.handle_event(&ev, now, gate.served_total());
344            dispatch(&actions, supervisor.closed, action_tx, gate)
345        }
346        WorkerCommand::ActionResult { action, ok } => {
347            let now = Instant::now();
348            let actions = supervisor.action_result(&action, ok, now);
349            dispatch(&actions, supervisor.closed, action_tx, gate)
350        }
351    };
352
353    publish_telemetry(supervisor, gate, telemetry);
354    keep_running
355}

357/// Apply gate control, forward `actions`, then report whether the worker should keep running.
358///
359/// The four places that produce actions — the two deadline paths in [`worker_loop`] and the two
360/// command arms in [`process_cmd`] — all had the same four-way branch spelled out: send, check
361/// whether the supervisor closed, poison, return. Fail-close is the property that matters most here
362/// and it was stated four times, so a correction to it had to be made four times.
363///
364/// Poisons the gate on every terminal path, so no caller has to remember to. `closed` is passed
365/// rather than read from the supervisor because the callers hold it by different borrows.
366///
367/// Terminality is derived from the *actions* as well as from `closed`, because the two do not
368/// always agree: several handlers return [`Close`](SessionPixAction::Close) without setting the
369/// supervisor's flag — both `InvalidTransition` paths in `on_ssa_request_sent`, `CounterRegression`
370/// and the share-order verdict in `on_recovery_progress`, and `on_unverifiable_shares`. Two of those
371/// are the anti-abuse verdicts. Reading only the flag would send the close, return `true`, and leave
372/// the gate open for as long as it takes the action driver to poison it on receipt — which is a real
373/// but bounded window, not a correct one. `Close` is also the one action the driver never reports
374/// back, so `action_result`'s own close path cannot cover for it.
375fn dispatch(actions: &[SessionPixAction], closed: bool, action_tx: &ActionTx, gate: &Arc<ServiceGate>) -> bool {
376    // Gate control is local state, not I/O. Apply it before touching the action channel so a safety
377    // transition cannot wait behind commitment generation, deposit-data lookup, or a network send in
378    // the action driver. Stop at Close to preserve the driver's rule that actions after a close are
379    // unreachable; poisoning below is stronger than every gate mode.
380    for action in actions
381        .iter()
382        .take_while(|action| !matches!(action, SessionPixAction::Close(_)))
383    {
384        match action {
385            SessionPixAction::ReleaseService => gate.release_service(),
386            SessionPixAction::WithholdService => gate.withhold_service(),
387            SessionPixAction::ProgressNotification => gate.notify_progress(),
388            _ => {}
389        }
390    }
391
392    let terminal = closed || actions.iter().any(|a| matches!(a, SessionPixAction::Close(_)));
393
394    // Sent before the verdict either way: a closing supervisor's last actions carry the reason it
395    // closed, and dropping them would leave the driver to infer it.
396    let delivered = send_actions(actions, action_tx);
397    if terminal || !delivered {
398        gate.poison();
399        return false;
400    }
401    true
402}
403
404/// Non-blocking forward of actions to the driver.
405///
406/// On `Disconnected`, the driver is gone — returns `false` so the caller
407/// can fail-close.
408///
409/// On `Full`:
410/// - **Coalescible** actions (`ProgressNotification`) are logged + skipped. They are idempotent and safe to drop — the
411///   next notification will replace them, and dropping here prevents transient load from killing a healthy session.
412/// - **Non-coalescible** actions (every other variant) are treated as fatal. If these cannot be delivered the channel
413///   is genuinely wedged.
414fn send_actions(actions: &[SessionPixAction], action_tx: &ActionTx) -> bool {
415    for action in actions {
416        match action_tx.try_send(action.clone()) {
417            Ok(()) => continue,
418            Err(TrySendError::Full(item)) => {
419                if is_coalescible(&item) {
420                    tracing::trace!(?action, "action channel full, dropping coalescible action");
421                    continue;
422                }
423                tracing::warn!(?action, "non-coalescible action dropped — channel full");
424                return false;
425            }
426            Err(TrySendError::Disconnected(_item)) => {
427                tracing::warn!(?action, "action driver disconnected");
428                return false;
429            }
430        }
431    }
432    true
433}
434
435/// Returns `true` for actions that are safe to drop when the action channel
436/// is transiently full.
437fn is_coalescible(action: &SessionPixAction) -> bool {
438    matches!(action, SessionPixAction::ProgressNotification)
439}

530    /// Wait for `condition` to hold, failing the test if it never does.
531    ///
532    /// The worker is a detached task, so every assertion about its effects is really an assertion
533    /// that it has been scheduled. A fixed sleep encodes a guess about the scheduler, and the guess
534    /// is wrong under CI load or on a busy single-core runner — the test then fails with nothing
535    /// wrong in the code. Polling asserts the same property and can only fail if it genuinely never
536    /// happens, which is why the deadline is generous: it is reached only on a real failure.
537    async fn poll_until(what: &str, mut condition: impl FnMut() -> bool) {
538        const DEADLINE: Duration = Duration::from_secs(5);
539        tokio::time::timeout(DEADLINE, async {
540            while !condition() {
541                tokio::time::sleep(Duration::from_millis(5)).await;
542            }
543        })
544        .await
545        .unwrap_or_else(|_| panic!("{what} — not observed within {DEADLINE:?}"));
546    }
