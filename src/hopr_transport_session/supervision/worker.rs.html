1//! Per-session actor for the [`SessionPixSupervisor`].
2//!
3//! The worker serializes lifecycle events through the deterministic core,
4//! manages the deadline timer, applies safety-critical gate actions locally,
5//! and forwards actions to an external I/O driver.
6//!
7//! Runtime-agnostic: uses crossfire channels and the runtime prelude from
8//! `hopr_utils` so no direct tokio dependency (tests use tokio freely).

39/// Cloneable handle to a running `SessionPixWorker`.
40#[derive(Clone)]
41pub struct SessionPixSupervisorHandle {
42    cmd_tx: CmdTx,
43    pub(crate) gate: Arc<ServiceGate>,
44}
45
46impl SessionPixSupervisorHandle {
47    /// Send a PIX event to the supervisor, awaiting capacity if the channel is
48    /// full.
49    ///
50    /// Returns `Err` if the worker is no longer running. Backpressures instead
51    /// of dropping events, so overflow cannot occur by construction.
52    pub async fn send_event(&self, ev: SessionPixEvent) -> Result<(), ()> {
53        match self.cmd_tx.send(WorkerCommand::Event(ev)).await {
54            Ok(()) => Ok(()),
55            Err(SendError(_)) => {
56                tracing::warn!("PIX supervisor command channel closed");
57                Err(())
58            }
59        }
60    }
61
62    /// Send an action result feedback to the supervisor, awaiting capacity if
63    /// the channel is full.
64    pub async fn send_action_result(&self, action: SessionPixAction, ok: bool) -> Result<(), ()> {
65        match self.cmd_tx.send(WorkerCommand::ActionResult { action, ok }).await {
66            Ok(()) => Ok(()),
67            Err(SendError(_)) => {
68                tracing::warn!("PIX supervisor result channel closed");
69                Err(())
70            }
71        }
72    }
73
74    /// Deliver a recovery-progress snapshot without blocking, dropping it if the channel is full.
75    ///
76    /// Returns `true` if the snapshot was queued.
77    ///
78    /// Unlike every other event, progress arrives once per acknowledgement batch rather than once
79    /// per SSA lifecycle transition, so it is the one input whose rate is set by traffic. Awaiting
80    /// capacity for it would put the supervisor's scheduling latency on the acknowledgement path —
81    /// where a stall costs share verification throughput for the whole Session.
82    ///
83    /// Dropping is safe by construction rather than by tolerance: snapshots carry absolute counters,
84    /// [`on_recovery_progress`](super::supervisor::SessionPixSupervisor) keeps the maximum it has
85    /// seen, and a lower or repeated `useful_shares` is already treated as benign — concurrent
86    /// acknowledgement batches make out-of-order snapshots ordinary. A dropped snapshot is therefore
87    /// indistinguishable from one that arrived out of order, and the next batch supersedes it.
88    ///
89    /// The one thing a drop delays is the [`ProgressNotification`](SessionPixAction::ProgressNotification)
90    /// that resets the gate's served-without-progress ceiling, so `max_served_without_progress` has
91    /// to leave room for a few missed snapshots. Since the command channel is drained by a state
92    /// machine that performs no I/O, drops require a genuinely wedged worker.
93    pub fn try_send_progress(&self, progress: SsaRecoveryProgress<HoprPseudonym>) -> bool {
94        match self
95            .cmd_tx
96            .try_send(WorkerCommand::Event(SessionPixEvent::RecoveryProgress(progress)))
97        {
98            Ok(()) => true,
99            Err(TrySendError::Full(_)) => {
100                tracing::debug!(
101                    ssa_id = %progress.ssa_id,
102                    "supervisor command channel full — dropping progress snapshot"
103                );
104                false
105            }
106            Err(TrySendError::Disconnected(_)) => {
107                tracing::warn!("PIX supervisor command channel closed");
108                false
109            }
110        }
111    }
112}

127/// Spawn a supervisor worker and return its handle and action driver receiver.
128pub fn spawn_supervisor_worker(
129    cfg: SupervisorConfig,
130    dims: PixParams,
131    pseudonym: hopr_api::types::internal::prelude::HoprPseudonym,
132    now: Instant,
133) -> (SessionPixSupervisorHandle, ActionRx) {
134    let (cmd_tx, cmd_rx) = crossfire::mpsc::bounded_async::<WorkerCommand>(64);
135    let (action_tx, action_rx) = crossfire::mpsc::bounded_async::<SessionPixAction>(64);
136
137    let predeposit_budget = std::cmp::min(
138        dims.target_useful_shares().saturating_sub(1),
139        cfg.max_predeposit_packets,
140    );
141    let gate = ServiceGate::new(predeposit_budget, cfg.max_served_without_progress);
142
143    let handle = SessionPixSupervisorHandle {
144        cmd_tx,
145        gate: gate.clone(),
146    };
147
148    let (supervisor, initial_actions) = SessionPixSupervisor::new(cfg, dims, pseudonym, now);
149
150    spawn(worker_loop(supervisor, cmd_rx, action_tx, gate, initial_actions));
151
152    (handle, action_rx)
153}

155// ---------------------------------------------------------------------------
156// Worker loop
157// ---------------------------------------------------------------------------
158
159async fn worker_loop(
160    mut supervisor: SessionPixSupervisor,
161    cmd_rx: CmdRx,
162    action_tx: ActionTx,
163    gate: Arc<ServiceGate>,
164    initial_actions: Vec<SessionPixAction>,
165) {
166    // Emit initial actions. `false` says only that a freshly built supervisor has not flagged
167    // itself, which it cannot have: `new` does not close. It is not a claim that these actions are
168    // non-terminal — `dispatch` reads the payload for that, so a construction-time `Close` would
169    // still fail closed here rather than being waved through on the strength of the flag.
170    if !dispatch(&initial_actions, false, &action_tx, &gate) {
171        return;
172    }
173
174    loop {
175        // Two independent timers, one wake-up. The deadline sweep is the lifecycle's, the fill tick
176        // is the rate planner's, and neither may starve the other — so the loop always sleeps until
177        // the earlier of the two and lets `handle_timers` decide which of them was actually due.
178        // `next_fill_tick` is `None` unless there is something to plan or something to stop, so a
179        // Session with no funded cycle waits on its command channel exactly as it did before.
180        let deadline = match (supervisor.next_deadline(), supervisor.next_fill_tick()) {
181            (Some(dl), Some(tick)) => Some(dl.min(tick)),
182            (dl, tick) => dl.or(tick),
183        };
184
185        if let Some(dl) = deadline {
186            let now = Instant::now();
187            if now >= dl {
188                let actions = supervisor.handle_timers(now, gate.served_total());
189                report_fill_stall(&mut supervisor);
190                if !dispatch(&actions, supervisor.closed, &action_tx, &gate) {
191                    return;
192                }
193                continue;
194            }
195
196            let duration = dl.saturating_duration_since(Instant::now());
197
198            match cmd_rx
199                .recv()
200                .timeout(futures_time::time::Duration::from(duration))
201                .await
202            {
203                Ok(result) => {
204                    if !process_cmd(result.ok(), &mut supervisor, &action_tx, &gate).await {
205                        return;
206                    }
207                }
208                Err(_) => {
209                    let now = Instant::now();
210                    let actions = supervisor.handle_timers(now, gate.served_total());
211                    report_fill_stall(&mut supervisor);
212                    if !dispatch(&actions, supervisor.closed, &action_tx, &gate) {
213                        return;
214                    }
215                }
216            }
217        } else {
218            let cmd = cmd_rx.recv().await.ok();
219            if !process_cmd(cmd, &mut supervisor, &action_tx, &gate).await {
220                return;
221            }
222        }
223    }
224}
225
226/// Counts a fill stall, if one just began.
227///
228/// The planner cannot do this itself — it is a pure function of its inputs, and a metric is neither —
229/// so the edge is latched there and drained here, in the actor that is allowed to have effects. Not
230/// folded into `dispatch`, because the stall is not an action: it produces no work for the driver,
231/// only a number for the operator.
232fn report_fill_stall(supervisor: &mut SessionPixSupervisor) {
233    if supervisor.take_fill_stall() {
234        #[cfg(feature = "telemetry")]
235        crate::telemetry::record_pix_fill_backoff(crate::telemetry::PixFillBackoff::Stalled);
236    }
237}

239/// Handle a received command from the handle.
240///
241/// Returns `false` to signal the worker loop to stop.
242async fn process_cmd(
243    cmd: Option<WorkerCommand>,
244    supervisor: &mut SessionPixSupervisor,
245    action_tx: &ActionTx,
246    gate: &Arc<ServiceGate>,
247) -> bool {
248    let cmd = match cmd {
249        Some(c) => c,
250        None => {
251            // All senders dropped — close. Terminal by construction, so `dispatch` is told the
252            // supervisor is closed and its `false` is the verdict.
253            let actions = vec![SessionPixAction::Close(SessionPixCloseReason::SupervisorUnavailable)];
254            return dispatch(&actions, true, action_tx, gate);
255        }
256    };
257
258    match cmd {
259        WorkerCommand::Event(ev) => {
260            let now = Instant::now();
261            let actions = supervisor.handle_event(&ev, now, gate.served_total());
262            if !dispatch(&actions, supervisor.closed, action_tx, gate) {
263                return false;
264            }
265        }
266        WorkerCommand::ActionResult { action, ok } => {
267            let now = Instant::now();
268            let actions = supervisor.action_result(&action, ok, now);
269            if !dispatch(&actions, supervisor.closed, action_tx, gate) {
270                return false;
271            }
272        }
273    }
274    true
275}

277/// Apply gate control, forward `actions`, then report whether the worker should keep running.
278///
279/// The four places that produce actions — the two deadline paths in [`worker_loop`] and the two
280/// command arms in [`process_cmd`] — all had the same four-way branch spelled out: send, check
281/// whether the supervisor closed, poison, return. Fail-close is the property that matters most here
282/// and it was stated four times, so a correction to it had to be made four times.
283///
284/// Poisons the gate on every terminal path, so no caller has to remember to. `closed` is passed
285/// rather than read from the supervisor because the callers hold it by different borrows.
286///
287/// Terminality is derived from the *actions* as well as from `closed`, because the two do not
288/// always agree: several handlers return [`Close`](SessionPixAction::Close) without setting the
289/// supervisor's flag — both `InvalidTransition` paths in `on_ssa_request_sent`, `CounterRegression`
290/// and the share-order verdict in `on_recovery_progress`, and `on_unverifiable_shares`. Two of those
291/// are the anti-abuse verdicts. Reading only the flag would send the close, return `true`, and leave
292/// the gate open for as long as it takes the action driver to poison it on receipt — which is a real
293/// but bounded window, not a correct one. `Close` is also the one action the driver never reports
294/// back, so `action_result`'s own close path cannot cover for it.
295fn dispatch(actions: &[SessionPixAction], closed: bool, action_tx: &ActionTx, gate: &Arc<ServiceGate>) -> bool {
296    // Gate control is local state, not I/O. Apply it before touching the action channel so a safety
297    // transition cannot wait behind commitment generation, deposit-data lookup, or a network send in
298    // the action driver. Stop at Close to preserve the driver's rule that actions after a close are
299    // unreachable; poisoning below is stronger than every gate mode.
300    for action in actions
301        .iter()
302        .take_while(|action| !matches!(action, SessionPixAction::Close(_)))
303    {
304        match action {
305            SessionPixAction::ReleaseService => gate.release_service(),
306            SessionPixAction::WithholdService => gate.withhold_service(),
307            SessionPixAction::ProgressNotification => gate.notify_progress(),
308            _ => {}
309        }
310    }
311
312    let terminal = closed || actions.iter().any(|a| matches!(a, SessionPixAction::Close(_)));
313
314    // Sent before the verdict either way: a closing supervisor's last actions carry the reason it
315    // closed, and dropping them would leave the driver to infer it.
316    let delivered = send_actions(actions, action_tx);
317    if terminal || !delivered {
318        gate.poison();
319        return false;
320    }
321    true
322}
323
324/// Non-blocking forward of actions to the driver.
325///
326/// On `Disconnected`, the driver is gone — returns `false` so the caller
327/// can fail-close.
328///
329/// On `Full`:
330/// - **Coalescible** actions (`ProgressNotification`) are logged + skipped. They are idempotent and safe to drop — the
331///   next notification will replace them, and dropping here prevents transient load from killing a healthy session.
332/// - **Non-coalescible** actions (every other variant) are treated as fatal. If these cannot be delivered the channel
333///   is genuinely wedged.
334fn send_actions(actions: &[SessionPixAction], action_tx: &ActionTx) -> bool {
335    for action in actions {
336        match action_tx.try_send(action.clone()) {
337            Ok(()) => continue,
338            Err(TrySendError::Full(item)) => {
339                if is_coalescible(&item) {
340                    tracing::trace!(?action, "action channel full, dropping coalescible action");
341                    continue;
342                }
343                tracing::warn!(?action, "non-coalescible action dropped — channel full");
344                return false;
345            }
346            Err(TrySendError::Disconnected(_item)) => {
347                tracing::warn!(?action, "action driver disconnected");
348                return false;
349            }
350        }
351    }
352    true
353}
354
355/// Returns `true` for actions that are safe to drop when the action channel
356/// is transiently full.
357fn is_coalescible(action: &SessionPixAction) -> bool {
358    matches!(action, SessionPixAction::ProgressNotification)
359}

404    /// Wait for `condition` to hold, failing the test if it never does.
405    ///
406    /// The worker is a detached task, so every assertion about its effects is really an assertion
407    /// that it has been scheduled. A fixed sleep encodes a guess about the scheduler, and the guess
408    /// is wrong under CI load or on a busy single-core runner — the test then fails with nothing
409    /// wrong in the code. Polling asserts the same property and can only fail if it genuinely never
410    /// happens, which is why the deadline is generous: it is reached only on a real failure.
411    async fn poll_until(what: &str, mut condition: impl FnMut() -> bool) {
412        const DEADLINE: Duration = Duration::from_secs(5);
413        tokio::time::timeout(DEADLINE, async {
414            while !condition() {
415                tokio::time::sleep(Duration::from_millis(5)).await;
416            }
417        })
418        .await
419        .unwrap_or_else(|_| panic!("{what} — not observed within {DEADLINE:?}"));
420    }

917    /// Drains the action channel until a `SetFillRate` arrives, or the deadline passes.
918    ///
919    /// A funded cycle also produces `ReleaseService` and a `ProgressNotification` or two, so a test
920    /// that read only the next action would be asserting on whichever one the worker happened to
921    /// emit first.
922    async fn next_fill_rate(action_rx: &ActionRx, within: Duration) -> Option<FillRate> {
923        tokio::time::timeout(within, async {
924            loop {
925                match action_rx.recv().await {
926                    Ok(SessionPixAction::SetFillRate(rate)) => break Some(rate),
927                    Ok(_) => continue,
928                    Err(_) => break None,
929                }
930            }
931        })
932        .await
933        .ok()
934        .flatten()
935    }

950    /// The planner is only useful if the worker actually wakes for it.
951    ///
952    /// Everything else about fill is tested against a supervisor driven by hand with explicit
953    /// instants. This is the one property that cannot be: that the worker's own timer — which
954    /// previously woke only for lifecycle deadlines, none of which is due within an hour of a cycle
955    /// funding — now also wakes for the sampling interval and delivers what the planner decided.
956    #[tokio::test]
957    async fn the_worker_emits_a_fill_rate_within_two_ticks_of_funding() -> anyhow::Result<()> {
958        let p = HoprPseudonym::random();
959        let (handle, action_rx) = spawn_supervisor_worker(default_cfg(), dims(), p, Instant::now());
960
961        // Two sampling intervals, so the check spans a tick the planner could have emitted on. At
962        // 200 ms the first tick had not happened yet, so the assertion held for a Session that fills
963        // everything as readily as for one that fills nothing.
964        let initial = next_fill_rate(&action_rx, 2 * SAMPLING_INTERVAL).await;
965        assert_eq!(None, initial, "an unfunded Session must not be filled");
966
967        fund_first_cycle(&handle, p)
968            .await
969            .map_err(|()| anyhow::anyhow!("worker stopped"))?;
970
971        let rate = next_fill_rate(&action_rx, 2 * SAMPLING_INTERVAL + Duration::from_secs(1))
972            .await
973            .ok_or_else(|| anyhow::anyhow!("no fill rate was emitted within two sampling intervals of funding"))?;
974        assert!(!rate.is_zero(), "a funded cycle must be filled, got {rate:?}");
975        Ok(())
976    }

979    ///
980    /// `dispatch` stops applying actions at a `Close` and the driver breaks its loop on one, so a
981    /// zero that arrived after would never be applied — the Session would be torn down with its fill
982    /// stream still running at whatever rate it was last set to.
983    #[tokio::test]
984    async fn the_worker_delivers_a_zero_fill_rate_ahead_of_the_close() -> anyhow::Result<()> {
985        let p = HoprPseudonym::random();

987
988        fund_first_cycle(&handle, p)
989            .await
990            .map_err(|()| anyhow::anyhow!("worker stopped"))?;
991        let rate = next_fill_rate(&action_rx, 2 * SAMPLING_INTERVAL + Duration::from_secs(1))
992            .await
993            .ok_or_else(|| anyhow::anyhow!("fill must be running before the close is meaningful"))?;

995
996        handle
997            .send_event(SessionPixEvent::UnverifiableShares {
998                ssa_id: SsaId::new(p, SsaIndex::new(1).expect("index one is non-zero")),
999                observed_total: 1,

1001            .await
1002            .map_err(|()| anyhow::anyhow!("worker stopped"))?;
1003

1005        let ordering = tokio::time::timeout(Duration::from_secs(2), async {

1007                match action_rx.recv().await {
1008                    Ok(SessionPixAction::SetFillRate(rate)) => saw_zero |= rate.is_zero(),
1009                    Ok(SessionPixAction::Close(_)) => break saw_zero,
1010                    Ok(_) => continue,
1011                    Err(_) => break saw_zero,

1013            }
1014        })
1015        .await
1016        .map_err(|_| anyhow::anyhow!("no close was delivered"))?;

1019            ordering,
1020            "the zero fill rate must precede the close on the action channel"
1021        );
1022        Ok(())
