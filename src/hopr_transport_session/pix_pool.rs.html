1//! A thread pool of its own for the CPU-heavy PIX commitment work.
2//!
3//! On the shared CPU pool, the jobs of one commitment burst took every thread, and packets queued
4//! behind them until the burst was done (hoprnet#8765).
5
6use std::{sync::LazyLock, time::Instant};
7
8use futures::channel::oneshot;
9use hopr_utils::parallelize::cpu::{self, rayon};
10
11/// PIX commitment work run by [`spawn_pix_blocking`], as `hopr_pix_pool_execution_seconds` labels it.
12#[derive(Clone, Copy, Debug, PartialEq, Eq, strum::Display)]
13#[strum(serialize_all = "snake_case")]
14pub(crate) enum PixPoolOperation {
15    /// An Exit decoding and storing the commitments of one `SsaCommit` message.
16    CommitmentIngest,
17    /// An Entry generating the polynomials and commitments of a new SSA cycle.
18    CommitmentGeneration,
19}
20
21/// Half as many threads as the shared CPU pool, and at least one.
22static PIX_POOL: LazyLock<Result<rayon::ThreadPool, rayon::ThreadPoolBuildError>> = LazyLock::new(|| {
23    let shared_threads = match cpu::pool_thread_count() {
24        0 => std::thread::available_parallelism().map_or(1, |n| n.get() / 2),
25        n => n,
26    };
27    rayon::ThreadPoolBuilder::new()
28        .num_threads((shared_threads / 2).max(1))
29        .thread_name(|i| format!("pix-{i}"))
30        .build()
31});
32
33/// Runs `f` on the PIX pool and returns its result, re-raising a panic of `f` in the caller.
34///
35/// Parallel iterators inside `f` stay on the PIX pool. The job is skipped if the returned future is
36/// dropped before it starts.
37pub(crate) async fn spawn_pix_blocking<R: Send + 'static>(
38    operation: PixPoolOperation,
39    f: impl FnOnce() -> R + Send + 'static,
40) -> anyhow::Result<R> {
41    let pool = PIX_POOL
42        .as_ref()
43        .map_err(|error| anyhow::anyhow!("the PIX thread pool could not be built: {error}"))?;
44
45    let (tx, rx) = oneshot::channel();
46    let span = tracing::Span::current();
47    let submitted_at = Instant::now();
48    pool.spawn_fifo(move || {
49        if tx.is_canceled() {
50            return;
51        }
52        let _span = span.entered();
53        let queue_wait = submitted_at.elapsed();
54        let started_at = Instant::now();
55        let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(f));
56        #[cfg(feature = "telemetry")]
57        crate::telemetry::pix::record_pix_pool_job(operation, queue_wait, started_at.elapsed());
58        #[cfg(not(feature = "telemetry"))]
59        let _ = (operation, queue_wait, started_at);
60        let _ = tx.send(result);
61    });
62
63    match rx.await {
64        Ok(Ok(value)) => Ok(value),
65        Ok(Err(panic)) => std::panic::resume_unwind(panic),
66        Err(oneshot::Canceled) => Err(anyhow::anyhow!("a PIX pool job ended without a result")),
67    }
68}

70#[cfg(test)]
71mod tests {
72    use futures::FutureExt;
73
74    use super::*;
75
76    #[tokio::test]
77    async fn a_pix_job_returns_its_result() -> anyhow::Result<()> {
78        let result = spawn_pix_blocking(PixPoolOperation::CommitmentIngest, || 42).await?;
79        assert_eq!(result, 42);
80        Ok(())
81    }
82
83    #[tokio::test]
84    async fn parallel_work_inside_a_pix_job_stays_on_the_pix_pool() -> anyhow::Result<()> {
85        use rayon::prelude::*;
86
87        let thread_names = spawn_pix_blocking(PixPoolOperation::CommitmentGeneration, || {
88            (0..256)
89                .into_par_iter()
90                .map(|_| std::thread::current().name().map(str::to_owned))
91                .collect::<Vec<_>>()
92        })
93        .await?;
94
95        assert!(
96            thread_names
97                .iter()
98                .all(|name| name.as_deref().is_some_and(|name| name.starts_with("pix-"))),
99            "every item must run on a PIX pool thread: {thread_names:?}"
100        );
101        Ok(())
102    }
103
104    #[cfg(panic = "unwind")]
105    #[tokio::test]
106    async fn a_panic_in_a_pix_job_reaches_the_caller() {
107        let outcome = std::panic::AssertUnwindSafe(spawn_pix_blocking(PixPoolOperation::CommitmentIngest, || {
108            panic!("pix job panic")
109        }))
110        .catch_unwind()
111        .await;
112        assert!(outcome.is_err(), "the panic must be re-raised in the caller");
113    }
