9/// Monotonic origin for the degraded-return-path deadline.
10///
11/// An `Instant` cannot live in an atomic, and the deadline is written by one layer and read by
12/// another, so it travels as milliseconds elapsed from a fixed point. Monotonic rather than
13/// wall-clock, so a clock adjustment cannot extend or cancel the window.
14static EPOCH: LazyLock<Instant> = LazyLock::new(Instant::now);

60/// Configuration for the `SurbBalancer`.
61#[derive(Clone, Copy, Debug, PartialEq, smart_default::SmartDefault)]
62pub struct SurbBalancerConfig {
63    /// The desired number of SURBs to be always kept as a buffer locally or at the Session counterparty.
64    ///
65    /// The `SurbBalancer` will try to maintain approximately this number of SURBs
66    /// locally or remotely (at the counterparty) at all times.
67    ///
68    /// The local buffer is maintained by regulating (`SurbFlowController`) the egress from the Session.
69    /// The remote buffer (at session counterparty) is maintained by regulating the flow of non-organic SURBs via
70    /// keep-alive messages.
71    ///
72    /// It does not make sense to set this value higher than the [`max_surb_buffer_size`](crate::SessionManagerConfig)
73    /// configuration at the counterparty.
74    ///
75    /// Default is 7000 SURBs.
76    #[default(7_000)]
77    pub target_surb_buffer_size: u64,
78    /// Maximum outflow of SURBs.
79    ///
80    /// - In the context of the local SURB buffer (Entry), this is the maximum egress Session traffic (= SURB
81    ///   consumption).
82    /// - In the context of the remote SURB buffer (Exit), this is the maximum egress of keep-alive messages to the
83    ///   counterparty (= artificial SURB production).
84    ///
85    /// The default is 5000 (which is 2500 packets/second currently)
86    #[default(5_000)]
87    pub max_surbs_per_sec: u64,
88
89    /// Sets what percentage of the target buffer size should be discarded at each window.
90    ///
91    /// The `SurbBalancer` will discard the given percentage of `target_surb_buffer_size` at each
92    /// window with the given `Duration`.
93    ///
94    /// The default is `(60, 0.05)` (5% of the target buffer size is discarded every 60 seconds).
95    #[default(_code = "Some((Duration::from_secs(60), 0.05))")]
96    pub surb_decay: Option<(Duration, f64)>,
97
98    /// Keeps producing SURBs while the return path is known to be failing, instead of reading the
99    /// resulting silence as a full counterparty buffer.
100    ///
101    /// The remote buffer is estimated as *produced − consumed*, and consumption is only observed
102    /// when a reply reaches us. A return path that drops every reply therefore looks exactly like a
103    /// counterparty that is well stocked, so production is throttled at the very moment the
104    /// counterparty is in fact draining towards empty and needs more.
105    ///
106    /// Distinguishing that from a peer which simply has nothing to say is impossible from here --
107    /// both show no consumption -- so this only takes effect once an outside observer marks the
108    /// return path degraded, and it expires on its own if no further evidence arrives.
109    ///
110    /// Off by default: sustaining production spends bandwidth on a path that may be genuinely idle,
111    /// which is only worth it for sessions that value recovery latency over that bandwidth.
112    #[default(false)]
113    pub sustain_on_return_path_loss: bool,
114}

118    #[inline]
119    pub fn as_controller_bounds(&self) -> BalancerControllerBounds {
120        BalancerControllerBounds::new(self.target_surb_buffer_size, self.max_surbs_per_sec)
121    }

124/// Runtime state of the `SurbBalancer`.
125#[derive(Debug, Default)]
126pub struct BalancerStateValues {
127    pub target_surb_buffer_size: AtomicU64,
128    pub max_surbs_per_sec: AtomicU64,
129    pub decay_duration_msec: AtomicU64,
130    pub decay_volume_pct: AtomicU8,
131    pub buffer_level: AtomicU64,
132    /// Whether this session opted into sustaining production through return-path loss.
133    pub sustain_on_return_path_loss: AtomicBool,
134    /// How many SURBs the counterparty can physically hold, or 0 when unknown.
135    ///
136    /// The estimate is `produced - consumed`, and consumption is only observed once a reply
137    /// arrives -- so a return path that drops replies lets the believed level grow without bound.
138    /// The counterparty's store is a ring buffer that evicts the oldest entry on overflow, so
139    /// everything above its capacity was discarded on arrival and was never a real level. Measured
140    /// during an outage: 51 917 believed against a 15 000-entry store.
141    pub counterparty_buffer_capacity: AtomicU64,
142    /// Milliseconds from the crate-internal `EPOCH` monotonic origin until which the return path
143    /// counts as degraded.
144    ///
145    /// A deadline rather than a flag: it is set by a layer that observes the return path and read
146    /// here, and nothing is guaranteed to come back and clear it. Expiring on its own bounds the
147    /// damage of a marker that is never withdrawn to a short over-production instead of a session
148    /// that mints forever.
149    pub return_path_degraded_until_ms: AtomicU64,
150}
151
152impl BalancerStateValues {
153    /// Constructor from a [`SurbBalancerConfig`].
154    pub fn new(cfg: SurbBalancerConfig) -> Self {
155        let state = Self::default();
156        state.update(&cfg);
157        state
158    }
159
160    /// Performs update of the [`BalancerStateValues`] from the [`SurbBalancerConfig`] and
161    /// enables it.
162    pub fn update(&self, cfg: &SurbBalancerConfig) {
163        self.target_surb_buffer_size
164            .store(cfg.target_surb_buffer_size, std::sync::atomic::Ordering::Relaxed);
165        self.max_surbs_per_sec
166            .store(cfg.max_surbs_per_sec, std::sync::atomic::Ordering::Relaxed);
167        self.decay_duration_msec.store(
168            cfg.surb_decay
169                .map(|(d, _)| d.as_millis().min(u64::MAX as u128) as u64)
170                .unwrap_or_default(),
171            std::sync::atomic::Ordering::Relaxed,
172        );
173        self.decay_volume_pct.store(
174            cfg.surb_decay
175                .map(|(_, p)| (p.clamp(0.0, 1.0) * 100.0).round() as u8)
176                .unwrap_or_default(),
177            std::sync::atomic::Ordering::Relaxed,
178        );
179        self.sustain_on_return_path_loss
180            .store(cfg.sustain_on_return_path_loss, std::sync::atomic::Ordering::Relaxed);
181    }
182
183    /// Declares how many SURBs the counterparty's store can hold, bounding the level estimate.
184    ///
185    /// Taken from the session manager's `maximum_surb_buffer_size`, which is the same capacity
186    /// already used to clamp a counterparty's requested target. Zero leaves the estimate unbounded.
187    pub fn set_counterparty_buffer_capacity(&self, capacity: u64) {
188        self.counterparty_buffer_capacity
189            .store(capacity, std::sync::atomic::Ordering::Relaxed);
190    }
191
192    /// Caps `level` at what the counterparty can actually hold.
193    ///
194    /// Never below the configured target: a target above the counterparty's capacity is
195    /// unreachable by construction, and clamping to capacity there would hold the error permanently
196    /// negative and pin production at maximum forever -- a worse failure than the unbounded
197    /// estimate this exists to prevent. In that configuration the capacity figure is simply not
198    /// usable for this session.
199    fn clamp_to_counterparty_capacity(&self, level: u64) -> u64 {
200        match self
201            .counterparty_buffer_capacity
202            .load(std::sync::atomic::Ordering::Relaxed)
203        {
204            0 => level,
205            capacity => {
206                level.min(capacity.max(self.target_surb_buffer_size.load(std::sync::atomic::Ordering::Relaxed)))
207            }
208        }
209    }
210
211    /// Marks the return path as degraded for the next `grace` period.
212    ///
213    /// Called by whichever layer can actually tell a dead return path from a quiet peer -- from
214    /// here the two are indistinguishable, since neither delivers replies. Re-marking simply
215    /// extends the window.
216    pub fn mark_return_path_degraded(&self, grace: Duration) {
217        let until = EPOCH.elapsed().saturating_add(grace).as_millis().min(u64::MAX as u128) as u64;
218        self.return_path_degraded_until_ms
219            .fetch_max(until, std::sync::atomic::Ordering::Relaxed);
220    }
221
222    /// Whether [`buffer_level`](Self::buffer_level) is currently an instruction rather than a
223    /// measurement.
224    ///
225    /// While this holds, the controller deliberately writes `0` into the level to drive production
226    /// to its maximum. That zero says "produce flat out", not "the counterparty holds nothing", so
227    /// anything reading the level as a *supply ceiling* must consult this first or it will read the
228    /// instruction as an order to send nothing.
229    ///
230    /// True only when both the opt-in (`sustain_on_return_path_loss`) and live evidence
231    /// ([`mark_return_path_degraded`](Self::mark_return_path_degraded), within its window) are
232    /// present: without the opt-in this is not our behaviour to change, and without evidence there
233    /// is nothing to tell a dead return path from an idle one.
234    ///
235    /// `pub` because it is not only the controller's business — hence the emphasis above on what
236    /// the flag does *not* mean. It is not a general "the return path is degraded" signal.
237    pub fn return_path_estimate_is_stale(&self) -> bool {
238        self.should_sustain_through_return_path_loss()
239    }
240
241    fn should_sustain_through_return_path_loss(&self) -> bool {
242        let deadline = self
243            .return_path_degraded_until_ms
244            .load(std::sync::atomic::Ordering::Relaxed);
245
246        // Zero is "never marked", not "marked at the epoch" -- otherwise every session that opted
247        // in would start out believing its return path was already dead.
248        deadline > 0
249            && (EPOCH.elapsed().as_millis() as u64) < deadline
250            && self
251                .sustain_on_return_path_loss
252                .load(std::sync::atomic::Ordering::Relaxed)
253    }
254
255    /// Extracts the [`SurbBalancerConfig`] from the [`BalancerStateValues`].
256    pub fn as_config(&self) -> SurbBalancerConfig {
257        SurbBalancerConfig {
258            target_surb_buffer_size: self.target_surb_buffer_size.load(std::sync::atomic::Ordering::Relaxed),
259            max_surbs_per_sec: self.max_surbs_per_sec.load(std::sync::atomic::Ordering::Relaxed),
260            surb_decay: self.surb_decay(),
261            sustain_on_return_path_loss: self
262                .sustain_on_return_path_loss
263                .load(std::sync::atomic::Ordering::Relaxed),
264        }
265    }
266
267    /// Checks if SURB balancing is disabled (no target buffer size set).
268    pub fn is_disabled(&self) -> bool {
269        self.target_surb_buffer_size.load(std::sync::atomic::Ordering::Relaxed) == 0
270    }
271
272    /// Extracts the SURB decay configuration from the [`BalancerStateValues`].
273    pub fn surb_decay(&self) -> Option<(Duration, f64)> {
274        Some((
275            self.decay_duration_msec.load(std::sync::atomic::Ordering::Relaxed),
276            self.decay_volume_pct.load(std::sync::atomic::Ordering::Relaxed),
277        ))
278        .filter(|&(d, p)| d > 0 && p > 0)
279        .map(|(d, p)| (Duration::from_millis(d), p as f64 / 100.0))
280    }
281
282    /// Gets the current estimated SURB buffer level.
283    #[inline]
284    pub fn buffer_level(&self) -> u64 {
285        self.buffer_level.load(std::sync::atomic::Ordering::Relaxed)
286    }
287
288    /// Returns the current `BalancerControllerBounds` from the [`BalancerStateValues`].
289    #[inline]
290    pub fn controller_bounds(&self) -> BalancerControllerBounds {
291        BalancerControllerBounds::new(
292            self.target_surb_buffer_size.load(std::sync::atomic::Ordering::Relaxed),
293            self.max_surbs_per_sec.load(std::sync::atomic::Ordering::Relaxed),
294        )
295    }

304/// Runs a continuous process that attempts to [evaluate](SurbFlowEstimator) and
305/// [regulate](SurbFlowController) the flow of SURBs to the Session counterparty,
306/// to keep the number of SURBs locally or at the counterparty at a certain level.
307///
308/// Internally, the Balancer uses an implementation of [`SurbBalancerController`] to
309/// control the rate of SURBs consumed or sent to the counterparty
310/// each time the [`update`](SurbBalancer::update) method is called:
311///
312/// 1. The size of the SURB buffer at locally or at the counterparty is estimated using [`SurbFlowEstimator`].
313/// 2. Error against a set-point given in [`SurbBalancerConfig`] is evaluated in the `SurbBalancerController`.
314/// 3. The `SurbBalancerController` applies a new SURB flow rate value using the [`SurbFlowController`].
315///
316/// In the local context, the `SurbFlowController` might simply regulate the egress traffic from the
317/// Session, slowing it down to avoid fast SURB drainage.
318///
319/// In the remote context, the `SurbFlowController` might regulate the flow of non-organic SURBs via
320/// Start protocol's `KeepAlive` messages to deliver additional
321/// SURBs to the counterparty.
322pub struct SurbBalancer<C, E, F> {
323    session_id: SessionId,
324    controller: C,
325    surb_estimator: E,
326    flow_control: F,
327    state: Arc<BalancerStateValues>,
328    last_estimator_state: SimpleSurbFlowEstimator,
329    last_update: std::time::Instant,
330    last_decay: std::time::Instant,
331    was_below_target: bool,
332    /// Whether the previous update ran in open loop, so both edges can be acted on.
333    was_degraded: bool,
334    /// DIAGNOSTIC: when the last balancer-state line was emitted, to rate-limit it.
335    last_report: std::time::Instant,
336}

341    E: SurbFlowEstimator + Send + Sync + 'static,
342    F: SurbFlowController + Send + Sync + 'static,
343{

352        {

354            METRIC_TARGET_ERROR_ESTIMATE.set(&[sid], 0.0);
355            METRIC_CONTROL_OUTPUT.set(&[sid], 0.0);
356        }
357
358        controller.set_target_and_limit(state.controller_bounds());
359
360        Self {
361            surb_estimator,
362            flow_control,
363            controller,
364            session_id,
365            state,
366            last_estimator_state: Default::default(),
367            last_update: std::time::Instant::now(),
368            last_decay: std::time::Instant::now(),
369            was_below_target: true,
370            was_degraded: false,
371            last_report: std::time::Instant::now(),
372        }
373    }

375    /// Computes the next control update and adjusts the [`SurbFlowController`] rate accordingly.
376    #[tracing::instrument(level = "trace", skip_all)]
377    fn update(&mut self) -> u64 {
378        let dt = self.last_update.elapsed();
379
380        // Load the updated current buffer level
381        let mut current = self.state.buffer_level.load(std::sync::atomic::Ordering::Acquire);
382
383        if dt < Duration::from_millis(10) {
384            tracing::debug!("time elapsed since last update is too short, skipping update");
385            return current;
386        }
387
388        self.last_update = std::time::Instant::now();
389
390        // Take a snapshot of the active SURB estimator and calculate the balance change
391        let snapshot = SimpleSurbFlowEstimator::from(&self.surb_estimator);
392        let Some(target_buffer_change) = snapshot.estimated_surb_buffer_change(&self.last_estimator_state) else {
393            tracing::error!("non-monotonic change in SURB estimators");
394            return current;
395        };
396
397        self.last_estimator_state = snapshot;
398        current = current.saturating_add_signed(target_buffer_change);
399
400        // If SURB decaying is enabled, check if the decay window has elapsed
401        // and calculate the number of SURBs that will be discarded
402        if let Some(num_decayed_surbs) = self
403            .state
404            .surb_decay()
405            .filter(|(decay_window, _)| &self.last_decay.elapsed() >= decay_window)
406            .map(|(_, decay_coeff)| (self.controller.bounds().target() as f64 * decay_coeff).round() as u64)
407        {
408            current = current.saturating_sub(num_decayed_surbs);
409            self.last_decay = std::time::Instant::now();
410            tracing::trace!(num_decayed_surbs, "SURBs were discarded due to automatic decay");
411        }
412
413        // Believing a level the counterparty cannot hold keeps production throttled long after
414        // the surplus was evicted on arrival, so the estimate is bounded by the store it describes.
415        let believed = current;
416        current = self.state.clamp_to_counterparty_capacity(current);
417        if current != believed {
418            // Not the configured capacity: the bound applied is `max(capacity, target)`, so name
419            // the clamped level and the capacity separately rather than conflating them.
420            tracing::debug!(
421                believed,
422                clamped_to = current,
423                counterparty_capacity = self
424                    .state
425                    .counterparty_buffer_capacity
426                    .load(std::sync::atomic::Ordering::Relaxed),
427                "counterparty SURB estimate exceeded its store; the surplus was never held"
428            );
429        }
430
431        let degraded = self.state.should_sustain_through_return_path_loss();
432        if degraded != self.was_degraded {
433            // The estimate stops meaning what it meant on both edges: entering, it is inflated by
434            // production nobody was seen to consume; leaving, it is a level that was never
435            // observed. Either way the accumulated error belongs to a regime that has ended.
436            self.controller.reset();
437            self.was_degraded = degraded;
438
439            if !degraded {
440                // Coming back, treat the counterparty as freshly started rather than as whatever
441                // the outage left behind. It really did drain while replies were lost, and this is
442                // the estimate that self-corrects: consumption is observable again, so the buffer
443                // level climbs on its own as production outruns it.
444                current = 0;
445                self.last_decay = std::time::Instant::now();
446                tracing::debug!("return path recovered; restarting closed-loop SURB control");
447            }
448        }
449
450        if degraded {
451            // While replies are being lost there is no valid estimate to act on: every SURB the
452            // counterparty spends is invisible from here, so the accumulated `produced - consumed`
453            // reads as a filling buffer precisely when it is emptying. Drop to open loop and assume
454            // the worst, which drives production to the maximum until replies resume.
455            tracing::debug!(
456                believed = current,
457                "return path degraded; ignoring the counterparty buffer estimate"
458            );
459            // Reads as "produce flat out" to the controller below. `SurbSupply` must not read it
460            // as "the buffer is empty, admit nothing" -- see `return_path_estimate_is_stale`.
461            current = 0;
462        }
463
464        self.state
465            .buffer_level
466            .store(current, std::sync::atomic::Ordering::Release);
467
468        // Error from the desired target SURB buffer size at counterparty
469        let error = current as i64 - self.controller.bounds().target() as i64;
470
471        if self.was_below_target && error >= 0 {
472            tracing::trace!(current, "reached target SURB buffer size");
473            self.was_below_target = false;
474        } else if !self.was_below_target && error < 0 {
475            tracing::trace!(current, "SURB buffer size is below target");
476            self.was_below_target = true;
477        }
478
479        tracing::trace!(
480            ?dt,
481            delta = target_buffer_change,
482            rate = target_buffer_change as f64 / dt.as_secs_f64(),
483            current,
484            error,
485            "estimated SURB buffer change"
486        );
487
488        let output = self.controller.next_control_output(current);
489        tracing::trace!(output, "next balancer control output for session");
490
491        // Both ends run this same loop -- the Entry with the PID driving production, the Exit with
492        // the proportional controller gating egress -- so one line covers both and the session id
493        // tells them apart. Rate-limited to one per second so it can run under a full-rate session.
494        //
495        // At `debug` rather than `info`: one line per session per second is fine for a handful of
496        // sessions and is a lot of formatting work for a node carrying many, none of which an
497        // operator needs to see during healthy operation.
498        if self.last_report.elapsed() >= Duration::from_secs(1) {
499            self.last_report = std::time::Instant::now();
500            tracing::debug!(
501                session = %self.session_id,
502                level = current,
503                target = self.controller.bounds().target(),
504                output,
505                produced = self.surb_estimator.estimate_surbs_produced(),
506                consumed = self.surb_estimator.estimate_surbs_consumed(),
507                degraded,
508                "surb balancer state"
509            );
510        }
511
512        self.flow_control.adjust_surb_flow(output as usize);
513
514        #[cfg(all(feature = "telemetry", not(test)))]
515        {
516            let sid: &str = self.session_id.as_ref();
517            METRIC_CURRENT_BUFFER.set(&[sid], current as f64);
518            METRIC_CURRENT_TARGET.set(&[sid], self.controller.bounds().target() as f64);
519            METRIC_TARGET_ERROR_ESTIMATE.set(&[sid], error as f64);
520            METRIC_CONTROL_OUTPUT.set(&[sid], output as f64);
521            METRIC_SURB_RATE.set(&[sid], target_buffer_change as f64 / dt.as_secs_f64());
522        }
523
524        current
525    }

527    /// Spawns a new task that performs updates of the given [`SurbBalancer`] at the given `sampling_interval`.
528    ///
529    /// If `cfg_feedback` is given, [`SurbBalancerConfig`] can be queried for updates and also updated
530    /// if the underlying [`SurbBalancerController`] also does target updates.
531    ///
532    /// Returns a stream of current estimated buffer levels, and also an `AbortHandle`
533    /// to terminate the loop. If `abort_reg` was given, the returned `AbortHandle` corresponds
534    /// to it.
535    #[instrument(level = "debug", skip(self), fields(session_id = %self.session_id))]
536    pub fn start_control_loop(
537        mut self,
538        sampling_interval: Duration,
539    ) -> (impl futures::Stream<Item = u64>, AbortHandle) {
540        let (abort_handle, abort_reg) = AbortHandle::new_pair();
541
542        // Start an interval stream at which the balancer will sample and perform updates
543
544        // DropAbortable not needed because the stream only generates items when polled
545        let sampling_stream = futures::stream::Abortable::new(
546            futures_time::stream::interval(sampling_interval.max(MIN_BALANCER_SAMPLING_INTERVAL).into()),
547            abort_reg,
548        );
549
550        let balancer_level_capacity = std::env::var("HOPR_INTERNAL_SESSION_BALANCER_LEVEL_CAPACITY")
551            .ok()
552            .and_then(|s| s.trim().parse::<usize>().ok())
553            .filter(|&c| c > 0)
554            .unwrap_or(32_768);
555
556        tracing::debug!(
557            capacity = balancer_level_capacity,
558            "Creating session balancer level channel"
559        );
560        let (mut level_tx, level_rx) = futures::channel::mpsc::channel(balancer_level_capacity);
561        hopr_utils::runtime::prelude::spawn(
562            async move {
563                pin_mut!(sampling_stream);
564                while sampling_stream.next().await.is_some() {
565                    // Check if the balancer controller needs to be reconfigured
566                    let current_bounds = self.state.controller_bounds();
567                    if current_bounds != self.controller.bounds() {
568                        self.controller.set_target_and_limit(current_bounds);
569                        tracing::debug!(new_cfg = ?self.state.as_config(), "surb balancer has been reconfigured");
570                    }
571
572                    // Perform controller update (this internally samples the SurbFlowEstimator)
573                    // and send an update about the current level to the outgoing stream.
574                    // If the other party has closed the stream, we don't care about the update.
575                    let level = self.update();
576                    if !level_tx.is_closed()
577                        && let Err(error) = level_tx.try_send(level)
578                    {
579                        tracing::error!(%error, "cannot send balancer level update");
580                    }
581                }
582
583                tracing::debug!("balancer done");
584            }
585            .in_current_span(),
586        );
587
588        (level_rx, abort_handle)
589    }
590}

841    /// A balancer whose production follows its own control output, as it does in a live Session.
842    ///
843    /// Returns the balancer, the shared estimator and the latest control output. Production must be
844    /// fed back rather than held constant: with production pinned to consumption the buffer never
845    /// fills, maximum output is the correct answer, and every phase of the test reads the same.
846    #[allow(clippy::type_complexity)]
847    fn balancer_with_feedback(
848        cfg: SurbBalancerConfig,
849    ) -> (
850        SurbBalancer<PidBalancerController, AtomicSurbFlowEstimator, MockSurbFlowController>,
851        AtomicSurbFlowEstimator,
852        Arc<BalancerStateValues>,
853        Arc<AtomicU64>,
854    ) {
855        let output = Arc::new(AtomicU64::new(0));
856        let output_clone = output.clone();
857        let mut controller = MockSurbFlowController::new();
858        controller.expect_adjust_surb_flow().returning(move |r| {
859            output_clone.store(r as u64, std::sync::atomic::Ordering::Relaxed);
860        });
861
862        let surb_estimator = AtomicSurbFlowEstimator::default();
863        let state: Arc<BalancerStateValues> = Arc::new(cfg.into());
864        let balancer = SurbBalancer::new(
865            HoprPseudonym::random(),
866            PidBalancerController::default(),
867            surb_estimator.clone(),
868            controller,
869            state.clone(),
870        );
871
872        (balancer, surb_estimator, state, output)
873    }

875    /// One sampling interval: mint at the rate last commanded, and consume `consumed` of them.
876    fn tick(
877        balancer: &mut SurbBalancer<PidBalancerController, AtomicSurbFlowEstimator, MockSurbFlowController>,
878        surb_estimator: &AtomicSurbFlowEstimator,
879        output: &AtomicU64,
880        consumed: u64,
881    ) {
882        let step = Duration::from_millis(50);
883        std::thread::sleep(step);
884
885        let minted = output.load(std::sync::atomic::Ordering::Relaxed) * step.as_millis() as u64 / 1000;
886        surb_estimator
887            .produced
888            .fetch_add(minted, std::sync::atomic::Ordering::Relaxed);
889        surb_estimator
890            .consumed
891            .fetch_add(consumed, std::sync::atomic::Ordering::Relaxed);
892        balancer.update();
893    }

895    /// SURBs the counterparty spends per interval while it is answering normally.
896    const REPLIES_PER_TICK: u64 = 40;
897
898    /// Drives a balancer through a healthy stretch, then through one where no reply comes back.
899    ///
900    /// Returns the control output at the end of each stretch. The two phases are deliberately
901    /// indistinguishable from inside the balancer -- consumption simply stops -- which is the whole
902    /// point: only the caller's `sustain` choice separates a dead return path from an idle peer.
903    fn drive_until_replies_stop(cfg: SurbBalancerConfig, mark_degraded: bool) -> (u64, u64) {
904        let (mut balancer, surb_estimator, state, output) = balancer_with_feedback(cfg);
905
906        for _ in 0..40 {
907            tick(&mut balancer, &surb_estimator, &output, REPLIES_PER_TICK);
908        }
909        let healthy = output.load(std::sync::atomic::Ordering::Relaxed);
910
911        if mark_degraded {
912            state.mark_return_path_degraded(Duration::from_secs(30));
913        }
914
915        // Replies stop while production continues.
916        for _ in 0..20 {
917            tick(&mut balancer, &surb_estimator, &output, 0);
918        }
919
920        (healthy, output.load(std::sync::atomic::Ordering::Relaxed))
921    }

1004    /// After the outage the counterparty must be refilled, not merely un-throttled.
1005    ///
1006    /// Returning to closed loop is only half the claim: production has to actually climb the curve
1007    /// again and restore the buffer. Resetting the controller is what makes that prompt -- the error
1008    /// accumulated while the estimate was meaningless would otherwise have to be unwound first.
1009    #[test_log::test]
1010    fn surb_balancer_should_refill_the_counterparty_after_the_return_path_recovers() {
1011        let cfg = sustaining_config(true);
1012        let (mut balancer, surb_estimator, state, output) = balancer_with_feedback(cfg);
1013
1014        for _ in 0..40 {
1015            tick(&mut balancer, &surb_estimator, &output, REPLIES_PER_TICK);
1016        }
1017        // A band, not the setpoint itself: the controller oscillates around its target, so a
1018        // sample taken at an arbitrary tick legitimately sits either side of it.
1019        let refilled = cfg.target_surb_buffer_size / 2;
1020        assert!(
1021            state.buffer_level.load(std::sync::atomic::Ordering::Relaxed) >= refilled,
1022            "the healthy phase must reach the setpoint band before an outage means anything"
1023        );
1024
1025        // The return path dies: replies stop, and open loop takes over.
1026        state.mark_return_path_degraded(Duration::from_millis(400));
1027        for _ in 0..8 {
1028            tick(&mut balancer, &surb_estimator, &output, 0);
1029        }
1030
1031        // The mark lapses, the counterparty answers again, and the belief restarts from empty.
1032        let mut ticks_to_refill = None;
1033        for n in 1..=60 {
1034            tick(&mut balancer, &surb_estimator, &output, REPLIES_PER_TICK);
1035            if state.buffer_level.load(std::sync::atomic::Ordering::Relaxed) >= refilled {
1036                ticks_to_refill = Some(n);
1037                break;
1038            }
1039        }
1040
1041        let ticks = ticks_to_refill.expect("the counterparty must be refilled to the setpoint after recovery");
1042        tracing::info!(ticks, "refilled the counterparty after the outage");
1043
1044        // Each tick is one sampling interval; the balancer samples far more often than this in a
1045        // live Session, so a bound in ticks is a bound in sampling intervals, not in wall clock.
1046        assert!(
1047            ticks <= 30,
1048            "refilling must ramp rather than crawl: took {ticks} sampling intervals"
1049        );
1050    }

1163    // --- return-path-degraded deadline primitive (Test C, gap 1) ---------------------------------
1164    //
1165    // The behavioural loop tests above (sustain / ignore-unless-opted-in / return-to-closed-loop /
1166    // refill) already cover how the balancer *reacts* to the signal. What they do not isolate are
1167    // three edge cases of the `mark_return_path_degraded` / `return_path_estimate_is_stale`
1168    // deadline primitive itself, each guarding a specific correctness invariant.
1169
1170    /// The zero-guard: never marked is never degraded. `return_path_degraded_until_ms == 0` means
1171    /// "never marked", not "marked at the epoch" — without the explicit `deadline > 0` check every
1172    /// opted-in session would believe its return path was dead from the first packet.
1173    #[test]
1174    fn return_path_should_not_be_stale_before_it_is_ever_marked() {
1175        let state = BalancerStateValues::new(SurbBalancerConfig {
1176            sustain_on_return_path_loss: true,
1177            ..Default::default()
1178        });
1179        assert!(!state.return_path_estimate_is_stale());
1180    }
1181
1182    /// The window expires on its own, independently of any recovery signal. The behavioural tests
1183    /// clear the degraded state by resuming replies; this pins the other exit — a marker nobody
1184    /// ever withdraws must still lapse, bounding over-production to the grace window. Marking with a
1185    /// zero grace sets the deadline to "now"; the monotonic clock only moves forward, so the
1186    /// subsequent read is already past it — no sleep, no flake.
1187    #[test]
1188    fn an_expired_degraded_window_should_no_longer_be_stale() {
1189        let state = BalancerStateValues::new(SurbBalancerConfig {
1190            sustain_on_return_path_loss: true,
1191            ..Default::default()
1192        });
1193        state.mark_return_path_degraded(Duration::ZERO);
1194        assert!(!state.return_path_estimate_is_stale());
1195    }
1196
1197    /// Re-marking extends the window: an already-expired deadline followed by a fresh mark is stale
1198    /// again. Guards the `fetch_max` in `mark_return_path_degraded`.
1199    #[test]
1200    fn re_marking_should_reopen_an_expired_window() {
1201        let state = BalancerStateValues::new(SurbBalancerConfig {
1202            sustain_on_return_path_loss: true,
1203            ..Default::default()
1204        });
1205        state.mark_return_path_degraded(Duration::ZERO);
1206        assert!(!state.return_path_estimate_is_stale());

1208        assert!(state.return_path_estimate_is_stale());
