1//! This crate defines the Start sub-protocol used for HOPR Session initiation and management.
2//!
3//! The Start protocol is used to establish Session as described in HOPR
4//! [`RFC-0012`](https://github.com/hoprnet/rfc/tree/main/rfcs/RFC-0012-session-start-protocol).
5//! and is implemented via the [`StartProtocol`] enum.
6//!
7//! The protocol is defined via generic arguments `I` (for Session ID), `T` (for Session Target),
8//! `C` (for Session capabilities) and `G` (for Session Stealth Address commitment representation).
9//!
10//! Per `RFC-0012`, the types `I` and `T` are serialized/deserialized to the CBOR binary format
11//! (see [`RFC7049`](https://datatracker.ietf.org/doc/html/rfc7049)) and therefore must implement
12//! `serde::Serialize + serde::Deserialize`.
13//! The capability type `C` must be expressible as a single unsigned byte.
14//!
15//! The `G` type is used to represent the Session Stealth Address commitment representation.
16//! It is typically a [`PixGroupRepr`](hopr_protocol_pix::PixGroupRepr)
17//!
18//! See [`StartProtocol`] docs for the protocol diagram.
19

78/// The session initiation message of the Start protocol.
79///
80/// ## Generic parameters
81/// - `T` is the session target
82/// - `C` are session capabilities
83///
84/// The `additional_data` are set dependent on the `capabilities`
85/// or set to `0x0000000000000000` to be ignored.
86#[derive(Debug, Clone, PartialEq, Eq)]
87pub struct StartInitiation<T, C> {
88    /// Random challenge for this initiation.
89    pub challenge: StartChallenge,
90    /// Target of the session, i.e., what should the other party do with the traffic.
91    pub target: T,
92    /// Requested capabilities of the session.
93    ///
94    /// This might also contain information required for the PIX protocol.
95    pub capabilities: C,
96    /// Additional options (might be `capabilities` dependent), ignored if `0x0000000000000000`.
97    ///
98    /// When PIX is offered, the upper 32 bits are the
99    /// [`PixParams`](hopr_protocol_pix::PixParams) word — see
100    /// [`PixParams::into_additional_data`](hopr_protocol_pix::PixParams::into_additional_data) —
101    /// and the lower 32 bits are the SURB balancer target. The field is then fully allocated;
102    /// there is no room left in it to negotiate anything further.
103    pub additional_data: u64,
104}

119/// Lists all messages of the Start protocol for a session establishment.
120///
121/// ## Generic parameters
122/// - `I` is the session identifier.
123/// - `T` is the session target.
124/// - `C` are session capabilities.
125/// - `G` is the type of the commitment to the Session Stealth Address (SSA).
126/// - `K` is the wire form of the proof of knowledge going with a client SSA commitment.
127/// - `D` is the type of the deposit data carried in the Server's SSA commitment.
128///
129/// # Diagram of the protocol
130/// ```mermaid
131/// sequenceDiagram
132///      Entry->>Exit: SessionInitiation (Challenge)
133///      alt If Exit can accept a new session
134///      Note right of Exit: SessionID [Pseudonym, Tag]
135///      Exit->>Entry: SessionEstablished (Challenge, SessionID)
136///      Note left of Entry: SessionID [Pseudonym, Tag]
137///      Exit->>Entry: SsaRequest (SessionID, SsaIndex, ServerCommitments)
138///      Entry->>Exit: SsaCommit (SessionID, SsaIndex, CoeffIndex, PolyCoeffs)
139///      Entry->>Exit: KeepAlive (SessionID)
140///      Exit->>Entry: KeepAlive (SessionID)
141///      Note over Entry,Exit: Data
142///      else If Exit cannot accept a new session
143///      Exit->>Entry: SessionError (Challenge, Reason)
144///      end
145///      opt If initiation attempt times out
146///      Note left of Entry: Failure
147///      end
148/// ```
149#[derive(Debug, Clone, PartialEq, Eq, strum::EnumDiscriminants)]
150#[strum_discriminants(vis(pub))]
151#[strum_discriminants(derive(strum::FromRepr, strum::EnumCount), repr(u8))]
152pub enum StartProtocol<I, T, C, G, K, D> {
153    /// Request to initiate a new session.
154    StartSession(StartInitiation<T, C>),
155    /// Confirmation that a new session has been established by the counterparty.
156    SessionEstablished(StartEstablished<I>),
157    /// Client's message to fill Client commitments to establish a Session Stealth Address (SSA).
158    SsaCommit(SsaClientCommitmentMessage<I, G, K>),
159    /// Server-side commitment to Session Stealth Address (SSA).
160    SsaRequest(SsaServerCommitmentMessage<I, G, D>),
161    /// Counterparty could not establish a new session due to an error.
162    SessionError(StartErrorType<I>),
163    /// A ping message to keep the session alive.
164    KeepAlive(KeepAliveMessage<I>),
165}

167/// Filling up the Client's commitment to the Session Stealth Address (SSA).
168///
169/// The generic argument `G` typically represents a [`PixGroupRepr`](hopr_protocol_pix::PixGroupRepr).
170///
171/// The overall Client's commitment to a single new SSA usually requires multiple messages, all
172/// sharing the same [`SsaIndex`](hopr_protocol_pix::SsaIndex).
173///
174/// Each of these messages contains commitments to polynomial coefficients that all belong
175/// to the same coefficient in each polynomial.
176///
177/// In practice every message carries `coefficient_index == 0`: PIX commits to each polynomial's
178/// constant term and nothing else, so the constant-term pass *is* the whole commitment. The field
179/// (and phase 2 of [`new_multiple`](Self::new_multiple), which now has nothing to emit) is retained
180/// because the wire format still admits higher coefficient indices; a peer that sends them merely
181/// wastes bandwidth, since the receiver ignores them.
182///
183/// See [`hopr_protocol_pix::SsaPartCommitment`] for why the non-constant coefficient commitments
184/// were dropped and what that costs.
185#[derive(Debug, Clone, PartialEq, Eq)]
186pub struct SsaClientCommitmentMessage<I, G, K> {
187    /// Session ID.
188    pub session_id: I,
189    /// Index of the Session Stealth Address (SSA) that is being committed.
190    pub ssa_index: hopr_protocol_pix::SsaIndex,
191    /// Index of the polynomial coefficient that is being committed.
192    ///
193    /// Zero value indicates the polynomial constant term commitment, which when summed over
194    /// all polynomials for a given [`SsaIndex`](hopr_protocol_pix::SsaIndex)
195    /// results in the Client's SSA commitment.
196    ///
197    /// This is always zero in practice — see the type-level documentation.
198    pub coefficient_index: u16,
199    /// Proof that the Client knows the discrete logarithm of its SSA commitment.
200    ///
201    /// Present exactly when `coefficient_index == 0`: the commitment it opens is the *sum* of all
202    /// constant terms, so it belongs with the messages that deliver them, and there is no
203    /// separate presence flag on the wire — the coefficient index decides it. Every constant-term
204    /// message carries it rather than only one, so no single lost packet can strand a cycle that
205    /// would otherwise be recoverable, and the receiver keeps the first it sees.
206    ///
207    /// Without it the receiver cannot distinguish a genuine commitment from one crafted so the
208    /// *combined* deposit key is known to the sender alone. See
209    /// [`hopr_protocol_pix::SsaCommitmentProof`].
210    pub commitment_proof: Option<K>,
211    /// Contains the serialized coefficient commitments of multiple polynomials,
212    /// all belonging to the same `coefficient_index` in each polynomial.
213    ///
214    /// This might not be the complete set yet and might require multiple messages to deliver
215    /// the complete commitment to the given coefficient of all polynomials for the given SSA.
216    pub coefficient_commitments: std::collections::HashMap<hopr_protocol_pix::PolynomialIndex, G>,
217}

219impl<I: serde::Serialize + Clone, G: Clone, K: Clone> SsaClientCommitmentMessage<I, G, K> {
220    /// Uses given the `session_id` and an [`SsaCommitment`] that will be split across multiple messages.
221    ///
222    /// The returned messages are ordered by coefficient index, making sure the constant terms
223    /// of the polynomials are delivered first.
224    pub fn new_multiple<S: hopr_protocol_pix::PixSpec>(
225        session_id: I,
226        commitment: SsaCommitment<S, S::Pseudonym>,
227    ) -> Result<Vec<Self>, StartProtocolError>
228    where
229        G: From<hopr_protocol_pix::PixGroupRepr<S>>,
230        K: From<hopr_protocol_pix::SsaCommitmentProof<S>>,
231    {
232        let ssa_index = commitment.ssa_id.ssa_index();
233        let commitment_proof = K::from(commitment.commitment_proof);
234
235        // A single message can only carry a limited number of coefficient commitments so that the
236        // resulting encoded message still fits within a HOPR packet payload. The commitments of a
237        // single coefficient (across all polynomials) might therefore need to be split across
238        // multiple messages. The encode layout this is derived from is documented on the helper.
239        let SsaCommitChunking {
240            max_commitments_per_message,
241            max_constant_terms_per_message,
242        } = StartProtocol::<I, (), (), G, K, ()>::ssa_commit_chunking(&session_id)?;
243
244        // Group the transposed verifiers by coefficient index, each group sorted by polynomial
245        // index. A `BTreeMap` keeps the coefficient order deterministic; the inner sort is what
246        // lets a block of polynomials be addressed as the same slice range in every coefficient.
247        let mut by_coefficient: std::collections::BTreeMap<u16, Vec<(hopr_protocol_pix::PolynomialIndex, G)>> =
248            std::collections::BTreeMap::new();
249
250        for (coefficient_index, coefficients) in commitment.verifiers {
251            let entry = by_coefficient.entry(coefficient_index).or_default();
252            for (poly_index, coefficient_commitment) in coefficients {
253                entry.push((poly_index, G::from(coefficient_commitment)));
254            }
255        }
256        for coefficients in by_coefficient.values_mut() {
257            coefficients.sort_unstable_by_key(|(poly_index, _)| *poly_index);
258        }
259
260        let mut messages = Vec::new();
261        let mut push_chunk = |coefficient_index: u16, chunk: &[(hopr_protocol_pix::PolynomialIndex, G)]| {
262            messages.push(Self {
263                session_id: session_id.clone(),
264                ssa_index,
265                coefficient_index,
266                // Presence is decided by the coefficient index, matching what the decoder expects.
267                commitment_proof: (coefficient_index == 0).then(|| commitment_proof.clone()),
268                coefficient_commitments: chunk.iter().cloned().collect(),
269            });
270        };
271
272        // Phase 1 — every polynomial's constant term, so the Exit can derive the SSA deposit
273        // address as early as possible. This is the one coefficient that must be delivered across
274        // *all* polynomials before anything else, because the address is their sum.
275        let constant_terms = by_coefficient.remove(&0).unwrap_or_default();
276        for chunk in constant_terms.chunks(max_constant_terms_per_message) {
277            push_chunk(0, chunk);
278        }
279
280        // Phase 2 — any remaining coefficients, a block of polynomials at a time.
281        //
282        // Dead in practice: PIX commits to constant terms only, so `by_coefficient` is empty after
283        // the `remove(&0)` above and this loop body never runs. It is kept because the wire format
284        // still admits higher coefficient indices, and because the block-major layout is the
285        // non-obvious part — emitting one polynomial's row at a time would waste most of each
286        // packet, while walking coefficient-major would complete no polynomial until the very last
287        // message.
288        let num_polys = constant_terms
289            .len()
290            .max(by_coefficient.values().map(|c| c.len()).max().unwrap_or(0));
291        for block_start in (0..num_polys).step_by(max_commitments_per_message) {
292            for (&coefficient_index, coefficients) in &by_coefficient {
293                if block_start >= coefficients.len() {
294                    continue;
295                }
296                let block_end = (block_start + max_commitments_per_message).min(coefficients.len());
297                push_chunk(coefficient_index, &coefficients[block_start..block_end]);
298            }
299        }
300
301        Ok(messages)
302    }
303}

305/// Sent by the Server to deliver the commitment to possibly multiple new Session Stealth Addresses (SSAs).
306///
307/// This message is typically sent for the first time right after the [`StartEstablished`] message
308/// if PIX capabilities are indicated in the [`StartInitiation`] message, and the Server accepts it.
309///
310/// It is then subsequently sent every time the Server needs the next batch of SSAs
311/// (with indices strictly greater than in the last batch) to be committed to.
312#[derive(Debug, Clone, PartialEq, Eq)]
313pub struct SsaServerCommitmentMessage<I, G, D> {
314    /// Session ID.
315    pub session_id: I,
316    /// Parameters of the PIX protocol the server requires, packed by
317    /// [`PixParams::to_u32`](hopr_protocol_pix::PixParams::to_u32).
318    pub params: u32,
319    /// Per-SSA deposit/payment data, carried in CBOR as a single map keyed by the same
320    /// [`SsaIndex`](hopr_protocol_pix::SsaIndex) as [`commitments`](Self::commitments).
321    pub deposit_data: std::collections::HashMap<hopr_protocol_pix::SsaIndex, D>,
322    /// Server's serialized commitments to the SSAs, ordered by the SSA index.
323    pub commitments: std::collections::BTreeMap<hopr_protocol_pix::SsaIndex, G>,
324}

336            params: params.to_u32(),
337            deposit_data: deposit_data.into_iter().collect(),

339        }
340    }

342    /// The PIX dimensions this request was made under, and the curve suite they are dimensions of.
343    ///
344    /// Fails if the peer packed values outside the protocol ranges, or named a curve suite that does

348        hopr_protocol_pix::PixParams::try_from_u32(self.params)
349            .map_err(|error| StartProtocolError::ParseError(format!("ssa_req.params: {error}")))
350    }

404impl<I, T, C, G, K, D> StartProtocol<I, T, C, G, K, D> {
405    /// Maximum size of the CBOR-serialized
406    /// [`deposit_data`](SsaServerCommitmentMessage::deposit_data) map an
407    /// [`SsaRequest`](StartProtocol::SsaRequest) message can carry.
408    pub const MAX_DEPOSIT_DATA_SIZE: usize = ApplicationData::PAYLOAD_SIZE.saturating_sub(
409        4 + size_of::<u32>()
410            + size_of::<u16>()
411            + size_of::<hopr_protocol_pix::SsaIndex>()
412            + Self::PIX_COEFF_COMMITMENT_REPR_SIZE
413            + 1,
414    );
415    /// Maximum number of SSAs that can be requested in a single SsaRequest message.
416    pub const MAX_SSAS_PER_REQUEST: u16 = ((ApplicationData::PAYLOAD_SIZE - 11)
417        / (size_of::<hopr_protocol_pix::SsaIndex>() + Self::PIX_COEFF_COMMITMENT_REPR_SIZE))
418        as u16;
419    /// Size of the PIX coefficient commitment representation in bytes.
420    pub const PIX_COEFF_COMMITMENT_REPR_SIZE: usize = size_of::<G>();
421    /// Size of the serialized client SSA commitment proof of knowledge in bytes.
422    ///
423    /// Carried only by constant-term `SsaCommit` messages, so it reduces the entry budget of those
424    /// and of no others.
425    pub const PIX_COMMITMENT_PROOF_SIZE: usize = size_of::<K>();
426    /// Size of the Start protocol message header in bytes.
427    pub const START_HEADER_SIZE: usize =
428        size_of_val(&Self::START_PROTOCOL_MESSAGE_TAG) + size_of::<u8>() + size_of::<u16>();
429    /// Fixed [`Tag`] of every protocol message.
430    pub const START_PROTOCOL_MESSAGE_TAG: Tag = Tag::Reserved(ReservedTag::SessionStart as u64);
431    /// Current version of the Start protocol.
432    pub const START_PROTOCOL_VERSION: u8 = 0x03;
433
434    /// How many commitment entries one [`SsaCommit`](StartProtocol::SsaCommit) message can carry,
435    /// for each of the two delivery phases.
436    ///
437    /// This is the bound
438    /// [`SsaClientCommitmentMessage::new_multiple`] chunks by, exposed so that a caller predicting
439    /// the message count for a commitment does not have to restate the encode layout. Restating it
440    /// is error-prone in a way that hides: the copy this replaced used `SsaIndex` (4 B) where an
441    /// `SsaCommit` entry prefix is a `PolynomialIndex` (2 B), against a different fixed overhead,
442    /// and agreed with the encoder only because the two mistakes floored to the same divisor at the
443    /// dimensions it ran at.
444    ///
445    /// Takes the `session_id` itself rather than a precomputed length: its CBOR encoding is part of
446    /// the layout, so leaving the caller to measure it is the same footgun one level up.
447    pub fn ssa_commit_chunking(session_id: &I) -> Result<SsaCommitChunking, StartProtocolError>
448    where
449        I: serde::Serialize,
450    {
451        // A single message can only carry a limited number of coefficient commitments
452        // so that the resulting encoded message still fits within a HOPR packet payload.
453        // The commitments of a single coefficient (across all polynomials) might therefore
454        // need to be split across multiple messages.
455        //
456        // The bound is derived from the actual SsaCommit encode layout:
457        //   header:     version(1) + disc(1) + data_len(2) = 4
458        //   fixed:      ssa_index(4) + coefficient_index(2) + poly_count(2) = 8
459        //   proof:      PIX_COMMITMENT_PROOF_SIZE, constant-term messages only
460        //   per-entry:  PolynomialIndex(2) + commitment_repr(PIX_COEFF_COMMITMENT_REPR_SIZE)
461        //   trailer:    CBOR-encoded session_id
462        let header_and_fixed: usize = 4 + 4 + 2 + 2; // header + ssa_index + coeff_index + num_polys
463        let per_entry = size_of::<hopr_protocol_pix::PolynomialIndex>() + Self::PIX_COEFF_COMMITMENT_REPR_SIZE;
464        // Compute the exact CBOR size of the session_id for this instantiation.
465        let cbor_session_id_size = serde_cbor_2::to_vec(session_id)?.len();
466        let budget = ApplicationData::PAYLOAD_SIZE.saturating_sub(header_and_fixed + cbor_session_id_size);
467
468        Ok(SsaCommitChunking {
469            max_commitments_per_message: (budget / per_entry).max(1),
470            // Constant-term messages additionally carry the proof, so they fit fewer entries. Phase
471            // 2 keeps the full budget, which is what the block size in `new_multiple` is aligned to;
472            // only phase 1 pays for the proof, costing a handful of extra messages per cycle.
473            max_constant_terms_per_message: (budget.saturating_sub(Self::PIX_COMMITMENT_PROOF_SIZE) / per_entry).max(1),
474        })
475    }
476}

478/// How many commitment entries one `SsaCommit` message can carry, per delivery phase.
479///
480/// Returned by [`StartProtocol::ssa_commit_chunking`]. The two are easy to confuse and differ only
481/// by the proof of knowledge, so they are named rather than returned as a pair.
482#[derive(Debug, Clone, Copy, PartialEq, Eq)]
483pub struct SsaCommitChunking {
484    /// Phase 2 — higher-coefficient messages, which get the full entry budget.
485    pub max_commitments_per_message: usize,
486    /// Phase 1 — constant-term messages, which also carry the proof of knowledge and so fit fewer.
487    pub max_constant_terms_per_message: usize,
488}

490impl<I, T, C, G, K, D> StartProtocol<I, T, C, G, K, D>
491where
492    I: serde::Serialize,
493    T: serde::Serialize,
494    C: Into<u8>,
495    G: AsRef<[u8]>,
496    K: AsRef<[u8]>,
497    D: serde::Serialize,
498{
499    /// Tries to encode the message into binary format and [`Tag`]
500    pub fn encode(self) -> errors::Result<(Tag, Box<[u8]>)> {
501        let mut out = Vec::with_capacity(ApplicationData::PAYLOAD_SIZE);
502        out.push(Self::START_PROTOCOL_VERSION);
503        out.push(StartProtocolDiscriminants::from(&self) as u8);
504
505        let mut data = Vec::with_capacity(ApplicationData::PAYLOAD_SIZE - 2);
506        match self {
507            StartProtocol::StartSession(init) => {
508                data.extend_from_slice(&init.challenge.to_be_bytes());
509                data.push(init.capabilities.into());
510                data.extend_from_slice(&init.additional_data.to_be_bytes());
511                let target = serde_cbor_2::to_vec(&init.target)?;
512                data.extend_from_slice(&target);
513            }
514            StartProtocol::SessionEstablished(est) => {
515                data.extend_from_slice(&est.orig_challenge.to_be_bytes());
516                let session_id = serde_cbor_2::to_vec(&est.session_id)?;
517                data.extend(session_id);
518            }
519            StartProtocol::SessionError(err) => {
520                match err.identifier {
521                    ErrorIdentifier::Challenge(challenge) => {
522                        data.push(ErrorIdentifierDiscriminants::Challenge as u8);
523                        data.extend_from_slice(&challenge.to_be_bytes());
524                    }
525                    ErrorIdentifier::SessionId(id) => {
526                        data.push(ErrorIdentifierDiscriminants::SessionId as u8);
527                        let id_bytes = serde_cbor_2::to_vec(&id)?;
528                        data.extend(id_bytes);
529                    }
530                }
531                data.push(err.reason as u8);
532            }
533            StartProtocol::KeepAlive(ping) => {
534                data.push(ping.flags.bits());
535                data.extend_from_slice(&ping.additional_data.to_be_bytes());
536                let session_id = serde_cbor_2::to_vec(&ping.session_id)?;
537                data.extend(session_id);
538            }
539            StartProtocol::SsaCommit(commit) => {
540                data.extend_from_slice(&commit.ssa_index.get().to_be_bytes());
541                data.extend_from_slice(&commit.coefficient_index.to_be_bytes());
542
543                let num_polys = commit.coefficient_commitments.len() as hopr_protocol_pix::PolynomialIndex;
544                data.extend_from_slice(&num_polys.to_be_bytes());
545
546                // The proof goes immediately after the fixed prefix, and only on constant-term
547                // messages. Its presence is implied by `coefficient_index == 0` rather than by a
548                // flag, so the two must agree — a mismatch would desynchronise the decoder.
549                match (commit.coefficient_index, &commit.commitment_proof) {
550                    (0, Some(proof)) => {
551                        let proof = proof.as_ref();
552                        if proof.len() != Self::PIX_COMMITMENT_PROOF_SIZE {
553                            return Err(StartProtocolError::ParseError("commitment_proof_size".into()));
554                        }
555                        data.extend_from_slice(proof);
556                    }
557                    (0, None) => return Err(StartProtocolError::ParseError("missing commitment_proof".into())),
558                    (_, Some(_)) => {
559                        return Err(StartProtocolError::ParseError(
560                            "commitment_proof on a non-constant term".into(),
561                        ));
562                    }
563                    (_, None) => {}
564                }
565
566                let session_id = serde_cbor_2::to_vec(&commit.session_id)?;
567                let total_coeff_commit_len = (size_of::<hopr_protocol_pix::PolynomialIndex>()
568                    + Self::PIX_COEFF_COMMITMENT_REPR_SIZE)
569                    * commit.coefficient_commitments.len();
570
571                // Remaining payload budget: the final `out` buffer contains
572                // version (1) + disc (1) + data_len (2) + data contents = 4 + data.len(),
573                // which must fit within PAYLOAD_SIZE.  Check using explicit arithmetic
574                // rather than Vec::spare_capacity_mut() which reflects pre-allocation.
575                let avail_space = ApplicationData::PAYLOAD_SIZE.saturating_sub(4 + data.len() + session_id.len());
576                if commit.coefficient_commitments.is_empty() || total_coeff_commit_len > avail_space {
577                    return Err(StartProtocolError::NumberOfCommitments);
578                }
579
580                for (index, commitment) in commit.coefficient_commitments {
581                    let commitment_repr = commitment.as_ref();
582                    if commitment_repr.len() != Self::PIX_COEFF_COMMITMENT_REPR_SIZE {
583                        return Err(StartProtocolError::ParseError("commitment_repr_size".into()));
584                    }
585
586                    // Prepending 16-bit representation of the polynomial index
587                    // will possibly consume less space than putting an entire 1024-bit bitmap
588                    // of polynomials present in each message (assuming 1024 polynomials per SSA).
589                    data.extend_from_slice(&index.to_be_bytes());
590                    data.extend_from_slice(commitment_repr);
591                }
592
593                data.extend(session_id);
594            }
595            StartProtocol::SsaRequest(req) => {
596                data.extend_from_slice(&req.params.to_be_bytes());
597
598                let deposit_data =
599                    serde_cbor_2::to_vec(&req.deposit_data.iter().collect::<std::collections::BTreeMap<_, _>>())?;
600
601                if deposit_data.len() > Self::MAX_DEPOSIT_DATA_SIZE {
602                    return Err(StartProtocolError::DepositDataTooLarge {
603                        size: deposit_data.len(),
604                        max: Self::MAX_DEPOSIT_DATA_SIZE,
605                    });
606                }
607
608                data.extend_from_slice(&deposit_data);
609
610                let num_commitments = req.commitments.len() as u16;
611                data.extend_from_slice(&num_commitments.to_be_bytes());
612
613                let session_id = serde_cbor_2::to_vec(&req.session_id)?;
614
615                let required_size = (size_of::<hopr_protocol_pix::SsaIndex>() + Self::PIX_COEFF_COMMITMENT_REPR_SIZE)
616                    * req.commitments.len();
617
618                // Remaining payload budget: the final `out` buffer contains
619                // version (1) + disc (1) + data_len (2) + data contents = 4 + data.len(),
620                // which must fit within PAYLOAD_SIZE.  Check using explicit arithmetic
621                // rather than Vec::spare_capacity_mut() which reflects pre-allocation.
622                //
623                // `data` already holds the serialized deposit data, so a `D` that passed
624                // `MAX_DEPOSIT_DATA_SIZE` above still narrows this budget: past a certain size it
625                // is the commitment count that has to give way, which is what this reports.
626                let avail_space = ApplicationData::PAYLOAD_SIZE.saturating_sub(4 + data.len() + session_id.len());
627                if req.commitments.is_empty() || required_size > avail_space {
628                    return Err(StartProtocolError::NumberOfCommitments);
629                }
630
631                for (ssa_index, commitment) in req.commitments {
632                    let commitment_repr = commitment.as_ref();
633                    if commitment_repr.len() != Self::PIX_COEFF_COMMITMENT_REPR_SIZE {
634                        return Err(StartProtocolError::ParseError("commitment_repr_size".into()));
635                    }
636
637                    data.extend_from_slice(&ssa_index.get().to_be_bytes());
638                    data.extend_from_slice(commitment_repr);
639                }
640
641                data.extend(session_id);
642            }
643        }
644
645        out.extend_from_slice(&(data.len() as u16).to_be_bytes());
646        out.extend(data);
647
648        Ok((Self::START_PROTOCOL_MESSAGE_TAG, out.into_boxed_slice()))
649    }
650}

652/// Reads the per-SSA deposit data map, refusing one that announces more entries than a request can
653/// carry.
654fn decode_deposit_data<'de, D, De>(
655    deserializer: De,
656    max_entries: usize,
657) -> Result<std::collections::HashMap<hopr_protocol_pix::SsaIndex, D>, De::Error>
658where
659    D: serde::Deserialize<'de>,
660    De: serde::Deserializer<'de>,
661{
662    struct BoundedMap<D> {
663        max_entries: usize,
664        _deposit_data: std::marker::PhantomData<D>,
665    }
666
667    impl<'de, D: serde::Deserialize<'de>> serde::de::Visitor<'de> for BoundedMap<D> {
668        type Value = std::collections::HashMap<hopr_protocol_pix::SsaIndex, D>;
669
670        fn expecting(&self, f: &mut std::fmt::Formatter) -> std::fmt::Result {
671            write!(f, "at most {} per-SSA deposit data entries", self.max_entries)
672        }
673
674        fn visit_map<A: serde::de::MapAccess<'de>>(self, mut map: A) -> Result<Self::Value, A::Error> {
675            if let Some(declared) = map.size_hint()
676                && declared > self.max_entries
677            {
678                return Err(serde::de::Error::invalid_length(declared, &self));
679            }
680
681            // Safe to trust now: whatever it says is at or below the bound.
682            let mut out = std::collections::HashMap::with_capacity(map.size_hint().unwrap_or(0));
683            // Counted as they arrive, not by what the map ends up holding: repeating one key
684            // overwrites in place, so `out.len()` would sit at one however many entries followed.
685            let mut entries = 0;
686            while let Some((ssa_index, data)) = map.next_entry()? {
687                entries += 1;
688                if entries > self.max_entries {
689                    return Err(serde::de::Error::invalid_length(entries, &self));
690                }
691                out.insert(ssa_index, data);
692            }
693            Ok(out)
694        }
695    }
696
697    deserializer.deserialize_map(BoundedMap {
698        max_entries,
699        _deposit_data: std::marker::PhantomData,
700    })
701}

703impl<I, T, C, G, K, D> StartProtocol<I, T, C, G, K, D>
704where
705    I: for<'de> serde::Deserialize<'de>,
706    T: for<'de> serde::Deserialize<'de>,
707    C: TryFrom<u8>,
708    G: for<'a> TryFrom<&'a [u8]>,
709    K: for<'a> TryFrom<&'a [u8]>,
710    D: for<'de> serde::Deserialize<'de>,
711{
712    /// Tries to decode the message from the binary representation and [`Tag`].
713    ///
714    /// The `tag` must be currently [`START_PROTOCOL_MESSAGE_TAG`](Self::START_PROTOCOL_MESSAGE_TAG)
715    /// and version [`START_PROTOCOL_VERSION`](Self::START_PROTOCOL_VERSION).
716    pub fn decode(tag: Tag, data: &[u8]) -> errors::Result<Self> {
717        if tag != Self::START_PROTOCOL_MESSAGE_TAG {
718            return Err(StartProtocolError::UnknownTag);
719        }
720
721        if data.len() < 5 {
722            return Err(StartProtocolError::InvalidLength);
723        }
724
725        if data[0] != Self::START_PROTOCOL_VERSION {
726            return Err(StartProtocolError::InvalidVersion);
727        }
728
729        let disc = data[1];
730        let len = u16::from_be_bytes(
731            data[2..4]
732                .try_into()
733                .map_err(|_| StartProtocolError::ParseError("len".into()))?,
734        ) as usize;
735        let data_offset = 2 + size_of::<u16>();
736
737        if data.len() != data_offset + len {
738            return Err(StartProtocolError::InvalidLength);
739        }
740        let body = &data[data_offset..data_offset + len];
741
742        Ok(
743            match StartProtocolDiscriminants::from_repr(disc).ok_or(StartProtocolError::UnknownMessage)? {
744                StartProtocolDiscriminants::StartSession => {
745                    if body.len() < size_of::<StartChallenge>() + 1 + size_of::<u64>() {
746                        return Err(StartProtocolError::InvalidLength);
747                    }
748
749                    StartProtocol::StartSession(StartInitiation {
750                        challenge: StartChallenge::from_be_bytes(
751                            body[..size_of::<StartChallenge>()]
752                                .try_into()
753                                .map_err(|_| StartProtocolError::ParseError("init.challenge".into()))?,
754                        ),
755                        capabilities: body[size_of::<StartChallenge>()]
756                            .try_into()
757                            .map_err(|_| StartProtocolError::ParseError("init.capabilities".into()))?,
758                        additional_data: u64::from_be_bytes(
759                            body[size_of::<StartChallenge>() + 1..size_of::<StartChallenge>() + 1 + size_of::<u64>()]
760                                .try_into()
761                                .map_err(|_| StartProtocolError::ParseError("init.additional_data".into()))?,
762                        ),
763                        target: serde_cbor_2::from_slice(&body[size_of::<StartChallenge>() + 1 + size_of::<u64>()..])?,
764                    })
765                }
766                StartProtocolDiscriminants::SessionEstablished => {
767                    if body.len() < size_of::<StartChallenge>() {
768                        return Err(StartProtocolError::InvalidLength);
769                    }
770                    StartProtocol::SessionEstablished(StartEstablished {
771                        orig_challenge: StartChallenge::from_be_bytes(
772                            body[..size_of::<StartChallenge>()]
773                                .try_into()
774                                .map_err(|_| StartProtocolError::ParseError("est.challenge".into()))?,
775                        ),
776                        session_id: serde_cbor_2::from_slice(&body[size_of::<StartChallenge>()..])?,
777                    })
778                }
779                StartProtocolDiscriminants::SessionError => {
780                    if body.is_empty() {
781                        return Err(StartProtocolError::InvalidLength);
782                    }
783                    let (identifier, reason_start) = match ErrorIdentifierDiscriminants::from_repr(body[0])
784                        .ok_or(StartProtocolError::ParseError("err.identifier_tag".into()))?
785                    {
786                        ErrorIdentifierDiscriminants::Challenge => {
787                            if body.len() < 1 + size_of::<StartChallenge>() + 1 {
788                                return Err(StartProtocolError::InvalidLength);
789                            }
790                            let challenge = StartChallenge::from_be_bytes(
791                                body[1..1 + size_of::<StartChallenge>()]
792                                    .try_into()
793                                    .map_err(|_| StartProtocolError::ParseError("err.challenge".into()))?,
794                            );
795                            (ErrorIdentifier::Challenge(challenge), 1 + size_of::<StartChallenge>())
796                        }
797                        ErrorIdentifierDiscriminants::SessionId => {
798                            if body.len() < 2 {
799                                return Err(StartProtocolError::InvalidLength);
800                            }
801                            // Reason byte is the last byte; CBOR session_id is everything
802                            // between the tag byte and the reason.
803                            let reason_start = body.len().saturating_sub(1);
804                            let session_id: I = serde_cbor_2::from_slice(&body[1..reason_start])?;
805                            (ErrorIdentifier::SessionId(session_id), reason_start)
806                        }
807                    };
808                    StartProtocol::SessionError(StartErrorType {
809                        identifier,
810                        // An unrecognized reason is read as [`StartErrorReason::Unknown`] rather
811                        // than failing the message, which is the whole purpose of that variant.
812                        // The alternative loses the error entirely: a peer that added a reason this
813                        // build predates would have its `SessionError` rejected as malformed, and
814                        // the Session it was closing would die on a timeout instead of being
815                        // reported. Refusing to parse tells this node *less* than reading the byte
816                        // it does not recognize as "some reason", so a reason may be added without
817                        // a protocol version bump.
818                        reason: StartErrorReason::from_repr(body[reason_start]).unwrap_or(StartErrorReason::Unknown),
819                    })
820                }
821                StartProtocolDiscriminants::KeepAlive => {
822                    if body.len() < 1 + size_of::<u64>() {
823                        return Err(StartProtocolError::InvalidLength);
824                    }
825
826                    StartProtocol::KeepAlive(KeepAliveMessage {
827                        flags: KeepAliveFlags::new(body[0])
828                            .map_err(|_| StartProtocolError::ParseError("ka.flags".into()))?,
829                        additional_data: u64::from_be_bytes(
830                            body[1..1 + size_of::<u64>()]
831                                .try_into()
832                                .map_err(|_| StartProtocolError::ParseError("ka.additional_data".into()))?,
833                        ),
834                        session_id: serde_cbor_2::from_slice(&body[1 + size_of::<u64>()..])?,
835                    })
836                }
837                StartProtocolDiscriminants::SsaCommit => {
838                    if body.len()
839                        <= size_of::<hopr_protocol_pix::SsaIndex>()
840                            + size_of::<hopr_protocol_pix::CoefficientIndex>()
841                            + 2 * size_of::<hopr_protocol_pix::PolynomialIndex>()
842                            + Self::PIX_COEFF_COMMITMENT_REPR_SIZE
843                    {
844                        return Err(StartProtocolError::InvalidLength);
845                    }
846
847                    let ssa: hopr_protocol_pix::SsaIndex = hopr_protocol_pix::RawSsaIndex::from_be_bytes(
848                        body[..size_of::<hopr_protocol_pix::SsaIndex>()]
849                            .try_into()
850                            .map_err(|_| StartProtocolError::ParseError("ssa_index".into()))?,
851                    )
852                    .try_into()
853                    .map_err(|_| StartProtocolError::ParseError("ssa_index is 0".into()))?;
854                    let coefficient_index = hopr_protocol_pix::CoefficientIndex::from_be_bytes(
855                        body[size_of::<hopr_protocol_pix::SsaIndex>()
856                            ..size_of::<hopr_protocol_pix::SsaIndex>()
857                                + size_of::<hopr_protocol_pix::CoefficientIndex>()]
858                            .try_into()
859                            .map_err(|_| StartProtocolError::ParseError("coefficient_index".into()))?,
860                    );
861                    let num_polys =
862                        hopr_protocol_pix::PolynomialIndex::from_be_bytes(
863                            body[size_of::<hopr_protocol_pix::SsaIndex>()
864                                + size_of::<hopr_protocol_pix::CoefficientIndex>()
865                                ..size_of::<hopr_protocol_pix::SsaIndex>()
866                                    + size_of::<hopr_protocol_pix::CoefficientIndex>()
867                                    + size_of::<hopr_protocol_pix::PolynomialIndex>()]
868                                .try_into()
869                                .map_err(|_| StartProtocolError::ParseError("polynomial_index".into()))?,
870                        );
871                    if num_polys == 0 || num_polys > MAX_POLYS_PER_SSA {
872                        return Err(StartProtocolError::NumberOfCommitments);
873                    }
874
875                    let mut fixed_prefix_size = size_of::<hopr_protocol_pix::SsaIndex>()
876                        + size_of::<hopr_protocol_pix::CoefficientIndex>()
877                        + size_of::<hopr_protocol_pix::PolynomialIndex>();
878
879                    // The proof sits right after the fixed prefix on constant-term messages only,
880                    // with no presence flag — the coefficient index is what says whether it is
881                    // there, so the encoder is required to keep the two in agreement.
882                    let commitment_proof = if coefficient_index == 0 {
883                        if body.len() <= fixed_prefix_size + Self::PIX_COMMITMENT_PROOF_SIZE {
884                            return Err(StartProtocolError::InvalidLength);
885                        }
886                        let proof =
887                            K::try_from(&body[fixed_prefix_size..fixed_prefix_size + Self::PIX_COMMITMENT_PROOF_SIZE])
888                                .map_err(|_| StartProtocolError::ParseError("commitment_proof".into()))?;
889                        fixed_prefix_size += Self::PIX_COMMITMENT_PROOF_SIZE;
890                        Some(proof)
891                    } else {
892                        None
893                    };
894
895                    let mut coefficient_commitments = {
896                        // Derive the maximum number of polynomial entries this packet
897                        // can carry from the same per-entry constraints as the encoder
898                        // (see `new_multiple`).  Enforce it before allocating, so
899                        // unreasonably large `num_polys` values that pass the
900                        // `MAX_POLYS_PER_SSA` guard are still caught by the wire limit.
901                        let per_entry =
902                            size_of::<hopr_protocol_pix::PolynomialIndex>() + Self::PIX_COEFF_COMMITMENT_REPR_SIZE;
903                        // Availability: remaining body after the fixed prefix fields
904                        // (ssa_index + coefficient_index + num_polys), reserving at
905                        // least one byte for the trailing CBOR session_id.
906                        let avail = body.len().saturating_sub(fixed_prefix_size);
907                        let max_by_payload = avail.saturating_sub(1) / per_entry;
908                        if num_polys as usize > max_by_payload {
909                            return Err(StartProtocolError::NumberOfCommitments);
910                        }
911                        std::collections::HashMap::with_capacity(num_polys as usize)
912                    };
913                    let mut next_offset = fixed_prefix_size;
914                    for _ in 0..num_polys {
915                        // Still needs to be space left for Session ID at the end of commitments
916                        if body.len()
917                            <= next_offset
918                                + size_of::<hopr_protocol_pix::PolynomialIndex>()
919                                + Self::PIX_COEFF_COMMITMENT_REPR_SIZE
920                        {
921                            return Err(StartProtocolError::InvalidLength);
922                        }
923
924                        let index = hopr_protocol_pix::PolynomialIndex::from_be_bytes(
925                            body[next_offset..next_offset + size_of::<hopr_protocol_pix::PolynomialIndex>()]
926                                .try_into()
927                                .map_err(|_| StartProtocolError::ParseError("polynomial_index".into()))?,
928                        );
929                        next_offset += size_of::<hopr_protocol_pix::PolynomialIndex>();
930
931                        let commitment =
932                            G::try_from(&body[next_offset..next_offset + Self::PIX_COEFF_COMMITMENT_REPR_SIZE])
933                                .map_err(|_| StartProtocolError::ParseError("commitment".into()))?;
934                        next_offset += Self::PIX_COEFF_COMMITMENT_REPR_SIZE;
935
936                        if coefficient_commitments.insert(index, commitment).is_some() {
937                            return Err(StartProtocolError::DuplicateCommitment);
938                        }
939                    }
940
941                    StartProtocol::SsaCommit(SsaClientCommitmentMessage {
942                        session_id: serde_cbor_2::from_slice(&body[next_offset..])?,
943                        ssa_index: ssa,
944                        coefficient_index,
945                        commitment_proof,
946                        coefficient_commitments,
947                    })
948                }
949                StartProtocolDiscriminants::SsaRequest => {
950                    if body.len() <= size_of::<u32>() + 1 {
951                        return Err(StartProtocolError::InvalidLength);
952                    }
953
954                    let params = u32::from_be_bytes(
955                        body[..size_of::<u32>()]
956                            .try_into()
957                            .map_err(|_| StartProtocolError::ParseError("params".into()))?,
958                    );
959
960                    // deposit_data is CBOR — decode using a deserializer that tracks its
961                    // byte offset so we can skip only its size and leave the rest of the body
962                    // (num_commitments + entries + session_id) untouched. This holds for the map
963                    // just as it did for a single value: the offset is where the one CBOR item
964                    // ended, whatever its shape.
965                    //
966                    // The `SsaIndex` keys are `NonZero`, so a peer sending index 0 is rejected here
967                    // by serde rather than reaching the session layer — the same guard the
968                    // commitment entries get below, for free.
969                    let mut de = serde_cbor_2::Deserializer::from_slice(&body[size_of::<u32>()..]);
970                    let deposit_data = decode_deposit_data::<D, _>(&mut de, Self::MAX_SSAS_PER_REQUEST as usize)
971                        .map_err(|e| StartProtocolError::ParseError(format!("deposit_data: {e}")))?;
972                    let deposit_data_len = de.byte_offset();
973                    let mut next_offset = size_of::<u32>() + deposit_data_len;
974
975                    if body.len() <= next_offset + size_of::<u16>() {
976                        return Err(StartProtocolError::InvalidLength);
977                    }
978
979                    let num_commitments = u16::from_be_bytes(
980                        body[next_offset..next_offset + size_of::<u16>()]
981                            .try_into()
982                            .map_err(|_| StartProtocolError::ParseError("num_commitments".into()))?,
983                    );
984                    if num_commitments == 0 || num_commitments > Self::MAX_SSAS_PER_REQUEST {
985                        return Err(StartProtocolError::NumberOfCommitments);
986                    }
987                    next_offset += size_of::<u16>();
988
989                    let mut commitments = std::collections::BTreeMap::new();
990                    for _ in 0..num_commitments {
991                        if body.len()
992                            <= next_offset
993                                + size_of::<hopr_protocol_pix::SsaIndex>()
994                                + Self::PIX_COEFF_COMMITMENT_REPR_SIZE
995                        {
996                            return Err(StartProtocolError::InvalidLength);
997                        }
998
999                        let ssa_index: hopr_protocol_pix::SsaIndex = hopr_protocol_pix::RawSsaIndex::from_be_bytes(
1000                            body[next_offset..next_offset + size_of::<hopr_protocol_pix::SsaIndex>()]
1001                                .try_into()
1002                                .map_err(|_| StartProtocolError::ParseError("ssa_index".into()))?,
1003                        )
1004                        .try_into()
1005                        .map_err(|_| StartProtocolError::ParseError("ssa_index is 0".into()))?;
1006                        next_offset += size_of::<hopr_protocol_pix::SsaIndex>();
1007
1008                        let commitment =
1009                            G::try_from(&body[next_offset..next_offset + Self::PIX_COEFF_COMMITMENT_REPR_SIZE])
1010                                .map_err(|_| StartProtocolError::ParseError("commitment".into()))?;
1011                        next_offset += Self::PIX_COEFF_COMMITMENT_REPR_SIZE;
1012
1013                        if commitments.insert(ssa_index, commitment).is_some() {
1014                            return Err(StartProtocolError::DuplicateCommitment);
1015                        }
1016                    }
1017
1018                    StartProtocol::SsaRequest(SsaServerCommitmentMessage {
1019                        session_id: serde_cbor_2::from_slice(&body[next_offset..])?,
1020                        params,
1021                        deposit_data,
1022                        commitments,
1023                    })
1024                }
1025            },
1026        )
1027    }
1028}

1732    /// Type the hand-framed bodies are decoded as: 33-byte commitments and a 65-byte proof, the
1733    /// same shape the production instantiation uses.

1736    /// Wraps a raw body in the Start protocol envelope (version, discriminant, 16-bit body length)
1737    /// exactly as `encode` does. Malformed-input tests need bodies the encoder would refuse to
1738    /// produce, so they frame them by hand.
1739    fn frame(disc: StartProtocolDiscriminants, body: &[u8]) -> Vec<u8> {
1740        let mut out = vec![Decoder::START_PROTOCOL_VERSION, disc as u8];
1741        out.extend_from_slice(&(body.len() as u16).to_be_bytes());
1742        out.extend_from_slice(body);
1743        out
1744    }

1746    /// Builds an `SsaCommit` body: fixed prefix, optional proof, entries, trailing CBOR session id.
1747    /// `declared_polys` is written to the wire as-is, so it can disagree with `entries` — which is
1748    /// the whole point of the tests that use it.
1749    fn ssa_commit_body(
1750        coefficient_index: u16,
1751        declared_polys: u16,
1752        proof: Option<&[u8]>,
1753        entries: &[(u16, [u8; 33])],
1754    ) -> Vec<u8> {
1755        let mut body = Vec::new();
1756        body.extend_from_slice(&1u32.to_be_bytes());
1757        body.extend_from_slice(&coefficient_index.to_be_bytes());
1758        body.extend_from_slice(&declared_polys.to_be_bytes());
1759        if let Some(proof) = proof {
1760            body.extend_from_slice(proof);
1761        }
1762        for (index, commitment) in entries {
1763            body.extend_from_slice(&index.to_be_bytes());
1764            body.extend_from_slice(commitment);
1765        }
1766        body.extend(serde_cbor_2::to_vec(&MALFORMED_SESSION_ID).expect("session id must serialize"));
1767        body
1768    }

2048    /// A map header is the peer's word for how many entries follow, and a packet of a few dozen
2049    /// bytes is free to claim billions. Sizing an allocation from that claim is the whole problem:
2050    /// the count a request can carry is bounded by its commitments, and a header above that bound is
2051    /// refused on the header, before it has been used to size anything.
2052    #[test]
2053    fn start_protocol_decode_should_reject_a_deposit_data_map_larger_than_a_request_can_carry() {
2054        let over_the_bound = |context: &str| context.starts_with("deposit_data") && context.contains("at most");
2055
2056        // A definite-length CBOR map (0xBB) announcing every entry the address space can hold, in a
2057        // body that carries none of them.
2058        let mut announced = vec![0xbb];
2059        announced.extend_from_slice(&u64::MAX.to_be_bytes());
2060        let body = ssa_request_body_with_deposit_data(&announced, 1, &[(1, [0u8; 33])]);
2061        assert!(
2062            matches!(decode_framed(StartProtocolDiscriminants::SsaRequest, &body), Err(StartProtocolError::ParseError(ref c)) if over_the_bound(c)),
2063            "a map header claiming {} entries must be refused on the claim: {:?}",
2064            u64::MAX,
2065            decode_framed(StartProtocolDiscriminants::SsaRequest, &body)
2066        );
2067
2068        // Complete and well-formed, one entry over the bound — the case a header check alone would
2069        // let through, and the only difference from the accepted body below.
2070        let cbor_map = |entries: u16| {
2071            serde_cbor_2::to_vec(
2072                &(1..=entries as u32)
2073                    .map(|i| (i, ()))
2074                    .collect::<std::collections::BTreeMap<_, _>>(),
2075            )
2076            .expect("deposit data map must serialize")
2077        };
2078
2079        let body =
2080            ssa_request_body_with_deposit_data(&cbor_map(Decoder::MAX_SSAS_PER_REQUEST + 1), 1, &[(1, [0u8; 33])]);
2081        assert!(
2082            matches!(decode_framed(StartProtocolDiscriminants::SsaRequest, &body), Err(StartProtocolError::ParseError(ref c)) if over_the_bound(c)),
2083            "a complete map one entry over the bound must be refused: {:?}",
2084            decode_framed(StartProtocolDiscriminants::SsaRequest, &body)
2085        );
2086
2087        // An indefinite-length map (0xbf … 0xff) announces no count at all, so the header check has
2088        // nothing to refuse and the per-entry one is the only bound left. Repeating a single key
2089        // makes the map's own size useless as a counter: every entry overwrites the last, so the
2090        // decoder holds one pair no matter how many went past it. Only the entries themselves are
2091        // hand-framed; the pairs come from the serializer.
2092        let mut repeated_key = vec![0xbf];
2093        for _ in 0..=Decoder::MAX_SSAS_PER_REQUEST {
2094            repeated_key.extend(serde_cbor_2::to_vec(&1u32).expect("an ssa index must serialize"));
2095            repeated_key.extend(serde_cbor_2::to_vec(&MinimalDeposit::default()).expect("deposit data must serialize"));
2096        }
2097        repeated_key.push(0xff);
2098
2099        let body = ssa_request_body_with_deposit_data(&repeated_key, 1, &[(1, [0u8; 33])]);
2100        assert!(
2101            matches!(decode_framed(StartProtocolDiscriminants::SsaRequest, &body), Err(StartProtocolError::ParseError(ref c)) if over_the_bound(c)),
2102            "an indefinite map must be bounded by the entries it carries, not the keys it keeps: {:?}",
2103            decode_framed(StartProtocolDiscriminants::SsaRequest, &body)
2104        );
2105
2106        // And the bound itself still parses, entries and all.
2107        let body = ssa_request_body_with_deposit_data(&cbor_map(Decoder::MAX_SSAS_PER_REQUEST), 1, &[(1, [0u8; 33])]);
2108        let decoded = decode_framed(StartProtocolDiscriminants::SsaRequest, &body);
2109        assert!(
2110            matches!(decoded, Ok(StartProtocol::SsaRequest(ref req)) if req.deposit_data.len() == Decoder::MAX_SSAS_PER_REQUEST as usize),
2111            "a map at the bound must decode with all of its entries: {decoded:?}"
2112        );
2113    }
