1//! Judges forward relayers from the receipts an Exit sends for the Session packets it received.
2//!
3//! Each Session packet sent over a forward path is recorded with its secret-derived tag. A receipt lists the tags
4//! that reached the Exit; a packet still unreceipted once later packets of the same Session are (`FORWARD_GRACE`)
5//! is overdue, and counts against the relayers on its path that nothing has been delivered through since. A relayer
6//! that the overdue packets point at, after `explain_away` and `FORWARD_CONFIRMATION`, is excluded from the forward
7//! draws to that Exit until the verdict is lifted. Everything is judged within one pseudonym, so a Session's receipts
8//! never judge another Session's packets.
9

26/// The most packets recorded at once; the oldest is evicted first.
27const MAX_RECORDS: usize = 16_384;
28/// How long a packet is waited for, after which its record is dropped and counts for nothing.
29const RECORD_TTL: Duration = Duration::from_secs(10);
30/// How long a packet may trail a receipted one, sent after it, before it is overdue.
31const FORWARD_GRACE: Duration = Duration::from_millis(500);
32/// How long a verdict excludes a relayer.
33const FORWARD_VERDICT_TTL: Duration = Duration::from_secs(30);
34/// The most relayers excluded for one Exit at once, which bounds how far a lying Exit can steer.
35const MAX_FORWARD_VERDICTS_PER_EXIT: usize = 2;
36/// How long a suspect stands as a black hole before it is judged.
37const FORWARD_CONFIRMATION: Duration = Duration::from_millis(500);
38/// How long a delivery is kept without news.
39const STATE_TTL: Duration = Duration::from_secs(60);
40/// How long a receipt stream is kept without news: longer than the default idle timeout (3 min), so that the first
41/// receipt after an idle continues it.
42const STREAM_TTL: Duration = Duration::from_secs(600);
43/// How old a stream must be to be reset by a reused pseudonym: a copy of an early receipt, delayed on a stalled return
44/// path, is no new Session.
45const RESET_AFTER: Duration = RECORD_TTL;
46/// The most receipts one receipt is held to have skipped over; a larger gap is a loss.
47const MAX_MISSING: u32 = 16;
48/// The highest `seq` a new Session sends first (see the Exit's overflow jump).
49const NEW_SESSION_SEQ_MAX: u32 = 3;
50
51type Key = OffchainPublicKey;
52
53struct Record {
54    exit: Key,
55    /// The relayers, nearest to us first.
56    relayers: [Option<Key>; RoutingOptions::MAX_INTERMEDIATE_HOPS],
57    sent_at: Instant,
58    /// Another packet of the pseudonym had the same tag: a receipt for it says nothing about either, so it credits
59    /// nobody and the packets are never overdue.
60    ambiguous: bool,
61}
62
63impl Record {
64    fn relayers(&self) -> impl Iterator<Item = &Key> {
65        self.relayers.iter().flatten()
66    }
67}

69/// The receipts of one pseudonym.
70struct Stream {
71    last_seq: u32,
72    /// Packets sent before it are no evidence: receipts may have been lost.
73    floor: Instant,
74    /// The latest send time among the packets receipted.
75    watermark: Option<Instant>,
76    seen: Instant,
77    /// When the stream began.
78    started: Instant,
79    /// Receipts skipped over, and since when; they may only be late, so nothing is judged while they are awaited.
80    missing: Vec<(u32, Instant)>,
81}

83/// A relayer that has been a black hole since `since`, over packets from `oldest` on.
84struct Suspicion {
85    since: Instant,
86    oldest: Instant,
87    pseudonym: HoprPseudonym,
88}

91struct State {
92    /// By pseudonym, so that judging a Session touches only its packets.
93    records: HashMap<HoprPseudonym, HashMap<u32, Record>>,
94    /// How many records there are in all.
95    count: usize,
96    /// Records by send time; entries of records gone are skipped.
97    order: VecDeque<(HoprPseudonym, u32, Instant)>,
98    streams: HashMap<HoprPseudonym, Stream>,
99    /// When the packet sent last got through each (exit, relayer).
100    delivered: HashMap<(Key, Key), Instant>,
101    /// The (exit, relayer) pairs excluded, and since when.
102    verdicts: HashMap<(Key, Key), Instant>,
103    /// When each verdict was last lifted; a relapse within the TTL is judged without the wait.
104    lifted: HashMap<(Key, Key), Instant>,
105    /// The relayers that are black holes but have not yet stood for `FORWARD_CONFIRMATION`.
106    suspected: HashMap<(Key, Key), Suspicion>,
107    /// Whether the verdicts changed since they were last published.
108    changed: bool,
109}

111/// Records forward Session packets and judges their relayers from the Exit's receipts.
112#[derive(Default)]
113pub struct ForwardReceipts {
114    any_verdict: AtomicBool,
115    /// The relayers excluded per Exit, for the planner.
116    excluded: RwLock<HashMap<Key, Vec<Key>>>,
117    /// The pseudonyms whose Session asked for receipts, with the Session instance: only theirs are recorded, so that
118    /// Sessions without receipts cannot crowd out the evidence of the others. Kept from the Session's start to its
119    /// end, however long it is idle.
120    active: DashMap<HoprPseudonym, u64>,
121    state: Mutex<State>,
122}

130    /// Judges `relayer` for `exit` as of `at`, for the tests of the reply rule.
131    #[cfg(test)]
132    pub(crate) fn judge_at(&self, exit: Key, relayer: Key, at: Instant) {
133        let mut state = self.state.lock();
134        state.verdicts.insert((exit, relayer), at);
135        state.changed = true;
136        state.publish(&self.excluded, &self.any_verdict);
137    }

139    /// The relayers excluded from forward draws to `exit`; the only cost without verdicts is an atomic load.
140    pub fn excluded_relayers(&self, exit: &Key) -> Vec<Key> {
141        if !self.any_verdict.load(Ordering::Relaxed) {
142            return Vec::new();
143        }
144        self.excluded.read().get(exit).cloned().unwrap_or_default()
145    }
146
147    /// The (exit, relayer) pairs with a forward verdict; cheap while there is none.
148    pub fn judged_pairs(&self) -> std::collections::HashSet<(Key, Key)> {
149        if !self.any_verdict.load(Ordering::Relaxed) {
150            return Default::default();
151        }
152        self.excluded
153            .read()
154            .iter()
155            .flat_map(|(exit, relayers)| relayers.iter().map(|relayer| (*exit, *relayer)))
156            .collect()
157    }

159    /// Records a Session packet sent over `path`, the relayers and then the Exit, if its Session asked for receipts.
160    pub fn record(&self, pseudonym: HoprPseudonym, tag: u32, path: &[Key]) {
161        if !self.active.contains_key(&pseudonym) {
162            return;
163        }
164        self.record_at(pseudonym, tag, path, Instant::now());
165    }

167    /// Starts recording the packets of `pseudonym`, forgetting what an earlier Session under it left.
168    fn start_session(&self, pseudonym: HoprPseudonym, instance: u64) {
169        {
170            let mut state = self.state.lock();
171            state.remove_where_owned(pseudonym);
172            state.streams.remove(&pseudonym);
173        }
174        self.active.insert(pseudonym, instance);
175    }

177    fn record_at(&self, pseudonym: HoprPseudonym, tag: u32, path: &[Key], sent_at: Instant) {
178        let Some((exit, relayers)) = path.split_last() else {
179            return;
180        };
181        let mut slots = [None; RoutingOptions::MAX_INTERMEDIATE_HOPS];
182        for (slot, relayer) in slots.iter_mut().zip(relayers) {
183            *slot = Some(*relayer);
184        }
185        let mut state = self.state.lock();
186        while state.count >= MAX_RECORDS || state.order.len() >= 2 * MAX_RECORDS {
187            let Some((owner, tag, at)) = state.order.pop_front() else {
188                break;
189            };
190            if state.is_live(owner, tag, at) {
191                state.remove(owner, tag);
192            }
193        }
194        let record = Record {
195            exit: *exit,
196            relayers: slots,
197            sent_at,
198            ambiguous: false,
199        };
200        if state.insert(pseudonym, tag, record) {
201            state.order.push_back((pseudonym, tag, sent_at));
202        }
203    }

205    /// Takes the receipt numbered `seq` for `pseudonym`, which lists the `tags` that reached the Exit in the
206    /// intervals `first_seq..=seq`.
207    fn receipts_at(&self, pseudonym: HoprPseudonym, seq: u32, first_seq: u32, tags: &[u32], now: Instant) {
208        let first_seq = first_seq.min(seq);
209        let mut state = self.state.lock();
210
211        let mut gap = 0;
212        let before = state
213            .streams
214            .get(&pseudonym)
215            .map(|stream| (stream.last_seq, stream.missing.clone()));
216        match state.streams.get_mut(&pseudonym) {
217            // The pseudonym was reused: a new Session numbers from 1, or a little above it. Not a copy of an early
218            // receipt that arrives late, though.
219            Some(stream)
220                if now.saturating_duration_since(stream.started) > RESET_AFTER
221                    && ((seq == 1 && stream.last_seq > 1)
222                        || (seq <= NEW_SESSION_SEQ_MAX && seq.saturating_add(2) < stream.last_seq)) =>
223            {
224                *stream = Stream::new(seq, now);
225            }
226            Some(stream) => {
227                // The seqs after the last one and before the first this receipt covers are skipped over: maybe late
228                let next = stream.last_seq.saturating_add(1);
229                gap = first_seq.saturating_sub(next);
230                if gap > MAX_MISSING {
231                    stream.floor = now;
232                } else {
233                    stream.missing.extend((next..first_seq).map(|missing| (missing, now)));
234                }
235                stream
236                    .missing
237                    .retain(|(missing, _)| !(first_seq..=seq).contains(missing));
238                // Still missing after the grace: lost. It can only have carried packets that reached the Exit
239                // before the jump, so the evidence from then on stays.
240                let lost = stream
241                    .missing
242                    .iter()
243                    .filter(|(_, since)| now.saturating_duration_since(*since) > FORWARD_GRACE)
244                    .map(|(_, since)| *since)
245                    .max();
246                if let Some(since) = lost {
247                    stream.floor = stream.floor.max(since);
248                    stream
249                        .missing
250                        .retain(|(_, since)| now.saturating_duration_since(*since) <= FORWARD_GRACE);
251                }
252                stream.last_seq = stream.last_seq.max(seq);
253                stream.seen = now;
254            }
255            None => {
256                state.streams.insert(pseudonym, Stream::new(seq, now));
257            }
258        }
259
260        // A packet is timely if it is not far behind what was receipted before this receipt
261        let watermark_before = state.streams.get(&pseudonym).and_then(|stream| stream.watermark);
262        let mut matched = 0;
263        let mut lifts = Vec::new();
264        // Nothing to match for a pseudonym with no records
265        let to_match = if state.records.contains_key(&pseudonym) {
266            tags
267        } else {
268            &[]
269        };
270        for tag in to_match {
271            let Some(record) = state.remove(pseudonym, *tag) else {
272                continue;
273            };
274            if record.ambiguous || now.saturating_duration_since(record.sent_at) > RECORD_TTL {
275                continue;
276            }
277            matched += 1;
278            if let Some(stream) = state.streams.get_mut(&pseudonym) {
279                stream.watermark = stream.watermark.max(Some(record.sent_at));
280            }
281            let timely = watermark_before.is_some_and(|watermark| record.sent_at + FORWARD_GRACE > watermark);
282            for relayer in record.relayers() {
283                let key = (record.exit, *relayer);
284                let delivered = state.delivered.entry(key).or_insert(record.sent_at);
285                *delivered = (*delivered).max(record.sent_at);
286                // Only a timely packet sent after the verdict says the relayer is delivering again; a late one
287                // may have been drawn just before the verdict and be what it was about.
288                if timely
289                    && state
290                        .verdicts
291                        .get(&key)
292                        .is_some_and(|verdict| record.sent_at > *verdict)
293                {
294                    lifts.push(key);
295                }
296            }
297        }
298        for (exit, relayer) in lifts {
299            state.lift(exit, relayer, now);
300        }
301        let after = state
302            .streams
303            .get(&pseudonym)
304            .map(|stream| (stream.last_seq, stream.missing.clone()));
305        let waiting = state
306            .streams
307            .get(&pseudonym)
308            .is_some_and(|stream| !stream.missing.is_empty());
309        // A copy of a receipt already taken tells nothing new: it must not let a confirmation mature ahead of the
310        // next real receipt
311        let nothing_new = matched == 0 && before == after;
312        let outstanding = if waiting || nothing_new {
313            state.records.get(&pseudonym).map_or(0, HashMap::len)
314        } else {
315            state.evaluate(pseudonym, now)
316        };
317        state.publish(&self.excluded, &self.any_verdict);
318        tracing::debug!(%pseudonym, seq, first_seq, tags = tags.len(), matched, gap, outstanding, "forward receipts received");
319    }

321    /// Lifts the verdicts that have run their course and forgets what has expired. Called once a second.
322    pub fn refresh(&self) {
323        self.refresh_at(Instant::now());
324    }
325
326    fn refresh_at(&self, now: Instant) {
327        let mut state = self.state.lock();
328        let expired: Vec<_> = state
329            .verdicts
330            .iter()
331            .filter(|(_, since)| now.saturating_duration_since(**since) >= FORWARD_VERDICT_TTL)
332            .map(|(key, _)| *key)
333            .collect();
334        for (exit, relayer) in expired {
335            state.lift(exit, relayer, now);
336        }
337        state.remove_where(|record| now.saturating_duration_since(record.sent_at) > RECORD_TTL);
338        state.prune_order(now);
339        state
340            .streams
341            .retain(|_, stream| now.saturating_duration_since(stream.seen) <= STREAM_TTL);
342        state
343            .delivered
344            .retain(|_, at| now.saturating_duration_since(*at) <= STATE_TTL);
345        state
346            .lifted
347            .retain(|_, at| now.saturating_duration_since(*at) <= FORWARD_VERDICT_TTL);
348        state
349            .suspected
350            .retain(|_, suspicion| now.saturating_duration_since(suspicion.since) <= RECORD_TTL);
351        state.publish(&self.excluded, &self.any_verdict);
352    }

368impl State {
369    fn is_live(&self, owner: HoprPseudonym, tag: u32, at: Instant) -> bool {
370        self.records
371            .get(&owner)
372            .and_then(|records| records.get(&tag))
373            .is_some_and(|record| record.sent_at == at)
374    }
375
376    /// Inserts the record, unless the tag is taken: then the earlier record is marked ambiguous and kept.
377    /// Returns whether it was inserted.
378    fn insert(&mut self, owner: HoprPseudonym, tag: u32, record: Record) -> bool {
379        match self.records.entry(owner).or_default().entry(tag) {
380            std::collections::hash_map::Entry::Occupied(mut taken) => {
381                taken.get_mut().ambiguous = true;
382                false
383            }
384            std::collections::hash_map::Entry::Vacant(free) => {
385                free.insert(record);
386                self.count += 1;
387                true
388            }
389        }
390    }
391
392    fn remove(&mut self, owner: HoprPseudonym, tag: u32) -> Option<Record> {
393        let records = self.records.get_mut(&owner)?;
394        let record = records.remove(&tag)?;
395        if records.is_empty() {
396            self.records.remove(&owner);
397        }
398        self.count -= 1;
399        Some(record)
400    }
401
402    /// Drops every record of `owner`.
403    fn remove_where_owned(&mut self, owner: HoprPseudonym) {
404        if let Some(records) = self.records.remove(&owner) {
405            self.count -= records.len();
406        }
407    }
408
409    fn remove_where(&mut self, gone: impl Fn(&Record) -> bool) {
410        for records in self.records.values_mut() {
411            records.retain(|_, record| !gone(record));
412        }
413        self.records.retain(|_, records| !records.is_empty());
414        self.count = self.records.values().map(HashMap::len).sum();
415    }
416
417    /// Drops the order entries of records gone or expired.
418    fn prune_order(&mut self, now: Instant) {
419        while let Some((owner, tag, at)) = self.order.front().copied() {
420            let live = self.is_live(owner, tag, at);
421            if live && now.saturating_duration_since(at) <= RECORD_TTL {
422                break;
423            }
424            if live {
425                self.remove(owner, tag);
426            }
427            self.order.pop_front();
428        }
429    }
430
431    /// Lifts the verdict on `relayer` for `exit`, dropping the packets sent through it before the lift.
432    fn lift(&mut self, exit: Key, relayer: Key, now: Instant) {
433        if self.verdicts.remove(&(exit, relayer)).is_none() {
434            return;
435        }
436        tracing::info!(%relayer, %exit, "forward verdict lifted");
437        self.changed = true;
438        self.lifted.insert((exit, relayer), now);
439        self.suspected.remove(&(exit, relayer));
440        self.remove_where(|record| record.exit == exit && record.relayers().any(|other| *other == relayer));
441    }
442
443    /// Judges the packets of `pseudonym` that are overdue; returns how many it has outstanding.
444    fn evaluate(&mut self, pseudonym: HoprPseudonym, now: Instant) -> usize {
445        let (Some(stream), Some(records)) = (self.streams.get(&pseudonym), self.records.get(&pseudonym)) else {
446            return 0;
447        };
448        let (floor, watermark) = (stream.floor, stream.watermark);
449        let live = || {
450            records
451                .iter()
452                .filter(|(_, record)| !record.ambiguous && now.saturating_duration_since(record.sent_at) <= RECORD_TTL)
453        };
454
455        // Per (exit, relayer): the unanswered overdue packets.
456        let mut unanswered: HashMap<(Key, Key), Vec<(HoprSurbId, Instant)>> = HashMap::new();
457        let mut outstanding = 0;
458        for (tag, record) in live() {
459            outstanding += 1;
460            let overdue = watermark
461                .is_some_and(|watermark| record.sent_at >= floor && record.sent_at + FORWARD_GRACE <= watermark);
462            if !overdue {
463                continue;
464            }
465            for relayer in record.relayers() {
466                let key = (record.exit, *relayer);
467                if !self.delivered.get(&key).is_some_and(|at| record.sent_at <= *at) {
468                    unanswered.entry(key).or_default().push((id_of(*tag), record.sent_at));
469                }
470            }
471        }
472        unanswered.retain(|_, packets| packets.len() >= MIN_RACES);
473
474        // The packets outstanding through each of those, with the relayer's position on their paths.
475        let mut open: HashMap<(Key, Key), HashMap<HoprSurbId, usize>> = HashMap::new();
476        if !unanswered.is_empty() {
477            for (tag, record) in live() {
478                for (position, relayer) in record.relayers().enumerate() {
479                    let key = (record.exit, *relayer);
480                    if unanswered.contains_key(&key) {
481                        open.entry(key).or_default().insert(id_of(*tag), position);
482                    }
483                }
484            }
485        }
486
487        let mut suspects_by_exit: HashMap<Key, Vec<Suspect<Key>>> = HashMap::new();
488        let mut oldest: HashMap<(Key, Key), Instant> = HashMap::new();
489        for (key, packets) in unanswered {
490            if let Some(at) = packets.iter().map(|(_, at)| *at).min() {
491                oldest.insert(key, at);
492            }
493            suspects_by_exit.entry(key.0).or_default().push(Suspect {
494                relayer: key.1,
495                unanswered: packets.into_iter().map(|(id, _)| id).collect(),
496                open: open.remove(&key).unwrap_or_default(),
497                needed: MIN_RACES,
498            });
499        }
500
501        let mut black_holes: HashSet<(Key, Key)> = HashSet::new();
502        for (exit, suspects) in suspects_by_exit {
503            let known_dead = |relayer: &Key| self.verdicts.contains_key(&(exit, *relayer));
504            for relayer in explain_away(suspects, known_dead) {
505                black_holes.insert((exit, relayer));
506            }
507        }
508
509        for key in &black_holes {
510            if self.verdicts.contains_key(key) {
511                continue;
512            }
513            let Some(oldest) = oldest.get(key).copied() else {
514                continue;
515            };
516            let delivered = self.delivered.get(key).copied();
517            let suspicion = self.suspected.entry(*key).or_insert(Suspicion {
518                since: now,
519                oldest,
520                pseudonym,
521            });
522            if delivered.is_some_and(|at| at >= suspicion.oldest) {
523                // It delivers after all.
524                *suspicion = Suspicion {
525                    since: now,
526                    oldest,
527                    pseudonym,
528                };
529                continue;
530            }
531            let relapse = self
532                .lifted
533                .get(key)
534                .is_some_and(|at| now.saturating_duration_since(*at) < FORWARD_VERDICT_TTL);
535            if !relapse && now.saturating_duration_since(suspicion.since) < FORWARD_CONFIRMATION {
536                continue;
537            }
538            if self.verdicts.keys().filter(|(exit, _)| *exit == key.0).count() >= MAX_FORWARD_VERDICTS_PER_EXIT {
539                continue;
540            }
541            let (exit, relayer) = *key;
542            tracing::info!(%relayer, %exit, over = "forward", "relayer judged dead");
543            self.suspected.remove(key);
544            self.verdicts.insert(*key, now);
545            self.changed = true;
546        }
547        // A suspicion that is no black hole any more has not stood.
548        self.suspected
549            .retain(|key, suspicion| suspicion.pseudonym != pseudonym || black_holes.contains(key));
550        outstanding
551    }
552
553    /// Publishes the verdicts for the planner, if they changed.
554    fn publish(&mut self, excluded: &RwLock<HashMap<Key, Vec<Key>>>, any_verdict: &AtomicBool) {
555        if !std::mem::take(&mut self.changed) {
556            return;
557        }
558        let mut view: HashMap<Key, Vec<Key>> = HashMap::new();
559        for (exit, relayer) in self.verdicts.keys() {
560            view.entry(*exit).or_default().push(*relayer);
561        }
562        *excluded.write() = view;
563        any_verdict.store(!self.verdicts.is_empty(), Ordering::Relaxed);
564    }

567/// The record of a tag, in the form `explain_away` keys its replies by.
568fn id_of(tag: u32) -> HoprSurbId {
569    let mut id = [0u8; SURB_ID_SIZE];
570    id[..size_of::<u32>()].copy_from_slice(&tag.to_be_bytes());
571    id
572}

574impl ForwardReceiptSink for ForwardReceipts {

579    fn session_started(&self, pseudonym: HoprPseudonym, instance: u64) {

648    /// A Session sending a packet every 10 ms and receipted every 250 ms, with the Exit's view of time in
649    /// milliseconds from `t0`.
650    struct Sim {
651        fr: ForwardReceipts,
652        pseudonym: HoprPseudonym,
653        exit: Key,
654        t0: Instant,
655        next_tag: u32,
656        seq: u32,
657        /// (tag, arrival at the Exit in ms, if it arrives)
658        sent: Vec<(u32, Option<u64>)>,
659        /// When the receipts were made, the lost ones included.
660        receipted: Vec<u64>,
661        /// How many intervals a receipt covers.
662        cover: usize,
663        /// The seq of the last overflow jump, and when the receipt before it was made: older intervals are not
664        /// repeated.
665        jump_base: (u32, u64),
666        verdict_ms: Option<u64>,
667        /// How long the receipt with a `seq` is delayed, and the receipts delayed: (release in ms, seq, first seq,
668        /// tags).
669        late: Vec<(u32, u64)>,
670        held: Vec<(u64, u32, u32, Vec<u32>)>,
671    }
672
673    impl Sim {
674        fn new() -> Self {
675            Self::sharing(ForwardReceipts::default(), key(), Instant::now())
676        }
677
678        fn sharing(fr: ForwardReceipts, exit: Key, t0: Instant) -> Self {
679            Self {
680                fr,
681                pseudonym: HoprPseudonym::random(),
682                exit,
683                t0,
684                next_tag: 0,
685                seq: 0,
686                sent: Vec::new(),
687                receipted: Vec::new(),
688                cover: 2,
689                jump_base: (0, 0),
690                verdict_ms: None,
691                late: Vec::new(),
692                held: Vec::new(),
693            }
694        }
695
696        fn at(&self, ms: u64) -> Instant {
697            self.t0 + Duration::from_millis(ms)
698        }
699
700        fn send(&mut self, ms: u64, relayers: &[Key], delay: Option<u64>) {
701            let mut path = relayers.to_vec();
702            path.push(self.exit);
703            self.fr.record_at(self.pseudonym, self.next_tag, &path, self.at(ms));
704            self.sent.push((self.next_tag, delay.map(|delay| ms + delay)));
705            self.next_tag += 1;
706        }
707
708        /// Makes the receipt at `ms` with the next `seq` plus `jump`, covering `cover` intervals but none from before
709        /// an Exit overflow (a jump), which repeats nothing.
710        fn receipt(&mut self, ms: u64, jump: u32) {
711            self.seq += 1 + jump;
712            let covered = if jump > 0 { 1 } else { self.cover };
713            if jump > 0 {
714                self.jump_base = (self.seq, self.receipted.last().copied().unwrap_or(0));
715            }
716            let first_seq = self.seq.saturating_sub(covered as u32 - 1).max(1).max(self.jump_base.0);
717            let cut = self
718                .receipted
719                .len()
720                .checked_sub(covered)
721                .map_or(0, |i| self.receipted[i])
722                .max(self.jump_base.1);
723            let tags: Vec<u32> = self
724                .sent
725                .iter()
726                .filter(|(_, arrival)| arrival.is_some_and(|a| a <= ms && a > cut))
727                .map(|(tag, _)| *tag)
728                .collect();
729            self.receipted.push(ms);
730            match self.late.iter().find(|(seq, _)| *seq == self.seq) {
731                Some((_, delay)) => self.held.push((ms + delay, self.seq, first_seq, tags)),
732                None => self.deliver(ms, self.seq, first_seq, &tags),
733            }
734        }
735
736        fn deliver(&mut self, ms: u64, seq: u32, first_seq: u32, tags: &[u32]) {
737            self.fr.receipts_at(self.pseudonym, seq, first_seq, tags, self.at(ms));
738            if self.verdict_ms.is_none() && !self.fr.excluded_relayers(&self.exit).is_empty() {
739                self.verdict_ms = Some(ms);
740            }
741        }
742
743        /// Delivers the delayed receipts that are due at `ms`.
744        fn release(&mut self, ms: u64) {
745            let (due, held): (Vec<_>, Vec<_>) = std::mem::take(&mut self.held)
746                .into_iter()
747                .partition(|(at, ..)| *at <= ms);
748            self.held = held;
749            for (_, seq, first_seq, tags) in due {
750                self.deliver(ms, seq, first_seq, &tags);
751            }
752        }
753
754        /// Runs from `from` to `to` ms: a packet every 10 ms over `route(i)` that arrives after `fate(path)`, and
755        /// a receipt every 250 ms unless `lost(seq)`.
756        fn run(
757            &mut self,
758            from: u64,
759            to: u64,
760            route: impl Fn(usize, &[Key]) -> Vec<Key>,
761            fate: impl Fn(&[Key]) -> Option<u64>,
762            lost: impl Fn(u32) -> bool,
763        ) {
764            for ms in (from..to).step_by(10) {
765                self.release(ms);
766                let excluded = self.fr.excluded_relayers(&self.exit);
767                let path = route((ms / 10) as usize, &excluded);
768                let delay = fate(&path);
769                self.send(ms, &path, delay);
770                if ms % 250 == 0 && ms > 0 {
771                    if lost(self.seq + 1) {
772                        self.seq += 1;
773                        self.receipted.push(ms);
774                    } else {
775                        self.receipt(ms, 0);
776                    }
777                }
778            }
779        }
780    }

782    /// Round-robin over `paths`, skipping the paths through an excluded relayer.
783    fn rotate(paths: Vec<Vec<Key>>) -> impl Fn(usize, &[Key]) -> Vec<Key> {
784        move |i, excluded| {
785            let allowed: Vec<&Vec<Key>> = paths
786                .iter()
787                .filter(|path| !path.iter().any(|relayer| excluded.contains(relayer)))
788                .collect();
789            let pool: Vec<&Vec<Key>> = if allowed.is_empty() {
790                paths.iter().collect()
791            } else {
792                allowed
793            };
794            pool[i % pool.len()].clone()
795        }
796    }

798    /// Runs `scenario` under a capturing subscriber until its log has a line with `expect`. A scoped subscriber can
799    /// miss events while other tests install theirs, so a miss runs the scenario again.
800    fn logged<T>(expect: &str, scenario: impl Fn() -> anyhow::Result<T>) -> anyhow::Result<(T, Vec<String>)> {
801        for _ in 0..20 {
802            let logs = Logs::default();
803            let out = tracing::subscriber::with_default(logs.clone(), &scenario)?;
804            let lines = logs.0.lock().clone();
805            if lines.iter().any(|line| line.contains(expect)) {
806                return Ok((out, lines));
807            }
808        }
809        anyhow::bail!("`{expect}` was never logged")
810    }
