31/// Default concurrency for the incoming acknowledgement processing pipeline when not overridden
32/// via [`AcknowledgementPipelineConfig::ack_input_concurrency`].
33const DEFAULT_ACK_INPUT_CONCURRENCY: usize = 10;
34/// Default concurrency for the outgoing acknowledgement processing pipeline when not overridden
35/// via [`AcknowledgementPipelineConfig::ack_output_concurrency`].
36const DEFAULT_ACK_OUTPUT_CONCURRENCY: usize = 10;
37const QUEUE_SEND_TIMEOUT: std::time::Duration = std::time::Duration::from_millis(50);
38// Set well above the worst-case rayon queue wait so packets are never silently
39// dropped due to CPU contention.  The formula (Sphinx ~21 ms/packet, up to 64
40// concurrent tasks, 8-core rayon pool) gives a worst-case queue wait of
41// 56 × 21 ms / 8 ≈ 147 ms — uncomfortably close to the old 150 ms ceiling.
42// Under any real CPU load (container overhead, background node traffic) encoding
43// regularly exceeded 150 ms, causing FuturesOrdered to emit None and silently
44// drop the packet, which then caused the session sequencer on the far side to
45// discard the next in-order frame after frame_timeout.
46//
47// 5 s gives headroom for sustained saturation while still providing a safety
48// net against truly hung encoding futures.  Normal encoding (~21 ms) is
49// unaffected; only pathological cases hit the ceiling.
50const PACKET_DECODING_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
51const PACKET_ENCODING_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
52
53/// Number of Rayon threads kept permanently free for outgoing packet encode (SURB generation).
54/// The ingress decode concurrency default is `pool_thread_count - ENCODE_RESERVED_THREADS` so
55/// that heavy download traffic cannot starve the upload/SURB replenishment path.
56const ENCODE_RESERVED_THREADS: usize = 2;
57
58/// Artificial per-packet delay injected into `wire_in` when the Rayon pool is detected as
59/// congested. 20 ms keeps the decode queue from growing while still allowing acks and keep-alive
60/// SURB packets to drain through at a reasonable pace.
61const INGRESS_THROTTLE_DELAY: std::time::Duration = std::time::Duration::from_millis(20);
62
63/// Pool outstanding-task watermark factor: the gate trips when
64/// `outstanding_tasks > pool_thread_count * INGRESS_POOL_HIGH_WATERMARK_FACTOR`.
65/// A factor of 3 gives one full pool worth of headroom above the cap.
66const INGRESS_POOL_HIGH_WATERMARK_FACTOR: usize = 3;

128/// Performs encoding of outgoing Application protocol packets into HOPR protocol outgoing packets.
129async fn start_outgoing_packet_pipeline<AppOut, A, E, WOut, WOutErr>(
130    app_outgoing: AppOut,
131    encoder: std::sync::Arc<E>,
132    exit_ack_proc: Option<A>,
133    wire_outgoing: WOut,
134    counters: PeerProtocolCounterRegistry,
135    concurrency: usize,
136) where
137    AppOut: futures::Stream<Item = (ResolvedTransportRouting<HoprSurb>, ApplicationDataOut)> + Send + 'static,
138    A: ExitAcknowledgementShareProcessor<HoprPixSpec> + Clone + Send + Sync + 'static,
139    E: PacketEncoder + Send + Sync + 'static,
140    WOut: futures::Sink<(PeerId, Bytes), Error = SinkTimeoutError<WOutErr>> + Clone + Unpin + Send + 'static,
141    WOutErr: std::error::Error,
142{
143    let res = app_outgoing
144        // Use `map().buffered()` (FuturesOrdered) instead of `then_concurrent()` (FuturesUnordered)
145        // so that encoded packets are emitted in submission order, preserving session frame sequence.
146        // Out-of-order sends to QUIC were causing frame reassembler discards on the receiver side.
147        .map(|(routing, data)| {
148            let encoder = encoder.clone();
149            let exit_ack_proc = exit_ack_proc.clone();
150            let counters = counters.clone();
151            async move {
152                hopr_transport_session::counters::ENCODE_STAGE_ENTRIES
153                    .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
154                match hopr_utils::parallelize::cpu::spawn_encode_blocking(
155                    move || {
156                        encoder.encode_packet(
157                            data.data.to_bytes(),
158                            routing,
159                            data.packet_info
160                                .map(|data| data.signals_to_destination)
161                                .unwrap_or_default(),
162                        )
163                    },
164                    "packet_encode",
165                )
166                .timeout(futures_time::time::Duration::from(PACKET_ENCODING_TIMEOUT))
167                .await
168                {
169                    Ok(Ok(Ok(packet))) => {
170                        #[cfg(all(feature = "telemetry", not(test)))]
171                        METRIC_PACKET_COUNT.increment(&["sent"]);
172                        counters.get_or_create(&packet.next_hop).record_message_sent();
173
174                        // If the pipeline has an exit acknowledgement processor (i.e., on an Exit node),
175                        // and the packet contains a non-empty encrypted partial SSA share (it is therefore an RP
176                        // packet), add it to the exit acknowledgement processor.
177                        if let Some(exit_ack_proc) = exit_ack_proc.as_ref()
178                            && let Some(encrypted_pix_share) =
179                                packet.encrypted_pix_share.filter(|s| !s.partial_share.is_empty())
180                        // Fail-safe to skip empty shares
181                        {
182                            let spi = encrypted_pix_share.ssa_polynomial_id();
183                            if let Err(error) = exit_ack_proc.insert_encrypted_share(
184                                &packet.next_hop,
185                                packet.ack_challenge,
186                                encrypted_pix_share,
187                            ) {
188                                #[cfg(all(feature = "telemetry", not(test)))]
189                                METRIC_SHARE_INSERT_FAILURES.increment();
190                                tracing::error!(
191                                    next_hop = packet.next_hop.to_peerid_str(),
192                                    ack_challenge = %packet.ack_challenge,
193                                    ssa_poly_id = ?spi,
194                                    %error,
195                                    "failed to insert encrypted share into the exit acknowledgement processor"
196                                );
197                            } else {
198                                tracing::trace!(
199                                    next_hop = packet.next_hop.to_peerid_str(),
200                                    "inserted new encrypted share into the exit acknowledgement processor"
201                                );
202                            }
203                        }
204
205                        tracing::trace!(peer = packet.next_hop.to_peerid_str(), "protocol message out");
206                        Some((packet.next_hop.into(), packet.data))
207                    }
208                    Ok(Ok(Err(error))) => {
209                        tracing::error!(%error, "outgoing packet could not be encoded");
210                        None
211                    }
212                    Ok(Err(error)) => {
213                        tracing::error!(%error, "parallel processing of the outgoing packet failed");
214                        None
215                    }
216                    Err(error) => {
217                        tracing::error!(%error, "timeout while processing the outgoing packet");
218                        hopr_utils::parallelize::cpu::ENCODE_TIMEOUT_DROPS
219                            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
220                        None
221                    }
222                }
223            }
224        })
225        .buffered(concurrency)
226        .filter_map(futures::future::ready)
227        .map(Ok)
228        .forward_to_timeout(wire_outgoing)
229        .in_current_span()
230        .await;
231
232    if let Err(error) = res {
233        tracing::error!(
234            task = "transport (protocol - msg egress)",
235            %error,
236            "long-running background task finished with error"
237        );
238    } else {
239        tracing::warn!(
240            task = "transport (protocol - msg egress)",
241            "long-running background task finished"
242        )
243    }
244}

246/// Performs HOPR protocol decoding of incoming packets into Application protocol packets.
247///
248/// `wire_incoming` --> `decoder` --> `ack_outgoing` (final + forwarded)
249///                             | --> `wire_outgoing` (forwarded)
250///                             | --> `ack_incoming` (forwarded)
251///                             | --> `app_incoming` (final)
252#[allow(clippy::too_many_arguments)]
253async fn start_incoming_packet_pipeline<WIn, WOut, D, T, TEvt, AckIn, AckOut, AppIn, AppInErr>(
254    (wire_outgoing, wire_incoming): (WOut, WIn),
255    decoder: std::sync::Arc<D>,
256    ticket_proc: std::sync::Arc<T>,
257    ticket_events: TEvt,
258    (ack_outgoing, ack_incoming): (AckOut, AckIn),
259    app_incoming: AppIn,
260    counters: super::counters::PeerProtocolCounterRegistry,
261    concurrency: usize,
262) where
263    WIn: futures::Stream<Item = (PeerId, Bytes)> + Send + 'static,
264    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
265    WOut::Error: std::error::Error,
266    D: PacketDecoder + Sync + Send + 'static,
267    T: UnacknowledgedTicketProcessor + Send + 'static,
268    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
269    TEvt::Error: std::error::Error,
270    AckIn: futures::Sink<(OffchainPublicKey, Vec<Acknowledgement>)> + Send + Unpin + Clone + 'static,
271    AckIn::Error: std::error::Error,
272    AckOut: futures::Sink<(OffchainPublicKey, Option<HalfKey>)> + Send + Unpin + Clone + 'static,
273    AckOut::Error: std::error::Error,
274    AppIn: futures::Sink<(HoprPseudonym, ApplicationDataIn), Error = SinkTimeoutError<AppInErr>> + Send + 'static,
275    AppInErr: std::error::Error,
276{
277    let ack_outgoing_success = ack_outgoing.clone();
278    let ack_outgoing_failure = ack_outgoing;
279    let ticket_proc_success = ticket_proc;
280
281    let res = wire_incoming
282        // Use `map().buffered()` (FuturesOrdered) instead of `then_concurrent()` (FuturesUnordered)
283        // so that decoded packets are emitted in the same order they arrive from the transport.
284        // Concurrent decode on rayon is preserved; only the output ordering guarantee changes.
285        // Out-of-order delivery was the root cause of session reassembler/sequencer discards.
286        .map(move |(peer, data)| {
287            let decoder = decoder.clone();
288            let mut ack_outgoing_failure = ack_outgoing_failure.clone();
289            let mut ticket_events_reject = ticket_events.clone();
290
291            tracing::trace!(%peer, "protocol message in");
292
293            async move {
294                match hopr_utils::parallelize::cpu::spawn_decode_blocking(move || decoder.decode(peer, data), "packet_decode")
295                    .timeout(futures_time::time::Duration::from(PACKET_DECODING_TIMEOUT))
296                    .await {
297                    Ok(Ok(Ok(packet))) => {
298                        tracing::trace!(%peer, ?packet, "successfully decoded incoming packet");
299                        Some(packet)
300                    },
301                    Ok(Ok(Err(IncomingPacketError::Undecodable(error)))) => {
302                        // Do not send an ack back if the packet could not be decoded at all
303                        //
304                        // Potentially adversarial behavior
305                        tracing::trace!(%peer, %error, "not sending ack back on undecodable packet - possible adversarial behavior");
306
307                        #[cfg(all(feature = "telemetry", not(test)))]
308                        METRIC_PACKET_REJECTED_COUNT.increment(&["undecodable"]);
309
310                        None
311                    },
312                    Ok(Ok(Err(IncomingPacketError::ProcessingError(sender, error)))) => {
313                        tracing::error!(%peer, %error, "failed to process the decoded packet");
314                        // On this failure, we send back a random acknowledgement
315                        ack_outgoing_failure
316                            .send((*sender, None))
317                            .await
318                            .unwrap_or_else(|error| {
319                                tracing::error!(%error, "failed to send ack to the egress queue");
320                            });
321
322                        #[cfg(all(feature = "telemetry", not(test)))]
323                        METRIC_PACKET_REJECTED_COUNT.increment(&["processing_error"]);
324
325                        None
326                    },
327                    Ok(Ok(Err(IncomingPacketError::InvalidTicket(sender, error)))) => {
328                        tracing::error!(%peer, %error, "failed to validate ticket on the received packet");
329                        if let Err(error) = ticket_events_reject
330                            .send(TicketEvent::RejectedTicket(error.ticket, error.issuer))
331                            .await {
332                            tracing::error!(%error, "failed to notify invalid ticket rejection");
333                        }
334                        // On this failure, we send back a random acknowledgement
335                        ack_outgoing_failure
336                            .send((*sender, None))
337                            .await
338                            .unwrap_or_else(|error| {
339                                tracing::error!(%error, "failed to send ack to the egress queue");
340                            });
341
342                        #[cfg(all(feature = "telemetry", not(test)))]
343                        {
344                            METRIC_VALIDATION_ERRORS.increment(&[error.kind.as_ref()]);
345                            METRIC_PACKET_REJECTED_COUNT.increment(&["invalid_ticket"]);
346                            METRIC_TICKETS_COUNT.increment(&["rejected"]);
347                        }
348
349                        None
350                    }
351                    Ok(Err(error)) => {
352                        tracing::error!(%error, "parallel processing of the incoming packet failed");
353                        None
354                    },
355                    Err(_) => {
356                        // If we cannot decode the packet within the time limit, just drop it
357                        tracing::error!(
358                            %peer,
359                            timeout_ms = PACKET_DECODING_TIMEOUT.as_millis() as u64,
360                            "dropped incoming packet: decode timeout - check the 'hopr_rayon_queue_wait_seconds' metric for pool saturation"
361                        );
362                        hopr_utils::parallelize::cpu::DECODE_TIMEOUT_DROPS
363                            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
364                        #[cfg(all(feature = "telemetry", not(test)))]
365                        {
366                            METRIC_PACKET_DECODE_TIMEOUTS.increment();
367                            METRIC_PACKET_REJECTED_COUNT.increment(&["timeout"]);
368                        }
369
370                        None
371                    }
372                }
373            }.instrument(tracing::debug_span!("incoming_packet_decode", %peer))
374        })
375        .buffered(concurrency)
376        .filter_map(futures::future::ready)
377        // Branch on the packet type BEFORE building the async future so each arm only clones
378        // the handles it actually needs. `futures::future::Either` lets us return three
379        // distinct async blocks from one closure without boxing.
380        // FuturesOrdered: preserve decode-stage order so session segments reach the socket in sequence.
381        .map(move |packet| {
382            match packet {
383                IncomingPacket::Acknowledgement(ack) => {
384                    let mut ack_incoming = ack_incoming.clone();
385                    let counters = counters.clone();
386                    Either::Left(async move {
387                        let IncomingAcknowledgementPacket { previous_hop, received_acks, .. } = *ack;
388                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), num_acks = received_acks.len(), "incoming acknowledgements");
389                        counters.get_or_create(&previous_hop).record_acks_received(received_acks.len() as u64);
390
391                        ack_incoming
392                            .send((previous_hop, received_acks))
393                            .await
394                            .unwrap_or_else(|error| {
395                                tracing::error!(%error, "failed dispatching received acknowledgement to the ticket ack queue");
396                            });
397
398                        // We do not acknowledge back acknowledgements.
399                        None
400                    })
401                }
402                IncomingPacket::Final(final_packet) => {
403                    let mut ack_outgoing_success = ack_outgoing_success.clone();
404                    Either::Right(Either::Left(async move {
405                        let IncomingFinalPacket {
406                            previous_hop,
407                            sender,
408                            plain_text,
409                            ack_key,
410                            info,
411                            ..
412                        } = *final_packet;
413                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), "incoming final packet");
414
415                        // Send acknowledgement back
416                        ack_outgoing_success
417                            .send((previous_hop, Some(ack_key)))
418                            .await
419                            .unwrap_or_else(|error| {
420                                tracing::error!(%error, "failed to send ack to the egress queue");
421                            });
422
423                        #[cfg(all(feature = "telemetry", not(test)))]
424                        METRIC_PACKET_COUNT.increment(&["received"]);
425
426                        Some((sender, plain_text, info))
427                    }))
428                }
429                IncomingPacket::Forwarded(fwd_packet) => {
430                    let ticket_proc = ticket_proc_success.clone();
431                    let mut wire_outgoing = wire_outgoing.clone();
432                    let mut ack_outgoing_success = ack_outgoing_success.clone();
433                    let counters = counters.clone();
434                    Either::Right(Either::Right(async move {
435                        let IncomingForwardedPacket {
436                            previous_hop,
437                            next_hop,
438                            data,
439                            ack_key_prev_hop,
440                            ack_challenge,
441                            received_ticket,
442                            ..
443                        } = *fwd_packet;
444                        // Per requirements, this call is not blocking
445                        if let Err(error) = ticket_proc.insert_unacknowledged_ticket(&next_hop, ack_challenge, received_ticket) {
446                            tracing::error!(
447                                previous_hop = previous_hop.to_peerid_str(),
448                                next_hop = next_hop.to_peerid_str(),
449                                %error,
450                                "failed to insert unacknowledged ticket into the ticket processor"
451                            );
452
453                            #[cfg(all(feature = "telemetry", not(test)))]
454                            METRIC_PACKET_REJECTED_COUNT.increment(&["unack_processing_error"]);
455
456                            return None;
457                        }
458
459                        // First, relay the packet to the next hop
460                        tracing::trace!(
461                            previous_hop = previous_hop.to_peerid_str(),
462                            next_hop = next_hop.to_peerid_str(),
463                            "forwarding packet to the next hop"
464                        );
465
466                        match wire_outgoing.send((next_hop.into(), data)).await {
467                            Ok(()) => {
468                                counters.get_or_create(&next_hop).record_message_sent();
469
470                                #[cfg(all(feature = "telemetry", not(test)))]
471                                METRIC_PACKET_COUNT.increment(&["forwarded"]);
472                            }
473                            Err(error) => {
474                                tracing::error!(%error, "failed to forward a packet to the transport layer");
475                                return None;
476                            }
477                        }
478
479                        // Send acknowledgement back
480                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), "acknowledging forwarded packet back");
481                        ack_outgoing_success
482                            .send((previous_hop, Some(ack_key_prev_hop)))
483                            .await
484                            .unwrap_or_else(|error| {
485                                tracing::error!(%error, "failed to send ack to the egress queue");
486                            });
487
488                        None
489                    }))
490                }
491            }
492        })
493        .buffered(concurrency)
494        .filter_map(|maybe_data| futures::future::ready(
495            // Create the ApplicationDataIn data structure for incoming data
496            maybe_data
497                .and_then(|(sender, data, aux_info)| ApplicationData::try_from(data.as_ref())
498                    .inspect_err(|error| tracing::error!(%sender, %error, "failed to decode application data"))
499                    .ok()
500                    .map(|data| (sender, ApplicationDataIn {
501                        data,
502                        packet_info: IncomingPacketInfo {
503                            signals_from_sender: aux_info.packet_signals,
504                            num_saved_surbs: aux_info.num_surbs,
505                        }
506                    })))
507        ))
508        .map(Ok)
509        .forward_to_timeout(app_incoming)
510        .in_current_span()
511        .await;
512
513    if let Err(error) = res {
514        tracing::error!(
515            task = "transport (protocol - msg ingress)",
516            %error,
517            "long-running background task finished with error"
518        );
519    } else {
520        tracing::warn!(
521            task = "transport (protocol - msg ingress)",
522            "long-running background task finished"
523        )
524    }

527async fn start_outgoing_ack_pipeline<AckOut, E, WOut>(
528    ack_outgoing: AckOut,
529    encoder: std::sync::Arc<E>,
530    cfg: AcknowledgementPipelineConfig,
531    packet_key: OffchainKeypair,
532    wire_outgoing: WOut,
533) where
534    AckOut: futures::Stream<Item = (OffchainPublicKey, Option<HalfKey>)> + Send + 'static,
535    E: PacketEncoder + Sync + Send + 'static,
536    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
537    WOut::Error: std::error::Error,
538{
539    ack_outgoing
540        .map(move |(destination, maybe_ack_key)| {
541            let packet_key = packet_key.clone();
542            // Sign acknowledgement with the given half-key or generate a signed random one
543            let ack = maybe_ack_key
544                .map(|ack_key| VerifiedAcknowledgement::new(ack_key, &packet_key))
545                .unwrap_or_else(|| VerifiedAcknowledgement::random(&packet_key));
546            (destination, ack)
547        })
548        .buffer(futures_time::time::Duration::from(cfg.ack_buffer_interval))
549        .filter(|acks| futures::future::ready(!acks.is_empty()))
550        // Group by sender, reusing the same HashMap across buffer cycles so we don't
551        // re-allocate its bucket storage every `cfg.ack_buffer_interval` (default 200ms).
552        //
553        // The halfbrown map uses a Vec backing for a small number of distinct senders
554        // (<32) and transitions to hashbrown otherwise — calling `drain()` keeps the
555        // underlying allocation, leaving us with only the per-group Vec<Ack> to allocate
556        // (which downstream consumes as owned values).
557        .scan(
558            halfbrown::HashMap::<OffchainPublicKey, Vec<VerifiedAcknowledgement>, ahash::RandomState>::with_capacity_and_hasher(
559                cfg.ack_grouping_capacity,
560                ahash::RandomState::default(),
561            ),
562            |groups, buffered_acks| {
563                for (dst, ack) in buffered_acks {
564                    groups
565                        .entry(dst)
566                        .and_modify(|v| v.push(ack))
567                        .or_insert_with(|| vec![ack]);
568                }
569                tracing::trace!(
570                    num_groups = groups.len(),
571                    num_acks = groups.values().map(|v| v.len()).sum::<usize>(),
572                    "acknowledgements grouped"
573                );
574                let drained: Vec<_> = groups.drain().collect();
575                futures::future::ready(Some(futures::stream::iter(drained)))
576            },
577        )
578        .flatten()
579        .for_each_concurrent(
580            cfg.ack_output_concurrency.filter(|&n| n > 0).unwrap_or(DEFAULT_ACK_OUTPUT_CONCURRENCY),
581            move |(destination, acks)| {
582                let encoder = encoder.clone();
583                let mut wire_outgoing = wire_outgoing.clone();
584                async move {
585                    // Make sure that the acknowledgements are sent in batches of at most MAX_ACKNOWLEDGEMENTS_BATCH_SIZE
586                    // TODO: find better strategy to avoid reallocations
587                    let c = acks.chunks(MAX_ACKNOWLEDGEMENTS_BATCH_SIZE).map(|c| c.to_vec()).collect::<Vec<_>>();
588                    for ack_chunk in c {
589                        let encoder = encoder.clone();
590                        #[cfg(all(feature = "telemetry", not(test)))]
591                        let ack_chunk_len = ack_chunk.len() as u64;
592                        match hopr_utils::parallelize::cpu::spawn_fifo_blocking(move || encoder.encode_acknowledgements(&ack_chunk, &destination), "ack_encode").await {
593                            Ok(Ok(ack_packet)) => {
594                                wire_outgoing
595                                    .feed((ack_packet.next_hop.into(), ack_packet.data))
596                                    .await
597                                    .unwrap_or_else(|error| {
598                                        tracing::error!(%error, "failed to forward an acknowledgement to the transport layer");
599                                    });
600
601                                #[cfg(all(feature = "telemetry", not(test)))]
602                                METRIC_SENT_ACKS.increment_by(ack_chunk_len);
603                            }
604                            Ok(Err(error)) => tracing::error!(%error, "failed to encode acknowledgements"),
605                            Err(error) => tracing::error!(%error, "parallel processing of the outgoing acknowledgements failed"),
606                        }
607                    }
608                    if let Err(error) = wire_outgoing.flush().await {
609                        tracing::error!(%error, "failed to flush acknowledgements batch to the transport layer");
610                    }
611                    tracing::trace!("acknowledgements out");
612                }.instrument(tracing::debug_span!("outgoing_ack_batch", peer = destination.to_peerid_str()))
613            }
614        )
615        .in_current_span()
616        .await;
617
618    tracing::warn!(
619        task = "transport (protocol - ack outgoing)",
620        "long-running background task finished"
621    );
622}

624/// Processes incoming acknowledgements using the [`ExitAcknowledgementShareProcessor`], sending
625/// all recovered SSAs to the given `recovered_ssa` sink.
626///
627/// Used by Exit nodes: they do not receive any incoming tickets but still use the incoming acknowledgements
628/// to recover SSAs and forward them to the `recovered_ssa` sink.
629async fn start_exit_incoming_ack_pipeline<S, AckIn, A, SEvt>(
630    ack_incoming: AckIn,
631    exit_proc: A,
632    recovered_ssa: SEvt,
633    concurrency: usize,
634) where
635    S: PixSpec,
636    AckIn: futures::Stream<Item = (OffchainPublicKey, Vec<Acknowledgement>)> + Send + 'static,
637    A: ExitAcknowledgementShareProcessor<S> + Clone + Send + Sync + 'static,
638    SEvt: futures::Sink<ShareResolution<S::Pseudonym, S::AddressPrivateKey>> + Clone + Unpin + Send + 'static,
639    SEvt::Error: std::error::Error,
640{
641    // Matches the concurrency of `start_relay_incoming_ack_pipeline`: each batch triggers up to
642    // `threshold` share verifications (an EC multi-scalar multiplication each), so processing
643    // batches one at a time would serialize all PIX recovery on a dedicated Exit node.
644    ack_incoming
645        .for_each_concurrent(concurrency, move |(peer, acks)| {
646            let exit_proc = exit_proc.clone();
647            let exit_proc_check = exit_proc.clone();
648            let mut recovered_ssa = recovered_ssa.clone();
649            async move {
650                tracing::trace!(%peer, num = acks.len(), "received acknowledgements");
651
652                if !exit_proc_check.has_pending_shares(&peer) {
653                    tracing::trace!(%peer, "no pending pix shares, skipping acknowledgement");
654                    return;
655                }
656
657                match hopr_utils::parallelize::cpu::spawn_fifo_blocking(
658                    move || exit_proc.acknowledge_shares(peer, acks),
659                    "exit_ack_decode",
660                )
661                .await
662                {
663                    Ok(Ok(ssa_priv_keys)) => {
664                        if let Err(error) = recovered_ssa
665                            .send_all(&mut futures::stream::iter(ssa_priv_keys.into_iter().map(Ok)))
666                            .await
667                        {
668                            tracing::error!(%peer, %error, "failed to send pix resolution");
669                        }
670                    }
671                    Ok(Err(ref error)) if exit_proc_check.is_expected_error(error) => {
672                        tracing::debug!(%peer, %error, "expected error while acknowledging pix share")
673                    }
674                    Ok(Err(error)) => {
675                        tracing::error!(%peer, %error, "failed to acknowledge pix share")
676                    }
677                    Err(error) => {
678                        tracing::error!(%peer, %error, "failed to spawn pix share acknowledgement")
679                    }
680                }
681            }
682        })
683        .in_current_span()
684        .await;
685
686    tracing::warn!(
687        task = "transport (protocol - pix share acknowledgement)",
688        "long-running background task finished"
689    );
690}

717async fn start_relay_incoming_ack_pipeline<AckIn, T, TEvt, A, SEvt>(
718    ack_incoming: AckIn,
719    ticket_events: TEvt,
720    ticket_proc: std::sync::Arc<T>,
721    exit_ack_proc: std::sync::Arc<A>,
722    recovered_ssa: SEvt,
723    concurrency: usize,
724) where
725    AckIn: futures::Stream<Item = (OffchainPublicKey, Vec<Acknowledgement>)> + Send + 'static,
726    T: UnacknowledgedTicketProcessor + Sync + Send + 'static,
727    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
728    TEvt::Error: std::error::Error,
729    A: ExitAcknowledgementShareProcessor<HoprPixSpec> + Send + Sync + 'static,
730    SEvt: futures::Sink<HoprShareResolution> + Clone + Unpin + Send + 'static,
731    SEvt::Error: std::error::Error,
732{
733    ack_incoming
734        .for_each_concurrent(concurrency, move |(peer, acks)| {
735            let ticket_proc = ticket_proc.clone();
736            let mut ticket_evt = ticket_events.clone();
737            let exit_proc = exit_ack_proc.clone();
738            let mut ssa_evt = recovered_ssa.clone();
739            async move {
740                tracing::trace!(num = acks.len(), "received acknowledgements");
741                if acks.is_empty() {
742                    return;
743                }
744
745                // Ticket processing always runs — every ack batch on a relay is checked
746                // against pending tickets even when none are expected.
747                // Unlike exit shares below, there is no has_pending_tickets shortcut:
748                // the acknowledge_tickets call itself does the lookup internally, so a
749                // separate check would just duplicate the work.
750
751                // When the peer has pending PIX shares (relay is also an Exit for this peer):
752                // clone acks for the ticket future and move the original into the exit future,
753                // then run both concurrently on the thread pool via join.
754                // When there are no pending shares, avoid the clone and move acks directly into
755                // the ticket future — a single blocking spawn, no concurrent join needed.
756                // Clone before the `if` so it stays alive for `is_expected_error` after the join.
757                let exit_proc_for_spawn = exit_proc.clone();
758                let (ticket_result, exit_result) = if exit_proc.has_pending_shares(&peer) {
759                    let ack_clone = acks.clone();
760                    let ticket_fut = hopr_utils::parallelize::cpu::spawn_fifo_blocking(
761                        move || ticket_proc.acknowledge_tickets(peer, ack_clone),
762                        "ack_decode",
763                    );
764                    let exit_fut = hopr_utils::parallelize::cpu::spawn_fifo_blocking(
765                        move || exit_proc_for_spawn.acknowledge_shares(peer, acks),
766                        "exit_ack_decode",
767                    );
768                    let (t_r, e_r) = futures::future::join(ticket_fut, exit_fut).await;
769                    (t_r, Some(e_r))
770                } else {
771                    let ticket_fut = hopr_utils::parallelize::cpu::spawn_fifo_blocking(
772                        move || ticket_proc.acknowledge_tickets(peer, acks),
773                        "ack_decode",
774                    );
775                    let t_r = ticket_fut.await;
776                    (t_r, None)
777                };
778
779                match ticket_result {
780                    Ok(Ok(resolutions)) if !resolutions.is_empty() => {
781                        let resolutions_iter = resolutions.into_iter().filter_map(|resolution| match resolution {
782                            ResolvedAcknowledgement::RelayingWin(redeemable_ticket) => {
783                                tracing::trace!("received ack for a winning ticket");
784                                #[cfg(all(feature = "telemetry", not(test)))]
785                                {
786                                    METRIC_RECEIVED_ACKS.increment(&["true"]);
787                                    METRIC_TICKETS_COUNT.increment(&["winning"]);
788                                }
789                                Some(Ok(TicketEvent::WinningTicket(redeemable_ticket)))
790                            }
791                            ResolvedAcknowledgement::RelayingLoss(_) => {
792                                // Losing tickets are not getting accounted for anywhere.
793                                tracing::trace!("received ack for a losing ticket");
794                                #[cfg(all(feature = "telemetry", not(test)))]
795                                {
796                                    METRIC_RECEIVED_ACKS.increment(&["true"]);
797                                    METRIC_TICKETS_COUNT.increment(&["losing"]);
798                                }
799                                None
800                            }
801                        });
802
803                        // All acknowledgements that resulted in winning tickets go upstream
804                        if let Err(error) = ticket_evt.send_all(&mut futures::stream::iter(resolutions_iter)).await {
805                            tracing::error!(%error, "failed to notify ticket resolutions");
806                        }
807                    }
808                    Ok(Ok(_)) => {
809                        tracing::debug!("acknowledgement batch could not acknowledge any ticket");
810                    }
811                    Ok(Err(TicketAcknowledgementError::UnexpectedAcknowledgement)) => {
812                        // Unexpected acknowledgements naturally happen
813                        // as acknowledgements of 0-hop packets
814                        tracing::trace!("received unexpected acknowledgement");
815                    }
816                    Ok(Err(error)) => {
817                        tracing::error!(%error, "failed to acknowledge ticket");
818                    }
819                    Err(error) => {
820                        tracing::error!(%error, "parallel processing of the incoming acknowledgements failed")
821                    }
822                }
823
824                // Process PIX share acknowledgements (skipped when no pending shares).
825                if let Some(result) = exit_result {
826                    match result {
827                        Ok(Ok(ssa_priv_keys)) if !ssa_priv_keys.is_empty() => {
828                            if let Err(error) = ssa_evt
829                                .send_all(&mut futures::stream::iter(ssa_priv_keys.into_iter().map(Ok)))
830                                .await
831                            {
832                                tracing::error!(%peer, %error, "failed to send pix resolution");
833                            }
834                        }
835                        Ok(Ok(_)) => {
836                            tracing::trace!(%peer, "empty pix share resolution batch");
837                        }
838                        Ok(Err(ref error)) if exit_proc.is_expected_error(error) => {
839                            tracing::trace!(%peer, %error, "pix share acknowledgement skipped");
840                        }
841                        Ok(Err(error)) => {
842                            tracing::error!(%peer, %error, "pix share acknowledgement failed");
843                        }
844                        Err(error) => {
845                            tracing::error!(%peer, %error, "failed to spawn pix share acknowledgement")
846                        }
847                    }
848                }
849            }
850            .instrument(tracing::debug_span!("incoming_ack_batch", peer = peer.to_peerid_str()))
851        })
852        .in_current_span()
853        .await;
854
855    tracing::warn!(
856        task = "transport (protocol - ticket acknowledgement)",
857        "long-running background task finished"
858    );
859}

966// NOTE: `PixScalar<HoprPixSpec>` normalizes to `HoprPixScalar` (i.e.
967// `<Secp256k1 as CurveArithmetic>::Scalar`). The bound is spelled in its
968// normalized form because rustc fails to equate the deeply nested

971
972/// Shared implementation of the packet pipeline used by [`PacketPipelineBuilder`]'s
973/// terminal `build_for_*` methods.
974#[allow(clippy::too_many_arguments)]
975#[tracing::instrument(skip_all, level = "trace", fields(me = packet_key.public().to_peerid_str()))]
976pub(super) fn run_packet_pipeline_inner<WIn, WOut, A, C, D, T, TEvt, SEvt, AppOut, AppIn>(
977    node_type: NodeType,
978    packet_key: OffchainKeypair,
979    wire_msg: (WOut, WIn),
980    codec: (C, D),
981    ticket_proc: T,
982    exit_ack_proc: A,
983    ticket_events: TEvt,
984    ssa_events: SEvt,
985    cfg: PacketPipelineConfig,
986    api: (AppOut, AppIn),
987    counters: PeerProtocolCounterRegistry,
988) -> AbortableList<PacketPipelineProcesses>
989where
990    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
991    WOut::Error: std::error::Error,
992    WIn: futures::Stream<Item = (PeerId, Bytes)> + Send + 'static,
993    C: PacketEncoder + Sync + Send + 'static,
994    D: PacketDecoder + Sync + Send + 'static,
995    A: ExitAcknowledgementShareProcessor<HoprPixSpec> + Send + Sync + 'static,
996    T: UnacknowledgedTicketProcessor + Sync + Send + 'static,
997    SEvt: futures::Sink<HoprShareResolution> + Clone + Unpin + Send + 'static,
998    SEvt::Error: std::error::Error,
999    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
1000    TEvt::Error: std::error::Error,
1001    AppOut: futures::Sink<(HoprPseudonym, ApplicationDataIn)> + Send + 'static,
1002    AppOut::Error: std::error::Error,
1003    AppIn: futures::Stream<Item = (ResolvedTransportRouting<HoprSurb>, ApplicationDataOut)> + Send + 'static,
1004{
1005    let mut processes = AbortableList::default();
1006
1007    #[cfg(all(feature = "telemetry", not(test)))]
1008    {
1009        // Initialize the lazy statics here
1010        lazy_static::initialize(&METRIC_PACKET_COUNT);
1011        lazy_static::initialize(&METRIC_PACKET_DECODE_TIMEOUTS);
1012        lazy_static::initialize(&METRIC_PACKET_REJECTED_COUNT);
1013        lazy_static::initialize(&METRIC_VALIDATION_ERRORS);
1014    }
1015
1016    let (outgoing_ack_tx, outgoing_ack_rx) = hopr_utils::network_types::crossfire_sink::bounded_sink_channel::<(
1017        OffchainPublicKey,
1018        Option<HalfKey>,
1019    )>(cfg.ack_config.ack_out_buffer_size);
1020
1021    let (incoming_ack_tx, incoming_ack_rx) = hopr_utils::network_types::crossfire_sink::bounded_sink_channel::<(
1022        OffchainPublicKey,
1023        Vec<Acknowledgement>,
1024    )>(cfg.ack_config.ticket_ack_buffer_size);
1025
1026    // Attach timeouts to all Sinks so that the pipelines are not blocked when
1027    // some channel is not being timely processed
1028    let (wire_out, wire_in) = (wire_msg.0.with_timeout(QUEUE_SEND_TIMEOUT), wire_msg.1);
1029    let (app_out, app_in) = (api.0.with_timeout(QUEUE_SEND_TIMEOUT), api.1);
1030    let incoming_ack_tx = incoming_ack_tx.with_timeout(QUEUE_SEND_TIMEOUT);
1031    let outgoing_ack_tx = outgoing_ack_tx.with_timeout(QUEUE_SEND_TIMEOUT);
1032    let ticket_events = ticket_events.with_timeout(QUEUE_SEND_TIMEOUT);
1033    let ssa_events = ssa_events.with_timeout(QUEUE_SEND_TIMEOUT);
1034
1035    let encoder = std::sync::Arc::new(codec.0);
1036    let decoder = std::sync::Arc::new(codec.1);
1037    let ticket_proc = std::sync::Arc::new(ticket_proc);
1038    let exit_ack_proc = std::sync::Arc::new(exit_ack_proc);
1039
1040    // Fallback concurrency when the Rayon pool has not been initialised yet.
1041    // Zero is normalised to 1 to prevent deadlock (0 concurrent tasks = no work done ever).
1042    let avail_concurrency = std::thread::available_parallelism()
1043        .ok()
1044        .map(|n| n.get())
1045        .unwrap_or(1)
1046        .max(1)
1047        * 8;
1048
1049    // Cap `output_concurrency` so the egress encode pipeline does not submit more tasks to Rayon
1050    // than it can drain within `PACKET_ENCODING_TIMEOUT`.  A deep queue (output_concurrency >> pool)
1051    // lets SURB keep-alive bursts queue hundreds of encode tasks, starving data packets that
1052    // arrive later and triggering the 150 ms timeout.
1053    //
1054    // The default cap mirrors the ingress watermark: `pool_threads * INGRESS_POOL_HIGH_WATERMARK_FACTOR`.
1055    // At encode time ≤ 5 ms and 14 threads: max queue wait ≈ (3×14 − 14)/14 × 5 ms = 10 ms, well
1056    // within the 150 ms budget.  Falls back to avail_concurrency when the pool is uninitialised.
1057    let pool_threads = hopr_utils::parallelize::cpu::pool_thread_count();
1058    let default_output_concurrency = if pool_threads > 0 {
1059        pool_threads * INGRESS_POOL_HIGH_WATERMARK_FACTOR
1060    } else {
1061        avail_concurrency
1062    };
1063    let output_concurrency = cfg
1064        .output_concurrency
1065        .filter(|&n| n > 0)
1066        .unwrap_or(default_output_concurrency);
1067
1068    // The ingress decode concurrency is deliberately capped below the Rayon pool size so that
1069    // ENCODE_RESERVED_THREADS are always available for outgoing encode / SURB generation.
1070    // Without this cap the FIFO pool fills up with decode work under heavy download traffic and
1071    // SURB replenishment starves, slowly collapsing the session's download throughput.
1072    // Note: pool_threads is already computed above for output_concurrency.
1073    let default_input_concurrency = if pool_threads > ENCODE_RESERVED_THREADS {
1074        pool_threads - ENCODE_RESERVED_THREADS
1075    } else if pool_threads > 0 {
1076        1 // pool is tiny but initialised — leave at least 1 decode slot
1077    } else {
1078        avail_concurrency // pool not initialised yet; fall back to the old behaviour
1079    };
1080    let input_concurrency = cfg
1081        .input_concurrency
1082        .filter(|&n| n > 0)
1083        .unwrap_or(default_input_concurrency);
1084
1085    // --- Ingress gate (safety-net backpressure) ---
1086    // Gate on DECODE_OUTSTANDING rather than the global outstanding_tasks() so that
1087    // encode work from the outgoing SURB/keep-alive pipeline (which runs concurrently
1088    // in the same process in cluster tests — and in production on the same node) does
1089    // not interfere with the relay's decision to throttle incoming packet decoding.
1090    // Using the decode-specific counter means only sustained decode congestion triggers
1091    // the delay; encode saturation alone does not.
1092    let high_watermark = input_concurrency * INGRESS_POOL_HIGH_WATERMARK_FACTOR;
1093    let wire_in = wire_in.then(move |(peer, data)| async move {
1094        if hopr_utils::parallelize::cpu::decode_outstanding_tasks() > high_watermark {
1095            hopr_utils::runtime::prelude::sleep(INGRESS_THROTTLE_DELAY).await;
1096        }
1097        (peer, data)
1098    });
1099
1100    processes.insert(
1101        PacketPipelineProcesses::MsgOut,
1102        hopr_utils::spawn_as_abortable!(
1103            start_outgoing_packet_pipeline(
1104                app_in,
1105                encoder.clone(),
1106                // Pass exit_ack_proc to any node that has an outgoing packet pipeline
1107                // (Relay, Exit, or Relay+Exit). Entry nodes never have a processor.
1108                // The Option inside already guards absent processors (e.g. pure Relay
1109                // without PIX uses a no-op), so this is safe to pass unconditionally.
1110                (node_type != NodeType::Entry).then(|| exit_ack_proc.clone()),
1111                wire_out.clone(),
1112                counters.clone(),
1113                output_concurrency
1114            )
1115            .in_current_span()
1116        ),
1117    );
1118
1119    processes.insert(
1120        PacketPipelineProcesses::MsgIn,
1121        hopr_utils::spawn_as_abortable!(
1122            start_incoming_packet_pipeline(
1123                (wire_out.clone(), wire_in),
1124                decoder,
1125                ticket_proc.clone(),
1126                ticket_events.clone(),
1127                (outgoing_ack_tx, incoming_ack_tx),
1128                app_out,
1129                counters.clone(),
1130                input_concurrency,
1131            )
1132            .in_current_span()
1133        ),
1134    );
1135
1136    processes.insert(
1137        PacketPipelineProcesses::AckOut,
1138        hopr_utils::spawn_as_abortable!(
1139            start_outgoing_ack_pipeline(outgoing_ack_rx, encoder, cfg.ack_config, packet_key.clone(), wire_out,)
1140                .in_current_span()
1141        ),
1142    );
1143
1144    let ack_input_concurrency = cfg
1145        .ack_config
1146        .ack_input_concurrency
1147        .filter(|&n| n > 0)
1148        .unwrap_or(DEFAULT_ACK_INPUT_CONCURRENCY);
1149
1150    match node_type {
1151        NodeType::Relay => {
1152            processes.insert(
1153                PacketPipelineProcesses::AckIn,
1154                hopr_utils::spawn_as_abortable!(
1155                    start_relay_incoming_ack_pipeline(
1156                        incoming_ack_rx,
1157                        ticket_events,
1158                        ticket_proc,
1159                        exit_ack_proc.clone(),
1160                        ssa_events.clone(),
1161                        ack_input_concurrency,
1162                    )
1163                    .in_current_span()
1164                ),
1165            );
1166        }
1167        NodeType::Exit => {
1168            // Exit nodes still run the incoming acknowledgement pipeline (for future PIX use),
1169            // but only drain the stream — incoming acknowledgements are NOT forwarded to the
1170            // UnacknowledgedTicketProcessor because Exit nodes do not process tickets.
1171            let _ = (ticket_events, ticket_proc);
1172            processes.insert(
1173                PacketPipelineProcesses::AckIn,
1174                hopr_utils::spawn_as_abortable!(
1175                    start_exit_incoming_ack_pipeline(
1176                        incoming_ack_rx,
1177                        exit_ack_proc,
1178                        ssa_events,
1179                        ack_input_concurrency,
1180                    )
1181                    .in_current_span()
1182                ),
1183            );
1184        }
1185        NodeType::Entry => {
1186            // Entry nodes do not process tickets, but they DO receive ticket acknowledgements
1187            // (they pay relays for forwarding). The queue must be actively drained so the
1188            // inbound dispatcher can keep sending without hitting `SendError(disconnected)`.
1189            let _ = (ticket_events, ticket_proc, ack_input_concurrency);
1190            processes.insert(
1191                PacketPipelineProcesses::AckIn,
1192                hopr_utils::spawn_as_abortable!(start_drain_incoming_ack_pipeline(incoming_ack_rx).in_current_span()),
1193            );
1194        }
1195    }
1196
1197    processes
1198}

1723    /// Contract test pinning the ordering semantics the decode stages rely on.
1724    ///
1725    /// The incoming/outgoing packet decode stages use `map(..).buffered(concurrency)`
1726    /// (`FuturesOrdered`) rather than `then_concurrent(..)` (`FuturesUnordered`) precisely so decoded
1727    /// packets are emitted in arrival order even when per-packet decode latencies differ. Out-of-order
1728    /// emission was the root cause of session reassembler/sequencer frame discards under burst.
1729    ///
1730    /// NOTE: this reconstructs the `map(..).buffered(..).filter_map(ready)` shape locally rather than
1731    /// driving `start_incoming_packet_pipeline` (which would require mocking `PacketDecoder`, the
1732    /// ticket processor and five sinks plus constructing valid `IncomingFinalPacket` crypto types). It
1733    /// therefore documents and locks the ordering *contract* of the combinator the stage is built from,
1734    /// not the production wiring itself — a change from `buffered` to `then_concurrent` in the pipeline
1735    /// is caught end-to-end by the session loopback e2e test, not here. Latency is *inverted* w.r.t.
1736    /// position (item 0 slowest, last fastest) so completion order is the reverse of arrival order;
1737    /// `buffered` must still yield arrival order, whereas `FuturesUnordered` would yield completion
1738    /// order and fail this assertion.
1739    #[tokio::test]
1740    async fn buffered_decode_preserves_arrival_order_under_inverted_latency() {
1741        use std::time::Duration;
1742
1743        const N: usize = 16;
1744        let concurrency = N; // all decode futures in flight simultaneously
1745
1746        let output: Vec<usize> = futures::stream::iter(0..N)
1747            .map(|i| async move {
1748                // Invert latency: earlier items complete later than later items.
1749                tokio::time::sleep(Duration::from_millis(((N - i) * 4) as u64)).await;
1750                // `None` models an undecodable packet dropped by the `filter_map` below.
1751                if i == 3 { None } else { Some(i) }
1752            })
1753            .buffered(concurrency)
1754            .filter_map(futures::future::ready)
1755            .collect()
1756            .await;
1757
1758        let expected: Vec<usize> = (0..N).filter(|&i| i != 3).collect();
1759        assert_eq!(
1760            output, expected,
1761            "buffered(concurrency) must emit decoded packets in arrival order regardless of decode latency; got \
1762             {output:?}. A regression to then_concurrent()/FuturesUnordered would emit completion order and fail here."
1763        );
1764    }
