27/// Default concurrency for the incoming acknowledgement processing pipeline when not overridden
28/// via [`AcknowledgementPipelineConfig::ack_input_concurrency`].
29const DEFAULT_ACK_INPUT_CONCURRENCY: usize = 10;
30/// Default concurrency for the outgoing acknowledgement processing pipeline when not overridden
31/// via [`AcknowledgementPipelineConfig::ack_output_concurrency`].
32const DEFAULT_ACK_OUTPUT_CONCURRENCY: usize = 10;
33const QUEUE_SEND_TIMEOUT: std::time::Duration = std::time::Duration::from_millis(50);
34// Set well above the worst-case rayon queue wait so packets are never silently
35// dropped due to CPU contention.  The formula (Sphinx ~21 ms/packet, up to 64
36// concurrent tasks, 8-core rayon pool) gives a worst-case queue wait of
37// 56 × 21 ms / 8 ≈ 147 ms — uncomfortably close to the old 150 ms ceiling.
38// Under any real CPU load (container overhead, background node traffic) encoding
39// regularly exceeded 150 ms, causing FuturesOrdered to emit None and silently
40// drop the packet, which then caused the session sequencer on the far side to
41// discard the next in-order frame after frame_timeout.
42//
43// 5 s gives headroom for sustained saturation while still providing a safety
44// net against truly hung encoding futures.  Normal encoding (~21 ms) is
45// unaffected; only pathological cases hit the ceiling.
46const PACKET_DECODING_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
47const PACKET_ENCODING_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
48
49/// Number of Rayon threads kept permanently free for outgoing packet encode (SURB generation).
50/// The ingress decode concurrency default is `pool_thread_count - ENCODE_RESERVED_THREADS` so
51/// that heavy download traffic cannot starve the upload/SURB replenishment path.
52const ENCODE_RESERVED_THREADS: usize = 2;
53
54/// Artificial per-packet delay injected into `wire_in` when the Rayon pool is detected as
55/// congested. 20 ms keeps the decode queue from growing while still allowing acks and keep-alive
56/// SURB packets to drain through at a reasonable pace.
57const INGRESS_THROTTLE_DELAY: std::time::Duration = std::time::Duration::from_millis(20);
58
59/// Pool outstanding-task watermark factor: the gate trips when
60/// `outstanding_tasks > pool_thread_count * INGRESS_POOL_HIGH_WATERMARK_FACTOR`.
61/// A factor of 3 gives one full pool worth of headroom above the cap.
62const INGRESS_POOL_HIGH_WATERMARK_FACTOR: usize = 3;

119/// Performs encoding of outgoing Application protocol packets into HOPR protocol outgoing packets.
120async fn start_outgoing_packet_pipeline<AppOut, E, WOut, WOutErr>(
121    app_outgoing: AppOut,
122    encoder: std::sync::Arc<E>,
123    wire_outgoing: WOut,
124    counters: super::counters::PeerProtocolCounterRegistry,
125    concurrency: usize,
126) where
127    AppOut: futures::Stream<Item = (ResolvedTransportRouting<HoprSurb>, ApplicationDataOut)> + Send + 'static,
128    E: PacketEncoder + Send + Sync + 'static,
129    WOut: futures::Sink<(PeerId, Bytes), Error = SinkTimeoutError<WOutErr>> + Clone + Unpin + Send + 'static,
130    WOutErr: std::error::Error,
131{
132    let res = app_outgoing
133        // Use `map().buffered()` (FuturesOrdered) instead of `then_concurrent()` (FuturesUnordered)
134        // so that encoded packets are emitted in submission order, preserving session frame sequence.
135        // Out-of-order sends to QUIC were causing frame reassembler discards on the receiver side.
136        .map(|(routing, data)| {
137            let encoder = encoder.clone();
138            let counters = counters.clone();
139            async move {
140                hopr_transport_session::counters::ENCODE_STAGE_ENTRIES
141                    .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
142                match hopr_utils::parallelize::cpu::spawn_encode_blocking(
143                    move || {
144                        encoder.encode_packet(
145                            data.data.to_bytes(),
146                            routing,
147                            data.packet_info
148                                .map(|data| data.signals_to_destination)
149                                .unwrap_or_default(),
150                        )
151                    },
152                    "packet_encode",
153                )
154                .timeout(futures_time::time::Duration::from(PACKET_ENCODING_TIMEOUT))
155                .await
156                {
157                    Ok(Ok(Ok(packet))) => {
158                        #[cfg(all(feature = "telemetry", not(test)))]
159                        METRIC_PACKET_COUNT.increment(&["sent"]);
160
161                        counters.get_or_create(&packet.next_hop).record_message_sent();
162                        tracing::trace!(peer = packet.next_hop.to_peerid_str(), "protocol message out");
163                        Some((packet.next_hop.into(), packet.data))
164                    }
165                    Ok(Ok(Err(error))) => {
166                        tracing::error!(%error, "outgoing packet could not be encoded");
167                        None
168                    }
169                    Ok(Err(error)) => {
170                        tracing::error!(%error, "parallel processing of the outgoing packet failed");
171                        None
172                    }
173                    Err(error) => {
174                        tracing::error!(%error, "timeout while processing the outgoing packet");
175                        hopr_utils::parallelize::cpu::ENCODE_TIMEOUT_DROPS
176                            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
177                        None
178                    }
179                }
180            }
181        })
182        .buffered(concurrency)
183        .filter_map(futures::future::ready)
184        .map(Ok)
185        .forward_to_timeout(wire_outgoing)
186        .in_current_span()
187        .await;
188
189    if let Err(error) = res {
190        tracing::error!(
191            task = "transport (protocol - msg egress)",
192            %error,
193            "long-running background task finished with error"
194        );
195    } else {
196        tracing::warn!(
197            task = "transport (protocol - msg egress)",
198            "long-running background task finished"
199        )
200    }
201}

203/// Performs HOPR protocol decoding of incoming packets into Application protocol packets.
204///
205/// `wire_incoming` --> `decoder` --> `ack_outgoing` (final + forwarded)
206///                             | --> `wire_outgoing` (forwarded)
207///                             | --> `ack_incoming` (forwarded)
208///                             | --> `app_incoming` (final)
209#[allow(clippy::too_many_arguments)]
210async fn start_incoming_packet_pipeline<WIn, WOut, D, T, TEvt, AckIn, AckOut, AppIn, AppInErr>(
211    (wire_outgoing, wire_incoming): (WOut, WIn),
212    decoder: std::sync::Arc<D>,
213    ticket_proc: std::sync::Arc<T>,
214    ticket_events: TEvt,
215    (ack_outgoing, ack_incoming): (AckOut, AckIn),
216    app_incoming: AppIn,
217    counters: super::counters::PeerProtocolCounterRegistry,
218    concurrency: usize,
219) where
220    WIn: futures::Stream<Item = (PeerId, Bytes)> + Send + 'static,
221    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
222    WOut::Error: std::error::Error,
223    D: PacketDecoder + Sync + Send + 'static,
224    T: UnacknowledgedTicketProcessor + Send + 'static,
225    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
226    TEvt::Error: std::error::Error,
227    AckIn: futures::Sink<(OffchainPublicKey, Vec<Acknowledgement>)> + Send + Unpin + Clone + 'static,
228    AckIn::Error: std::error::Error,
229    AckOut: futures::Sink<(OffchainPublicKey, Option<HalfKey>)> + Send + Unpin + Clone + 'static,
230    AckOut::Error: std::error::Error,
231    AppIn: futures::Sink<(HoprPseudonym, ApplicationDataIn), Error = SinkTimeoutError<AppInErr>> + Send + 'static,
232    AppInErr: std::error::Error,
233{
234    let ack_outgoing_success = ack_outgoing.clone();
235    let ack_outgoing_failure = ack_outgoing;
236    let ticket_proc_success = ticket_proc;
237
238    let res = wire_incoming
239        // Use `map().buffered()` (FuturesOrdered) instead of `then_concurrent()` (FuturesUnordered)
240        // so that decoded packets are emitted in the same order they arrive from the transport.
241        // Concurrent decode on rayon is preserved; only the output ordering guarantee changes.
242        // Out-of-order delivery was the root cause of session reassembler/sequencer discards.
243        .map(move |(peer, data)| {
244            let decoder = decoder.clone();
245            let mut ack_outgoing_failure = ack_outgoing_failure.clone();
246            let mut ticket_events_reject = ticket_events.clone();
247
248            tracing::trace!(%peer, "protocol message in");
249
250            async move {
251                match hopr_utils::parallelize::cpu::spawn_decode_blocking(move || decoder.decode(peer, data), "packet_decode")
252                    .timeout(futures_time::time::Duration::from(PACKET_DECODING_TIMEOUT))
253                    .await {
254                    Ok(Ok(Ok(packet))) => {
255                        tracing::trace!(%peer, ?packet, "successfully decoded incoming packet");
256                        Some(packet)
257                    },
258                    Ok(Ok(Err(IncomingPacketError::Undecodable(error)))) => {
259                        // Do not send an ack back if the packet could not be decoded at all
260                        //
261                        // Potentially adversarial behavior
262                        tracing::trace!(%peer, %error, "not sending ack back on undecodable packet - possible adversarial behavior");
263
264                        #[cfg(all(feature = "telemetry", not(test)))]
265                        METRIC_PACKET_REJECTED_COUNT.increment(&["undecodable"]);
266
267                        None
268                    },
269                    Ok(Ok(Err(IncomingPacketError::ProcessingError(sender, error)))) => {
270                        tracing::error!(%peer, %error, "failed to process the decoded packet");
271                        // On this failure, we send back a random acknowledgement
272                        ack_outgoing_failure
273                            .send((*sender, None))
274                            .await
275                            .unwrap_or_else(|error| {
276                                tracing::error!(%error, "failed to send ack to the egress queue");
277                            });
278
279                        #[cfg(all(feature = "telemetry", not(test)))]
280                        METRIC_PACKET_REJECTED_COUNT.increment(&["processing_error"]);
281
282                        None
283                    },
284                    Ok(Ok(Err(IncomingPacketError::InvalidTicket(sender, error)))) => {
285                        tracing::error!(%peer, %error, "failed to validate ticket on the received packet");
286                        if let Err(error) = ticket_events_reject
287                            .send(TicketEvent::RejectedTicket(error.ticket, error.issuer))
288                            .await {
289                            tracing::error!(%error, "failed to notify invalid ticket rejection");
290                        }
291                        // On this failure, we send back a random acknowledgement
292                        ack_outgoing_failure
293                            .send((*sender, None))
294                            .await
295                            .unwrap_or_else(|error| {
296                                tracing::error!(%error, "failed to send ack to the egress queue");
297                            });
298
299                        #[cfg(all(feature = "telemetry", not(test)))]
300                        {
301                            METRIC_VALIDATION_ERRORS.increment(&[error.kind.as_ref()]);
302                            METRIC_PACKET_REJECTED_COUNT.increment(&["invalid_ticket"]);
303                            METRIC_TICKETS_COUNT.increment(&["rejected"]);
304                        }
305
306                        None
307                    }
308                    Ok(Err(error)) => {
309                        tracing::error!(%error, "parallel processing of the incoming packet failed");
310                        None
311                    },
312                    Err(_) => {
313                        // If we cannot decode the packet within the time limit, just drop it
314                        tracing::error!(
315                            %peer,
316                            timeout_ms = PACKET_DECODING_TIMEOUT.as_millis() as u64,
317                            "dropped incoming packet: decode timeout - check the 'hopr_rayon_queue_wait_seconds' metric for pool saturation"
318                        );
319                        hopr_utils::parallelize::cpu::DECODE_TIMEOUT_DROPS
320                            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
321                        #[cfg(all(feature = "telemetry", not(test)))]
322                        {
323                            METRIC_PACKET_DECODE_TIMEOUTS.increment();
324                            METRIC_PACKET_REJECTED_COUNT.increment(&["timeout"]);
325                        }
326
327                        None
328                    }
329                }
330            }.instrument(tracing::debug_span!("incoming_packet_decode", %peer))
331        })
332        .buffered(concurrency)
333        .filter_map(futures::future::ready)
334        // Branch on the packet type BEFORE building the async future so each arm only clones
335        // the handles it actually needs. `futures::future::Either` lets us return three
336        // distinct async blocks from one closure without boxing.
337        // FuturesOrdered: preserve decode-stage order so session segments reach the socket in sequence.
338        .map(move |packet| {
339            match packet {
340                IncomingPacket::Acknowledgement(ack) => {
341                    let mut ack_incoming = ack_incoming.clone();
342                    let counters = counters.clone();
343                    Either::Left(async move {
344                        let IncomingAcknowledgementPacket { previous_hop, received_acks, .. } = *ack;
345                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), num_acks = received_acks.len(), "incoming acknowledgements");
346                        counters.get_or_create(&previous_hop).record_acks_received(received_acks.len() as u64);
347
348                        ack_incoming
349                            .send((previous_hop, received_acks))
350                            .await
351                            .unwrap_or_else(|error| {
352                                tracing::error!(%error, "failed dispatching received acknowledgement to the ticket ack queue");
353                            });
354
355                        // We do not acknowledge back acknowledgements.
356                        None
357                    })
358                }
359                IncomingPacket::Final(final_packet) => {
360                    let mut ack_outgoing_success = ack_outgoing_success.clone();
361                    Either::Right(Either::Left(async move {
362                        let IncomingFinalPacket {
363                            previous_hop,
364                            sender,
365                            plain_text,
366                            ack_key,
367                            info,
368                            ..
369                        } = *final_packet;
370                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), "incoming final packet");
371
372                        // Send acknowledgement back
373                        ack_outgoing_success
374                            .send((previous_hop, Some(ack_key)))
375                            .await
376                            .unwrap_or_else(|error| {
377                                tracing::error!(%error, "failed to send ack to the egress queue");
378                            });
379
380                        #[cfg(all(feature = "telemetry", not(test)))]
381                        METRIC_PACKET_COUNT.increment(&["received"]);
382
383                        Some((sender, plain_text, info))
384                    }))
385                }
386                IncomingPacket::Forwarded(fwd_packet) => {
387                    let ticket_proc = ticket_proc_success.clone();
388                    let mut wire_outgoing = wire_outgoing.clone();
389                    let mut ack_outgoing_success = ack_outgoing_success.clone();
390                    let counters = counters.clone();
391                    Either::Right(Either::Right(async move {
392                        let IncomingForwardedPacket {
393                            previous_hop,
394                            next_hop,
395                            data,
396                            ack_key_prev_hop,
397                            ack_challenge,
398                            received_ticket,
399                            ..
400                        } = *fwd_packet;
401                        // Per requirements, this call is not blocking
402                        if let Err(error) = ticket_proc.insert_unacknowledged_ticket(&next_hop, ack_challenge, received_ticket) {
403                            tracing::error!(
404                                previous_hop = previous_hop.to_peerid_str(),
405                                next_hop = next_hop.to_peerid_str(),
406                                %error,
407                                "failed to insert unacknowledged ticket into the ticket processor"
408                            );
409
410                            #[cfg(all(feature = "telemetry", not(test)))]
411                            METRIC_PACKET_REJECTED_COUNT.increment(&["unack_processing_error"]);
412
413                            return None;
414                        }
415
416                        // First, relay the packet to the next hop
417                        tracing::trace!(
418                            previous_hop = previous_hop.to_peerid_str(),
419                            next_hop = next_hop.to_peerid_str(),
420                            "forwarding packet to the next hop"
421                        );
422
423                        match wire_outgoing.send((next_hop.into(), data)).await {
424                            Ok(()) => {
425                                counters.get_or_create(&next_hop).record_message_sent();
426
427                                #[cfg(all(feature = "telemetry", not(test)))]
428                                METRIC_PACKET_COUNT.increment(&["forwarded"]);
429                            }
430                            Err(error) => {
431                                tracing::error!(%error, "failed to forward a packet to the transport layer");
432                                return None;
433                            }
434                        }
435
436                        // Send acknowledgement back
437                        tracing::trace!(previous_hop = previous_hop.to_peerid_str(), "acknowledging forwarded packet back");
438                        ack_outgoing_success
439                            .send((previous_hop, Some(ack_key_prev_hop)))
440                            .await
441                            .unwrap_or_else(|error| {
442                                tracing::error!(%error, "failed to send ack to the egress queue");
443                            });
444
445                        None
446                    }))
447                }
448            }
449        })
450        .buffered(concurrency)
451        .filter_map(|maybe_data| futures::future::ready(
452            // Create the ApplicationDataIn data structure for incoming data
453            maybe_data
454                .and_then(|(sender, data, aux_info)| ApplicationData::try_from(data.as_ref())
455                    .inspect_err(|error| tracing::error!(%sender, %error, "failed to decode application data"))
456                    .ok()
457                    .map(|data| (sender, ApplicationDataIn {
458                        data,
459                        packet_info: IncomingPacketInfo {
460                            signals_from_sender: aux_info.packet_signals,
461                            num_saved_surbs: aux_info.num_surbs,
462                        }
463                    })))
464        ))
465        .map(Ok)
466        .forward_to_timeout(app_incoming)
467        .in_current_span()
468        .await;
469
470    if let Err(error) = res {
471        tracing::error!(
472            task = "transport (protocol - msg ingress)",
473            %error,
474            "long-running background task finished with error"
475        );
476    } else {
477        tracing::warn!(
478            task = "transport (protocol - msg ingress)",
479            "long-running background task finished"
480        )
481    }
482}

484async fn start_outgoing_ack_pipeline<AckOut, E, WOut>(
485    ack_outgoing: AckOut,
486    encoder: std::sync::Arc<E>,
487    cfg: AcknowledgementPipelineConfig,
488    packet_key: OffchainKeypair,
489    wire_outgoing: WOut,
490) where
491    AckOut: futures::Stream<Item = (OffchainPublicKey, Option<HalfKey>)> + Send + 'static,
492    E: PacketEncoder + Sync + Send + 'static,
493    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
494    WOut::Error: std::error::Error,
495{
496    ack_outgoing
497        .map(move |(destination, maybe_ack_key)| {
498            let packet_key = packet_key.clone();
499            // Sign acknowledgement with the given half-key or generate a signed random one
500            let ack = maybe_ack_key
501                .map(|ack_key| VerifiedAcknowledgement::new(ack_key, &packet_key))
502                .unwrap_or_else(|| VerifiedAcknowledgement::random(&packet_key));
503            (destination, ack)
504        })
505        .buffer(futures_time::time::Duration::from(cfg.ack_buffer_interval))
506        .filter(|acks| futures::future::ready(!acks.is_empty()))
507        // Group by sender, reusing the same HashMap across buffer cycles so we don't
508        // re-allocate its bucket storage every `cfg.ack_buffer_interval` (default 200ms).
509        //
510        // The halfbrown map uses a Vec backing for a small number of distinct senders
511        // (<32) and transitions to hashbrown otherwise — calling `drain()` keeps the
512        // underlying allocation, leaving us with only the per-group Vec<Ack> to allocate
513        // (which downstream consumes as owned values).
514        .scan(
515            halfbrown::HashMap::<OffchainPublicKey, Vec<VerifiedAcknowledgement>, ahash::RandomState>::with_capacity_and_hasher(
516                cfg.ack_grouping_capacity,
517                ahash::RandomState::default(),
518            ),
519            |groups, buffered_acks| {
520                for (dst, ack) in buffered_acks {
521                    groups
522                        .entry(dst)
523                        .and_modify(|v| v.push(ack))
524                        .or_insert_with(|| vec![ack]);
525                }
526                tracing::trace!(
527                    num_groups = groups.len(),
528                    num_acks = groups.values().map(|v| v.len()).sum::<usize>(),
529                    "acknowledgements grouped"
530                );
531                let drained: Vec<_> = groups.drain().collect();
532                futures::future::ready(Some(futures::stream::iter(drained)))
533            },
534        )
535        .flatten()
536        .for_each_concurrent(
537            cfg.ack_output_concurrency.filter(|&n| n > 0).unwrap_or(DEFAULT_ACK_OUTPUT_CONCURRENCY),
538            move |(destination, acks)| {
539                let encoder = encoder.clone();
540                let mut wire_outgoing = wire_outgoing.clone();
541                async move {
542                    // Make sure that the acknowledgements are sent in batches of at most MAX_ACKNOWLEDGEMENTS_BATCH_SIZE
543                    // TODO: find better strategy to avoid reallocations
544                    let c = acks.chunks(MAX_ACKNOWLEDGEMENTS_BATCH_SIZE).map(|c| c.to_vec()).collect::<Vec<_>>();
545                    for ack_chunk in c {
546                        let encoder = encoder.clone();
547                        #[cfg(all(feature = "telemetry", not(test)))]
548                        let ack_chunk_len = ack_chunk.len() as u64;
549                        match hopr_utils::parallelize::cpu::spawn_fifo_blocking(move || encoder.encode_acknowledgements(&ack_chunk, &destination), "ack_encode").await {
550                            Ok(Ok(ack_packet)) => {
551                                wire_outgoing
552                                    .feed((ack_packet.next_hop.into(), ack_packet.data))
553                                    .await
554                                    .unwrap_or_else(|error| {
555                                        tracing::error!(%error, "failed to forward an acknowledgement to the transport layer");
556                                    });
557
558                                #[cfg(all(feature = "telemetry", not(test)))]
559                                METRIC_SENT_ACKS.increment_by(ack_chunk_len);
560                            }
561                            Ok(Err(error)) => tracing::error!(%error, "failed to encode acknowledgements"),
562                            Err(error) => tracing::error!(%error, "parallel processing of the outgoing acknowledgements failed"),
563                        }
564                    }
565                    if let Err(error) = wire_outgoing.flush().await {
566                        tracing::error!(%error, "failed to flush acknowledgements batch to the transport layer");
567                    }
568                    tracing::trace!("acknowledgements out");
569                }.instrument(tracing::debug_span!("outgoing_ack_batch", peer = destination.to_peerid_str()))
570            }
571        )
572        .in_current_span()
573        .await;
574
575    tracing::warn!(
576        task = "transport (protocol - ack outgoing)",
577        "long-running background task finished"
578    );
579}
580
581/// Drains incoming acknowledgements without forwarding them to an [`UnacknowledgedTicketProcessor`].
582///
583/// Used by Entry and Exit nodes — neither processes incoming ticket acknowledgements.
584/// Entry nodes receive acks from relays (they pay for forwarding), Exit nodes keep
585/// the pipeline alive for future PIX use. In both cases the queue must be actively
586/// drained; dropping the receiver causes every inbound ack dispatch to fail with
587/// `SendError(disconnected)`.
588async fn start_drain_incoming_ack_pipeline<AckIn>(ack_incoming: AckIn)
589where
590    AckIn: futures::Stream<Item = (OffchainPublicKey, Vec<Acknowledgement>)> + Send + 'static,
591{
592    ack_incoming
593        .for_each(move |(peer, acks)| {
594            tracing::trace!(%peer, num = acks.len(), "received acknowledgements (drained, not processed)");
595            futures::future::ready(())
596        })
597        .in_current_span()
598        .await;
599
600    tracing::warn!(
601        task = "transport (protocol - ticket acknowledgement drain)",
602        "long-running background task finished"
603    );
604}
605
606async fn start_relay_incoming_ack_pipeline<AckIn, T, TEvt>(
607    ack_incoming: AckIn,
608    ticket_events: TEvt,
609    ticket_proc: std::sync::Arc<T>,
610    concurrency: usize,
611) where
612    AckIn: futures::Stream<Item = (OffchainPublicKey, Vec<Acknowledgement>)> + Send + 'static,
613    T: UnacknowledgedTicketProcessor + Sync + Send + 'static,
614    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
615    TEvt::Error: std::error::Error,
616{
617    ack_incoming
618        .for_each_concurrent(concurrency, move |(peer, acks)| {
619            let ticket_proc = ticket_proc.clone();
620            let mut ticket_evt = ticket_events.clone();
621            async move {
622                tracing::trace!(num = acks.len(), "received acknowledgements");
623                match hopr_utils::parallelize::cpu::spawn_fifo_blocking(
624                    move || ticket_proc.acknowledge_tickets(peer, acks),
625                    "ack_decode",
626                )
627                .await
628                {
629                    Ok(Ok(resolutions)) if !resolutions.is_empty() => {
630                        let resolutions_iter = resolutions.into_iter().filter_map(|resolution| match resolution {
631                            ResolvedAcknowledgement::RelayingWin(redeemable_ticket) => {
632                                tracing::trace!("received ack for a winning ticket");
633                                #[cfg(all(feature = "telemetry", not(test)))]
634                                {
635                                    METRIC_RECEIVED_ACKS.increment(&["true"]);
636                                    METRIC_TICKETS_COUNT.increment(&["winning"]);
637                                }
638                                Some(Ok(TicketEvent::WinningTicket(redeemable_ticket)))
639                            }
640                            ResolvedAcknowledgement::RelayingLoss(_) => {
641                                // Losing tickets are not getting accounted for anywhere.
642                                tracing::trace!("received ack for a losing ticket");
643                                #[cfg(all(feature = "telemetry", not(test)))]
644                                {
645                                    METRIC_RECEIVED_ACKS.increment(&["true"]);
646                                    METRIC_TICKETS_COUNT.increment(&["losing"]);
647                                }
648                                None
649                            }
650                        });
651
652                        // All acknowledgements that resulted in winning tickets go upstream
653                        if let Err(error) = ticket_evt.send_all(&mut futures::stream::iter(resolutions_iter)).await {
654                            tracing::error!(%error, "failed to notify ticket resolutions");
655                        }
656                    }
657                    Ok(Ok(_)) => {
658                        tracing::debug!("acknowledgement batch could not acknowledge any ticket");
659                    }
660                    Ok(Err(TicketAcknowledgementError::UnexpectedAcknowledgement)) => {
661                        // Unexpected acknowledgements naturally happen
662                        // as acknowledgements of 0-hop packets
663                        tracing::trace!("received unexpected acknowledgement");
664                    }
665                    Ok(Err(error)) => {
666                        tracing::error!(%error, "failed to acknowledge ticket");
667                    }
668                    Err(error) => {
669                        tracing::error!(%error, "parallel processing of the incoming acknowledgements failed")
670                    }
671                }
672            }
673            .instrument(tracing::debug_span!("incoming_ack_batch", peer = peer.to_peerid_str()))
674        })
675        .in_current_span()
676        .await;
677
678    tracing::warn!(
679        task = "transport (protocol - ticket acknowledgement)",
680        "long-running background task finished"
681    );
682}
683/// Node type for which the packet processing pipeline is being constructed.
684///
685/// The three HOPR node types differ in how they treat tickets and incoming acknowledgements:
686/// * [`Relay`](NodeType::Relay) — full pipeline, processes tickets and incoming acknowledgements.
687/// * [`Entry`](NodeType::Entry) — does not process tickets and does not even start the incoming acknowledgement
688///   pipeline.
689/// * [`Exit`](NodeType::Exit) — does not process tickets, but still runs the incoming acknowledgement pipeline (which
690///   only drains the stream) for future use.
691#[derive(Debug, Copy, Clone, PartialEq, Eq, Hash)]
692pub enum NodeType {
693    Relay,
694    Entry,
695    Exit,
696}

698/// No-op [`UnacknowledgedTicketProcessor`] used by node types that do not process tickets
699/// (Entry and Exit). All methods are unreachable because the inner pipeline never invokes
700/// them on those node types (Entry skips the ack pipeline entirely, Exit uses the drain
701/// variant, and the forwarded packet branch never fires on a terminal/source node).
702#[derive(Debug, Default, Copy, Clone)]
703#[doc(hidden)]
704pub struct NoopTicketProcessor;

706impl UnacknowledgedTicketProcessor for NoopTicketProcessor {

711        &self,
712        _: &OffchainPublicKey,
713        _: HalfKeyChallenge,
714        _: UnacknowledgedTicket,

716        Ok(())
717    }
718
719    #[inline]
720    fn acknowledge_tickets(
721        &self,
722        _: OffchainPublicKey,
723        _: Vec<Acknowledgement>,
724    ) -> Result<Vec<ResolvedAcknowledgement>, TicketAcknowledgementError<Self::Error>> {
725        Ok(Vec::with_capacity(0))
726    }

728/// Shared implementation of the packet pipeline used by [`PacketPipelineBuilder`]'s
729/// terminal `build_for_*` methods.
730#[allow(clippy::too_many_arguments)]
731#[tracing::instrument(skip_all, level = "trace", fields(me = packet_key.public().to_peerid_str()))]
732pub(super) fn run_packet_pipeline_inner<WIn, WOut, C, D, T, TEvt, AppOut, AppIn>(
733    node_type: NodeType,
734    packet_key: OffchainKeypair,
735    wire_msg: (WOut, WIn),
736    codec: (C, D),
737    ticket_proc: T,
738    ticket_events: TEvt,
739    cfg: PacketPipelineConfig,
740    api: (AppOut, AppIn),
741    counters: PeerProtocolCounterRegistry,
742) -> AbortableList<PacketPipelineProcesses>
743where
744    WOut: futures::Sink<(PeerId, Bytes)> + Clone + Unpin + Send + 'static,
745    WOut::Error: std::error::Error,
746    WIn: futures::Stream<Item = (PeerId, Bytes)> + Send + 'static,
747    C: PacketEncoder + Sync + Send + 'static,
748    D: PacketDecoder + Sync + Send + 'static,
749    T: UnacknowledgedTicketProcessor + Sync + Send + 'static,
750    TEvt: futures::Sink<TicketEvent> + Clone + Unpin + Send + 'static,
751    TEvt::Error: std::error::Error,
752    AppOut: futures::Sink<(HoprPseudonym, ApplicationDataIn)> + Send + 'static,
753    AppOut::Error: std::error::Error,
754    AppIn: futures::Stream<Item = (ResolvedTransportRouting<HoprSurb>, ApplicationDataOut)> + Send + 'static,
755{
756    let mut processes = AbortableList::default();
757
758    #[cfg(all(feature = "telemetry", not(test)))]
759    {
760        // Initialize the lazy statics here
761        lazy_static::initialize(&METRIC_PACKET_COUNT);
762        lazy_static::initialize(&METRIC_PACKET_DECODE_TIMEOUTS);
763        lazy_static::initialize(&METRIC_PACKET_REJECTED_COUNT);
764        lazy_static::initialize(&METRIC_VALIDATION_ERRORS);
765    }
766
767    let (outgoing_ack_tx, outgoing_ack_rx) = hopr_utils::network_types::crossfire_sink::bounded_sink_channel::<(
768        OffchainPublicKey,
769        Option<HalfKey>,
770    )>(cfg.ack_config.ack_out_buffer_size);
771
772    let (incoming_ack_tx, incoming_ack_rx) = hopr_utils::network_types::crossfire_sink::bounded_sink_channel::<(
773        OffchainPublicKey,
774        Vec<Acknowledgement>,
775    )>(cfg.ack_config.ticket_ack_buffer_size);
776
777    // Attach timeouts to all Sinks so that the pipelines are not blocked when
778    // some channel is not being timely processed
779    let (wire_out, wire_in) = (wire_msg.0.with_timeout(QUEUE_SEND_TIMEOUT), wire_msg.1);
780    let (app_out, app_in) = (api.0.with_timeout(QUEUE_SEND_TIMEOUT), api.1);
781    let incoming_ack_tx = incoming_ack_tx.with_timeout(QUEUE_SEND_TIMEOUT);
782    let outgoing_ack_tx = outgoing_ack_tx.with_timeout(QUEUE_SEND_TIMEOUT);
783    let ticket_events = ticket_events.with_timeout(QUEUE_SEND_TIMEOUT);
784
785    let encoder = std::sync::Arc::new(codec.0);
786    let decoder = std::sync::Arc::new(codec.1);
787    let ticket_proc = std::sync::Arc::new(ticket_proc);
788
789    // Fallback concurrency when the Rayon pool has not been initialised yet.
790    // Zero is normalised to 1 to prevent deadlock (0 concurrent tasks = no work done ever).
791    let avail_concurrency = std::thread::available_parallelism()
792        .ok()
793        .map(|n| n.get())
794        .unwrap_or(1)
795        .max(1)
796        * 8;
797
798    // Cap `output_concurrency` so the egress encode pipeline does not submit more tasks to Rayon
799    // than it can drain within `PACKET_ENCODING_TIMEOUT`.  A deep queue (output_concurrency >> pool)
800    // lets SURB keep-alive bursts queue hundreds of encode tasks, starving data packets that
801    // arrive later and triggering the 150 ms timeout.
802    //
803    // The default cap mirrors the ingress watermark: `pool_threads * INGRESS_POOL_HIGH_WATERMARK_FACTOR`.
804    // At encode time ≤ 5 ms and 14 threads: max queue wait ≈ (3×14 − 14)/14 × 5 ms = 10 ms, well
805    // within the 150 ms budget.  Falls back to avail_concurrency when the pool is uninitialised.
806    let pool_threads = hopr_utils::parallelize::cpu::pool_thread_count();
807    let default_output_concurrency = if pool_threads > 0 {
808        pool_threads * INGRESS_POOL_HIGH_WATERMARK_FACTOR
809    } else {
810        avail_concurrency
811    };
812    let output_concurrency = cfg
813        .output_concurrency
814        .filter(|&n| n > 0)
815        .unwrap_or(default_output_concurrency);
816
817    // The ingress decode concurrency is deliberately capped below the Rayon pool size so that
818    // ENCODE_RESERVED_THREADS are always available for outgoing encode / SURB generation.
819    // Without this cap the FIFO pool fills up with decode work under heavy download traffic and
820    // SURB replenishment starves, slowly collapsing the session's download throughput.
821    // Note: pool_threads is already computed above for output_concurrency.
822    let default_input_concurrency = if pool_threads > ENCODE_RESERVED_THREADS {
823        pool_threads - ENCODE_RESERVED_THREADS
824    } else if pool_threads > 0 {
825        1 // pool is tiny but initialised — leave at least 1 decode slot
826    } else {
827        avail_concurrency // pool not initialised yet; fall back to the old behaviour
828    };
829    let input_concurrency = cfg
830        .input_concurrency
831        .filter(|&n| n > 0)
832        .unwrap_or(default_input_concurrency);
833
834    // --- Ingress gate (safety-net backpressure) ---
835    // Gate on DECODE_OUTSTANDING rather than the global outstanding_tasks() so that
836    // encode work from the outgoing SURB/keep-alive pipeline (which runs concurrently
837    // in the same process in cluster tests — and in production on the same node) does
838    // not interfere with the relay's decision to throttle incoming packet decoding.
839    // Using the decode-specific counter means only sustained decode congestion triggers
840    // the delay; encode saturation alone does not.
841    let high_watermark = input_concurrency * INGRESS_POOL_HIGH_WATERMARK_FACTOR;
842    let wire_in = wire_in.then(move |(peer, data)| async move {
843        if hopr_utils::parallelize::cpu::decode_outstanding_tasks() > high_watermark {
844            hopr_utils::runtime::prelude::sleep(INGRESS_THROTTLE_DELAY).await;
845        }
846        (peer, data)
847    });
848
849    processes.insert(
850        PacketPipelineProcesses::MsgOut,
851        hopr_utils::spawn_as_abortable!(
852            start_outgoing_packet_pipeline(
853                app_in,
854                encoder.clone(),
855                wire_out.clone(),
856                counters.clone(),
857                output_concurrency
858            )
859            .in_current_span()
860        ),
861    );
862
863    processes.insert(
864        PacketPipelineProcesses::MsgIn,
865        hopr_utils::spawn_as_abortable!(
866            start_incoming_packet_pipeline(
867                (wire_out.clone(), wire_in),
868                decoder,
869                ticket_proc.clone(),
870                ticket_events.clone(),
871                (outgoing_ack_tx, incoming_ack_tx),
872                app_out,
873                counters.clone(),
874                input_concurrency,
875            )
876            .in_current_span()
877        ),
878    );
879
880    processes.insert(
881        PacketPipelineProcesses::AckOut,
882        hopr_utils::spawn_as_abortable!(
883            start_outgoing_ack_pipeline(outgoing_ack_rx, encoder, cfg.ack_config, packet_key.clone(), wire_out,)
884                .in_current_span()
885        ),
886    );
887
888    let ack_input_concurrency = cfg
889        .ack_config
890        .ack_input_concurrency
891        .filter(|&n| n > 0)
892        .unwrap_or(DEFAULT_ACK_INPUT_CONCURRENCY);
893
894    match node_type {
895        NodeType::Relay => {
896            processes.insert(
897                PacketPipelineProcesses::AckIn,
898                hopr_utils::spawn_as_abortable!(
899                    start_relay_incoming_ack_pipeline(
900                        incoming_ack_rx,
901                        ticket_events,
902                        ticket_proc,
903                        ack_input_concurrency
904                    )
905                    .in_current_span()
906                ),
907            );
908        }
909        NodeType::Exit => {
910            // Exit nodes still run the incoming acknowledgement pipeline (for future PIX use),
911            // but only drain the stream — incoming acknowledgements are NOT forwarded to the
912            // UnacknowledgedTicketProcessor because Exit nodes do not process tickets.
913            let _ = (ticket_events, ticket_proc, ack_input_concurrency);
914            processes.insert(
915                PacketPipelineProcesses::AckIn,
916                hopr_utils::spawn_as_abortable!(start_drain_incoming_ack_pipeline(incoming_ack_rx).in_current_span()),
917            );
918        }
919        NodeType::Entry => {
920            // Entry nodes do not process tickets, but they DO receive ticket acknowledgements
921            // (they pay relays for forwarding). The queue must be actively drained so the
922            // inbound dispatcher can keep sending without hitting `SendError(disconnected)`.
923            let _ = (ticket_events, ticket_proc, ack_input_concurrency);
924            processes.insert(
925                PacketPipelineProcesses::AckIn,
926                hopr_utils::spawn_as_abortable!(start_drain_incoming_ack_pipeline(incoming_ack_rx).in_current_span()),
927            );
928        }
929    }
930
931    processes
932}

1061    ///
1062    /// The incoming/outgoing packet decode stages use `map(..).buffered(concurrency)`
1063    /// (`FuturesOrdered`) rather than `then_concurrent(..)` (`FuturesUnordered`) precisely so decoded

1065    /// emission was the root cause of session reassembler/sequencer frame discards under burst.

1068    /// driving `start_incoming_packet_pipeline` (which would require mocking `PacketDecoder`, the
1069    /// ticket processor and five sinks plus constructing valid `IncomingFinalPacket` crypto types). It
1070    /// therefore documents and locks the ordering *contract* of the combinator the stage is built from,
1071    /// not the production wiring itself — a change from `buffered` to `then_concurrent` in the pipeline
1072    /// is caught end-to-end by the session loopback e2e test, not here. Latency is *inverted* w.r.t.
1073    /// position (item 0 slowest, last fastest) so completion order is the reverse of arrival order;
