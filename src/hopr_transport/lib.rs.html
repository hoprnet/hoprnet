141/// Fraction of the SURB ring buffer a Session's balancer target may occupy.
142///
143/// The remainder is headroom for balancer overshoot. The target used to be the ring buffer capacity
144/// itself, which left none: a Session whose buffer sat at target — the normal state when Entry → Exit
145/// traffic far exceeds the reverse, since the Exit then drains almost nothing — turned every
146/// overshot SURB into an immediate eviction of the oldest entry. Under PIX that is a lost SSA share,
147/// not just a lost SURB, because a share reaches the reconstructor only when its SURB is used.
148const SURB_BUFFER_TARGET_NUMERATOR: usize = 2;
149const SURB_BUFFER_TARGET_DENOMINATOR: usize = 3;
150
151/// Largest SURB buffer target a Session may request, given the ring buffer capacity backing it.
152///
153/// Kept strictly below `rb_capacity` so the balancer has somewhere to overshoot into before the
154/// ring buffer starts overwriting. See [`SURB_BUFFER_TARGET_NUMERATOR`].
155const fn surb_buffer_target_ceiling(rb_capacity: usize) -> usize {
156    // Saturating rather than wrapping: an operator-supplied capacity is only range-validated at the
157    // low end, and the multiplication would otherwise overflow on an absurd value.
158    rb_capacity.saturating_mul(SURB_BUFFER_TARGET_NUMERATOR) / SURB_BUFFER_TARGET_DENOMINATOR
159}

161/// PeerId -> OffchainPublicKey is a CPU-intensive blocking operation.
162///
163/// This helper uses a cached static object to speed up the lookup and avoid blocking the async
164/// runtime on repeated conversions for the same [`PeerId`]s.
165pub fn peer_id_to_public_key(peer_id: &PeerId) -> crate::errors::Result<OffchainPublicKey> {
166    PEER_ID_CACHE
167        .try_get_with_by_ref(peer_id, move || {
168            OffchainPublicKey::from_peerid(peer_id).map_err(|e| e.into())
169        })
170        .map_err(|e: Arc<HoprTransportError>| {
171            crate::errors::HoprTransportError::Other(anyhow::anyhow!(
172                "failed to convert peer_id ({:?}) to an offchain public key: {e}",
173                peer_id
174            ))
175        })
176}

209/// Allows configuration of one specific [`HoprSession`].
210///
211/// The configurator does not prevent the Session from being closed
212/// or the Session manager from being dropped.
213#[derive(Debug, Clone)]
214pub struct HoprSessionConfigurator {
215    id: SessionId,
216    // Makes sure configurator does not extend lifetime of the SessionManager.
217    smgr: std::sync::Weak<HoprSessionManager>,
218}
219
220impl HoprSessionConfigurator {
221    /// [`SessionId`] of the session this object can configure.
222    pub fn id(&self) -> &SessionId {
223        &self.id
224    }
225
226    /// Sends a Session Keep-Alive packet over the Session.
227    ///
228    /// NOTE: This usually carries at least 2 SURBs on the HOPR protocol level and can be
229    /// used for manual SURB balancing.
230    ///
231    /// NOTE: This operation only sends the Session Keep-Alive packet and **DOES NOT** guarantee the other side
232    /// has received it.
233    pub async fn ping(&self) -> errors::Result<()> {
234        Ok(self
235            .smgr
236            .upgrade()
237            .ok_or(HoprTransportError::Other(anyhow::anyhow!("session manager is dropped")))?
238            .ping_session(&self.id)
239            .await?)
240    }
241
242    /// Gets the configuration of the SURB balancer.
243    ///
244    /// Returns an error if the Session is closed, the Session manager is gone.
245    ///
246    /// Returns `Ok(None)` if the Session has been created without a SURB balancer.
247    pub fn get_surb_balancer_config(&self) -> errors::Result<Option<SurbBalancerConfig>> {
248        Ok(self
249            .smgr
250            .upgrade()
251            .ok_or(HoprTransportError::Other(anyhow::anyhow!("session manager is dropped")))?
252            .get_surb_balancer_config(&self.id)?)
253    }
254
255    /// Updates the configuration of the SURB balancer.
256    ///
257    /// Returns an error if the Session is closed, the Session manager is gone, or the
258    /// Session has been created without a SURB balancer.
259    pub fn update_surb_balancer_config(&self, config: SurbBalancerConfig) -> errors::Result<()> {
260        Ok(self
261            .smgr
262            .upgrade()
263            .ok_or(HoprTransportError::Other(anyhow::anyhow!("session manager is dropped")))?
264            .update_surb_balancer_config(&self.id, config)?)
265    }
266
267    /// Explicitly closes the underlying Session in the [`SessionManager`].
268    ///
269    /// Returns `true` if the session was found and closed, `false` if it was
270    /// already gone (or the manager is dropped). Frees the per-session state
271    /// (frame reassembly buffers, control channels, …) immediately rather than
272    /// waiting for the manager's idle-timeout eviction.
273    pub fn close(&self) -> bool {
274        match self.smgr.upgrade() {
275            Some(smgr) => smgr.close_session(&self.id),
276            None => false,
277        }
278    }

325    pub fn new(
326        identity: (&ChainKeypair, &OffchainKeypair),
327        resolver: Chain,
328        graph: Graph,
329        my_multiaddresses: Vec<Multiaddr>,
330        cfg: HoprProtocolConfig,
331    ) -> errors::Result<Self> {
332        // Validated at the constructor boundary, because this is the boundary. The shared node
333        // builder does validate before all four of its build methods reach here, but this function is
334        // `pub`, and a direct caller then skips every relational check on the way in — including
335        // `validate_pix_supervision_pairing`, which is a schema-level validator on
336        // `HoprProtocolConfig` and so is not reached by validating any single field.
337        //
338        // `run_inner` validates `self.cfg.pix` alone, which is the narrower of the two: it cannot see
339        // the supervisor-against-reconstructor pairing, and it runs at Session establishment rather
340        // than at construction — long after a caller could have been told.
341        validator::Validate::validate(&cfg)
342            .map_err(|error| HoprTransportError::Api(format!("invalid protocol configuration: {error}")))?;
343
344        let me_offchain = *identity.1.public();
345        let planner_config = cfg.path_planner;
346        let selector = HoprGraphPathSelector::new(
347            me_offchain,
348            graph.clone(),
349            planner_config.max_cached_paths,
350            planner_config.edge_penalty,
351            planner_config.min_ack_rate,
352            planner_config.min_paths_anonymity_floor,
353        );
354
355        let tag_allocators = hopr_transport_tag_allocator::create_allocators_from_config(&cfg.session.tag_allocator)?;
356
357        let mut session_telemetry_tag_allocator = None;
358        let mut probing_tag_allocator = None;
359        for (usage, alloc) in tag_allocators {
360            match usage {
361                // TODO: cleanup of Session tag allocators needed * (#8199)
362                hopr_transport_tag_allocator::Usage::Session => {}
363                hopr_transport_tag_allocator::Usage::SessionTerminalTelemetry => {
364                    session_telemetry_tag_allocator = Some(alloc)
365                }
366                hopr_transport_tag_allocator::Usage::ProvingTelemetry => probing_tag_allocator = Some(alloc),
367            }
368        }
369        let session_telemetry_tag_allocator = session_telemetry_tag_allocator
370            .ok_or_else(|| HoprTransportError::Api("session telemetry tag allocator missing".into()))?;
371        let probing_tag_allocator =
372            probing_tag_allocator.ok_or_else(|| HoprTransportError::Api("probing tag allocator missing".into()))?;
373
374        // A pseudonym's SURB ring buffer is dropped once no SURBs have arrived for
375        // `pseudonyms_lifetime` (600 s by default), which is the point at which that pseudonym's
376        // return path becomes permanently unresolvable. Nothing else in the node can be made to
377        // reach that state on a test timescale — the Session slot is evicted for idleness long
378        // before it — so without a way to shorten this timer, the behaviour past it is not
379        // observable end-to-end at all. Floored at the same `MINIMUM_SURB_LIFETIME` the config
380        // validator enforces, so this cannot reach a value the config file could not also express.
381        let surb_store_cfg = hopr_protocol_hopr::SurbStoreConfig {
382            pseudonyms_lifetime: std::env::var("HOPR_INTERNAL_SURB_PSEUDONYM_LIFETIME_MS")
383                .ok()
384                .and_then(|s| s.trim().parse::<u64>().ok())
385                .map(Duration::from_millis)
386                .map(|d| d.max(hopr_protocol_hopr::MINIMUM_SURB_LIFETIME))
387                .unwrap_or(cfg.packet.surb_store.pseudonyms_lifetime),
388            ..cfg.packet.surb_store
389        };
390
391        // Built before the session manager so the latter can be handed the seam that lets a
392        // session re-plan its return path on sustained loss.
393        let path_planner = PathPlanner::new(
394            me_offchain,
395            MemorySurbStore::new(surb_store_cfg),
396            resolver.clone(),
397            selector,
398            planner_config,
399        );
400
401        Ok(Self {
402            packet_key: identity.1.clone(),
403            chain_key: identity.0.clone(),
404            ping: Arc::new(OnceLock::new()),
405            network: Arc::new(OnceLock::new()),
406            graph,
407            path_planner,
408            my_multiaddresses,
409            smgr: Arc::new(SessionManager::new(SessionManagerConfig {
410                frame_mtu: std::env::var("HOPR_SESSION_FRAME_SIZE")
411                    .ok()
412                    .and_then(|s| s.parse::<usize>().ok())
413                    .unwrap_or_else(|| SessionManagerConfig::default().frame_mtu)
414                    .max(SESSION_MTU),
415                max_frame_timeout: std::env::var("HOPR_SESSION_FRAME_TIMEOUT_MS")
416                    .ok()
417                    .and_then(|s| s.parse::<u64>().ok().map(Duration::from_millis))
418                    .unwrap_or_else(|| SessionManagerConfig::default().max_frame_timeout)
419                    .max(Duration::from_millis(100)),
420                max_frames_behind_gap: cfg.session.max_frames_behind_gap,
421                max_buffered_segments: std::env::var("HOPR_SESSION_MAX_BUFFERED_SEGMENTS")
422                    .ok()
423                    .and_then(|s| s.parse::<usize>().ok())
424                    .unwrap_or_else(|| SessionManagerConfig::default().max_buffered_segments),
425                initiation_timeout_base: SESSION_INITIATION_TIMEOUT_BASE,
426                idle_timeout: cfg.session.idle_timeout,
427                balancer_sampling_interval: cfg.session.balancer_sampling_interval,
428                initial_return_session_egress_rate: 10,
429                minimum_surb_buffer_duration: cfg.session.balancer_minimum_surb_buffer_duration,
430                maximum_surb_buffer_size: surb_buffer_target_ceiling(cfg.packet.surb_store.rb_capacity),
431                // The lower bound is enforced once in `SessionManager::new` via
432                // `MIN_SURB_BUFFER_NOTIFICATION_PERIOD`; don't duplicate that floor as a literal here.
433                surb_balance_notify_period: std::env::var("HOPR_SESSION_SURB_BALANCE_NOTIFY_PERIOD_MS")
434                    .ok()
435                    .and_then(|s| s.parse::<u64>().ok())
436                    .map(|ms| Some(Duration::from_millis(ms)))
437                    .unwrap_or(cfg.session.surb_balance_notify_period),
438                surb_target_notify: true,
439                maximum_sessions: cfg.session.maximum_managed_sessions,
440                pix_config: cfg.incoming_session_pix_config.clone(),
441                max_ssas_per_ssa_request: cfg.pix.max_ssas_per_request,
442                ..Default::default()
443            })),
444            chain_api: resolver,
445            session_telemetry_tag_allocator,
446            probing_tag_allocator,
447            counters: PeerProtocolCounterRegistry::default(),

553        &self,
554        cover_traffic: Ct,
555        network: Net,
556        network_process: BoxedProcessFn,
557        ticket_factory: TFact,
558        pix_events: Option<PixEvt>,
559    ) -> errors::Result<(
560        HoprSocket<
561            futures::stream::BoxStream<'static, ApplicationDataIn>,
562            CrossfireSink<(DestinationRouting, ApplicationDataOut)>,
563        >,
564        AbortableList<HoprTransportProcess>,
565    )>
566    where
567        Ct: ProbingTrafficGeneration + CoverTrafficGeneration + Send + Sync + 'static,
568        TFact: TicketFactory + Clone + Send + Sync + 'static,
569        PixEvt: futures::Sink<PixEvent> + Clone + Unpin + Send + 'static,
570        PixEvt::Error: std::error::Error + Clone + Sync + Send + 'static,
571    {
572        self.run_inner(
573            protocol::NodeType::Entry,
574            cover_traffic,
575            network,
576            network_process,
577            futures::sink::drain(),
578            ticket_factory,
579            pix_events,
580            None,
581            None,
582        )
583        .await
584    }
585
586    /// Internal worker driving all node-type variants of `HoprTransport::run_*`.
587    ///
588    /// Branches on `role`:
589    /// - [`protocol::NodeType::Relay`]: full packet pipeline + SessionManager.
590    /// - [`protocol::NodeType::Exit`]: ack-drain pipeline + incoming Sessions.
591    /// - [`protocol::NodeType::Entry`]: no ack pipeline, no incoming Sessions.
592    #[allow(clippy::too_many_arguments)]
593    async fn run_inner<T, TFact, Ct, PixEvt>(
594        &self,
595        role: protocol::NodeType,
596        cover_traffic: Ct,
597        network: Net,
598        network_process: BoxedProcessFn,
599        ticket_events: T,
600        ticket_factory: TFact,
601        exit_ack_share: Option<PixEvt>,
602        on_incoming_session: Option<Sender<IncomingSession>>,
603        on_session_admission: Option<SessionAdmissionSink>,
604    ) -> errors::Result<(
605        HoprSocket<
606            futures::stream::BoxStream<'static, ApplicationDataIn>,
607            CrossfireSink<(DestinationRouting, ApplicationDataOut)>,
608        >,
609        AbortableList<HoprTransportProcess>,
610    )>
611    where
612        T: futures::Sink<hopr_api::node::TicketEvent> + Clone + Send + Unpin + 'static,
613        T::Error: std::error::Error + Clone + Send,
614        Ct: ProbingTrafficGeneration + CoverTrafficGeneration + Send + Sync + 'static,
615        TFact: TicketFactory + Clone + Send + Sync + 'static,
616        PixEvt: futures::Sink<PixEvent> + Clone + Unpin + Send + 'static,
617        PixEvt::Error: std::error::Error + Clone + Sync + Send + 'static,
618    {
619        let mut processes = AbortableList::<HoprTransportProcess>::default();
620
621        let (unresolved_routing_msg_tx, unresolved_routing_msg_rx) =
622            bounded_sink_channel::<(DestinationRouting, ApplicationDataOut)>(MAXIMUM_MSG_OUTGOING_BUFFER_SIZE);
623
624        // -- transport medium
625
626        let transport_network = network;
627        let transport_layer_process = network_process;
628
629        let msg_codec = crate::protocol::HoprBinaryCodec {};
630        let (wire_msg_tx, wire_msg_rx) =
631            protocol::stream::process_stream_protocol(msg_codec, transport_network.clone(), self.cfg.stream).await?;
632
633        // Shared mixing channel: all per-destination clones of `mixing_channel_tx` push into one
634        // heap, so cross-destination packets are mixed together rather than each destination
635        // getting its own independent delay queue. The single forwarder task owns the receiver
636        // (and therefore the heap timer) — no per-clone waker coordination is needed.
637        let mut mixer_cfg = self.cfg.mixer;
638        mixer_cfg.metric_delay_window = u64::try_from(5 * mixer_cfg.delay_range.as_millis())
639            .unwrap_or(u64::MAX)
640            .max(1);
641        let (mixing_channel_tx, mix_rx) = hopr_transport_mixer::channel(mixer_cfg);
642        let transit_latency_cfg = self.cfg.transit_latency;
643        processes.insert(
644            HoprTransportProcess::MixerForwarder,
645            hopr_utils::spawn_as_abortable!(async move {
646                let mut mix_rx = mix_rx;
647                let mut wire_sink = wire_msg_tx;
648
649                if let Some(lat) = transit_latency_cfg {
650                    // Concurrent transit-latency fan-out: spawn one short-lived task per
651                    // packet so each packet ages through its own ~mean delay independently.
652                    // A burst of N packets at mean=50 ms takes ~50 ms total, not N×50 ms.
653                    //
654                    // Without a tokio runtime the latency is silently skipped (pass-through).
655                    #[cfg(feature = "runtime-tokio")]
656                    {
657                        let (tx, rx) = futures::channel::mpsc::unbounded();
658                        let fan_out = async move {
659                            while let Some(item) = futures::StreamExt::next(&mut mix_rx).await {
660                                let mean_us = lat.mean.as_micros() as f64;
661                                let std_us = lat.std_dev.as_micros() as f64;
662                                let delay_us = if std_us > 0.0 {
663                                    use rand_distr::{Distribution, Normal};
664                                    Normal::new(mean_us, std_us)
665                                        .expect("transit latency Normal params are valid")
666                                        .sample(&mut rand::rng())
667                                        .max(0.0_f64)
668                                } else {
669                                    mean_us.max(0.0)
670                                };
671                                let delay = Duration::from_micros(delay_us as u64);
672                                let item_tx = tx.clone();
673                                hopr_utils::runtime::prelude::spawn(async move {
674                                    if !delay.is_zero() {
675                                        futures_timer::Delay::new(delay).await;
676                                    }
677                                    let _ = item_tx.unbounded_send(item);
678                                });
679                            }
680                            // `tx` drops here; the channel closes once all per-packet tasks send
681                        };
682                        let fan_in = async move {
683                            let mut rx = rx;
684                            while let Some(item) = futures::StreamExt::next(&mut rx).await {
685                                if wire_sink.send(item).await.is_err() {
686                                    tracing::error!(
687                                        task = %HoprTransportProcess::MixerForwarder,
688                                        "wire sink dropped — discarding transit-delayed packet"
689                                    );
690                                    break;
691                                }
692                            }
693                        };
694                        futures::join!(fan_out, fan_in);
695                    }
696                    #[cfg(not(feature = "runtime-tokio"))]
697                    {
698                        let _ = lat;
699                        while let Some(item) = futures::StreamExt::next(&mut mix_rx).await {
700                            if wire_sink.send(item).await.is_err() {
701                                tracing::error!(
702                                    task = %HoprTransportProcess::MixerForwarder,
703                                    "wire sink dropped — discarding mixed packet"
704                                );
705                            }
706                        }
707                    }
708                } else {
709                    while let Some(item) = futures::StreamExt::next(&mut mix_rx).await {
710                        if wire_sink.send(item).await.is_err() {
711                            tracing::error!(
712                                task = %HoprTransportProcess::MixerForwarder,
713                                "wire sink dropped — discarding mixed packet"
714                            );
715                        }
716                    }
717                }
718
719                tracing::warn!(
720                    task = %HoprTransportProcess::MixerForwarder,
721                    "long-running background task finished"
722                );
723            }),
724        );
725
726        // -- path cache background refresh (only when tokio runtime is available)
727        #[cfg(feature = "runtime-tokio")]
728        processes.insert(
729            HoprTransportProcess::PathRefresh,
730            hopr_utils::spawn_as_abortable!(self.path_planner.run_background_refresh()),
731        );
732
733        processes.insert(
734            HoprTransportProcess::Medium,
735            hopr_utils::spawn_as_abortable!(transport_layer_process().inspect(|_| tracing::warn!(
736                task = %HoprTransportProcess::Medium,
737                "long-running background task finished"
738            ))),
739        );
740
741        let msg_protocol_bidirectional_channel_capacity =
742            std::env::var("HOPR_INTERNAL_PROTOCOL_BIDIRECTIONAL_CHANNEL_CAPACITY")
743                .ok()
744                .and_then(|s| s.trim().parse::<usize>().ok())
745                .filter(|&c| c > 0)
746                .unwrap_or(16_384);
747
748        debug!(
749            capacity = msg_protocol_bidirectional_channel_capacity,
750            "creating protocol bidirectional channel"
751        );
752        let (tx_from_protocol, rx_from_protocol) =
753            bounded_sink_channel::<(HoprPseudonym, ApplicationDataIn)>(msg_protocol_bidirectional_channel_capacity);
754
755        // === START === cover traffic control
756        // Allocate a cover traffic tag from the session telemetry partition to avoid
757        // collisions with session and probing tags.
758        let cover_traffic_allocated_tag = self
759            .session_telemetry_tag_allocator
760            .allocate()
761            .ok_or_else(|| HoprTransportError::Api("failed to allocate cover traffic tag".into()))?;
762        let cover_traffic_tag: Tag = cover_traffic_allocated_tag.value().into();
763
764        // filter out the known cover traffic not to lose processing time with it
765        // The allocated tag is moved into the closure to keep it alive for the transport lifetime.
766        let rx_from_protocol = rx_from_protocol.filter_map(move |(pseudonym, data)| {
767            let _keep_alive = &cover_traffic_allocated_tag;
768            async move { (data.data.application_tag != cover_traffic_tag).then_some((pseudonym, data)) }
769        });
770
771        // prepare a cover traffic stream
772        let cover_traffic_stream = CoverTrafficGeneration::build(&cover_traffic).filter_map(move |routing| {
773            let start =
774                hopr_api::types::crypto_random::random_integer(0, Some((RANDOM_DATA.len() - 100) as u64)) as usize;
775            let data = &RANDOM_DATA[start..start + 100];
776
777            futures::future::ready(if let Ok(data) = ApplicationData::new(cover_traffic_tag, data) {
778                Some((routing, ApplicationDataOut::with_no_packet_info(data)))
779            } else {
780                tracing::error!("failed to construct cover traffic packet");
781                None
782            })
783        });
784
785        // merge cover traffic with other outgoing data
786        let merged_unresolved_output_data =
787            select_with_strategy(unresolved_routing_msg_rx, cover_traffic_stream, |_: &mut ()| {
788                futures::stream::PollNext::Left
789            });
790
791        // === END === cover traffic control
792
793        // We have to resolve DestinationRouting -> ResolvedTransportRouting before
794        // sending the external packets to the transport pipeline. Concurrency matches
795        // the encoder stage (output_concurrency) to avoid head-of-line blocking on
796        // cache-miss path lookups.
797        let path_planner = self.path_planner.clone();
798        let distress_threshold = self.cfg.packet.surb_store.distress_threshold;
799        let routing_concurrency = {
800            let avail = std::thread::available_parallelism()
801                .ok()
802                .map(|n| n.get())
803                .unwrap_or(1)
804                .max(1)
805                * 8;
806            self.cfg
807                .packet
808                .pipeline
809                .output_concurrency
810                .filter(|&n| n > 0)
811                .unwrap_or(avail)
812        };
813        let all_resolved_external_msg_rx = crate::path::resolve::resolve_routing_stage(
814            merged_unresolved_output_data,
815            move |size_hint, max_surbs, unresolved| {
816                let path_planner = path_planner.clone();
817                async move { path_planner.resolve_routing(size_hint, max_surbs, unresolved).await }
818            },
819            distress_threshold,
820            routing_concurrency,
821            crate::path::resolve::surb_resolution_wait(self.cfg.packet.pipeline.surb_resolution_wait),
822        );
823
824        let channels_dst = self
825            .chain_api
826            .domain_separators()
827            .await
828            .map_err(HoprTransportError::chain)?
829            .channel;
830
831        // The SSA generator is dimensioned from the global PIX config (not per
832        // session) because `handle_ssa_request` (SessionManager) validates that the
833        // Exit's negotiated quota matches the session's `pix_ssa_quota` before any
834        // client commitments are generated, and the Exit's `new_exit_commitment`
835        // bounds-checks polys_per_ssa and shares_per_poly.  The session quota is
836        // a subset of what the global generator covers, so one generator suffices.
837        //
838        // Validated here rather than left to the constructor: `PixGlobalConfig` carries more than
839        // the three fields `SsaGeneratorConfig` covers, and this used to be a SAFETY comment
840        // asserting that it had already been validated "before this code runs" via
841        // `#[validate(nested)]` — but nothing in this crate calls `validate()`, so the guarantee
842        // rested entirely on every caller remembering to.
843        validator::Validate::validate(&self.cfg.pix)
844            .map_err(|error| HoprTransportError::Api(format!("invalid PIX configuration: {error}")))?;
845        // Checked rather than `as`-cast: the validation above already bounds all three, but a
846        // truncating cast would turn a future widening of any of those ranges into a silently
847        // wrong-but-valid-looking dimension rather than a startup error.
848        fn narrow<T: TryFrom<usize>>(value: usize, field: &str) -> errors::Result<T> {
849            T::try_from(value).map_err(|_| HoprTransportError::Api(format!("PIX {field} out of range: {value}")))
850        }
851        // `try_new`, not `new`: the dimensions come from operator configuration, so a range
852        // violation is a startup error to report rather than a panic to take down the node.
853        let ssa_generator = Arc::new(
854            hopr_protocol_pix::SsaShareGenerator::<HoprPixSpec>::try_new(hopr_protocol_pix::SsaGeneratorConfig {
855                polynomials_per_ssa: narrow(self.cfg.pix.num_ssa_parts, "num_ssa_parts")?,
856                threshold: narrow(self.cfg.pix.ssa_part_size, "ssa_part_size")?,
857                surplus_shares: narrow(self.cfg.pix.surplus_shares(), "additional_shares")?,
858            })
859            .map_err(|error| HoprTransportError::Api(format!("invalid SSA generator configuration: {error}")))?,
860        );
861
862        let surb_round_trips = protocol::surb_telemetry::SurbRoundTripRegistry::default();
863
864        let pipeline_builder = HoprPacketPipelineBuilder::new()
865            .identity((&self.chain_key, &self.packet_key))
866            .transport((mixing_channel_tx, wire_msg_rx))
867            .api((tx_from_protocol, all_resolved_external_msg_rx))
868            .surb_store(self.path_planner.surb_store.clone())
869            .chain_api(self.chain_api.clone())
870            .ticket_factory(ticket_factory)
871            .ssa_generator(ssa_generator.clone())
872            .channels_dst(channels_dst)
873            .with_counters(self.counters.clone())
874            .with_surb_telemetry(
875                surb_round_trips.clone(),
876                protocol::surb_telemetry::path_slots_of(self.graph.clone()),
877            )
878            .with_config(self.cfg.packet);
879
880        // ── PixToolbox for the SessionManager ────────────────────────────
881        // The SessionManager needs a PixToolbox on all node types to handle
882        // PIX protocol messages (SsaRequest on Entry, SsaCommit on Exit).
883        // Only Exit nodes construct an SsaReconstructor and wire SSA recovery
884        // events back through the packet pipeline. Relay nodes do NOT
885        // participate in PIX share processing at the pipeline level.
886        // Entry nodes get a bare-bones PixToolbox (share_generator + dummy
887        // reconstructor) to handle SsaRequest, but do not use the
888        // reconstructor for SSA recovery.
889        let pix_toolbox = match (role, exit_ack_share) {
890            (protocol::NodeType::Exit, Some(ref ssa_events)) => {
891                let (pix_tools, pipeline_builder) = wire_exit_pix(
892                    pipeline_builder,
893                    ssa_generator.clone(),
894                    ssa_reconstructor(&self.cfg.pix)?,
895                    ssa_events,
896                    self.smgr.clone(),
897                    &mut processes,
898                );
899
900                let pipeline_processes = pipeline_builder.build_for_exit();
901                processes.extend_from(pipeline_processes);
902                Some(pix_tools)
903            }
904            (protocol::NodeType::Relay, None) => {
905                // Pure relay nodes do not participate in PIX — do not create a
906                // PixToolbox so the SessionManager rejects UsePIX
907                // pre-emptively during session initiation.
908                let pipeline_processes = pipeline_builder.with_ticket_events(ticket_events).build_for_relay();
909                processes.extend_from(pipeline_processes);
910                None
911            }
912            (protocol::NodeType::Relay, Some(ref ssa_events)) => {
913                // A relay that also acts as an Exit (has exit_ack_share) needs
914                // a full PixToolbox to handle the PIX handshake — the same wiring as the
915                // `(Exit, Some)` arm above, differing only in the terminal below.
916                let (pix_tools, pipeline_builder) = wire_exit_pix(
917                    pipeline_builder,
918                    ssa_generator.clone(),
919                    ssa_reconstructor(&self.cfg.pix)?,
920                    ssa_events,
921                    self.smgr.clone(),
922                    &mut processes,
923                );
924
925                let pipeline_processes = pipeline_builder.with_ticket_events(ticket_events).build_for_relay();
926                processes.extend_from(pipeline_processes);
927                Some(pix_tools)
928            }
929            (protocol::NodeType::Entry, Some(ref ssa_events)) => {
930                // Entry nodes need a bare-bones PixToolbox (share_generator only)
931                // to handle incoming SsaRequest messages from the Exit.
932                // No SSA reconstruction needed on Entry — forward events to the
933                // PIX event broadcast so subscribers (e.g. tests) see them.
934                // Configured like any other, not left on the defaults: this reconstructor is called
935                // "dummy" because Entry does not reconstruct, but it is not provably unreachable —
936                // the comment below records that an inbound `UsePIX` does reach `handle_ssa_commit`
937                // here. A knob that binds on some node roles and not others is the same defect
938                // shape as a comment quoting a constant it does not reference.
939                let dummy_reconstructor = ssa_reconstructor(&self.cfg.pix)?;
940                let (pix_tools, session_pix_events) = PixToolbox::new(ssa_generator.clone(), dummy_reconstructor);
941                processes.insert(
942                    HoprTransportProcess::PixEvents,
943                    hopr_utils::spawn_as_abortable!(
944                        // The same mapping as the Exit and Relay arms, deliberately: `DepositNeeded`
945                        // is not *expected* here, but it is reachable. `SessionManager` has no
946                        // notion of node role and this one runs with a live toolbox — only the
947                        // incoming-session notification is drained below — so an inbound
948                        // `StartSession` carrying `UsePIX` is accepted and ends in
949                        // `handle_ssa_commit` emitting it. Panicking would take the whole PIX event
950                        // task with it, stranding this node's own outbound deposits.
951                        session_pix_events
952                            .map(session_pix_event_to_pix_event)
953                            .map(Ok)
954                            .forward(ssa_events.clone().sink_map_err(HoprTransportError::other))
955                    ),
956                );
957
958                processes.extend_from(pipeline_builder.build_for_entry());
959                Some(pix_tools)
960            }
961            (_, None) => {
962                // Nodes without pix_events sink (no PIX configured at all)
963                let pipeline_processes = match role {
964                    protocol::NodeType::Relay => pipeline_builder.with_ticket_events(ticket_events).build_for_relay(),
965                    protocol::NodeType::Exit => pipeline_builder.build_for_exit(),
966                    protocol::NodeType::Entry => pipeline_builder.build_for_entry(),
967                };
968                processes.extend_from(pipeline_processes);
969                None
970            }
971        };
972
973        // ── Ssmgr startup ─────────────────────────────────────────────────
974        // Entry nodes don't accept incoming sessions, relay/exit nodes do.
975        let smgr_start_res = if role != protocol::NodeType::Entry {
976            self.smgr.start(
977                unresolved_routing_msg_tx.clone(),
978                on_incoming_session.ok_or_else(|| {
979                    HoprTransportError::Api("on_incoming_session channel is required for relay/exit nodes".into())
980                })?,
981                pix_toolbox,
982                on_session_admission,
983            )
984        } else {
985            // An Entry establishes Sessions rather than admitting them, so there is nothing to ask
986            // about even if the caller installed a session server.
987            self.smgr.start(
988                unresolved_routing_msg_tx.clone(),
989                futures::sink::drain(),
990                pix_toolbox,
991                None,
992            )
993        };
994
995        smgr_start_res
996            .map_err(|_| HoprTransportError::Api("failed to start session manager".into()))?
997            .into_iter()
998            .enumerate()
999            .map(|(i, jh)| (HoprTransportProcess::SessionsManagement(i + 1), jh))
1000            .for_each(|(k, v)| {
1001                processes.insert(k, v);
1002            });
1003        // -- periodic counter flush
1004        let flush_counters = self.counters.clone();
1005        let flush_graph = self.graph.clone();
1006        let flush_me = *self.packet_key.public();
1007        let flush_interval = self.cfg.counter_flush_interval;
1008        processes.insert(
1009            HoprTransportProcess::CounterFlush,
1010            hopr_utils::spawn_as_abortable!(async move {
1011                use hopr_api::graph::traits::{EdgeObservableWrite, EdgeWeightType};
1012
1013                futures_time::stream::interval(futures_time::time::Duration::from(flush_interval))
1014                    .for_each(|_| {
1015                        for (peer, num_packets, num_acks) in flush_counters.drain() {
1016                            tracing::trace!(
1017                                %peer,
1018                                num_packets,
1019                                num_acks,
1020                                "flushing protocol conformance counters"
1021                            );
1022                            flush_graph.upsert_edge(&flush_me, &peer, |obs| {
1023                                obs.record(EdgeWeightType::ImmediateProtocolConformance { num_packets, num_acks });
1024                            });
1025                        }
1026                        futures::future::ready(())
1027                    })
1028                    .await;
1029            }),
1030        );
1031
1032        // -- periodic SURB round-trip flush
1033        tracing::info!(?role, "starting surb round-trip flush task");
1034        // Long enough to outlast the silence gate that produced the evidence, so a path that stays
1035        // dead is re-marked before the previous mark lapses, and short enough that a path which
1036        // recovers unnoticed returns to closed-loop control promptly.
1037        const RETURN_PATH_DEGRADED_GRACE: std::time::Duration = std::time::Duration::from_secs(10);
1038        let surb_flush_graph = self.graph.clone();
1039        let surb_flush_interval = self.cfg.surb_flush_interval;
1040        let surb_flush_smgr = self.smgr.clone();
1041        let surb_flush_chain = self.chain_api.clone();
1042        let surb_flush_planner = self.path_planner.clone();
1043        processes.insert(
1044            HoprTransportProcess::SurbFlush,
1045            hopr_utils::spawn_as_abortable!(async move {
1046                let mut episodes = protocol::return_path_recovery::ReturnPathEpisodes::new(RETURN_PATH_DEGRADED_GRACE);
1047                let mut ticks = futures_time::stream::interval(futures_time::time::Duration::from(surb_flush_interval));
1048
1049                while ticks.next().await.is_some() {
1050                    // Re-plan first, refill only if re-planning moved traffic. Run the other way
1051                    // round the refill mints SURBs onto the very route the planner is abandoning --
1052                    // see `protocol::return_path_recovery`. Detection/flush/tick ordering lives in
1053                    // `run_flush_tick`.
1054                    //
1055                    // Borrowed rather than cloned per call: the callbacks are `FnMut`, so anything
1056                    // they capture has to survive being invoked once per silent destination.
1057                    let (planner, chain, smgr) = (&surb_flush_planner, &surb_flush_chain, &surb_flush_smgr);
1058                    let now_ms = hopr_utils::platform::time::native::current_time()
1059                        .as_unix_timestamp()
1060                        .as_millis();
1061                    for step in protocol::return_path_recovery::run_flush_tick(
1062                        &surb_round_trips,
1063                        &surb_flush_graph,
1064                        now_ms,
1065                        &mut episodes,
1066                        |destination| async move { planner.recompute_paths_from(&destination).await },
1067                        |destination| async move {
1068                            // Sessions name their destination by its chain address, this
1069                            // telemetry by its packet key, and a `NodeId` holding one is never
1070                            // equal to a `NodeId` holding the other -- so the match has to be
1071                            // made on a resolved form, not on the enum.
1072                            //
1073                            // The counterparty has to still be receiving SURBs to reply with,
1074                            // and its silence has by now convinced our balancer that it is well
1075                            // stocked -- so tell the Sessions routed there to stop believing
1076                            // that estimate while the evidence says otherwise.
1077                            match chain.packet_key_to_chain_key(&destination) {
1078                                Ok(Some(address)) => smgr.mark_return_path_degraded(
1079                                    &hopr_api::types::internal::prelude::NodeId::Chain(address),
1080                                    RETURN_PATH_DEGRADED_GRACE,
1081                                ),
1082                                // A resolver error is exactly the failure this recovery path exists
1083                                // to surface, so it must not be collapsed into "no chain key" — log
1084                                // it rather than silently marking zero sessions.
1085                                Err(error) => {
1086                                    tracing::warn!(%destination, %error, "could not resolve a silent destination's chain key to mark it degraded");
1087                                    0
1088                                }
1089                                Ok(None) => 0,
1090                            }
1091                        },
1092                    )
1093                    .await
1094                    {
1095                        match step {
1096                            protocol::return_path_recovery::RecoveryStep::Replanned { destination, moved } => {
1097                                tracing::info!(%destination, entries = moved, "return path silent, re-planned")
1098                            }
1099                            protocol::return_path_recovery::RecoveryStep::Refilled { destination, sessions } => {
1100                                tracing::info!(%destination, sessions, "refilling behind the re-plan")
1101                            }
1102                        }
1103                    }
1104                }
1105            }),
1106        );
1107
1108        // -- network probing
1109        let manual_ping_channel_capacity = std::env::var("HOPR_INTERNAL_MANUAL_PING_CHANNEL_CAPACITY")
1110            .ok()
1111            .and_then(|s| s.trim().parse::<usize>().ok())
1112            .filter(|&c| c > 0)
1113            .unwrap_or(128);
1114        debug!(capacity = manual_ping_channel_capacity, "Creating manual ping channel");
1115        let (manual_ping_tx, manual_ping_rx_raw) =
1116            crossfire::mpsc::bounded_async::<(OffchainPublicKey, PingQueryReplier)>(manual_ping_channel_capacity);
1117        let manual_ping_rx = manual_ping_rx_raw.into_stream();
1118
1119        let probe = Probe::new(self.cfg.probe, self.probing_tag_allocator.clone());
1120
1121        let (probing_processes, probe_classifier) = probe
1122            .continuously_scan(
1123                unresolved_routing_msg_tx.clone(),
1124                manual_ping_rx,
1125                cover_traffic,
1126                self.graph.clone(),
1127            )
1128            .await;
1129
1130        processes.flat_map_extend_from(probing_processes, HoprTransportProcess::Probing);
1131
1132        // manual ping
1133        self.ping
1134            .clone()
1135            .set(Pinger::new(
1136                PingConfig {
1137                    timeout: self.cfg.probe.timeout,
1138                },
1139                manual_ping_tx,
1140            ))
1141            .map_err(|_| HoprTransportError::Api("must set the ticket aggregation writer only once".into()))?;
1142
1143        // Wire incoming: cover-traffic-filtered stream → probe classify → (session dispatch).
1144        // This stage must run in a background task, so the pipeline drains even when the
1145        // caller discards the returned HoprSocket (e.g. edge-node builder).
1146        //
1147        // The channel uses a resilient for_each rather than .forward() so that a disconnected
1148        // receiver (HoprSocket dropped without consuming) logs an error and continues rather
1149        // than collapsing the entire ingress pipeline. Callers should use HoprSocket::reader()
1150        // and actively drain the stream; see hopr-lib builder for the reference drain.
1151        let (on_incoming_data_tx, on_incoming_data_rx) =
1152            bounded_sink_channel::<ApplicationDataIn>(msg_protocol_bidirectional_channel_capacity);
1153        let smgr = self.smgr.clone();
1154        let unresolved_routing_msg_tx_for_task = unresolved_routing_msg_tx.clone();
1155        processes.insert(
1156            HoprTransportProcess::SessionsManagement(0),
1157            hopr_utils::spawn_as_abortable!(async move {
1158                probe_classifier
1159                    .filter_stream(unresolved_routing_msg_tx_for_task, rx_from_protocol)
1160                    .filter_map(move |(pseudonym, data)| {
1161                        hopr_transport_session::counters::DISPATCH_MESSAGE_CALLS
1162                            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
1163                        // `dispatch_message` is synchronous and lock-free (moka::sync + crossfire
1164                        // try_send); it never blocks. However, the crossfire stream that feeds
1165                        // this loop does not participate in tokio's cooperative budget, so
1166                        // `future::ready` — which is always Poll::Ready — would let this task
1167                        // monopolize its worker thread under a saturated inbound queue (Processed
1168                        // items are filtered to None before the fold's tx.send().await, the only
1169                        // other suspension point on the hot path). Calling consume_budget() here
1170                        // integrates this loop into tokio's coop scheduler: it is a cheap
1171                        // thread-local decrement on the fast path and only actually yields
1172                        // (~every 128 polls) when the budget is exhausted, giving co-located
1173                        // tasks (including the SPHINX-crypto Rayon pipeline) a chance to run.
1174                        let result = match smgr.dispatch_message(pseudonym, data) {
1175                            Ok(DispatchResult::Processed) => {
1176                                tracing::trace!("message dispatch completed");
1177                                None
1178                            }
1179                            Ok(DispatchResult::Unrelated(data)) => {
1180                                tracing::trace!("unrelated message dispatch completed");
1181                                Some(data)
1182                            }
1183                            // Benign drop: the session's sink has closed, its inbox is momentarily
1184                            // full (backpressure), or the session is already deregistered. Counted
1185                            // in the session manager and logged quietly here so a departing
1186                            // counterparty cannot spam ERROR once per in-flight packet.
1187                            Ok(DispatchResult::Dropped(reason)) => {
1188                                tracing::trace!(?reason, "dropped session packet");
1189                                None
1190                            }
1191                            Err(error) => {
1192                                tracing::error!(%error, "error while dispatching packet in the session manager");
1193                                None
1194                            }
1195                        };
1196                        async move {
1197                            hopr_utils::runtime::prelude::consume_budget().await;
1198                            result
1199                        }
1200                    })
1201                    .fold(on_incoming_data_tx, |mut tx, data| async move {
1202                        if tx.send(data).await.is_err() {
1203                            tracing::error!(
1204                                task = %HoprTransportProcess::SessionsManagement(0),
1205                                "incoming-data channel disconnected — dropping unrelated packet; \
1206                                 HoprSocket must be consumed or drained by the caller"
1207                            );
1208                        }
1209                        tx
1210                    })
1211                    .await;
1212                tracing::warn!(
1213                    task = %HoprTransportProcess::SessionsManagement(0),
1214                    "long-running background task finished"
1215                );
1216            }),
1217        );
1218
1219        // Populate the OnceLock at the end, making sure everything before didn't fail.
1220        self.network
1221            .clone()
1222            .set(transport_network)
1223            .map_err(|_| HoprTransportError::Api("transport network viewer already set".into()))?;
1224
1225        Ok((
1226            (on_incoming_data_rx.boxed(), unresolved_routing_msg_tx).into(),
1227            processes,
1228        ))
1229    }

1432/// How many PIX share resolutions may be dispatched into the [`SessionManager`] at once.
1433///
1434/// A dispatch can block: lifecycle events are delivered to a per-session supervisor over a bounded
1435/// channel, so a session whose supervisor is busy backpressures its sender. All sessions share this
1436/// one stream, so dispatching sequentially would let one such session hold up PIX progress — and
1437/// therefore the pipelined next-SSA request — for every other session on the node.
1438const PIX_EVENT_DISPATCH_CONCURRENCY: usize = 64;
1439
1440/// Maps a [`HoprSessionOutPixEvent`] into the upper-layer [`PixEvent`] carrying the deposit
1441/// instruction for the funding strategy.
1442fn session_pix_event_to_pix_event(event: HoprSessionOutPixEvent) -> PixEvent {
1443    match event {
1444        HoprSessionOutPixEvent::ReadyToDeposit(AgreedSsaQuota {
1445            ssa_id,
1446            deposit_address,
1447            quota_per_ssa,
1448            deposit_data,
1449        }) => PixEvent::NewDepositAddress(PixNewDepositAddress {
1450            id: PixAddressId::new(ssa_id.pseudonym(), ssa_id.ssa_index()),
1451            address: deposit_address.into(),
1452            quota: quota_per_ssa,
1453            // What the Exit attached to this SSA in its `SsaRequest`, rebuilt by the Session layer
1454            // from the message. Empty when the Exit's pool produced none — `PixDepositData::is_empty`
1455            // is what tells the two apart, so there is nothing to signal here.
1456            deposit_data,
1457        }),
1458        HoprSessionOutPixEvent::DepositNeeded(
1459            AgreedSsaQuota {
1460                ssa_id,
1461                deposit_address,
1462                quota_per_ssa,
1463                deposit_data,
1464            },
1465            notifier,
1466        ) => PixEvent::DepositAddressReceived(PixDepositAddressReceived {
1467            id: PixAddressId::new(ssa_id.pseudonym(), ssa_id.ssa_index()),
1468            address: deposit_address.into(),
1469            quota: quota_per_ssa,
1470            // This node's own data coming back to it: the pool produced it for this SSA before the
1471            // request went out, and gets it returned now that the address it pays for is known.
1472            deposit_data,
1473            // Not optional as of hopr-api 4.0.1: the Exit always has a channel to report the deposit
1474            // on, and every `DepositAddressReceived` this maps from carries one.
1475            deposit_updated: notifier,
1476        }),
1477        // Straight through: `PixDepositDataRequest` is the pool's own request type, and the Session
1478        // layer neither adds to it nor reads the answers off the channel it carries.
1479        HoprSessionOutPixEvent::DepositDataRequest(request) => PixEvent::DepositDataRequest(request),
1480    }
1481}

1483/// Notifies the [`SessionManager`] about a single PIX share resolution and returns the
1484/// upper-layer [`PixEvent`] it produces, if any.
1485async fn dispatch_share_resolution(smgr: Arc<HoprSessionManager>, resolution: HoprShareResolution) -> Option<PixEvent> {
1486    match resolution {
1487        ShareResolution::RecoveredSsa(ssa_recovery_event) => {
1488            if let Err(error) = smgr
1489                .dispatch_pix_event(HoprSessionInPixEvent::SsaRecovered(ssa_recovery_event.ssa_id))
1490                .await
1491            {
1492                tracing::error!(%error, "failed to dispatch SSA recovery PIX event to the SessionManager");
1493            }
1494            Some(recovered_ssa_to_pix_event(&ssa_recovery_event))
1495        }
1496        ShareResolution::AlmostRecoveredSsa(ssa_id) => {
1497            if let Err(error) = smgr
1498                .dispatch_pix_event(HoprSessionInPixEvent::SsaAlmostRecovered(ssa_id))
1499                .await
1500            {
1501                tracing::error!(%error, %ssa_id, "failed to dispatch early SSA recovery event to the SessionManager");
1502            }
1503            None
1504        }
1505        ShareResolution::InvalidShares {
1506            peer,
1507            ssa_id,
1508            observed_total,
1509        } => {
1510            tracing::error!(
1511                %peer, %ssa_id, observed_total,
1512                "first RP relayer sent acknowledgement indicating invalid PIX share from Entry"
1513            );
1514            if let Err(error) = smgr
1515                .dispatch_pix_event(HoprSessionInPixEvent::UnverifiableShares { ssa_id, observed_total })
1516                .await
1517            {
1518                tracing::error!(%error, %ssa_id, "failed to dispatch invalid share PIX event to the SessionManager");
1519            }
1520            None
1521        }
1522        // Feeds the supervisor's recovery deadlines and the egress gate's progress ceiling. A funded
1523        // Session that stops making progress stops being served, so dropping these outright would
1524        // stall a healthy Session rather than merely lose a statistic.
1525        ShareResolution::Progress(progress) => {
1526            tracing::trace!(
1527                ssa_id = %progress.ssa_id,
1528                useful_shares = progress.useful_shares,
1529                shares_seen = progress.shares_seen,
1530                target = progress.target_useful_shares,
1531                recovered_polynomials = progress.recovered_polynomials,
1532                "pix recovery progress"
1533            );
1534            if let Err(error) = smgr
1535                .dispatch_pix_event(HoprSessionInPixEvent::RecoveryProgress(progress))
1536                .await
1537            {
1538                tracing::trace!(%error, ssa_id = %progress.ssa_id, "recovery progress for an unsupervised session");
1539            }
1540            None
1541        }
1542    }
1543}

1545/// Builds the Exit-side SSA reconstructor from operator configuration.
1546///
1547/// The single place `PixReconstructorConfig` becomes an `SsaReconstructorConfig`, called once per
1548/// node role that needs a reconstructor. Both production sites used to take
1549/// `SsaReconstructorConfig::default()` instead, which left all seven of the Exit's dials — capacity,
1550/// lifetimes, the acknowledgement window — unreachable from a config file.
1551///
1552/// `try_new` rather than `new`, for the same reason the generator uses it at the one call site
1553/// above: these values now come from an operator, so an out-of-range one is a startup error to
1554/// report and not a panic to take the node down with. `run_inner` has already validated
1555/// `self.cfg.pix` — `reconstructor` is `#[validate(nested)]` — so in practice this reports what that
1556/// validation would have caught anyway; it stays fallible because nothing in the type system says
1557/// the two must be called in that order.
1558fn ssa_reconstructor(cfg: &PixGlobalConfig) -> errors::Result<Arc<hopr_protocol_pix::SsaReconstructor<HoprPixSpec>>> {
1559    hopr_protocol_pix::SsaReconstructor::<HoprPixSpec>::try_new(cfg.reconstructor.into())
1560        .map(Arc::new)
1561        .map_err(|error| HoprTransportError::Api(format!("invalid SSA reconstructor configuration: {error}")))
1562}

1564/// Turns the reconstructor's synchronous resolution wake-up into an asynchronous stream.
1565///
1566/// The one-slot channel carries only wake-ups, never resolutions. A full channel therefore means a
1567/// wake-up is already pending and can be safely coalesced; the resolutions themselves remain in the
1568/// reconstructor until this stream or the acknowledgement fallback atomically takes them.
1569fn commitment_resolution_stream(
1570    ssa_reconstructor: Arc<hopr_protocol_pix::SsaReconstructor<HoprPixSpec>>,
1571) -> impl futures::Stream<Item = HoprShareResolution> + Send + 'static {
1572    let (ready_tx, ready_rx) = bounded_sink_channel::<()>(1);
1573    let registered = ssa_reconstructor.set_ready_resolution_notifier(move || match ready_tx.try_send(()) {
1574        Ok(()) | Err(crossfire::TrySendError::Full(())) => {}
1575        Err(crossfire::TrySendError::Disconnected(())) => {
1576            tracing::error!("PIX commitment-resolution notification stream is disconnected");
1577        }
1578    });
1579    if !registered {
1580        tracing::error!("PIX reconstructor already has a commitment-resolution notification stream");
1581    }
1582
1583    let ssa_reconstructor = Arc::downgrade(&ssa_reconstructor);
1584    ready_rx.flat_map(move |_| {
1585        futures::stream::iter(
1586            ssa_reconstructor
1587                .upgrade()
1588                .map(|reconstructor| reconstructor.take_ready_resolutions())
1589                .unwrap_or_default(),
1590        )
1591    })
1592}

1595/// [`HoprTransportProcess::PixEvents`] task, returning the toolbox and the pipeline builder with
1596/// share processing attached.
1597///
1598/// Shared by the `(Exit, Some)` and `(Relay, Some)` arms of `run_inner`, which differ only in the
1599/// terminal `build_for_*` call and the relay's extra `with_ticket_events` step. Both need the exact
1600/// same wiring, and keeping it in one place is not cosmetic: the previous two copies had already
1601/// drifted, with the comment on the reconstructor config fixed in the Exit copy only.
1602///
1603/// The reconstructor arrives built, like the generator, so that [`ssa_reconstructor`] stays the one

1608/// untouched, which is why none of them carry bounds here.
1609#[allow(clippy::type_complexity)]

1611    pipeline_builder: HoprPacketPipelineBuilder<WIn, WOut, Chain, S, TFact, G, AppOut, AppIn, TEvt>,
1612    ssa_generator: Arc<hopr_protocol_pix::SsaShareGenerator<HoprPixSpec>>,
1613    ssa_reconstructor: Arc<hopr_protocol_pix::SsaReconstructor<HoprPixSpec>>,
1614    ssa_events: &PixEvt,
1615    smgr: Arc<HoprSessionManager>,
1616    processes: &mut AbortableList<HoprTransportProcess>,
1617) -> (
1618    PixToolbox,
1619    HoprPacketPipelineBuilder<
1620        WIn,
1621        WOut,
1622        Chain,
1623        S,
1624        TFact,
1625        G,
1626        AppOut,

1629        Arc<hopr_protocol_pix::SsaReconstructor<HoprPixSpec>>,
1630        CrossfireSink<HoprShareResolution>,
1631    >,
1632)

1634    PixEvt: futures::Sink<PixEvent> + Clone + Unpin + Send + 'static,

1636{

1638    let (pix_tools, session_pix_events) = PixToolbox::new(ssa_generator, ssa_reconstructor.clone());
1639    let (ssa_share_resolution_events_tx, ssa_share_resolution_events_rx) = bounded_sink_channel(1024);
1640    processes.insert(
1641        HoprTransportProcess::PixEvents,

1643            pix_event_stream(
1644                session_pix_events,

1651    );

1654        pix_tools,
1655        pipeline_builder.with_exit_ack_share_processing(ssa_reconstructor, ssa_share_resolution_events_tx),
1656    )
1657}

1659/// Builds the merged PIX event stream feeding the upper layer.
1660///
1661/// Combines the Session-originated deposit instructions with the share resolutions coming out of
1662/// the packet pipeline. Share resolutions are dispatched into the `SessionManager` concurrently
1663/// (see [`PIX_EVENT_DISPATCH_CONCURRENCY`]).
1664///
1665/// Concurrency does not require ordering guarantees here: events for one session converge on that
1666/// session's supervisor, which is a single-threaded state machine whose handlers are idempotent per
1667/// phase and whose counters are absolute rather than incremental. Of the events belonging to one
1668/// cycle exactly one advances it and the rest are recognised as stale, whatever order they arrive
1669/// in.
1670fn pix_event_stream(
1671    session_pix_events: impl futures::Stream<Item = HoprSessionOutPixEvent> + Send + 'static,
1672    ssa_share_resolutions: impl futures::Stream<Item = HoprShareResolution> + Send + 'static,
1673    smgr: Arc<HoprSessionManager>,
1674) -> impl futures::Stream<Item = PixEvent> + Send + 'static {
1675    session_pix_events.map(session_pix_event_to_pix_event).merge(
1676        ssa_share_resolutions
1677            .then_concurrent(
1678                move |resolution| dispatch_share_resolution(smgr.clone(), resolution),
1679                PIX_EVENT_DISPATCH_CONCURRENCY,
1680            )
1681            .filter_map(futures::future::ready),
1682    )
1683}

1825        // defer waiting for their verifier and this acknowledgement call returns no resolution.
1826        let mut acks = Vec::new();
1827        while let Some((msg, share)) = {
1828            let msg = random_bytes::<20>();

1891    /// Verifies that the `filter_map` closure in `SessionsManagement(0)` does not monopolize
1892    /// the tokio worker thread under a continuously saturated inbound stream.
1893    ///
1894    /// ### Background
1895    ///
1896    /// The inbound dispatch loop (`transport/hopr/src/lib.rs`) drives:
1897    ///
1898    /// ```text
1899    /// rx_from_protocol (crossfire, no coop budget)
1900    ///   → filter_stream
1901    ///   → filter_map(sync dispatch_message + consume_budget().await)
1902    ///   → fold(tx.send().await)          ← only fires for Unrelated items
1903    /// ```
1904    ///
1905    /// On the `Processed` hot path (common case), items become `None` and are filtered
1906    /// out before the `fold`'s `tx.send().await`, so the only suspension point is the
1907    /// `consume_budget()` call inside `filter_map`.
1908    ///
1909    /// ### What this test proves
1910    ///
1911    /// On a `current_thread` runtime (single worker, fully cooperative scheduling) a tight
1912    /// stream loop that never returns `Poll::Pending` starves every other spawned task until
1913    /// the stream is drained.  By replacing `future::ready(result)` with an `async` block
1914    /// that calls `consume_budget().await` before returning, we give the scheduler a chance
1915    /// to preempt the loop every ≈128 polls.  The "canary" task — a plain `yield_now` loop —
1916    /// *must* make forward progress while the dispatch stream is draining for the fix to be
1917    /// considered effective.
1918    ///
1919    /// Reverting the `consume_budget().await` to `future::ready(result)` will make this
1920    /// test fail: the canary will never be scheduled and its counter will remain 0.
1921    #[tokio::test(flavor = "current_thread")]
1922    async fn sessions_management_dispatch_loop_yields_to_scheduler() {
1923        // N large enough to trigger ≥1 genuine yield (budget = 128 per reset).
1924        const N: usize = 1_000;
1925
1926        // Canary: counts how many times it was scheduled during the dispatch run.
1927        let counter = Arc::new(AtomicU64::new(0));
1928        let counter_clone = counter.clone();
1929        let canary = spawn(async move {
1930            loop {
1931                counter_clone.fetch_add(1, Ordering::Relaxed);
1932                yield_now().await;
1933            }
1934        });
1935
1936        // Dispatch loop — same shape as the fixed SessionsManagement(0) hot path:
1937        //   sync compute → consume_budget().await → None (Processed path)
1938        let unrelated_count = futures::stream::iter(0..N)
1939            .filter_map(|_item| {
1940                // Dispatch result computed synchronously (no clone); budget consumed async.
1941                let result: Option<()> = None; // every item is "Processed"
1942                async move {
1943                    consume_budget().await;
1944                    result
1945                }
1946            })
1947            .fold(0u64, |acc, _| async move { acc + 1 })
1948            .await;
1949
1950        assert_eq!(
1951            unrelated_count, 0,
1952            "all Processed items should be filtered out by filter_map"
1953        );
1954
1955        // The canary must have been scheduled at least once while the dispatch loop ran.
1956        // If consume_budget() is removed and future::ready() is used instead, the loop
1957        // never returns Poll::Pending on the current_thread executor and this assertion fails.
1958        let progress = counter.load(Ordering::Relaxed);
1959        assert!(
1960            progress > 0,
1961            "canary task made no forward progress during the {N}-item dispatch run (counter = {progress}); the \
1962             dispatch loop is monopolizing the executor thread — ensure filter_map returns `async move {{ \
1963             consume_budget().await; result }}` rather than `future::ready(result)`"
1964        );
1965
1966        canary.abort();
1967        canary.await.ok();
1968    }
