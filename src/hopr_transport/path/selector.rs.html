21/// Accumulated path cost and quality aggregates, folded edge-by-edge during DFS.
22///
23/// `PartialOrd` / `PartialEq` compare only the `cost` field so the DFS
24/// pruning threshold (`min_cost`) operates on the same scalar as before.

32}

59/// Returns the minimum of two `Option<T>` values, preferring `Some` over `None`.
60fn opt_min<T: PartialOrd>(a: Option<T>, b: Option<T>) -> Option<T> {
61    match (a, b) {
62        (Some(x), Some(y)) => Some(if x <= y { x } else { y }),
63        (x, y) => x.or(y),
64    }
65}

67/// Wraps an `EdgeValueFn<f64, W>` as a `ValueFn` whose `Value` type carries
68/// both the cost and per-path quality aggregates.
69///
70/// All cost semantics are delegated to the inner `EdgeValueFn`; the wrapper
71/// only adds the aggregate fold on the same `&Weight` reference already
72/// available during DFS, so no extra graph lookups are needed.
73struct MetricsValueFn<W: EdgeObservableRead> {
74    inner: EdgeValueFn<f64, W>,
75    /// Face value used to express each edge's balance as a count of single-hop tickets.
76    ///
77    /// Read once per query rather than stored on any edge: deriving a count at query time is
78    /// safe, whereas deriving it at edge-update time is what made a price change stale the
79    /// whole graph.
80    ticket_face_value: Option<hopr_api::graph::traits::Balance>,
81}
82
83impl<W> ValueFn for MetricsValueFn<W>
84where
85    W: EdgeObservableRead + Send + 'static,
86{
87    type Value = PathCostWithMetrics;
88    type Weight = W;
89
90    fn initial_value(&self) -> Self::Value {
91        PathCostWithMetrics {
92            cost: self.inner.initial_value(),
93            total_latency_ms: Some(0),
94            min_probe_success_rate: None,
95            min_ack_rate: None,
96            fundable_tickets_floor: None,
97        }
98    }
99
100    fn min_value(&self) -> Option<Self::Value> {
101        self.inner.min_value().map(|c| PathCostWithMetrics {
102            cost: c,
103            total_latency_ms: None,
104            min_probe_success_rate: None,
105            min_ack_rate: None,
106            fundable_tickets_floor: None,
107        })
108    }
109
110    fn into_value_fn(self) -> BasicValueFn<Self::Value, Self::Weight> {
111        let inner = self.inner.into_value_fn();
112        let ticket_face_value = self.ticket_face_value;
113        Arc::new(move |prev: PathCostWithMetrics, observed: &W, idx: usize| {
114            let cost = inner(prev.cost, observed, idx);
115
116            let edge_lat = observed
117                .immediate_qos()
118                .and_then(|m| m.average_latency())
119                .or_else(|| observed.intermediate_qos().and_then(|m| m.average_latency()));
120            let total_latency_ms = match (prev.total_latency_ms, edge_lat) {
121                (Some(acc), Some(lat)) => Some(((acc as u128 + lat.as_millis()).min(u32::MAX as u128)) as u32),
122                _ => None,
123            };
124
125            // Probe rate: taking the min of immediate and intermediate guards against nodes that
126            // look good on direct probes but degrade under multi-hop load.
127            // `and_then`, not `map`: a stream with no observations contributes nothing to the min
128            // rather than contributing a zero it never measured.
129            let edge_probe = observed
130                .immediate_qos()
131                .and_then(|m| m.average_probe_rate())
132                .into_iter()
133                .chain(observed.intermediate_qos().and_then(|m| m.average_probe_rate()))
134                .reduce(f64::min);
135            let min_probe_success_rate = opt_min(prev.min_probe_success_rate, edge_probe);
136
137            let edge_ack = observed.immediate_qos().and_then(|m| m.ack_rate());
138            let min_ack_rate = opt_min(prev.min_ack_rate, edge_ack);
139
140            // Expressed in single-hop tickets, not base units: the weighting factor below is a
141            // log ratio, and a base-unit balance would compress every realistic channel into a
142            // sliver of its output range.
143            //
144            // `zip`, not a fallback: an absent face value means the price is unknown, not that it
145            // is one. Dividing a base-unit balance by `1` yields a ticket count that saturates to
146            // `u128::MAX` and reads downstream as unlimited capacity — the opposite of what not
147            // knowing should imply. Contribute nothing to the floor until pricing arrives.
148            let edge_tickets = observed
149                .intermediate_qos()
150                .and_then(|m| m.balance())
151                .zip(ticket_face_value)
152                .map(|(balance, face_value)| {
153                    let tickets = if face_value.is_zero() {
154                        hopr_api::graph::traits::Balance::zero()
155                    } else {
156                        balance / face_value
157                    };
158                    if tickets > hopr_api::graph::traits::Balance::from(u128::MAX) {
159                        u128::MAX
160                    } else {
161                        tickets.low_u128()
162                    }
163                });
164            let fundable_tickets_floor = opt_min(prev.fundable_tickets_floor, edge_tickets);
165
166            PathCostWithMetrics {
167                cost,
168                total_latency_ms,
169                min_probe_success_rate,
170                min_ack_rate,
171                fundable_tickets_floor,
172            }
173        })
174    }
175}

177/// Trim the candidate set to lower median latency and minimise variance while
178/// preserving an anonymity floor and relayer diversity in every position.
179///
180/// Behaviour:
181/// - If `candidates.len() <= floor`, returns all candidates unchanged (`min(found_count, floor)` semantics — the floor
182///   is never a minimum to fabricate).
183/// - Sorts candidates with a known `total_latency_ms` ascending.
184/// - Keeps `floor` of them preferring distinct first relayers and bounding how often a relayer recurs in each later
185///   position (see `take_relayer_diverse` and `MIDDLE_HOP_SHARE`), lowest latency first within those rules.
186/// - If still over the floor with no populated candidates left, drops unpopulated candidates from the input-order tail.
187///
188/// A path is "fully measured" — and therefore preferred over unmeasured alternatives —
189/// when `total_latency_ms` is known AND either `hops == 0` (direct path, no channel
190/// expected) OR `fundable_tickets_floor` is also known.  This prevents 0-hop direct paths from
191/// being demoted simply because they carry no channel-capacity data.
192pub fn prune_for_consistency(candidates: Vec<PathWithMetrics>, floor: usize, hops: usize) -> Vec<PathWithMetrics> {
193    // floor == 0 means "no pruning" — caller opts out entirely.
194    if floor == 0 || candidates.len() <= floor {
195        return candidates;
196    }
197
198    let fully_measured =
199        |p: &PathWithMetrics| p.total_latency_ms.is_some() && (hops == 0 || p.fundable_tickets_floor.is_some());
200
201    let (mut populated, unpopulated): (Vec<_>, Vec<_>) = candidates.into_iter().partition(|p| fully_measured(p));
202
203    // Sort populated ascending by latency (lowest first → drop from the end).
204    populated.sort_by_key(|p| p.total_latency_ms.unwrap_or(u32::MAX));
205
206    // Prefer measured paths: keep as many populated paths as fit within the floor,
207    // then fill the remaining slots with unpopulated paths.  This ensures that
208    // latency-measured candidates are never discarded when unprobed paths alone
209    // would satisfy the floor.
210    let target_populated = populated.len().min(floor);
211    let populated = take_relayer_diverse(populated, target_populated);
212    let remaining = floor - target_populated;
213
214    let mut result = populated;
215    result.extend(unpopulated.into_iter().take(remaining));
216
217    result
218}

220/// A relayer may sit past the first position in at most `n.div_ceil(MIDDLE_HOP_SHARE)` of the `n`
221/// candidates `take_relayer_diverse` keeps, while the pool allows it, counted over every position past
222/// the first together: a relayer in the second position of some paths and the third of others costs
223/// as many paths when it dies as one in either alone.
224///
225/// Distinct first relayers alone leave every other position free to collapse: the best path from
226/// each first relayer often runs through the same second hop, so one relayer sat in the middle of
227/// every forward candidate, and when it died the whole forward direction went dark at once, with
228/// nothing left to corroborate a verdict against (measured: the busiest relay of a 2-hop session
229/// carried 29 % of all relayed packets; killed, the session never recovered). A third bounds what
230/// one relayer's death costs to about a third of the draws before the weights are even looked at.
231const MIDDLE_HOP_SHARE: usize = 3;
232
233/// Takes `n` candidates from `sorted`, preferring not-yet-represented first relayers and bounding
234/// how often any relayer recurs past the first position.
235///
236/// Plain truncation keeps the `n` lowest-latency paths, which in a well-connected network are often
237/// the *same* few relayers reached by different routes; every downstream draw then inherits that
238/// concentration. Relative order is preserved, so each pass stays latency-ordered. Four passes:
239/// paths with a new first relayer whose later relayers are all under the `MIDDLE_HOP_SHARE` cap;
240/// then any path under the cap; then, over the cap, paths with a new first relayer, so a pool whose
241/// every path shares a later relayer still keeps as many first relayers as it offers; then the best
242/// of the rest, so a thin pool still fills the floor.
243fn take_relayer_diverse(sorted: Vec<PathWithMetrics>, n: usize) -> Vec<PathWithMetrics> {
244    if sorted.len() <= n {
245        return sorted;
246    }
247
248    let cap = n.div_ceil(MIDDLE_HOP_SHARE).max(1);
249    let mut seen: Vec<OffchainPublicKey> = Vec::with_capacity(n);
250    let mut per_relayer: std::collections::HashMap<OffchainPublicKey, usize> = Default::default();
251    let mut taken = vec![false; sorted.len()];
252    let mut count = 0;
253
254    // The relayers of `path` past the first: `path` ends with the destination, which is no relayer.
255    let later_relayers = |p: &PathWithMetrics| {
256        p.path
257            .iter()
258            .skip(1)
259            .take(p.path.len().saturating_sub(2))
260            .copied()
261            .collect::<Vec<_>>()
262    };
263
264    let take = |i: usize,
265                p: &PathWithMetrics,
266                taken: &mut Vec<bool>,
267                per_relayer: &mut std::collections::HashMap<OffchainPublicKey, usize>| {
268        taken[i] = true;
269        for relayer in later_relayers(p) {
270            *per_relayer.entry(relayer).or_default() += 1;
271        }
272    };
273
274    // First pass: one path per distinct first relayer, best (lowest-latency) first, with every
275    // later relayer under the cap.
276    for (i, p) in sorted.iter().enumerate() {
277        if count == n {
278            break;
279        }
280        let Some(first) = p.path.first() else { continue };
281        let under_cap = later_relayers(p)
282            .into_iter()
283            .all(|relayer| per_relayer.get(&relayer).copied().unwrap_or(0) < cap);
284        if !seen.contains(first) && under_cap {
285            seen.push(*first);
286            take(i, p, &mut taken, &mut per_relayer);
287            count += 1;
288        }
289    }
290
291    // Second pass: any path whose later relayers are still under the cap, best first.
292    for (i, p) in sorted.iter().enumerate() {
293        if count == n {
294            break;
295        }
296        if taken[i] {
297            continue;
298        }
299        let under_cap = later_relayers(p)
300            .into_iter()
301            .all(|relayer| per_relayer.get(&relayer).copied().unwrap_or(0) < cap);
302        if under_cap {
303            if let Some(first) = p.path.first()
304                && !seen.contains(first)
305            {
306                seen.push(*first);
307            }
308            take(i, p, &mut taken, &mut per_relayer);
309            count += 1;
310        }
311    }
312
313    // Third pass: over the cap, a first relayer not yet represented still comes before a second path
314    // from one that is. Losing a later relayer then costs what the cap could not bound, but losing a
315    // first relayer costs no more than its own share.
316    for (i, p) in sorted.iter().enumerate() {
317        if count == n {
318            break;
319        }
320        if taken[i] {
321            continue;
322        }
323        let Some(first) = p.path.first() else { continue };
324        if !seen.contains(first) {
325            seen.push(*first);
326            take(i, p, &mut taken, &mut per_relayer);
327            count += 1;
328        }
329    }
330
331    // Fourth pass: fill any remaining slots with the best of the rest.
332    for slot in taken.iter_mut() {
333        if count == n {
334            break;
335        }
336        if !*slot {
337            *slot = true;
338            count += 1;
339        }
340    }
341
342    sorted
343        .into_iter()
344        .zip(taken)
345        .filter_map(|(p, keep)| keep.then_some(p))
346        .collect()
347}

349/// Number of distinct first relayers across `paths`.
350///
351/// The first relayer is the only node the destination gets to pick when it uses a SURB, so this is
352/// the count that decides whether the return-path degradation detector has a sibling to corroborate
353/// against: a value of 1 is the blind spot where a dead relayer is indistinguishable from a quiet
354/// peer. Paths with no hops (`path` empty) contribute nothing — a 0-hop return has no relayer.
355pub(crate) fn distinct_first_relayers(paths: &[PathWithMetrics]) -> usize {
356    paths
357        .iter()
358        .filter_map(|p| p.path.first())
359        .collect::<std::collections::HashSet<_>>()
360        .len()
361}

363/// Compute candidate paths from `src` to `dest` through `graph`.
364///
365/// `length` is the number of edges to traverse (= intermediate hops + 1).
366/// `take` caps the number of candidate paths returned.
367/// The graph crate returns only the intermediate nodes (both `src` and `dest` stripped);
368/// this function re-appends `dest` so callers receive `([intermediates..., dest], cost)`.
369fn compute_paths<G, W>(
370    graph: &G,
371    src: &OffchainPublicKey,
372    dest: &OffchainPublicKey,
373    length: std::num::NonZeroUsize,
374    take: usize,
375    value_fn: MetricsValueFn<W>,
376) -> (Vec<PathWithMetrics>, usize)
377where
378    G: NetworkGraphTraverse<NodeId = OffchainPublicKey, Observed = W>,
379    W: EdgeObservableRead + Send + 'static,
380{
381    let raw = graph.simple_paths(src, dest, length.get(), Some(take), value_fn);
382
383    let returned = raw.len();
384    let paths = raw
385        .into_iter()
386        .filter_map(|(path, _, metrics)| {
387            tracing::trace!(?path, cost = metrics.cost, "evaluating candidate path");
388            if metrics.cost > 0.0 {
389                let mut path = path;
390                path.push(*dest);
391                Some(PathWithMetrics::from((metrics, path)))
392            } else {
393                None
394            }
395        })
396        .collect();
397    (paths, returned)
398}

400/// Continues a fold that an earlier traversal started: it starts from `initial` and hands the
401/// wrapped function each edge's index shifted by `offset`.
402///
403/// A traversal from a path's second node, continued from the value after its first edge with an
404/// offset of one, costs every edge exactly as one traversal from the first node would: the value
405/// functions read an edge's position in the whole path, and that is what they are given.
406struct ContinuedValueFn<V, W> {
407    value_fn: BasicValueFn<V, W>,
408    initial: V,
409    min: Option<V>,
410    offset: usize,
411}
412
413impl<V, W> ValueFn for ContinuedValueFn<V, W>
414where
415    V: Clone + PartialOrd + Send + Sync + 'static,
416    W: EdgeObservableRead + Send + 'static,
417{
418    type Value = V;
419    type Weight = W;
420
421    fn initial_value(&self) -> Self::Value {
422        self.initial.clone()
423    }
424
425    fn min_value(&self) -> Option<Self::Value> {
426        self.min.clone()
427    }
428
429    fn into_value_fn(self) -> BasicValueFn<Self::Value, Self::Weight> {
430        let (value_fn, offset) = (self.value_fn, self.offset);
431        Arc::new(move |value, observed, index| value_fn(value, observed, index + offset))
432    }
433}

435/// How many times [`search_share`] searches one first relayer's continuations at most.
436///
437/// Each search asks for twice as many as the last, so the last asks for eight times the share,
438/// and all of them together find fewer paths than twice the last.
439const SHARE_SEARCHES: u32 = 4;
440
441/// One first relayer's continuations for its share of the pool.
442///
443/// Some continuations are unusable and dropped from the share: those of no value, and those that
444/// run through the source, which the single traversal from the source never returns but one that
445/// starts past it can. They must not take the places of usable ones, so while too few of the paths
446/// a search found are usable, the search is asked again for twice as many. The traversal shuffles
447/// each node's edges, so every search is a fresh draw that keeps what it found first.
448///
449/// Returns the paths of the first search that found `share` usable, or every continuation there
450/// is, or else those of the last of [`SHARE_SEARCHES`] searches. Doubling bounds the work where
451/// nearly every continuation is unusable, as in a subtree that reaches the destination only past
452/// the source: such a share stays short rather than paying a search per continuation.
453fn search_share<P>(share: usize, unusable: impl Fn(&P) -> bool, mut search: impl FnMut(usize) -> Vec<P>) -> Vec<P> {
454    let mut limit = share;
455    let mut searches = 1;
456    loop {
457        let found = search(limit);
458        let usable = found.iter().filter(|path| !unusable(path)).count();
459        if found.len() < limit || usable >= share {
460            return found;
461        }
462        if searches == SHARE_SEARCHES {
463            tracing::debug!(share, limit, usable, "share left short: most continuations unusable");
464            return found;
465        }
466        searches += 1;
467        limit = limit.saturating_mul(2);
468    }
469}

471/// Candidate paths from `src` to `dest`, spread evenly over `src`'s first relayers.
472///
473/// The traversal is depth-first and `take` keeps the paths it finds first, so a single traversal
474/// returns the whole subtree of one first relayer before it reaches the next. Over seven relayers
475/// a three-hop subtree holds 30 paths, so 50 candidates named one or two first relayers, and
476/// every path drawn until the next refresh went through them. Pruning for distinct first relayers
477/// cannot restore what the pool never held: a return pool of one first relayer put a Session's
478/// whole SURB stock behind it, and when that relayer died nothing the Exit sent got home.
479///
480/// So the first edges are found once, and each first relayer's continuations are searched
481/// separately within an even share of `take`, continued from the value after the first edge so
482/// that each path costs exactly what the single traversal would have made it cost. The shares
483/// are interleaved, so a pool cut short at `take` still keeps every first relayer.
484/// Returns the pool and the total number of continuations returned by all searches, including
485/// retries, before filtering and limiting each share or the pool.
486fn compute_paths_spread<G, W>(
487    graph: &G,
488    src: &OffchainPublicKey,
489    dest: &OffchainPublicKey,
490    length: std::num::NonZeroUsize,
491    take: usize,
492    value_fn: MetricsValueFn<W>,
493) -> (Vec<PathWithMetrics>, usize)
494where
495    G: NetworkGraphTraverse<NodeId = OffchainPublicKey, Observed = W>,
496    W: EdgeObservableRead + Send + 'static,
497{
498    // A direct path has no relayer to spread over.
499    if length.get() < 2 {
500        return compute_paths(graph, src, dest, length, take, value_fn);
501    }
502
503    let initial = value_fn.initial_value();
504    let min = value_fn.min_value();
505    let value_fn = value_fn.into_value_fn();
506
507    // Every first edge the full traversal would take, with the value after it: the minimum
508    // prunes here exactly as it does there.
509    let firsts = graph
510        .simple_paths_from(
511            src,
512            1,
513            None,
514            ContinuedValueFn {
515                value_fn: value_fn.clone(),
516                initial,
517                min: min.clone(),
518                offset: 0,
519            },
520        )
521        .into_iter()
522        .filter_map(|(nodes, _, after_first)| match nodes.as_slice() {
523            [first] if first != dest && first != src => Some((*first, after_first)),
524            _ => None,
525        })
526        .collect::<Vec<_>>();
527
528    if firsts.is_empty() {
529        return (Vec::new(), 0);
530    }
531
532    let share = take.div_ceil(firsts.len()).max(1);
533    let mut returned = 0;
534    let mut per_first = firsts
535        .into_iter()
536        .map(|(first, after_first)| {
537            let continuations = |limit| {
538                let found = graph.simple_paths(
539                    &first,
540                    dest,
541                    length.get() - 1,
542                    Some(limit),
543                    ContinuedValueFn {
544                        value_fn: value_fn.clone(),
545                        initial: after_first.clone(),
546                        min: min.clone(),
547                        offset: 1,
548                    },
549                );
550                returned += found.len();
551                found
552            };
553            // The continuations of no value or through `src`: `search_share` does not count them toward the
554            // share, and the pool drops them. One predicate for both, so the two cannot disagree.
555            let unusable = |(rest, _, metrics): &(Vec<OffchainPublicKey>, _, PathCostWithMetrics)| {
556                metrics.cost <= 0.0 || rest.contains(src)
557            };
558            search_share(share, unusable, continuations)
559                .into_iter()
560                .filter_map(move |continuation| {
561                    if unusable(&continuation) {
562                        return None;
563                    }
564                    let (rest, _, metrics) = continuation;
565                    let mut path = Vec::with_capacity(rest.len() + 2);
566                    path.push(first);
567                    path.extend(rest);
568                    path.push(*dest);
569                    tracing::trace!(?path, cost = metrics.cost, "evaluating candidate path");
570                    Some(PathWithMetrics::from((metrics, path)))
571                })
572                .take(share)
573                .collect::<Vec<_>>()
574                .into_iter()
575        })
576        .collect::<Vec<_>>();
577
578    // Sized by what the searches found: `take` is a configured bound, not a size.
579    let found = per_first.iter().map(|paths| paths.len()).sum::<usize>();
580    let mut spread = Vec::with_capacity(take.min(found));
581    while spread.len() < take {
582        let before = spread.len();
583        for paths in per_first.iter_mut() {
584            if spread.len() == take {
585                break;
586            }
587            spread.extend(paths.next());
588        }
589        if spread.len() == before {
590            break;
591        }
592    }
593    (spread, returned)
594}

596/// A lightweight graph-backed path selector.
597///
598/// Returns all candidate paths for a `(src, dest, hops)` query directly from
599/// the network graph — no caching is performed here.  The caller (typically
600/// [`crate::path::planner::PathPlanner`]) is responsible for caching, TTL management,
601/// background refresh, and final path selection.
602///
603/// Stores the planner's own identity (`me`) so that it can choose the
604/// appropriate cost function:
605/// - forward path (`src == me`): [`EdgeValueFn::forward`]
606/// - return path (`dest == me`): [`EdgeValueFn::returning`]

661    /// Extended forward path search: find shorter paths using
662    /// [`EdgeValueFn::forward_without_self_loopback`] and append `dest` to each one.
663    ///
664    /// This handles the case where the last edge (relay -> dest) has no graph edge
665    /// (e.g. no payment channel) but the path planner can still assume the last hop
666    /// is reachable. Paths already found by Phase 1 are excluded via `existing`.
667    ///
668    /// The cost from the shorter traversal is preserved as-is — the missing last
669    /// edge contributes a neutral `1.0` multiplier (no quality data available).
670    fn compute_extended_forward_paths(
671        &self,
672        src: &OffchainPublicKey,
673        dest: &OffchainPublicKey,
674        shorter_length: std::num::NonZeroUsize,
675        take: usize,
676        existing: &[PathWithMetrics],
677        // The caller's snapshot, so both phases of one query cost against the same face value.
678        ticket_face_value: Option<hopr_api::graph::traits::Balance>,
679    ) -> (Vec<PathWithMetrics>, usize) {
680        // `forward_without_self_loopback` counts the *finished* path length — the one that includes
681        // the `dest` edge appended below — not the shorter traversal depth. Passing the search depth
682        // understates it by one: for a 1-hop path it collapses `length` to 1, so the first
683        // (monetized) edge is funding-checked with zero remaining hops, waiving the check and
684        // selecting a channel that cannot fund the first-hop ticket (GNO-805). Search `shorter_length`
685        // edges, but cost them against the finished length.
686        let finished_length = shorter_length.saturating_add(1);
687        let value_fn = MetricsValueFn {
688            inner: EdgeValueFn::forward_without_self_loopback(
689                finished_length,
690                self.edge_penalty,
691                self.min_ack_rate,
692                ticket_face_value,
693            ),
694            ticket_face_value,
695        };
696        let raw = self
697            .graph
698            .simple_paths_from(src, shorter_length.get(), Some(take), value_fn);
699
700        let returned = raw.len();
701        let paths = raw
702            .into_iter()
703            .filter_map(|(path, _, metrics)| {
704                if metrics.cost <= 0.0 {
705                    return None;
706                }
707
708                // The graph crate already strips `src`; `path` is [intermediates…, terminator].
709                // Guard: if dest already appears as an intermediate or terminator, appending it
710                // would produce a non-adjacent duplicate that ValidatedPath::new rejects — skip early.
711                if path.contains(dest) {
712                    return None;
713                }
714                let mut candidate = path;
715                candidate.push(*dest);
716
717                // The missing last edge is assumed reachable only while the graph has no word on it. An
718                // edge it has measured dead (a score of exactly 0, which is how it reads a starved edge)
719                // is not missing, and adding the relay back would fill the floor with a path that
720                // delivers nothing, at a better cost than the healthy ones.
721                if let [.., relay, _] = candidate.as_slice()
722                    && self.graph.edge(relay, dest).and_then(|obs| obs.score()) == Some(0.0)
723                {
724                    return None;
725                }
726
727                if existing.iter().any(|pwm| pwm.path == candidate) {
728                    return None;
729                }
730
731                tracing::trace!(?candidate, cost = metrics.cost, "extended forward path candidate");
732                Some(PathWithMetrics::from((metrics, candidate)))
733            })
734            .take(take)
735            .collect();
736        (paths, returned)
737    }
738}

740impl<G> PathSelector for HoprGraphPathSelector<G>
741where
742    G: NetworkGraphTraverse<NodeId = OffchainPublicKey>
743        + NetworkGraphView<NodeId = OffchainPublicKey>
744        + Clone
745        + Send
746        + Sync
747        + 'static,
748    <G as NetworkGraphTraverse>::Observed: EdgeObservableRead + Send + 'static,
749{
750    /// Return all candidate paths from `src` to `dest` via `hops` relays.
751    ///
752    /// Each returned tuple contains a path `Vec<OffchainPublicKey>` of length
753    /// `hops + 1` (`[intermediates..., dest]`; `src` excluded) paired with its
754    /// accumulated traversal cost and per-path quality aggregates.
755    ///
756    /// Returns `Err(PathNotFound)` when the graph yields no positive-cost paths.
757    ///
758    /// The function has a potential to run expensive operations, it should be benchmarked
759    /// in a production environment and possibly guarded (e.g. by offloading the long execution
760    /// in an async executor to avoid blocking the caller).
761    #[tracing::instrument(
762        name = "selector.select_paths", level = "trace", skip_all,
763        fields(src = %src, dest = %dest, hops = hops, direction = if src == self.me { "forward" } else { "return" }),
764        ret, err(level = "trace")
765    )]
766    fn select_path(
767        &self,
768        src: OffchainPublicKey,
769        dest: OffchainPublicKey,
770        hops: usize,
771    ) -> Result<Vec<PathWithMetrics>> {
772        let direction = if src == self.me { "forward" } else { "return" };
773        tracing::trace!(%src, %dest, hops, direction, "computing paths from graph");
774
775        let length = std::num::NonZeroUsize::new(hops + 1)
776            .expect("can never fail, it is physically at least 1 after the addition");
777
778        // One read for the whole query. A concurrent `set_ticket_face_value` between two reads
779        // would let the traversal cost and the fundable-ticket floor come from different snapshots
780        // of the same graph, so `composite_weight` would rank a candidate against itself.
781        let ticket_face_value = self.graph.ticket_face_value();
782
783        let phase1;
784        let mut phase2 = None;
785        let paths = if src == self.me {
786            // Phase 1: search for full-length forward paths to dest.
787            let (mut found, returned) = compute_paths_spread(
788                &self.graph,
789                &src,
790                &dest,
791                length,
792                self.max_paths,
793                MetricsValueFn {
794                    inner: EdgeValueFn::forward(length, self.edge_penalty, self.min_ack_rate, ticket_face_value),
795                    ticket_face_value,
796                },
797            );
798            phase1 = PhaseCounts {
799                returned: returned as u64,
800                eligible: found.len() as u64,
801            };
802            tracing::trace!(
803                direction,
804                phase = 1,
805                count = found.len(),
806                "[forward] phase 1 candidates"
807            );
808
809            // Phase 2: if not enough paths, do an extended search with EdgeValueFn::forward_without_self_loopback
810            // for (length - 1) edges and assume the last hop can be done by anybody.
811            if found.len() < self.max_paths
812                && let Some(shorter) = std::num::NonZeroUsize::new(length.get() - 1)
813            {
814                let remaining = self.max_paths - found.len();
815                let (extended, returned) =
816                    self.compute_extended_forward_paths(&src, &dest, shorter, remaining, &found, ticket_face_value);
817                phase2 = Some(PhaseCounts {
818                    returned: returned as u64,
819                    eligible: extended.len() as u64,
820                });
821                tracing::trace!(
822                    direction,
823                    phase = 2,
824                    count = extended.len(),
825                    "[forward] phase 2 extended candidates"
826                );
827                found.extend(extended);
828            }
829
830            found
831        } else {
832            let (found, returned) = compute_paths_spread(
833                &self.graph,
834                &src,
835                &dest,
836                length,
837                self.max_paths,
838                MetricsValueFn {
839                    inner: EdgeValueFn::returning(length, self.edge_penalty, self.min_ack_rate, ticket_face_value),
840                    ticket_face_value,
841                },
842            );
843            phase1 = PhaseCounts {
844                returned: returned as u64,
845                eligible: found.len() as u64,
846            };
847            tracing::trace!(direction, count = found.len(), "[return] candidates");
848            found
849        };
850
851        for (i, pwm) in paths.iter().enumerate() {
852            tracing::trace!(
853                direction,
854                index = i,
855                path = ?pwm.path,
856                cost = pwm.cost,
857                total_latency_ms = ?pwm.total_latency_ms,
858                "[{direction}] candidate path"
859            );
860        }
861
862        let pruned = prune_for_consistency(paths, self.anonymity_floor, hops);
863        if let Some(observer) = &self.observer {
864            let (direction, destination) = if src == self.me {
865                (PathDirection::Forward, dest)
866            } else {
867                (PathDirection::Return, src)
868            };
869            observer(TransportEvent::PathSearched {
870                direction,
871                destination: hopr_api::types::internal::NodeId::Offchain(destination),
872                phase1,
873                phase2,
874                retained: pruned.len() as u64,
875            });
876        }
877        if pruned.is_empty() {
878            return Err(PathPlannerError::Path(PathError::PathNotFound(
879                hops,
880                src.to_string(),
881                dest.to_string(),
882            )));
883        }
884        if tracing::enabled!(tracing::Level::TRACE) {
885            let relayer_set = pruned
886                .iter()
887                .filter_map(|p| p.path.first())
888                .collect::<std::collections::HashSet<_>>();
889            tracing::trace!(
890                %src,
891                %dest,
892                hops,
893                direction,
894                survived = pruned.len(),
895                distinct_relayers = relayer_set.len(),
896                first_relayers = ?relayer_set,
897                "pruned candidate paths",
898            );
899        }
900        Ok(pruned)
901    }
902}

1188    async fn a_fast_dead_relay_should_not_keep_a_slot_after_pruning_for_zero_to_three_hops_in_both_directions()
1189    -> anyhow::Result<()> {
1190        const FLOOR: usize = 2;
1191        let me = pubkey(&[1; 32]);
1192        let dest = pubkey(&[2; 32]);
1193        let relay = |layer: usize, index: usize| pubkey(&[10 + (layer * 3 + index) as u8; 32]);
1194
1195        // 0 hops has no relay to judge: the direct route stays.
1196        for direction in ["forward", "return"] {
1197            let graph = ChannelGraph::new(me);
1198            graph.add_node(dest);
1199            let (src, dst) = if direction == "forward" { (me, dest) } else { (dest, me) };
1200            graph.add_edge(&src, &dst)?;
1201            mark_edge_full(&graph, &src, &dst);
1202            let paths = test_selector(me, graph, 50)
1203                .select_path(src, dst, 0)
1204                .context("direct path")?;
1205            assert_eq!(paths.len(), 1, "{direction}, 0 hops: the direct route must stay");
1206        }
1207
1208        let mut failures = Vec::new();
1209        for direction in ["forward", "return"] {
1210            for hops in 1..=3usize {
1211                for position in 1..=hops {
1212                    let graph = ChannelGraph::new(me);
1213                    // A node always knows the ticket price, and without it no path counts as fully measured.
1214                    graph.set_ticket_face_value(hopr_api::graph::traits::Balance::one());
1215                    graph.add_node(dest);
1216                    for layer in 1..=hops {
1217                        for index in 0..3 {
1218                            graph.add_node(relay(layer, index));
1219                        }
1220                    }
1221                    // Layer 0 is the sender, layer `hops + 1` the receiver.
1222                    let (sender, receiver) = if direction == "forward" { (me, dest) } else { (dest, me) };
1223                    let dead = relay(position, 0);
1224                    let node = |layer: usize, index: usize| match layer {
1225                        0 => sender,
1226                        l if l == hops + 1 => receiver,
1227                        l => relay(l, index),
1228                    };
1229                    for layer in 0..=hops {
1230                        for from in 0..if layer == 0 { 1 } else { 3 } {
1231                            for to in 0..if layer == hops { 1 } else { 3 } {
1232                                let (src, dst) = (node(layer, from), node(layer + 1, to));
1233                                graph.add_edge(&src, &dst)?;
1234                                if dst == dead {
1235                                    mark_edge_unlooped(&graph, &src, &dst);
1236                                } else {
1237                                    mark_edge_full(&graph, &src, &dst);
1238                                }
1239                            }
1240                        }
1241                    }
1242                    // The dead relay answers faster than the healthy ones, and none of its loops return.
1243                    for from in 0..if position == 1 { 1 } else { 3 } {
1244                        graph.upsert_edge(&node(position - 1, from), &dead, |obs| {
1245                            obs.record(EdgeWeightType::Immediate(Ok(Duration::from_millis(5))));
1246                            for _ in 0..5 {
1247                                obs.record(EdgeWeightType::Intermediate(Err(())));
1248                            }
1249                        });
1250                    }
1251
1252                    let selector = HoprGraphPathSelector::new(
1253                        me,
1254                        graph,
1255                        50,
1256                        PathPlannerConfig::default().edge_penalty,
1257                        PathPlannerConfig::default().min_ack_rate,
1258                        FLOOR,
1259                    );

1261                    let through_dead = kept.iter().filter(|pwm| pwm.path.contains(&dead)).count();
1262                    if kept.is_empty() || through_dead > 0 {
1263                        failures.push(format!(
1264                            "{direction}, {hops} hop(s), dead relay in position {position}: {through_dead} of {} kept \
1265                             paths run through the dead relay",
1266                            kept.len()
1267                        ));
1268                    }
1269                }
1270            }
1271        }
1272        assert!(failures.is_empty(), "{}", failures.join("\n"));
1273        Ok(())
1274    }

2411        OffchainPublicKey,
2412        Vec<OffchainPublicKey>,

2709    /// with nothing observed on it, forward, or with no ack rate, on return. Such a path is dropped
2710    /// and must not take the place of one of value. Every first relayer here reaches the destination
2711    /// over 6 probed relayers and 6 relayers whose last hop is like that, so half its continuations
2712    /// are of no value, and each share is as large as the other half.
2713    #[rstest::rstest]
2714    #[case::forward(true)]
2715    #[case::returning(false)]
2716    fn a_spread_pool_should_fill_each_share_with_paths_of_value(#[case] forward: bool) -> anyhow::Result<()> {
2717        let (me, dest, relays, graph) = mesh_graph(7, |_| 1000);
2718        // Ack rates where the value functions apply them, on the edge from me forward and on the
2719        // edge to me on return: without one, a zero penalty leaves every path of no value.
2720        for r in &relays {
2721            for (a, b) in [(me, *r), (*r, me)] {
2722                graph.upsert_edge(&a, &b, |obs| {
2723                    obs.record(EdgeWeightType::ImmediateProtocolConformance {
2724                        num_packets: 100,
2725                        num_acks: 100,
2726                    });
2727                });
2728            }
2729        }
2730        for _ in 0..6 {
2731            let unprobed = *OffchainKeypair::random().public();
2732            graph.add_node(unprobed);
2733            // Funded from every probed relayer, so these continuations are found.
2734            for r in &relays {
2735                graph.add_edge(r, &unprobed)?;
2736                graph.upsert_edge(r, &unprobed, |obs| {
2737                    obs.record(EdgeWeightType::Connected(true));
2738                    obs.record(EdgeWeightType::Balance(Some(hopr_api::graph::traits::Balance::from(
2739                        1000 * 100,
2740                    ))));
2741                });
2742            }
2743            // On to the destination with nothing observed, and to me connected with no ack rate.
2744            graph.add_edge(&unprobed, &dest)?;
2745            graph.add_edge(&unprobed, &me)?;
2746            graph.upsert_edge(&unprobed, &me, |obs| obs.record(EdgeWeightType::Connected(true)));
2747        }
2748        let (src, dst) = if forward { (me, dest) } else { (dest, me) };
2749        let length = std::num::NonZeroUsize::new(3).context("non-zero")?;
2750        let no_penalty = || {
2751            let (cfg, face) = (PathPlannerConfig::default(), graph.ticket_face_value());
2752            MetricsValueFn {
2753                inner: if forward {
2754                    EdgeValueFn::forward(length, 0.0, cfg.min_ack_rate, face)
2755                } else {
2756                    EdgeValueFn::returning(length, 0.0, cfg.min_ack_rate, face)
2757                },
2758                ticket_face_value: face,
2759            }
2760        };
2761        let take = 6 * relays.len();
2762        let (of_value, of_no_value): (Vec<_>, Vec<_>) = graph
2763            .simple_paths(&src, &dst, length.get(), None, no_penalty())
2764            .into_iter()
2765            .partition(|(.., metrics)| metrics.cost > 0.0);
2766        assert_eq!(
2767            take,
2768            of_value.len(),
2769            "the probed relayers carry 6 paths of value per first relayer"
2770        );
2771        assert_eq!(
2772            take,
2773            of_no_value.len(),
2774            "the unprobed relayers carry as many of no value"
2775        );
2776
2777        let (spread, returned) = compute_paths_spread(&graph, &src, &dst, length, take, no_penalty());
2778        assert!(spread.iter().all(|p| p.cost > 0.0), "a path of no value is dropped");
2779        assert_eq!(take, spread.len(), "every share filled with paths of value");
2780        assert!(returned > spread.len(), "diagnostics also count unusable continuations");
2781        Ok(())
2782    }
