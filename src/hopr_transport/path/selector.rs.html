20/// Accumulated path cost and quality aggregates, folded edge-by-edge during DFS.
21///
22/// `PartialOrd` / `PartialEq` compare only the `cost` field so the DFS
23/// pruning threshold (`min_cost`) operates on the same scalar as before.
24#[derive(Clone, Debug)]
25struct PathCostWithMetrics {
26    cost: f64,
27    total_latency_ms: Option<u32>,
28    min_probe_success_rate: Option<f64>,
29    min_ack_rate: Option<f64>,
30    fundable_tickets_floor: Option<u128>,
31}
32
33impl PartialOrd for PathCostWithMetrics {
34    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
35        self.cost.partial_cmp(&other.cost)
36    }
37}
38
39impl PartialEq for PathCostWithMetrics {
40    fn eq(&self, other: &Self) -> bool {
41        self.cost == other.cost
42    }
43}

58/// Returns the minimum of two `Option<T>` values, preferring `Some` over `None`.
59fn opt_min<T: PartialOrd>(a: Option<T>, b: Option<T>) -> Option<T> {
60    match (a, b) {
61        (Some(x), Some(y)) => Some(if x <= y { x } else { y }),
62        (x, y) => x.or(y),
63    }
64}

66/// Wraps an `EdgeValueFn<f64, W>` as a `ValueFn` whose `Value` type carries
67/// both the cost and per-path quality aggregates.
68///
69/// All cost semantics are delegated to the inner `EdgeValueFn`; the wrapper
70/// only adds the aggregate fold on the same `&Weight` reference already
71/// available during DFS, so no extra graph lookups are needed.
72struct MetricsValueFn<W: EdgeObservableRead> {
73    inner: EdgeValueFn<f64, W>,
74    /// Face value used to express each edge's balance as a count of single-hop tickets.
75    ///
76    /// Read once per query rather than stored on any edge: deriving a count at query time is
77    /// safe, whereas deriving it at edge-update time is what made a price change stale the
78    /// whole graph.
79    ticket_face_value: Option<hopr_api::graph::traits::Balance>,
80}
81
82impl<W> ValueFn for MetricsValueFn<W>
83where
84    W: EdgeObservableRead + Send + 'static,
85{
86    type Value = PathCostWithMetrics;
87    type Weight = W;
88
89    fn initial_value(&self) -> Self::Value {
90        PathCostWithMetrics {
91            cost: self.inner.initial_value(),
92            total_latency_ms: Some(0),
93            min_probe_success_rate: None,
94            min_ack_rate: None,
95            fundable_tickets_floor: None,
96        }
97    }
98
99    fn min_value(&self) -> Option<Self::Value> {
100        self.inner.min_value().map(|c| PathCostWithMetrics {
101            cost: c,
102            total_latency_ms: None,
103            min_probe_success_rate: None,
104            min_ack_rate: None,
105            fundable_tickets_floor: None,
106        })
107    }
108
109    fn into_value_fn(self) -> BasicValueFn<Self::Value, Self::Weight> {
110        let inner = self.inner.into_value_fn();
111        let ticket_face_value = self.ticket_face_value;
112        Arc::new(move |prev: PathCostWithMetrics, observed: &W, idx: usize| {
113            let cost = inner(prev.cost, observed, idx);
114
115            let edge_lat = observed
116                .immediate_qos()
117                .and_then(|m| m.average_latency())
118                .or_else(|| observed.intermediate_qos().and_then(|m| m.average_latency()));
119            let total_latency_ms = match (prev.total_latency_ms, edge_lat) {
120                (Some(acc), Some(lat)) => Some(((acc as u128 + lat.as_millis()).min(u32::MAX as u128)) as u32),
121                _ => None,
122            };
123
124            // Probe rate: taking the min of immediate and intermediate guards against nodes that
125            // look good on direct probes but degrade under multi-hop load.
126            // `and_then`, not `map`: a stream with no observations contributes nothing to the min
127            // rather than contributing a zero it never measured.
128            let edge_probe = observed
129                .immediate_qos()
130                .and_then(|m| m.average_probe_rate())
131                .into_iter()
132                .chain(observed.intermediate_qos().and_then(|m| m.average_probe_rate()))
133                .reduce(f64::min);
134            let min_probe_success_rate = opt_min(prev.min_probe_success_rate, edge_probe);
135
136            let edge_ack = observed.immediate_qos().and_then(|m| m.ack_rate());
137            let min_ack_rate = opt_min(prev.min_ack_rate, edge_ack);
138
139            // Expressed in single-hop tickets, not base units: the weighting factor below is a
140            // log ratio, and a base-unit balance would compress every realistic channel into a
141            // sliver of its output range.
142            //
143            // `zip`, not a fallback: an absent face value means the price is unknown, not that it
144            // is one. Dividing a base-unit balance by `1` yields a ticket count that saturates to
145            // `u128::MAX` and reads downstream as unlimited capacity — the opposite of what not
146            // knowing should imply. Contribute nothing to the floor until pricing arrives.
147            let edge_tickets = observed
148                .intermediate_qos()
149                .and_then(|m| m.balance())
150                .zip(ticket_face_value)
151                .map(|(balance, face_value)| {
152                    let tickets = if face_value.is_zero() {
153                        hopr_api::graph::traits::Balance::zero()
154                    } else {
155                        balance / face_value
156                    };
157                    if tickets > hopr_api::graph::traits::Balance::from(u128::MAX) {
158                        u128::MAX
159                    } else {
160                        tickets.low_u128()
161                    }
162                });
163            let fundable_tickets_floor = opt_min(prev.fundable_tickets_floor, edge_tickets);
164
165            PathCostWithMetrics {
166                cost,
167                total_latency_ms,
168                min_probe_success_rate,
169                min_ack_rate,
170                fundable_tickets_floor,
171            }
172        })
173    }
174}

176/// Trim the candidate set to lower median latency and minimise variance while
177/// preserving an anonymity floor.
178///
179/// Behaviour:
180/// - If `candidates.len() <= floor`, returns all candidates unchanged (`min(found_count, floor)` semantics — the floor
181///   is never a minimum to fabricate).
182/// - Sorts candidates with a known `total_latency_ms` ascending.
183/// - Drops from the high-latency tail until the total count equals `floor`, or until no populated candidates remain.
184/// - If still over the floor with no populated candidates left, drops unpopulated candidates from the input-order tail.
185///
186/// A path is "fully measured" — and therefore preferred over unmeasured alternatives —
187/// when `total_latency_ms` is known AND either `hops == 0` (direct path, no channel
188/// expected) OR `fundable_tickets_floor` is also known.  This prevents 0-hop direct paths from
189/// being demoted simply because they carry no channel-capacity data.
190pub fn prune_for_consistency(candidates: Vec<PathWithMetrics>, floor: usize, hops: usize) -> Vec<PathWithMetrics> {
191    // floor == 0 means "no pruning" — caller opts out entirely.
192    if floor == 0 || candidates.len() <= floor {
193        return candidates;
194    }
195
196    let fully_measured =
197        |p: &PathWithMetrics| p.total_latency_ms.is_some() && (hops == 0 || p.fundable_tickets_floor.is_some());
198
199    let (mut populated, unpopulated): (Vec<_>, Vec<_>) = candidates.into_iter().partition(|p| fully_measured(p));
200
201    // Sort populated ascending by latency (lowest first → drop from the end).
202    populated.sort_by_key(|p| p.total_latency_ms.unwrap_or(u32::MAX));
203
204    // Prefer measured paths: keep as many populated paths as fit within the floor,
205    // then fill the remaining slots with unpopulated paths.  This ensures that
206    // latency-measured candidates are never discarded when unprobed paths alone
207    // would satisfy the floor.
208    let target_populated = populated.len().min(floor);
209    let populated = take_relayer_diverse(populated, target_populated);
210    let remaining = floor - target_populated;
211
212    let mut result = populated;
213    result.extend(unpopulated.into_iter().take(remaining));
214
215    result
216}

218/// Takes `n` candidates from `sorted`, preferring not-yet-represented first relayers.
219///
220/// Plain truncation keeps the `n` lowest-latency paths, which in a well-connected network are often
221/// the *same* few relayers reached by different routes; every downstream draw then inherits that
222/// concentration. Relative order is preserved, so each pass stays latency-ordered.
223fn take_relayer_diverse(sorted: Vec<PathWithMetrics>, n: usize) -> Vec<PathWithMetrics> {
224    if sorted.len() <= n {
225        return sorted;
226    }
227
228    let mut seen: Vec<OffchainPublicKey> = Vec::with_capacity(n);
229    let mut taken = vec![false; sorted.len()];
230    let mut count = 0;
231
232    // First pass: one path per distinct first relayer, best (lowest-latency) first.
233    for (i, p) in sorted.iter().enumerate() {
234        if count == n {
235            break;
236        }
237        let Some(first) = p.path.first() else { continue };
238        if !seen.contains(first) {
239            seen.push(*first);
240            taken[i] = true;
241            count += 1;
242        }
243    }
244
245    // Second pass: fill any remaining slots with the best of the rest.
246    for slot in taken.iter_mut() {
247        if count == n {
248            break;
249        }
250        if !*slot {
251            *slot = true;
252            count += 1;
253        }
254    }
255
256    sorted
257        .into_iter()
258        .zip(taken)
259        .filter_map(|(p, keep)| keep.then_some(p))
260        .collect()
261}

263/// Compute candidate paths from `src` to `dest` through `graph`.
264///
265/// `length` is the number of edges to traverse (= intermediate hops + 1).
266/// `take` caps the number of candidate paths returned.
267/// The graph crate returns only the intermediate nodes (both `src` and `dest` stripped);
268/// this function re-appends `dest` so callers receive `([intermediates..., dest], cost)`.
269fn compute_paths<G, W>(
270    graph: &G,
271    src: &OffchainPublicKey,
272    dest: &OffchainPublicKey,
273    length: std::num::NonZeroUsize,
274    take: usize,
275    value_fn: MetricsValueFn<W>,
276) -> Vec<PathWithMetrics>
277where
278    G: NetworkGraphTraverse<NodeId = OffchainPublicKey, Observed = W>,
279    W: EdgeObservableRead + Send + 'static,
280{
281    let raw = graph.simple_paths(src, dest, length.get(), Some(take), value_fn);
282
283    raw.into_iter()
284        .filter_map(|(path, _, metrics)| {
285            tracing::trace!(?path, cost = metrics.cost, "evaluating candidate path");
286            if metrics.cost > 0.0 {
287                let mut path = path;
288                path.push(*dest);
289                Some(PathWithMetrics::from((metrics, path)))
290            } else {
291                None
292            }
293        })
294        .collect()
295}

297/// A lightweight graph-backed path selector.
298///
299/// Returns all candidate paths for a `(src, dest, hops)` query directly from
300/// the network graph — no caching is performed here.  The caller (typically
301/// [`crate::path::planner::PathPlanner`]) is responsible for caching, TTL management,
302/// background refresh, and final path selection.
303///
304/// Stores the planner's own identity (`me`) so that it can choose the
305/// appropriate cost function:
306/// - forward path (`src == me`): [`EdgeValueFn::forward`]
307/// - return path (`dest == me`): [`EdgeValueFn::returning`]
308#[derive(Clone)]
309pub struct HoprGraphPathSelector<G> {
310    me: OffchainPublicKey,
311    graph: G,
312    max_paths: usize,
313    edge_penalty: f64,
314    min_ack_rate: f64,
315    anonymity_floor: usize,
316}
317
318impl<G> HoprGraphPathSelector<G>
319where
320    G: NetworkGraphTraverse<NodeId = OffchainPublicKey>
321        + NetworkGraphView<NodeId = OffchainPublicKey>
322        + Clone
323        + Send
324        + Sync
325        + 'static,
326    <G as NetworkGraphTraverse>::Observed: EdgeObservableRead + Send + 'static,
327{
328    /// Create a new selector.
329    ///
330    /// * `me` – the planner's own offchain public key, used to determine path direction.
331    /// * `graph` – the network graph to query.
332    /// * `max_paths` – maximum number of candidate paths to return per query.
333    /// * `edge_penalty` – penalty multiplier for edges lacking probe-based quality observations.
334    /// * `min_ack_rate` – minimum acceptable message acknowledgment rate for path selection.
335    /// * `anonymity_floor` – minimum candidate count below which no latency-based pruning occurs.
336    pub fn new(
337        me: OffchainPublicKey,
338        graph: G,
339        max_paths: usize,
340        edge_penalty: f64,
341        min_ack_rate: f64,
342        anonymity_floor: usize,
343    ) -> Self {
344        Self {
345            me,
346            graph,
347            max_paths,
348            edge_penalty,
349            min_ack_rate,
350            anonymity_floor,
351        }
352    }
353
354    /// Extended forward path search: find shorter paths using
355    /// [`EdgeValueFn::forward_without_self_loopback`] and append `dest` to each one.
356    ///
357    /// This handles the case where the last edge (relay -> dest) has no graph edge
358    /// (e.g. no payment channel) but the path planner can still assume the last hop
359    /// is reachable. Paths already found by Phase 1 are excluded via `existing`.
360    ///
361    /// The cost from the shorter traversal is preserved as-is — the missing last
362    /// edge contributes a neutral `1.0` multiplier (no quality data available).
363    fn compute_extended_forward_paths(
364        &self,
365        src: &OffchainPublicKey,
366        dest: &OffchainPublicKey,
367        shorter_length: std::num::NonZeroUsize,
368        take: usize,
369        existing: &[PathWithMetrics],
370        // The caller's snapshot, so both phases of one query cost against the same face value.
371        ticket_face_value: Option<hopr_api::graph::traits::Balance>,
372    ) -> Vec<PathWithMetrics> {
373        let value_fn = MetricsValueFn {
374            inner: EdgeValueFn::forward_without_self_loopback(
375                shorter_length,
376                self.edge_penalty,
377                self.min_ack_rate,
378                ticket_face_value,
379            ),
380            ticket_face_value,
381        };
382        let raw = self
383            .graph
384            .simple_paths_from(src, shorter_length.get(), Some(take), value_fn);
385
386        raw.into_iter()
387            .filter_map(|(path, _, metrics)| {
388                if metrics.cost <= 0.0 {
389                    return None;
390                }
391
392                // The graph crate already strips `src`; `path` is [intermediates…, terminator].
393                // Guard: if dest already appears as an intermediate or terminator, appending it
394                // would produce a non-adjacent duplicate that ValidatedPath::new rejects — skip early.
395                if path.contains(dest) {
396                    return None;
397                }
398                let mut candidate = path;
399                candidate.push(*dest);
400
401                if existing.iter().any(|pwm| pwm.path == candidate) {
402                    return None;
403                }
404
405                tracing::trace!(?candidate, cost = metrics.cost, "extended forward path candidate");
406                Some(PathWithMetrics::from((metrics, candidate)))
407            })
408            .take(take)
409            .collect()
410    }
411}

413impl<G> PathSelector for HoprGraphPathSelector<G>
414where
415    G: NetworkGraphTraverse<NodeId = OffchainPublicKey>
416        + NetworkGraphView<NodeId = OffchainPublicKey>
417        + Clone
418        + Send
419        + Sync
420        + 'static,
421    <G as NetworkGraphTraverse>::Observed: EdgeObservableRead + Send + 'static,
422{
423    /// Return all candidate paths from `src` to `dest` via `hops` relays.
424    ///
425    /// Each returned tuple contains a path `Vec<OffchainPublicKey>` of length
426    /// `hops + 1` (`[intermediates..., dest]`; `src` excluded) paired with its
427    /// accumulated traversal cost and per-path quality aggregates.
428    ///
429    /// Returns `Err(PathNotFound)` when the graph yields no positive-cost paths.
430    ///
431    /// The function has a potential to run expensive operations, it should be benchmarked
432    /// in a production environment and possibly guarded (e.g. by offloading the long execution
433    /// in an async executor to avoid blocking the caller).
434    #[tracing::instrument(level = "trace", skip(self), fields(src = %src, dest = %dest, hops), ret, err)]
435    fn select_path(
436        &self,
437        src: OffchainPublicKey,
438        dest: OffchainPublicKey,
439        hops: usize,
440    ) -> Result<Vec<PathWithMetrics>> {
441        let direction = if src == self.me { "forward" } else { "return" };
442        tracing::debug!(%src, %dest, hops, direction, "computing paths from graph");
443
444        let length = std::num::NonZeroUsize::new(hops + 1)
445            .expect("can never fail, it is physically at least 1 after the addition");
446
447        // One read for the whole query. A concurrent `set_ticket_face_value` between two reads
448        // would let the traversal cost and the fundable-ticket floor come from different snapshots
449        // of the same graph, so `composite_weight` would rank a candidate against itself.
450        let ticket_face_value = self.graph.ticket_face_value();
451
452        let paths = if src == self.me {
453            // Phase 1: search for full-length forward paths to dest.
454            let mut found = compute_paths(
455                &self.graph,
456                &src,
457                &dest,
458                length,
459                self.max_paths,
460                MetricsValueFn {
461                    inner: EdgeValueFn::forward(length, self.edge_penalty, self.min_ack_rate, ticket_face_value),
462                    ticket_face_value,
463                },
464            );
465            tracing::debug!(
466                direction,
467                phase = 1,
468                count = found.len(),
469                "[forward] phase 1 candidates"
470            );
471
472            // Phase 2: if not enough paths, do an extended search with EdgeValueFn::forward_without_self_loopback
473            // for (length - 1) edges and assume the last hop can be done by anybody.
474            if found.len() < self.max_paths
475                && let Some(shorter) = std::num::NonZeroUsize::new(length.get() - 1)
476            {
477                let remaining = self.max_paths - found.len();
478                let extended =
479                    self.compute_extended_forward_paths(&src, &dest, shorter, remaining, &found, ticket_face_value);
480                tracing::debug!(
481                    direction,
482                    phase = 2,
483                    count = extended.len(),
484                    "[forward] phase 2 extended candidates"
485                );
486                found.extend(extended);
487            }
488
489            found
490        } else {
491            let found = compute_paths(
492                &self.graph,
493                &src,
494                &dest,
495                length,
496                self.max_paths,
497                MetricsValueFn {
498                    inner: EdgeValueFn::returning(length, self.edge_penalty, self.min_ack_rate, ticket_face_value),
499                    ticket_face_value,
500                },
501            );
502            tracing::debug!(direction, count = found.len(), "[return] candidates");
503            found
504        };
505
506        for (i, pwm) in paths.iter().enumerate() {
507            tracing::debug!(
508                direction,
509                index = i,
510                path = ?pwm.path,
511                cost = pwm.cost,
512                total_latency_ms = ?pwm.total_latency_ms,
513                "[{direction}] candidate path"
514            );
515        }
516
517        if paths.is_empty() {
518            Err(PathPlannerError::Path(PathError::PathNotFound(
519                hops,
520                src.to_string(),
521                dest.to_string(),
522            )))
523        } else {
524            Ok(prune_for_consistency(paths, self.anonymity_floor, hops))
525        }
526    }
527}
