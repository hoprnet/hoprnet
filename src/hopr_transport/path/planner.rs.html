35/// Rejects a temper exponent outside `(0, 1]`.
36///
37/// `w^γ` is only a flattening of the weights for `γ ∈ (0, 1]`.
38///
39/// `γ > 1` would sharpen instead — concentrating harder than raw path value, the opposite of the
40/// knob's purpose — and `γ <= 0` inverts or annihilates the ordering, sending traffic *preferentially*
41/// to the worst relays. Both are almost certainly typos, so refuse them rather than silently
42/// degrade every return path the node builds.
43fn validate_weight_temper(temper: f64) -> std::result::Result<(), ValidationError> {
44    if temper > 0.0 && temper <= 1.0 {
45        return Ok(());
46    }
47    let mut err = ValidationError::new("weight_temper_out_of_range");
48    err.message = Some(
49        format!(
50            "return_path_weight_temper ({temper}) must be in (0, 1]: 1.0 samples by raw path value, smaller values \
51             flatten towards uniform"
52        )
53        .into(),
54    );
55    Err(err)
56}

59#[derive(Debug, Clone, Copy, PartialEq, smart_default::SmartDefault, Validate)]

66pub struct PathPlannerConfig {
67    /// Maximum number of `(source, destination, options)` entries in the path cache.
68    #[default = 10_000]
69    pub max_cache_capacity: u64,
70    /// Time-to-live for a cached path list.  When an entry expires the next
71    /// [`PathPlanner::resolve_routing`] call transparently recomputes it (lazy refresh).
72    ///
73    /// Candidate weights are computed once, when the entry is filled, and frozen in the
74    /// [`hopr_utils::statistics::WeightedCollection`] -- so this bounds how stale the *weights* a
75    /// live session draws from can be, not merely how stale the candidate set is. At the previous
76    /// 60 s a relay that stopped delivering kept its full share of return-path draws for a minute
77    /// after the graph had already scored it down.
78    #[default(Duration::from_secs(10))]
79    #[cfg_attr(feature = "serde", serde(with = "humantime_serde"))]
80    pub cache_ttl: Duration,
81    /// Period between proactive background cache-refresh sweeps.
82    ///
83    /// Held at half the TTL so a steady-state session is normally served from an entry that was
84    /// re-weighted rather than one that expired under it.
85    #[default(Duration::from_secs(5))]
86    #[cfg_attr(feature = "serde", serde(with = "humantime_serde"))]
87    #[validate(custom(function = "validate_refresh_period"))]
88    pub refresh_period: Duration,
89    /// Maximum number of candidate paths the selector may return per query.
90    /// All returned candidates are validated and cached.
91    /// Must be at least 1: a zero limit leaves the selector unable to return any multi-hop path.
92    #[default = 50]
93    #[validate(range(min = 1))]
94    pub max_cached_paths: usize,
95    /// Penalty multiplier for edges lacking probe-based quality observations.
96    /// Applied during path cost evaluation to down-weight unprobed edges.
97    /// Must be finite and in `0.0..=1.0`.
98    #[default = 0.5]
99    #[validate(custom(function = "validate_unit_interval"))]
100    pub edge_penalty: f64,
101    /// Minimum acceptable message acknowledgment rate for path selection.
102    /// Edges with an ack rate below this threshold are excluded from candidate paths.
103    /// Must be finite and in `0.0..=1.0`.
104    #[default = 0.1]
105    #[validate(custom(function = "validate_unit_interval"))]
106    pub min_ack_rate: f64,
107    /// Candidate count below which no latency-based pruning occurs.
108    ///
109    /// When fewer paths than this value are found, the selector returns all of them
110    /// unchanged (`min(found_count, floor)` semantics — the floor is never a minimum
111    /// to fabricate).  Set to 0 to disable pruning entirely.
112    #[default = 8]
113    pub min_paths_anonymity_floor: usize,
114    /// Total path latency at which the latency factor in the composite weight equals 0.5.
115    /// Higher values make the weight less sensitive to latency differences.
116    #[default(Duration::from_millis(100))]
117    #[cfg_attr(feature = "serde", serde(with = "humantime_serde"))]
118    pub latency_halflife: Duration,
119    /// Reference channel balance used to scale the capacity factor in the composite weight.
120    /// `capacity_factor` saturates at 1.0 near this value.
121    /// Defaults to 10_000_000 (~10 MiB in wxHOPR tokens).
122    #[default = 10_000_000]
123    pub capacity_reference: u128,
124    /// Exponent applied to path weights before sampling, flattening the distribution. Despite the
125    /// name it applies to forward draws as well as to return draws (the name is kept for
126    /// configuration compatibility).
127    ///
128    /// Paths are drawn weighted-random by path value, which concentrates a session's SURBs, and its
129    /// forward packets, on the few highest-valued relays — losing one then costs far more than the
130    /// reliable-mode loss tolerance, and a forward path that collapses on one relayer goes dark
131    /// with it. Raising each weight to `γ ∈ (0, 1]` compresses the spread between good and bad
132    /// candidates without changing their order: `w' = w^γ`.
133    ///
134    /// `1.0` samples by raw path value (most traffic on the best relays, largest blast radius when
135    /// one dies). Values approaching `0.0` tend to a uniform draw (smallest blast radius, most
136    /// traffic on poor relays). With weights `(0.4, 0.3, 0.2, 0.1)`, `γ = 0.5` moves the busiest
137    /// relay's share from 40% to 33% and the ratio between busiest and least-busy from 4.0 to 2.0.
138    ///
139    /// Defaults to 0.5.
140    #[validate(custom(function = "validate_weight_temper"))]
141    #[default = 0.5]
142    pub return_path_weight_temper: f64,
143    /// Fraction of path draws, forward and return, made uniformly at random instead of by weight
144    /// (the name is kept for configuration compatibility).
145    ///
146    /// Weights come from observations, and observations only exist for paths that get selected — a
147    /// closed loop in which a path that falls out of favour stops being measured, so its score can
148    /// never recover and it is never chosen again. Spending a small share of draws uniformly keeps
149    /// the plausible candidates under observation, which is what lets a recovered one climb back.
150    ///
151    /// Plausible means a tempered weight of at least a tenth of the best candidate's. That leaves out
152    /// a relayer on probation and an edge probes measured dead, and also a healthy path slow or thin
153    /// enough to weigh less than a tenth. An exploring return draw takes one of those only within a
154    /// node-wide allowance of one a second, while an ordinary return draw keeps it at its tempered
155    /// weight: a reply through a relayer on probation is what revives it. A forward draw, exploring or
156    /// not, never takes one, as a forward packet probes nothing.
157    ///
158    /// Costs throughput in proportion: this share of draws, forward and return, deliberately ignores
159    /// which plausible candidate looks best. Candidates have already passed the selector's gates
160    /// (open channels, `min_ack_rate`, and so on) before reaching here, so an exploratory draw never
161    /// takes a route the cost function rejected. `0.0` disables it. Defaults to 0.1.
162    #[validate(custom(function = "validate_unit_interval"))]
163    #[default = 0.1]
164    pub return_path_exploration: f64,
165    /// Upper bound on a loopback probe's round-trip time considered plausible.
166    /// Measurements above this cap (clock skew, stale telemetry) are discarded
167    /// instead of poisoning the latency EMA with an absurd value.
168    #[default(Duration::from_secs(30))]
169    #[cfg_attr(feature = "serde", serde(with = "humantime_serde"))]
170    pub max_plausible_loopback_rtt: Duration,
171}

173/// Shortest accepted `refresh_period`. A zero period makes the sweep loop rebuild every cached
174/// multi-hop path back to back, keeping the selector and resolver continuously busy.
175const MIN_REFRESH_PERIOD: Duration = Duration::from_secs(1);
176
177fn validate_refresh_period(period: &Duration) -> std::result::Result<(), ValidationError> {
178    if *period >= MIN_REFRESH_PERIOD {
179        return Ok(());
180    }
181    let mut err = ValidationError::new("refresh_period_too_short");
182    err.message = Some(format!("refresh_period ({period:?}) must be at least {MIN_REFRESH_PERIOD:?}").into());
183    Err(err)
184}

186/// A sweep that runs less often than entries expire never reaches them: the sweep walks live entries
187/// only, so an expired one is refilled on demand, and a refill that finds no paths is `PathNotFound`
188/// where a refreshed entry would have kept its previous weights.
189fn validate_refresh_within_ttl(cfg: &PathPlannerConfig) -> std::result::Result<(), ValidationError> {
190    if cfg.refresh_period < cfg.cache_ttl {
191        return Ok(());
192    }
193    let mut err = ValidationError::new("refresh_period_not_below_cache_ttl");
194    err.message = Some(
195        format!(
196            "refresh_period ({:?}) must be shorter than cache_ttl ({:?})",
197            cfg.refresh_period, cfg.cache_ttl
198        )
199        .into(),
200    );
201    Err(err)
202}

204fn validate_unit_interval(value: f64) -> std::result::Result<(), ValidationError> {
205    if value.is_finite() && (0.0..=1.0).contains(&value) {
206        Ok(())
207    } else {
208        Err(ValidationError::new("value must be finite and in 0.0..=1.0"))
209    }

221/// Continuous, monotonically decreasing latency factor, bounded in (0, 1].
222///
223/// Returns 1.0 for zero latency and 0.5 when `latency == halflife`.
224fn latency_factor(latency: Duration, halflife: Duration) -> f64 {
225    let ms = latency.as_millis() as f64;
226    let h = halflife.as_millis().max(1) as f64;
227    1.0 / (1.0 + ms / h)
228}
229
230/// Continuous, monotonically increasing capacity factor, bounded in (0.05, 1.0].
231///
232/// Uses a log scale because channel balances span many orders of magnitude.
233/// Saturates at 1.0 near `reference`.
234fn capacity_factor(c: u128, reference: u128) -> f64 {
235    let log = (c as f64).max(1.0).log10();
236    let ref_log = (reference as f64).max(10.0).log10();
237    (log / ref_log).clamp(0.05, 1.0)
238}

240/// Composite selection weight for a candidate path.
241///
242/// Refines `pwc.cost` (the `EdgeValueFn` output) with latency and capacity factors
243/// derived from the per-path aggregates.  Factors are neutral (1.0) when the
244/// corresponding aggregate is unavailable to avoid penalising unprobed paths.
245/// For 0-hop routes (`hops == 0`) the capacity factor is always 1.0 — direct
246/// `me -> dest` packets use no payment channel, so `fundable_tickets_floor = None` is expected.
247fn composite_weight(pwc: &PathWithMetrics, hops: usize, params: WeightingParams) -> f64 {
248    let lat = pwc
249        .total_latency_ms
250        .map(|ms| latency_factor(Duration::from_millis(ms as u64), params.latency_halflife))
251        .unwrap_or(1.0);
252    let cap = if hops == 0 {
253        1.0
254    } else {
255        pwc.fundable_tickets_floor
256            .map(|c| capacity_factor(c, params.capacity_reference))
257            .unwrap_or(1.0)
258    };
259    pwc.cost * lat * cap
260}

262/// The weight a candidate is drawn by, as the walk valued it, tempered to spread the draws over
263/// relayers (see [`PathPlannerConfig::return_path_weight_temper`]), relayer health included.
264///
265/// Forward and return paths alike: an untempered forward draw put most of a session's packets on
266/// its single best path, and with the candidate prune diversifying first relayers only, on one
267/// middle relayer, whose death then took the whole forward direction with it (see
268/// `selector::MIDDLE_HOP_SHARE`).
269fn selection_weight(pwc: &PathWithMetrics, hops: usize, params: WeightingParams) -> f64 {
270    temper_weights(&[composite_weight(pwc, hops, params)], params.return_path_weight_temper)[0]
271}

273/// Picks an index into `weights` with probability proportional to the weight; `None` if all are
274/// non-positive.
275///
276/// Mirrors `WeightedCollection::pick_index`, which cannot be reused because the callers below
277/// select over *subsets* of a single collection.
278fn pick_weighted_index(weights: &[f64]) -> Option<usize> {
279    let total: f64 = weights.iter().map(|w| w.max(0.0)).sum();
280    if total <= 0.0 {
281        return None;
282    }
283
284    // Path selection is privacy-relevant, so draw from the CSPRNG rather than a thread RNG.
285    let r = hopr_api::types::crypto_random::random_float_in_range(0.0..total);
286    let mut cumulative = 0.0;
287    for (i, w) in weights.iter().enumerate() {
288        cumulative += w.max(0.0);
289        if r < cumulative {
290            return Some(i);
291        }
292    }
293
294    // Floating-point edge case: fall back to the last positive-weight entry.
295    weights.iter().rposition(|w| *w > 0.0)
296}

298/// Whether this draw should explore — ignore the weights and pick uniformly among the plausible
299/// candidates (see [`pick_exploring_index`]).
300///
301/// Weights are derived from observations, and observations only exist for paths that were selected.
302/// Left alone that is a closed loop: a path that falls out of favour stops being measured, so its
303/// score can never recover and it is never chosen again. Spending a small share of draws uniformly
304/// keeps the plausible candidates under observation, which is what lets one that has recovered climb
305/// back. An exploring draw takes a candidate below [`PLAUSIBLE_WEIGHT_FRACTION`] only as a return draw
306/// the [`ProbeBudget`] grants.
307fn should_explore(exploration: f64) -> bool {
308    exploration > 0.0 && hopr_api::types::crypto_random::random_float_in_range(0.0..1.0) < exploration
309}

311/// Picks a uniformly random index over `len` entries.
312fn pick_uniform_index(len: usize) -> Option<usize> {
313    (len > 0).then(|| (hopr_api::types::crypto_random::random_float_in_range(0.0..len as f64) as usize).min(len - 1))
314}

316/// The fraction of the best candidate's (tempered) weight below which a candidate is one the cost
317/// function all but rejects.
318///
319/// A relayer on probation is published at a health of a thousandth and an edge probes measured dead
320/// is floored at a billionth: tempered, they sit at about 3 % and 0.003 % of a healthy candidate. A
321/// stalled relayer still delivering at a hundredth of its health is at 10 %, and stays explorable.
322///
323/// Those figures are for the default `return_path_weight_temper` of 0.5. A weight tempered by `t`
324/// puts a relayer on probation at `0.001^t` of a healthy one, so a temper of 1/3 or less lifts it to
325/// this fraction or above: exploration then counts it as plausible, forward draws among them, and
326/// the probe budget no longer holds it back. Whoever lowers the temper that far trades this gate for
327/// the wider spread.
328const PLAUSIBLE_WEIGHT_FRACTION: f64 = 0.1;
329
330/// How often an exploring return draw may pick a candidate below [`PLAUSIBLE_WEIGHT_FRACTION`],
331/// node-wide.
332///
333/// That draw is the probe a relayer on probation is revived by, so it cannot be ruled out, but every
334/// one through a relayer that is really gone is a SURB lost at the counterparty — its PIX share with
335/// it — and each one is written off and replaced. Uniform exploration made a tenth of all draws such
336/// a probe whenever one candidate was implausible: 26 per second at production dimensions, and with
337/// them a few percent of the counterparty's stock dead, enough for a single lost control message to
338/// close a PIX Session. One a second is enough to notice a revival.
339const IMPLAUSIBLE_PROBE_INTERVAL: Duration = Duration::from_secs(1);
340
341/// The node-wide allowance of [`IMPLAUSIBLE_PROBE_INTERVAL`].
342#[derive(Debug)]
343struct ProbeBudget {
344    epoch: std::time::Instant,
345    /// Milliseconds since `epoch` of the last probe granted, plus one so that `0` is "none yet".
346    last_ms_plus_one: std::sync::atomic::AtomicU64,
347}
348
349impl Default for ProbeBudget {
350    fn default() -> Self {
351        Self {
352            epoch: std::time::Instant::now(),
353            last_ms_plus_one: std::sync::atomic::AtomicU64::new(0),
354        }
355    }
356}
357
358impl ProbeBudget {
359    /// Grants a probe if none was granted within [`IMPLAUSIBLE_PROBE_INTERVAL`].
360    fn take(&self) -> bool {
361        let now = u64::try_from(self.epoch.elapsed().as_millis())
362            .unwrap_or(u64::MAX - 1)
363            .saturating_add(1);
364        let interval = u64::try_from(IMPLAUSIBLE_PROBE_INTERVAL.as_millis()).unwrap_or(u64::MAX);
365        self.last_ms_plus_one
366            .fetch_update(
367                std::sync::atomic::Ordering::Relaxed,
368                std::sync::atomic::Ordering::Relaxed,
369                |last| (last == 0 || now.saturating_sub(last) >= interval).then_some(now),
370            )
371            .is_ok()
372    }
373}

375/// An exploring draw: uniform over the candidates whose weight is at least
376/// [`PLAUSIBLE_WEIGHT_FRACTION`] of the best, and over the others only when `allow_implausible`
377/// grants it (it is asked only when the draw lands on one).
378///
379/// Uniform over the plausible ones whatever is granted: a draw that lands on an implausible candidate
380/// it may not take is redrawn among the plausible ones.
381fn pick_exploring_index(weights: &[f64], allow_implausible: impl FnOnce() -> bool) -> Option<usize> {
382    let best = weights.iter().copied().fold(0.0, f64::max);
383    if best <= 0.0 {
384        return None;
385    }
386    let floor = best * PLAUSIBLE_WEIGHT_FRACTION;
387    let drawn = pick_uniform_index(weights.len())?;
388    if weights[drawn] >= floor || allow_implausible() {
389        return Some(drawn);
390    }
391    let plausible: Vec<usize> = (0..weights.len()).filter(|&i| weights[i] >= floor).collect();
392    pick_uniform_index(plausible.len()).map(|i| plausible[i])
393}

395/// Zeroes every weight below [`PLAUSIBLE_WEIGHT_FRACTION`] of the best, so that a weighted draw over
396/// `weights` never lands on a candidate the weights all but reject.
397///
398/// For ordinary forward draws. A return draw keeps such a candidate at its tempered weight, because a
399/// reply through it is what can end a relayer's probation. A forward packet probes nothing, so on such
400/// a candidate all it can do is be lost.
401fn gate_implausible(weights: &mut [f64]) {
402    let floor = weights.iter().copied().fold(0.0, f64::max) * PLAUSIBLE_WEIGHT_FRACTION;
403    for weight in weights.iter_mut().filter(|weight| **weight < floor) {
404        *weight = 0.0;
405    }
406}

408/// Flattens `weights` by raising each to `temper`, compressing the spread between good and bad
409/// candidates without reordering them.
410///
411/// `x^γ` is monotone for `γ > 0`, so the best candidate stays the best — only the *ratio* between
412/// them shrinks, which is what bounds how much of a session rides on any single relayer. Weights
413/// are clamped at zero first: a negative weight would make `powf` return NaN and poison the draw.
414fn temper_weights(weights: &[f64], temper: f64) -> Vec<f64> {
415    weights.iter().map(|w| w.max(0.0).powf(temper)).collect()
416}

431/// Records a diversity transition and logs it once: a WARN when a key first collapses to a single
432/// relayer, an INFO when it first recovers. Idempotent across repeated rebuilds of the same state,
433/// so the WARN fires on the transition, not on every rebuild.
434fn apply_diversity_transition(
435    collapsed_diversity: &Mutex<HashSet<PlannerCacheKey>>,
436    src_key: &OffchainPublicKey,
437    kind: &str,
438    obs: &DiversityObservation,
439) {
440    let mut collapsed = collapsed_diversity.lock();
441    if obs.collapsed {
442        if collapsed.insert(obs.key) {
443            tracing::warn!(
444                src = %src_key,
445                kind,
446                distinct_relayers = obs.distinct_relayers,
447                candidates = obs.candidates,
448                "return-path relayer diversity collapsed to a single relayer",
449            );
450        }
451    } else if collapsed.remove(&obs.key) {
452        tracing::info!(
453            src = %src_key,
454            kind,
455            distinct_relayers = obs.distinct_relayers,
456            candidates = obs.candidates,
457            "return-path relayer diversity recovered for this destination",
458        );
459    }
460}

467/// What a cache entry is rebuilt from, shared by the planner and its background sweep.
468struct PathBuilder<R, S> {
469    resolver: R,
470    selector: S,
471    weighting: WeightingParams,
472    me: OffchainPublicKey,
473    /// Keys currently in the collapsed return-path diversity state, so the WARN fires once on the
474    /// transition into it (and an INFO once on recovery) rather than on every rebuild (#8502).
475    collapsed_diversity: Arc<Mutex<HashSet<PlannerCacheKey>>>,
476}
477
478impl<R, S> PathBuilder<R, S>
479where
480    R: ChainKeyOperations + ChainReadChannelOperations + Send + Sync,
481    S: PathSelector,
482{
483    /// Rebuilds the weighted candidate collection for one `(source, destination, hops)` triple from
484    /// whatever the graph currently says.
485    ///
486    /// Shared by the lazy cache fill, the background sweep and the on-demand recompute, so all three
487    /// necessarily agree: a divergence here would make a session's weights depend on which of the three
488    /// happened to run last.
489    ///
490    /// `Ok(None)` means the selector offered nothing, or nothing survived validation. Callers decide
491    /// what that means — a fill turns it into `PathNotFound`, a refresh drops the existing entry.
492    /// `Err` is reserved for a selector that actually failed.
493    async fn rebuild(
494        &self,
495        src_key: OffchainPublicKey,
496        dest_key: OffchainPublicKey,
497        hops: usize,
498        kind: &'static str,
499    ) -> Result<Option<RebuiltPaths>> {
500        let (resolver, weighting, me) = (&self.resolver, self.weighting, self.me);
501        // Dead relayers are pruned by the traversal itself, and stalled ones cost less there: the SURB
502        // telemetry records their health in the graph.
503        let candidates = self.selector.select_path(src_key, dest_key, hops)?;
504
505        let chain_resolver = ChainPathResolver::from(resolver);
506        let mut valid_paths: Vec<(ValidatedPath, f64)> = Vec::with_capacity(candidates.len());
507        let mut path_metrics: Vec<PathWithMetrics> = Vec::with_capacity(candidates.len());
508        for pwc in candidates {
509            // Cloned, not taken: `distinct_first_relayers` below needs the path intact (#8502).
510            let node_ids: Vec<NodeId> = pwc.path.iter().cloned().map(NodeId::Offchain).collect::<Vec<_>>();
511            match ValidatedPath::new(NodeId::Offchain(src_key), node_ids, &chain_resolver).await {
512                Ok(vp) => {
513                    // Return paths are planned from the destination back to us.
514                    valid_paths.push((vp, selection_weight(&pwc, hops, weighting)));
515                    path_metrics.push(pwc);
516                }
517                Err(e) => tracing::debug!(kind, error = %e, "path candidate failed validation"),
518            }
519        }
520
521        // A return path that resolves to a single first relayer puts every SURB of the destination
522        // behind it. Checked after validation: only the survivors reflect what a session can draw.
523        let diversity = (src_key != me && hops > 0 && !path_metrics.is_empty()).then(|| {
524            let distinct_relayers = super::selector::distinct_first_relayers(&path_metrics);
525            DiversityObservation {
526                key: (src_key, dest_key, hops as u32),
527                collapsed: distinct_relayers <= 1,
528                distinct_relayers,
529                candidates: path_metrics.len(),
530            }
531        });
532
533        if valid_paths.is_empty() {
534            return Ok(None);
535        }
536
537        let weighted = hopr_utils::statistics::WeightedCollection::new(valid_paths);
538        let total_wt = weighted.total_weight();
539        // Return draws spend `exploration` of their picks uniformly, so log what a draw actually does.
540        let exploration = if dest_key == me {
541            weighting.return_path_exploration
542        } else {
543            0.0
544        };
545        let candidates = weighted.iter().count() as f64;
546        for ((vp, w), pwm) in weighted.iter().zip(path_metrics.iter()) {
547            tracing::debug!(
548                kind,
549                destination = %dest_key,
550                hops,
551                path = %vp,
552                cost = pwm.cost,
553                selection_weight = w,
554                sampling_probability = if total_wt > 0.0 {
555                    (1.0 - exploration) * w.max(0.0) / total_wt + exploration / candidates
556                } else {
557                    0.0
558                },
559                total_latency_ms = ?pwm.total_latency_ms,
560                min_probe_success_rate = ?pwm.min_probe_success_rate,
561                min_ack_rate = ?pwm.min_ack_rate,
562                fundable_tickets_floor = ?pwm.fundable_tickets_floor,
563                "weighted candidate path",
564            );
565        }
566        Ok(Some((weighted, diversity)))
567    }
568
569    fn apply_diversity(&self, kind: &str, diversity: Option<DiversityObservation>) {
570        if let Some(obs) = diversity {
571            apply_diversity_transition(&self.collapsed_diversity, &obs.key.0, kind, &obs);
572        }
573    }

576/// Smallest change in a candidate's share of the draws that counts as traffic having moved.
577///
578/// Weights are recomputed from live observations, so they jitter constantly without meaning
579/// anything. One percentage point of a candidate's share is well below the shift a relayer going
580/// silent produces (measured: 33% to near zero) and well above the noise of an idle graph.
581const MIN_SHARE_SHIFT: f64 = 0.01;
582
583/// Each candidate's share of the total weight, keyed by the route it takes.
584fn shares_by_route(paths: &hopr_utils::statistics::WeightedCollection<ValidatedPath>) -> Vec<(String, f64)> {
585    let total: f64 = paths.iter().map(|(_, w)| w.max(0.0)).sum();
586    paths
587        .iter()
588        .map(|(vp, w)| {
589            let share = if total > 0.0 { w.max(0.0) / total } else { 0.0 };
590            (vp.to_string(), share)
591        })
592        .collect()
593}
594
595/// Whether re-weighting would send a materially different share of the draws somewhere else.
596///
597/// Shares rather than raw weights, because the draw normalises over the collection: every weight
598/// halving changes nothing about where traffic goes. Callers use this to decide whether a re-plan
599/// achieved anything, and a re-plan that moved nothing is a reason *not* to act on it -- so
600/// answering "yes" by default would defeat the check it exists for.
601fn weights_moved(
602    before: &hopr_utils::statistics::WeightedCollection<ValidatedPath>,
603    after: &hopr_utils::statistics::WeightedCollection<ValidatedPath>,
604) -> bool {
605    let (before, after) = (shares_by_route(before), shares_by_route(after));
606    if before.len() != after.len() {
607        return true;
608    }
609    before.iter().any(|(route, was)| {
610        // A route that vanished has lost its whole share, which is the largest move there is.
611        after
612            .iter()
613            .find(|(other, _)| other == route)
614            .is_none_or(|(_, now)| (now - was).abs() >= MIN_SHARE_SHIFT)
615    })
616}

618/// Cache key for the path planner: `(source, destination, hops)`.
619///
620/// Only the `Hops` variant of [`RoutingOptions`] is cached (explicit intermediate
621/// paths bypass the cache), so the key stores the hop count as a plain `u32`.
622///
623/// Keyed on resolved offchain keys rather than [`NodeId`], because a `NodeId` naming a node by its
624/// chain address is never equal to one naming the same node by its packet key. Callers hold
625/// whichever form their layer happens to use -- Sessions carry chain addresses, the SURB telemetry
626/// reports packet keys -- so a raw-`NodeId` key silently stores the same route twice. Resolving
627/// first makes lookup and insertion agree by construction.
628type PlannerCacheKey = (OffchainPublicKey, OffchainPublicKey, u32);
629type PlannerCacheValue = Arc<hopr_utils::statistics::WeightedCollection<ValidatedPath>>;
630
631/// Writes `value` under `key` only if `swept` is still the entry cached there, returning whether it
632/// was.
633///
634/// The background refresh reads an entry, rebuilds it from the graph across awaits, then writes. A
635/// [`PathPlanner::recompute_paths_from`] that ran in between built from newer SURB evidence, so the
636/// refresh's older selection must not land on top of it. The check and the write are one
637/// `and_compute_with`, which moka serialises against every other compute call on the key, and the
638/// recompute writes through one too — a plain insert would otherwise slip between this read and write.
639/// A key that went vacant meanwhile is left to the next fill.
640async fn put_if_unchanged(
641    cache: &moka::future::Cache<PlannerCacheKey, PlannerCacheValue>,
642    key: PlannerCacheKey,
643    swept: &PlannerCacheValue,
644    value: PlannerCacheValue,
645) -> bool {
646    let written = cache
647        .entry(key)
648        .and_compute_with(|current| {
649            futures::future::ready(match current {
650                Some(current) if Arc::ptr_eq(current.value(), swept) => moka::ops::compute::Op::Put(value),
651                _ => moka::ops::compute::Op::Nop,
652            })
653        })
654        .await;
655    matches!(written, moka::ops::compute::CompResult::ReplacedWith(_))
656}
657
658async fn drop_if_unchanged(
659    cache: &moka::future::Cache<PlannerCacheKey, PlannerCacheValue>,
660    key: PlannerCacheKey,
661    swept: &PlannerCacheValue,
662) {
663    cache
664        .entry(key)
665        .and_compute_with(|current| {
666            futures::future::ready(match current {
667                Some(current) if Arc::ptr_eq(current.value(), swept) => moka::ops::compute::Op::Remove,
668                _ => moka::ops::compute::Op::Nop,
669            })
670        })
671        .await;
672}

674impl<R, S> PathBuilder<R, S>
675where
676    R: ChainKeyOperations + ChainReadChannelOperations + Send + Sync,
677    S: PathSelector,
678{
679    /// Rebuilds every cached multi-hop entry from the current graph, one pass of the background sweep.
680    async fn sweep(&self, cache: &moka::future::Cache<PlannerCacheKey, PlannerCacheValue>) {
681        for (key, swept) in cache.iter() {
682            let (src_key, dest_key, hops_u32) = *key.as_ref();
683            if hops_u32 == 0 {
684                continue;
685            }
686
687            let rebuilt = self
688                .rebuild(src_key, dest_key, hops_u32 as usize, "background-refresh")
689                .await;
690            // Only if this is still the entry the sweep read: a `recompute_paths_from` that ran
691            // during the rebuild built from newer evidence, and its route must not be overwritten.
692            match rebuilt {
693                Ok(Some((weighted, diversity))) => {
694                    // A rejected stale refresh must not move the diversity tracker (#8502).
695                    if put_if_unchanged(cache, (src_key, dest_key, hops_u32), &swept, Arc::new(weighted)).await {
696                        self.apply_diversity("background-refresh", diversity);
697                    }
698                }
699                // No route left, e.g. every candidate runs through a dead relayer: serving the old one
700                // would keep drawing it.
701                Ok(None) | Err(PathPlannerError::Path(PathError::PathNotFound(..))) => {
702                    drop_if_unchanged(cache, (src_key, dest_key, hops_u32), &swept).await
703                }
704                Err(error) => tracing::debug!(%error, "background refresh failed, keeping the cached paths"),
705            }
706        }
707    }
708}

710/// Path planner that resolves [`DestinationRouting`] to [`ResolvedTransportRouting`].
711///
712/// The planner delegates path *discovery* to any [`PathSelector`] implementation and
713/// owns the `moka` cache of fully-validated [`ValidatedPath`] objects paired with
714/// their traversal cost, keyed by `(source: NodeId, destination: NodeId, hops: u32)`.
715///
716/// On a cache miss the planner calls the selector, validates every candidate against
717/// the chain resolver, and stores an `Arc<WeightedCollection<ValidatedPath>>` in the
718/// cache. On a cache hit a candidate is picked via weighted random selection (higher
719/// cost = higher quality = higher probability).
720///
721/// A background sweep (`background_refresh`) can be spawned to
722/// proactively re-warm the cache for all previously-seen keys.
723#[derive(Clone)]
724pub struct PathPlanner<Surb, R, S> {
725    me: OffchainPublicKey,
726    pub surb_store: Surb,
727    paths: Arc<PathBuilder<R, S>>,
728    cache: moka::future::Cache<PlannerCacheKey, PlannerCacheValue>,
729    refresh_period: Duration,
730    /// See [`IMPLAUSIBLE_PROBE_INTERVAL`]; shared by every clone of the planner.
731    probe_budget: Arc<ProbeBudget>,
732}
733
734impl<Surb, R, S> PathPlanner<Surb, R, S>
735where
736    Surb: SurbStore + Send + Sync + 'static,
737    R: ChainKeyOperations + ChainReadChannelOperations + Send + Sync + 'static,
738    S: PathSelector + Send + Sync + 'static,

740    /// Create a new path planner.
741    ///
742    /// `me` is this node's [`OffchainPublicKey`]; it is used as the source in path queries.
743    pub fn new(me: OffchainPublicKey, surb_store: Surb, resolver: R, selector: S, config: PathPlannerConfig) -> Self {
744        // Bound the diversity tracker to the live cache: when a key's cached paths are evicted by
745        // capacity or expire by TTL, drop its collapsed-state record too, so the set cannot grow
746        // without bound as distinct `(source, destination, hops)` keys churn. A `Replaced` eviction
747        // (a recompute or refresh installing a newer value for the same key) leaves the key live, so
748        // its record must stay. Cleanup is silent -- it is not a diversity recovery, so no INFO.
749        let collapsed_diversity: Arc<Mutex<HashSet<PlannerCacheKey>>> = Arc::new(Mutex::new(HashSet::new()));
750        let tracker_for_eviction = collapsed_diversity.clone();
751        let cache = moka::future::Cache::builder()
752            .max_capacity(config.max_cache_capacity)
753            .time_to_live(config.cache_ttl)
754            .eviction_listener(move |key: Arc<PlannerCacheKey>, _value, cause| {
755                if cause != moka::notification::RemovalCause::Replaced {
756                    tracker_for_eviction.lock().remove(key.as_ref());
757                }
758            })
759            .build();
760
761        Self {
762            me,
763            surb_store,
764            paths: Arc::new(PathBuilder {
765                resolver,
766                selector,
767                weighting: WeightingParams {
768                    latency_halflife: config.latency_halflife,
769                    capacity_reference: config.capacity_reference,
770                    return_path_weight_temper: config.return_path_weight_temper,
771                    return_path_exploration: config.return_path_exploration,
772                },
773                me,
774                collapsed_diversity,
775            }),
776            cache,
777            refresh_period: config.refresh_period,
778            probe_budget: Arc::default(),
779        }
780    }

806    #[tracing::instrument(level = "trace", skip(self))]
807    async fn resolve_path(
808        &self,
809        source: NodeId,
810        destination: NodeId,
811        options: RoutingOptions,
812    ) -> Result<ValidatedPath> {
813        let path = match options {
814            RoutingOptions::IntermediatePath(explicit_path) => {
815                tracing::debug!(
816                    direction = "loopback",
817                    ?source,
818                    ?destination,
819                    ?explicit_path,
820                    "resolving intermediate path"
821                );
822                let resolver = ChainPathResolver::from(&self.paths.resolver);
823                ValidatedPath::new(
824                    source,
825                    explicit_path
826                        .into_iter()
827                        .chain(std::iter::once(destination))
828                        .collect::<Vec<_>>(),
829                    &resolver,
830                )
831                .await?
832            }
833
834            RoutingOptions::Hops(hops) if u32::from(hops) == 0 => {
835                trace!(hops = 0, "resolving zero-hop direct path");
836                let resolver = ChainPathResolver::from(&self.paths.resolver);
837                ValidatedPath::new(source, vec![destination], &resolver).await?
838            }
839
840            RoutingOptions::Hops(hops) => {
841                let hops_usize: usize = hops.into();
842                let paths = self.cached_paths(source, destination, hops).await?;
843
844                // Format from the `NodeId`s rather than re-resolving them: `cached_paths` has
845                // already done that, and for `NodeId::Chain` each resolution is a resolver lookup.
846                //
847                // Explored like a return draw (see `return_path_exploration`): weights exist only for
848                // paths that get drawn, so without it a forward candidate that fell out of favour is
849                // never measured again, and the stream settles on one path.
850                //
851                // Over plausible candidates only, exploring or not: a forward packet through a relayer
852                // on probation, or over an edge probes measured dead, is data lost, PIX shares and
853                // SURBs with it, and it probes nothing (a probation ends on a reply through the
854                // relayer, which only a return path carries). Tempered, a relayer on probation keeps
855                // about 3 % of a healthy one's weight, so the ordinary draw is gated too.
856                let mut weights = paths.iter().map(|(_, weight)| *weight).collect::<Vec<_>>();
857                let drawn = if should_explore(self.paths.weighting.return_path_exploration) {
858                    pick_exploring_index(&weights, || false)
859                } else {
860                    gate_implausible(&mut weights);
861                    pick_weighted_index(&weights)
862                }
863                .and_then(|i| paths.iter().nth(i))
864                .map(|(vp, _)| vp.clone());
865                drawn.ok_or_else(|| {
866                    PathPlannerError::Path(PathError::PathNotFound(
867                        hops_usize,
868                        source.to_string(),
869                        destination.to_string(),
870                    ))
871                })?
872            }
873        };
874
875        #[cfg(all(feature = "telemetry", not(test)))]
876        {
877            hopr_api::types::telemetry::SimpleHistogram::observe(&METRIC_PATH_LENGTH, (path.num_hops() - 1) as f64);
878        }
879
880        trace!(%path, "validated resolved path");
881        Ok(path)
882    }

884    /// Cached weighted collection of validated `hops`-hop paths from `source` to `destination`,
885    /// computed on a miss.
886    ///
887    /// Single-path callers draw with `pick_one`; batch callers (see
888    /// [`PathPlanner::resolve_diverse_return_paths`]) work over the collection directly, paying one
889    /// cache lookup instead of one per path.
890    #[tracing::instrument(level = "trace", skip(self))]
891    async fn cached_paths(
892        &self,
893        source: NodeId,
894        destination: NodeId,
895        hops: hopr_api::types::primitive::bounded::BoundedSize<{ RoutingOptions::MAX_INTERMEDIATE_HOPS }>,
896    ) -> Result<PlannerCacheValue> {
897        let hops_usize: usize = hops.into();
898        trace!(hops = hops_usize, "resolving path via planner cache");
899
900        let src_key = self.resolve_node_id_to_offchain_key(&source).await?;
901        let dest_key = self.resolve_node_id_to_offchain_key(&destination).await?;
902
903        let cache_key: PlannerCacheKey = (src_key, dest_key, u32::from(hops));
904
905        let paths = self.paths.clone();
906
907        self.cache
908            .try_get_with(cache_key, async move {
909                trace!(hops = hops_usize, "path cache miss, querying selector");
910                let (weighted, diversity) =
911                    paths
912                        .rebuild(src_key, dest_key, hops_usize, "fill")
913                        .await?
914                        .ok_or_else(|| {
915                            PathPlannerError::Path(PathError::PathNotFound(
916                                hops_usize,
917                                src_key.to_hex(),
918                                dest_key.to_hex(),
919                            ))
920                        })?;
921                // A fill always becomes the cache entry, so its transition can be applied here.
922                paths.apply_diversity("fill", diversity);
923                Ok(Arc::new(weighted))
924            })
925            .await
926            .map_err(PathPlannerError::CacheError)
927    }

935    /// Rebuilds every cached entry whose paths originate at `source`, replacing each in place.
936    ///
937    /// Return paths are cached under `(counterparty, me, hops)`, so this is how a caller that has
938    /// just learned a counterparty's return traffic went silent forces those weights to be rebuilt
939    /// from the current graph, instead of waiting up to
940    /// [`PathPlannerConfig::refresh_period`] for the background sweep to reach them.
941    ///
942    /// Entries are replaced, never dropped: one that rebuilds to nothing keeps serving what it
943    /// already holds. Dropping them was measured to collapse a healthy session from 100 % to
944    /// 0.14 %, because a live session draws its next return path from this very entry.
945    ///
946    /// Returns how many entries came back with a materially different **share** of the draws --
947    /// not how many were rebuilt. Callers use this to decide whether the re-plan achieved anything
948    /// worth acting on, and on a healthy graph a rebuild always succeeds, so counting rebuilds
949    /// would answer "something moved" every time.
950    pub async fn recompute_paths_from(&self, source: &OffchainPublicKey) -> usize {
951        // 0-hop entries name a direct route with nothing to re-weight, exactly as in the sweep.
952        let keys = self
953            .cache
954            .iter()
955            .map(|(key, _)| *key.as_ref())
956            .filter(|(src, _, hops)| src == source && *hops > 0)
957            .collect::<Vec<PlannerCacheKey>>();
958
959        let mut moved = 0usize;
960        for (src_key, dest_key, hops) in keys {
961            if let Ok(Some((weighted, diversity))) =
962                self.paths.rebuild(src_key, dest_key, hops as usize, "recompute").await
963            {
964                let key = (src_key, dest_key, hops);
965                // A fresh entry counts as moved -- there was no previous distribution to compare
966                // against, so nothing here can say the traffic stayed put.
967                let mut shifted = true;
968                // Through `and_compute_with`, like the background refresh's `put_if_unchanged`: moka
969                // serialises only compute calls on a key, so a plain insert here could land between
970                // the refresh's read and write and be overwritten by the older entry it rebuilt.
971                self.cache
972                    .entry(key)
973                    .and_compute_with(|previous| {
974                        if let Some(previous) = previous {
975                            shifted = weights_moved(previous.value(), &weighted);
976                        }
977                        futures::future::ready(moka::ops::compute::Op::Put(Arc::new(weighted)))
978                    })
979                    .await;
980                // The recompute always installs, so its diversity transition always applies.
981                self.paths.apply_diversity("recompute", diversity);
982                if shifted {
983                    moved += 1;
984                }
985            }
986        }
987
988        tracing::debug!(%source, moved, "recomputed cached paths originating at peer");
989        moved
990    }

992    /// Resolves `count` return paths from `destination` back to this node, drawn weighted-random
993    /// over [`PathPlannerConfig::return_path_weight_temper`]-flattened path values.
994    ///
995    /// Tempering is what bounds the blast radius of losing one relay: raw path values concentrate a
996    /// session's SURBs on the few best candidates, so the flatter the effective distribution, the
997    /// smaller any single relay's share of the return stream.
998    ///
999    /// Draws are independent — deliberately. Spreading a *batch* over K distinct relayers was tried
1000    /// and cannot work: a batch is the SURBs that fit in one packet (`HoprPacket::PAYLOAD_SIZE /
1001    /// HoprSurb::SIZE`, a handful at most), so K was capped by it whatever the configuration said,
1002    /// while each packet re-drew independently regardless. Tempering has no such ceiling.
1003    ///
1004    /// Only `Hops` routing has alternatives to draw over — an explicit path resolves to itself, so
1005    /// those fall back to plain repeated resolution.
1006    #[tracing::instrument(level = "trace", skip(self))]
1007    async fn resolve_diverse_return_paths(
1008        &self,
1009        destination: NodeId,
1010        options: RoutingOptions,
1011        count: usize,
1012    ) -> Result<Vec<ValidatedPath>> {
1013        // No return paths requested (e.g. the message fills the payload, leaving no room for
1014        // SURBs). Resolve nothing — querying the planner here would turn "none wanted" into a
1015        // `PathNotFound` error.
1016        if count == 0 {
1017            return Ok(Vec::new());
1018        }
1019
1020        let me = NodeId::Offchain(self.me);
1021
1022        let hops = match options {
1023            RoutingOptions::Hops(hops) if u32::from(hops) > 0 => hops,
1024            // A fixed path or a direct return has no alternatives to weight over.
1025            other => {
1026                return (0..count)
1027                    .map(|_| self.resolve_path(destination, me, other.clone()))
1028                    .collect::<FuturesUnordered<_>>()
1029                    .try_collect::<Vec<_>>()
1030                    .await;
1031            }
1032        };
1033
1034        let candidates = self.cached_paths(destination, me, hops).await?;
1035        let items: Vec<&(ValidatedPath, f64)> = candidates.iter().collect();
1036        let weights = items.iter().map(|(_, w)| *w).collect::<Vec<_>>();
1037
1038        if weights.iter().all(|w| *w <= 0.0) {
1039            return Err(PathPlannerError::Path(PathError::PathNotFound(
1040                hops.into(),
1041                destination.to_string(),
1042                me.to_string(),
1043            )));
1044        }
1045
1046        // Distinct first relayers in the candidate pool: the return-relayer diversity actually
1047        // available for this destination. A value of 1 is the corroboration blind spot (a dead
1048        // relayer cannot be told from a quiet peer) — the same condition `PathBuilder::rebuild` WARNs
1049        // on. This draw runs per-packet, so both sets below are built only when DEBUG is actually
1050        // recorded -- not on every call.
1051        let log_relayer_diversity = tracing::enabled!(tracing::Level::DEBUG);
1052        let candidate_relayers = log_relayer_diversity.then(|| {
1053            items
1054                .iter()
1055                .filter_map(|item| item.0.first())
1056                .collect::<std::collections::HashSet<_>>()
1057        });
1058
1059        tracing::debug!(
1060            %destination,
1061            count,
1062            candidates = items.len(),
1063            distinct_relayers = candidate_relayers.as_ref().map_or(0, |r| r.len()),
1064            temper = self.paths.weighting.return_path_weight_temper,
1065            exploration = self.paths.weighting.return_path_exploration,
1066            "drawing return paths from tempered weights"
1067        );
1068
1069        // An exploring draw may land on a candidate the weights all but reject only within the
1070        // node-wide probe allowance: see `IMPLAUSIBLE_PROBE_INTERVAL`.
1071        let drawn = (0..count)
1072            .filter_map(|_| {
1073                if should_explore(self.paths.weighting.return_path_exploration) {
1074                    pick_exploring_index(&weights, || self.probe_budget.take())
1075                } else {
1076                    pick_weighted_index(&weights)
1077                }
1078                .map(|i| items[i].0.clone())
1079            })
1080            .collect::<Vec<_>>();
1081
1082        // How many distinct relayers the actual SURBs went to. Fewer than the candidate pool means
1083        // the weighting concentrated the stream — expected — but a persistent 1 here while
1084        // `candidates` > 1 says the draw itself is starving the siblings the detector relies on.
1085        let drawn_relayers = log_relayer_diversity.then(|| {
1086            drawn
1087                .iter()
1088                .filter_map(|vp| vp.first())
1089                .collect::<std::collections::HashSet<_>>()
1090        });
1091        tracing::debug!(
1092            %destination,
1093            drawn = drawn.len(),
1094            distinct_relayers = drawn_relayers.as_ref().map_or(0, |r| r.len()),
1095            of_candidates = candidate_relayers.as_ref().map_or(0, |r| r.len()),
1096            "drew return paths"
1097        );
1098
1099        Ok(drawn)
1100    }

1102    /// Resolve a [`DestinationRouting`] to a [`ResolvedTransportRouting`].
1103    ///
1104    /// Returns the resolved routing and, for `Return` variants, the number of remaining SURBs.
1105    #[tracing::instrument(level = "trace", skip(self))]
1106    pub async fn resolve_routing(
1107        &self,
1108        size_hint: usize,
1109        max_surbs: usize,
1110        routing: DestinationRouting,
1111    ) -> Result<(ResolvedTransportRouting<HoprSurb>, Option<usize>, Option<u8>)> {
1112        match routing {
1113            DestinationRouting::Forward {
1114                destination,
1115                pseudonym,
1116                forward_options,
1117                return_options,
1118            } => {
1119                tracing::debug!(direction = "forward", %destination, "resolving forward path");
1120
1121                let forward_path = self
1122                    .resolve_path(NodeId::Offchain(self.me), *destination, forward_options)
1123                    .await?;
1124                tracing::debug!(direction = "forward", %destination, path = %forward_path, "resolved path");
1125
1126                let resolved_pseudonym = pseudonym.unwrap_or_else(HoprPseudonym::random);
1127
1128                // Capture the SURB-batch generation here — with the return-path plan, not at encode
1129                // time — because the replying side supersedes SURBs by generation, so a batch's
1130                // generation must match the plan it was minted for.
1131                //
1132                // Crucially, read it *before* resolving the return paths, not after. The flush loop
1133                // recomputes paths and only *then* bumps the generation (`recompute_paths_from` then
1134                // `bump_generation` in `HoprTransport`'s SURB-flush task), so a bump racing this
1135                // resolution has, by the time it happens, already moved the planner to the new route.
1136                // Reading first therefore makes any residual mislabel *under*-label — a fresh route
1137                // stamped with the previous generation, which the peer merely appends and then clears
1138                // on the next batch (a few wasted SURBs) — and never *over*-label — a stale route
1139                // stamped with the newer generation, which would make the peer drop live SURBs and
1140                // hand out dead ones. Reading after the resolve is what would allow the harmful
1141                // over-label (choose old route, bump, read new generation), so the order matters and a
1142                // lock on the per-packet hot path is not needed.
1143                let (return_paths, generation) = if let Some(return_options) = return_options {
1144                    let generation = self.surb_store.current_generation(&resolved_pseudonym);
1145
1146                    let num_possible_surbs = HoprPacket::max_surbs_with_message(size_hint).min(max_surbs);
1147                    trace!(
1148                        %destination,
1149                        %num_possible_surbs,
1150                        data_len = size_hint,
1151                        max_surbs,
1152                        "resolving packet return paths"
1153                    );
1154
1155                    let return_paths: Vec<_> = self
1156                        .resolve_diverse_return_paths(*destination, return_options, num_possible_surbs)
1157                        .await?
1158                        .into_iter()
1159                        .enumerate()
1160                        .inspect(|(i, rp)| {
1161                            tracing::debug!(direction = "return", %destination, index = i, path = %rp, "resolved return path");
1162                        })
1163                        .map(|(_, rp)| rp)
1164                        .collect();
1165
1166                    // Only stamp a generation if return paths (hence SURBs) were actually produced.
1167                    let generation = (!return_paths.is_empty()).then_some(generation);
1168                    (return_paths, generation)
1169                } else {
1170                    (vec![], None)
1171                };
1172
1173                trace!(%destination, num_surbs = return_paths.len(), data_len = size_hint, "resolved packet");
1174
1175                Ok((
1176                    ResolvedTransportRouting::Forward {
1177                        pseudonym: resolved_pseudonym,
1178                        forward_path,
1179                        return_paths,
1180                    },
1181                    None,
1182                    generation,
1183                ))
1184            }
1185
1186            DestinationRouting::Return(matcher) => {
1187                let FoundSurb {
1188                    sender_id,
1189                    surb,
1190                    remaining,
1191                } = self
1192                    .surb_store
1193                    .find_surb(matcher)
1194                    .ok_or_else(|| PathPlannerError::Surb(format!("no surb for pseudonym {}", matcher.pseudonym())))?;
1195                Ok((ResolvedTransportRouting::Return(sender_id, surb), Some(remaining), None))
1196            }
1197        }
1198    }
1199}

1201impl<Surb, R, S> BackgroundPathCacheRefreshable for PathPlanner<Surb, R, S>
1202where
1203    Surb: SurbStore + Send + Sync + 'static,
1204    R: ChainKeyOperations + ChainReadChannelOperations + Send + Sync + 'static,
1205    S: PathSelector + Send + Sync + 'static,
1206{
1207    /// Returns a future that runs the background path-cache refresh loop.
1208    ///
1209    /// The returned future iterates over all keys currently in the planner's cache
1210    /// and recomputes their paths on a configurable schedule, so that steady-state
1211    /// traffic is always served from cache.
1212    fn run_background_refresh(&self) -> impl std::future::Future<Output = ()> + Send + 'static {
1213        // Clone only the fields we need — avoids requiring R: Clone + S: Clone.
1214        let cache = self.cache.clone();
1215        let paths = self.paths.clone();
1216        let refresh_period = self.refresh_period;
1217
1218        // run at a non-zero interval
1219        futures_time::stream::interval(futures_time::time::Duration::from_millis(
1220            refresh_period.as_millis() as u64 + 1u64,
1221        ))
1222        .for_each(move |_| {
1223            let cache = cache.clone();
1224            let paths = paths.clone();
1225            async move { paths.sweep(&cache).await }
1226        })
1227    }
1228}

2769    /// Share of the collection's total weight held by candidates whose first hop is `relayer`.
2770    ///
2771    /// This -- not the raw weight -- is what decides how much of a session's return stream rides on
2772    /// one relay, because the draw normalises over the collection.
2773    fn share_of(paths: &PlannerCacheValue, relayer: &OffchainPublicKey) -> f64 {
2774        let total: f64 = paths.iter().map(|(_, w)| *w).sum();
2775        let held: f64 = paths
2776            .iter()
2777            .filter(|(vp, _)| vp.first() == Some(relayer))
2778            .map(|(_, w)| *w)
2779            .sum();
2780        if total > 0.0 { held / total } else { 0.0 }
2781    }
