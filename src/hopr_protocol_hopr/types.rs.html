6/// Packet that is being sent out by us.
7pub struct OutgoingPacket {
8    /// Offchain public key of the next hop.
9    pub next_hop: OffchainPublicKey,
10    /// Challenge to be solved from the acknowledgement of the next hop.
11    pub ack_challenge: HalfKeyChallenge,
12    /// Optional encrypted partial SSA share for PIX protocol.
13    pub encrypted_pix_share: Option<TaggedEncryptedPartialSsaShare<HoprPixSpec>>,
14    /// Encoded HOPR packet.
15    pub data: Bytes,
16    /// SURBs minted onto this packet, in the order of the return paths that produced them.
17    ///
18    /// Surfaced so a layer above can pair each SURB with the return path it encodes — the ids exist
19    /// only inside packet construction, and the routing that produced them only outside it, so this
20    /// is the single point where the two can be associated. Empty for packets carrying no SURBs.
21    pub minted_surbs: Vec<HoprSurbId>,
22}

34/// Contains some miscellaneous information about a received packet.
35#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
36pub struct AuxiliaryPacketInfo {
37    /// Packet signals that the packet carried.
38    ///
39    /// Zero if no signal flags were specified.
40    pub packet_signals: PacketSignals,
41    /// Number of SURBs that the packet carried.
42    pub num_surbs: usize,
43    /// How many of those SURBs the store dropped on arrival because the per-pseudonym buffer was
44    /// already full: the SURBs that left the store, or never entered it, to make room. A share-less
45    /// newcomer the store turned away counts as well.
46    ///
47    /// This is the only point at which an overflow is visible per packet. The store gives up
48    /// share-less SURBs first, and refuses a share-less newcomer before it evicts a share-bearing
49    /// SURB, so what an eviction costs depends on what the buffer holds. A share-less SURB is only a
50    /// lost return path. A share-bearing one carries a partial SSA share that reaches the
51    /// reconstructor only when the SURB is *used*, so evicting it destroys the share. That only
52    /// happens once the buffer holds nothing but shares, which makes this count an upper bound on the
53    /// shares lost. The exact figure is [`SurbInsertOutcome::evicted_shares`]. It is deliberately not
54    /// carried here: it is the store's own business, reported in its eviction summary (the
55    /// `ring_share` value of the `cache` label of `hopr_surb_store_evictions_count`, and a warning per
56    /// report interval). The redundancy budget that absorbs those losses is a fixed surplus per
57    /// polynomial, and emission is windowed, so on deployed dimensions a burst of roughly
58    /// `surplus × SHARE_EMISSION_WINDOW` evicted share-bearing SURBs is enough to put polynomials
59    /// below their threshold — and a cycle short of recovery is worth nothing at all. See
60    /// [`SurbStoreConfig::rb_capacity`](crate::SurbStoreConfig::rb_capacity) and
61    /// `hopr_protocol_pix::SHARE_EMISSION_WINDOW`.
62    ///
63    /// The Exit's Session books this count into its SURB flow estimate — an evicted SURB has left the
64    /// buffer just as a spent one has — so the meaning above must not change: it is the total of
65    /// SURBs that left or never entered the store, not the number of lost shares. See
66    /// `counterparty_buffer_capacity` in `hopr-transport-session`.
67    pub num_evicted_surbs: usize,
68}

155/// Incoming HOPR packet.
156#[derive(Debug, strum::EnumTryAs)]
157pub enum IncomingPacket {
158    /// Packet is intended for us
159    Final(Box<IncomingFinalPacket>),
160    /// Packet must be forwarded
161    Forwarded(Box<IncomingForwardedPacket>),
162    /// The packet contains acknowledgements of delivered packets.
163    Acknowledgement(Box<IncomingAcknowledgementPacket>),
164}
165
166impl IncomingPacket {
167    /// Tag identifying the packet.
168    pub fn packet_tag(&self) -> &PacketTag {
169        match self {
170            IncomingPacket::Final(f) => &f.packet_tag,
171            IncomingPacket::Forwarded(f) => &f.packet_tag,
172            IncomingPacket::Acknowledgement(f) => &f.packet_tag,
173        }
174    }
175
176    /// Previous hop that sent us the packet.
177    pub fn previous_hop(&self) -> &OffchainPublicKey {
178        match self {
179            IncomingPacket::Final(f) => &f.previous_hop,
180            IncomingPacket::Forwarded(f) => &f.previous_hop,
181            IncomingPacket::Acknowledgement(f) => &f.previous_hop,
182        }
183    }
184}

197/// What storing SURBs via `SurbStore::insert_surbs` did to the ring buffer.
198///
199/// The `evicted` count exists because an overflow is otherwise entirely silent: the buffer drops a
200/// SURB to make room — a share-less one whenever there is one to drop — and the caller sees only that
201/// the insert "succeeded". That count is the only local evidence that the sender is producing faster
202/// than this side can hold. It does not say what the overflow cost, though. Under PIX a share reaches
203/// the reconstructor only when its SURB is *used*, so an evicted share-bearing SURB takes its share
204/// with it permanently, whereas a share-less one is only a lost return path. `evicted_shares` is the
205/// exact number of shares lost to overflow, which is what tells routine share-less churn from a loss
206/// that matters.
207#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
208pub struct SurbInsertOutcome {
209    /// Number of SURBs held for the pseudonym after the insert.
210    pub retained: usize,
211    /// Number of SURBs that left, or never entered, the buffer to make room during this insert:
212    /// share-less ones first, oldest first, and the oldest share-bearing one only once none remain. A
213    /// share-less SURB turned away because the buffer held nothing but shares counts as well.
214    pub evicted: usize,
215    /// Of [`evicted`](Self::evicted), how many carried a PIX share: the exact number of shares this
216    /// insert lost to overflow, each of them for good. It is zero for as long as the buffer still has
217    /// share-less SURBs to give up, and for every refused share-less newcomer.
218    ///
219    /// `evicted - evicted_shares` is the number of share-less SURBs dropped or refused, which cost a
220    /// return path and nothing more.
221    pub evicted_shares: usize,
222}
