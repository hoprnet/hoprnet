6/// Packet that is being sent out by us.
7pub struct OutgoingPacket {
8    /// Offchain public key of the next hop.
9    pub next_hop: OffchainPublicKey,
10    /// Challenge to be solved from the acknowledgement of the next hop.
11    pub ack_challenge: HalfKeyChallenge,
12    /// Optional encrypted partial SSA share for PIX protocol.
13    pub encrypted_pix_share: Option<TaggedEncryptedPartialSsaShare<HoprPixSpec>>,
14    /// Encoded HOPR packet.
15    pub data: Bytes,
16    /// SURBs minted onto this packet, in the order of the return paths that produced them.
17    ///
18    /// Surfaced so a layer above can pair each SURB with the return path it encodes — the ids exist
19    /// only inside packet construction, and the routing that produced them only outside it, so this
20    /// is the single point where the two can be associated. Empty for packets carrying no SURBs.
21    pub minted_surbs: Vec<HoprSurbId>,
22}

30            .finish_non_exhaustive()
31    }

34/// Contains some miscellaneous information about a received packet.
35#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
36pub struct AuxiliaryPacketInfo {
37    /// Packet signals that the packet carried.
38    ///
39    /// Zero if no signal flags were specified.
40    pub packet_signals: PacketSignals,
41    /// Number of SURBs that the packet carried.
42    pub num_surbs: usize,
43    /// How many of those SURBs the store dropped on arrival because the per-pseudonym buffer was
44    /// already full.
45    ///
46    /// This is the only point at which an overflow is visible, and the cost is higher than a wasted
47    /// SURB: each one carries a partial SSA share that reaches the reconstructor only when the SURB
48    /// is *used*, so an eviction destroys the share. The redundancy budget that absorbs those losses
49    /// is a fixed surplus per polynomial, and emission is windowed, so on deployed dimensions a
50    /// burst of roughly `surplus × SHARE_EMISSION_WINDOW` evictions is enough to put polynomials
51    /// below their threshold — and a cycle short of recovery is worth nothing at all. See
52    /// [`SurbStoreConfig::rb_capacity`](crate::SurbStoreConfig::rb_capacity) and
53    /// `hopr_protocol_pix::SHARE_EMISSION_WINDOW`.
54    ///
55    /// Carried for observability. Sessions deliberately do not subtract it from their SURB level
56    /// estimate, because the imprecise estimate is the one that fails safe — see
57    /// `counterparty_buffer_capacity` in `hopr-transport-session`.
58    pub num_evicted_surbs: usize,
59}

146/// Incoming HOPR packet.
147#[derive(Debug, strum::EnumTryAs)]
148pub enum IncomingPacket {
149    /// Packet is intended for us
150    Final(Box<IncomingFinalPacket>),
151    /// Packet must be forwarded
152    Forwarded(Box<IncomingForwardedPacket>),
153    /// The packet contains acknowledgements of delivered packets.
154    Acknowledgement(Box<IncomingAcknowledgementPacket>),
155}
156
157impl IncomingPacket {
158    /// Tag identifying the packet.
159    pub fn packet_tag(&self) -> &PacketTag {
160        match self {
161            IncomingPacket::Final(f) => &f.packet_tag,
162            IncomingPacket::Forwarded(f) => &f.packet_tag,
163            IncomingPacket::Acknowledgement(f) => &f.packet_tag,
164        }
165    }
166
167    /// Previous hop that sent us the packet.
168    pub fn previous_hop(&self) -> &OffchainPublicKey {
169        match self {
170            IncomingPacket::Final(f) => &f.previous_hop,
171            IncomingPacket::Forwarded(f) => &f.previous_hop,
172            IncomingPacket::Acknowledgement(f) => &f.previous_hop,
173        }
174    }
175}

188/// What storing SURBs via `SurbStore::insert_surbs` did to the ring buffer.
189///
190/// The `evicted` count exists because an overflow is otherwise entirely silent: the buffer drops its
191/// oldest entry and the caller sees only that the insert "succeeded". That count is the only local
192/// evidence that the sender is producing faster than this side can hold, and under PIX it is also a
193/// tally of destroyed SSA shares — a share reaches the reconstructor only when its SURB is *used*,
194/// so an evicted SURB takes its share with it permanently.
195#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
196pub struct SurbInsertOutcome {
197    /// Number of SURBs held for the pseudonym after the insert.
198    pub retained: usize,
199    /// Number of SURBs dropped to make room during this insert, oldest first.
200    pub evicted: usize,
201}
