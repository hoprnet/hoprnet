83/// Which end of the per-pseudonym buffer a pop consumes from. Replying side only.
84///
85/// Chooses the consumption order *within* a SURB generation. It is orthogonal to the per-SURB
86/// generation tag (`SurbReceiverInfo::generation`), which handles an explicit return-path re-plan by
87/// clearing superseded SURBs wholesale on the next push. Overflow always evicts the oldest SURB, in
88/// either order.
89#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash, strum::EnumString, strum::Display)]
90#[cfg_attr(feature = "serde", derive(serde::Deserialize, serde::Serialize))]
91#[strum(serialize_all = "lowercase")]
92#[cfg_attr(feature = "serde", serde(rename_all = "lowercase"))]
93pub enum SurbPopOrder {
94    /// Oldest first. Default; preserves the historical behaviour.
95    #[default]
96    Fifo,
97    /// Newest first, so a return-path change applies sooner instead of only after the buffered SURBs
98    /// drain. Stale ones are shed from the other end on overflow.
99    Lifo,
100}

102/// Configuration for the SURB cache.
103///
104/// The configuration options affect both the sending side (SURB creator) and the
105/// replying side (SURB consumer).
106///
107/// In the classical scenario (`Entry - Relay 1 -... - Exit`), the sending side is
108/// the `Entry` and the replying side is the `Exit`.
109#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, smart_default::SmartDefault, validator::Validate)]
110#[cfg_attr(
111    feature = "serde",
112    derive(serde::Deserialize, serde::Serialize),
113    serde(deny_unknown_fields)
114)]
115pub struct SurbStoreConfig {
116    /// Size of the SURB ring buffer per pseudonym.
117    ///
118    /// Affects only the replying side.
119    ///
120    /// This indicates how many SURBs can be at most held to be used to send a reply
121    /// back to the sending side.
122    ///
123    /// Once the buffer is full, a push overwrites the oldest SURBs, which are then never used.
124    /// With PIX that is not merely a wasted SURB: each one carries a partial SSA share that is only
125    /// delivered to the reconstructor when the SURB is *used*, so an overwrite is a permanently
126    /// lost share. The capacity is therefore sized well above what any Session's SURB balancer
127    /// targets — see `maximum_surb_buffer_size` in `hopr-transport`, which is derived from this
128    /// value with headroom left for balancer overshoot.
129    ///
130    /// This is a ceiling rather than a reservation: the internal `SurbRingBuffer` allocates with
131    /// occupancy, so a pseudonym holding three SURBs costs three, and only one that genuinely fills
132    /// up pays `rb_capacity` × ~400 B. That property is what makes a large default safe here — see
133    /// the "Why the capacity is a ceiling, not a reservation" section on `SurbRingBuffer`.
134    ///
135    /// Default is 100 000.
136    // `SurbRingBuffer` is deliberately unlinked above: it is crate-private, so an intra-doc link
137    // from this public field trips `rustdoc::private_intra_doc_links`, which CI builds as an error.
138    #[default(default_rb_capacity())]
139    #[validate(range(min = 1024, message = "rb_capacity must be at least 1024"))]
140    #[cfg_attr(feature = "serde", serde(default = "default_rb_capacity"))]
141    pub rb_capacity: usize,
142    /// Which end of the per-pseudonym buffer a pop consumes from; see [`SurbPopOrder`].
143    ///
144    /// Affects only the replying side. Default is [`SurbPopOrder::Fifo`].
145    #[cfg_attr(feature = "serde", serde(default))]
146    pub pop_order: SurbPopOrder,
147    /// Threshold for the number of SURBs in the ring buffer, below which it is
148    /// considered low ("SURB distress").
149    ///
150    /// Default is 500.
151    #[default(default_distress_threshold())]
152    #[validate(range(min = 10, message = "distress_threshold must be at least 10"))]
153    #[cfg_attr(feature = "serde", serde(default = "default_distress_threshold"))]
154    pub distress_threshold: usize,
155    /// Maximum number of reply openers (SURB counterparts) per pseudonym.
156    ///
157    /// Affects only the sending side when decrypting a received reply.
158    ///
159    /// This mostly affects Sessions, as they use a fixed pseudonym.
160    /// It reflects how many reply openers the initiator-side of a Session can hold,
161    /// until the oldest ones are dropped. If the other party uses a SURB corresponding
162    /// to a dropped reply opener, the reply message will be undecryptable by the initiator-side.
163    ///
164    /// Default is 100 000.
165    #[default(default_max_openers_per_pseudonym())]
166    #[validate(range(min = 100, message = "max_openers_per_pseudonym must be at least 100"))]
167    #[cfg_attr(feature = "serde", serde(default = "default_max_openers_per_pseudonym"))]
168    pub max_openers_per_pseudonym: usize,
169    /// The maximum number of distinct pseudonyms for which we hold a SURB ringbuffer.
170    ///
171    /// Affects only the replying side.
172    ///
173    /// For each pseudonym, there is a ring-buffer with capacity `rb_capacity`.
174    ///
175    /// Default is 10 000.
176    #[default(default_max_pseudonyms())]
177    #[validate(range(min = 100, message = "max_pseudonyms must be at least 100"))]
178    #[cfg_attr(feature = "serde", serde(default = "default_max_pseudonyms"))]
179    pub max_pseudonyms: usize,
180    /// Maximum lifetime of ring-buffer for each pseudonym.
181    ///
182    /// # Effects on sending side
183    /// This is the period for which we hold all reply openers for a pseudonym.
184    /// If no more messages carrying SURBs are sent during this period, the entire stash of
185    /// reply openers is dropped. Preventing receiving any more replies for that pseudonym.
186    ///
187    /// # Effects on replying side
188    /// If a pseudonym has not received any SURBs for this period,
189    /// the entire ring buffer with `rb_capacity` (= all SURBs for this pseudonym) is dropped.
190    /// Preventing from sending any more replies for that pseudonym.
191    ///
192    /// Default is 600 seconds.
193    #[default(default_pseudonyms_lifetime())]
194    #[validate(custom(function = "validate_pseudonyms_lifetime"))]
195    #[cfg_attr(
196        feature = "serde",
197        serde(default = "default_pseudonyms_lifetime", with = "humantime_serde")
198    )]
199    pub pseudonyms_lifetime: Duration,
200    /// Maximum lifetime of a reply opener.
201    ///
202    /// Affects only the sending side.
203    ///
204    /// A reply opener is distinguished using [`HoprSurbId`] and a pseudonym it belongs to.
205    /// If a reply opener is not used to decrypt the received packet within this period,
206    /// it is dropped. If the replying side uses the corresponding SURB to send a reply,
207    /// it won't be possible to decrypt it when received.
208    ///
209    /// Default is 3600 seconds.
210    #[default(default_reply_opener_lifetime())]
211    #[validate(custom(function = "validate_reply_opener_lifetime"))]
212    #[cfg_attr(
213        feature = "serde",
214        serde(default = "default_reply_opener_lifetime", with = "humantime_serde")
215    )]
216    pub reply_opener_lifetime: Duration,
217    /// How often cache evictions are summarised in the log, per cache and cause, instead of per entry.
218    ///
219    /// Affects both sides. Default is 60 seconds.
220    #[default(default_eviction_report_interval())]
221    #[validate(custom(function = "validate_eviction_report_interval"))]
222    #[cfg_attr(
223        feature = "serde",
224        serde(default = "default_eviction_report_interval", with = "humantime_serde")
225    )]
226    pub eviction_report_interval: Duration,
227    /// Evictions from one cache per report interval above which the summary is a warning rather than debug.
228    ///
229    /// Affects both sides. Default is 1000.
230    #[default(default_eviction_report_threshold())]
231    #[cfg_attr(feature = "serde", serde(default = "default_eviction_report_threshold"))]
232    pub eviction_report_threshold: u64,

235/// Basic [`SurbStore`] implementation based on an in-memory cache.
236///
237/// This SURB store offers no persistence, and all SURBs and Reply Openers are lost once dropped.
238///
239/// The instance can be cheaply cloned.
240#[derive(Clone)]
241pub struct MemorySurbStore {
242    pseudonym_openers: moka::sync::Cache<HoprPseudonym, moka::sync::Cache<HoprSurbId, ReplyOpener>>,
243    surbs_per_pseudonym: moka::sync::Cache<HoprPseudonym, SurbRingBuffer<HoprSurb>>,
244    /// Relayers this node can no longer pay. Holds at most a handful of entries (our own closing
245    /// channels), so a plain set behind an `RwLock` beats a concurrent map on this read-heavy path.
246    invalidated_relayers: Arc<parking_lot::RwLock<std::collections::HashSet<HoprKeyIdent>>>,
247    /// Current SURB-batch generation per pseudonym we originate for (sending side). Advanced on a
248    /// return-path change so the replying side can drop SURBs for the superseded path.
249    ///
250    /// This is sending-side state, so it is retained like the reply openers
251    /// ([`pseudonym_openers`](Self::pseudonym_openers)) — **not** like the receiving-side
252    /// [`surbs_per_pseudonym`](Self::surbs_per_pseudonym). If this entry were evicted while the peer
253    /// still held SURBs of generation `N`, [`current_generation`](Self::current_generation) would
254    /// fall back to `0`; the peer's [`SurbRingBuffer::push`] then reads `generation_is_newer(0, N)`
255    /// as false (for `1 <= N <= 128`) and silently discards every fresh batch, stranding the reply
256    /// path. Both dimensions are therefore taken from the sending-side reply-opener config, not the
257    /// receiving-side SURB config: `reply_opener_lifetime` bounds how long we expect replies (hence
258    /// outstanding SURBs) for a pseudonym, and it is `>=` the peer's SURB idle window, so the serial
259    /// outlives the SURBs it numbers; and the capacity matches the reply-opener pseudonym bound
260    /// (`max_openers_per_pseudonym`, which covers `maximum_managed_sessions`) so LRU pressure from the
261    /// unrelated receiving-side `max_pseudonyms` cannot evict a live sender's generation.
262    generations: moka::sync::Cache<HoprPseudonym, Arc<std::sync::atomic::AtomicU8>>,
263    stats: Arc<EvictionStats>,
264    cfg: Arc<SurbStoreConfig>,
265}

316/// Evictions from one cache in one report interval, by cause.
317#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
318struct EvictionCounts {
319    expired: u64,
320    size: u64,
321    replaced: u64,
322}
323
324impl EvictionCounts {
325    /// `Explicit` removals are deliberate (a used opener, an invalidation), not evictions, so they are skipped.
326    fn bump(&mut self, cause: RemovalCause) {
327        match cause {
328            RemovalCause::Expired => self.expired += 1,
329            RemovalCause::Size => self.size += 1,
330            RemovalCause::Replaced => self.replaced += 1,
331            RemovalCause::Explicit => {}
332        }
333    }
334
335    fn total(&self) -> u64 {
336        self.expired + self.size + self.replaced
337    }
338}

340/// The interval being counted right now.
341#[derive(Debug)]
342struct IntervalState {
343    started_at: Instant,
344    /// Parallel to [`EvictedCache::ALL`].
345    counts: [EvictionCounts; 4],

348/// Eviction counts per cache and cause, reported once per interval (GNO-793: per-entry lines flooded the log).
349struct EvictionStats {
350    interval: Duration,
351    warn_threshold: u64,
352    state: parking_lot::Mutex<IntervalState>,
353}
354
355impl EvictionStats {
356    fn new(cfg: &SurbStoreConfig) -> Self {
357        Self {
358            interval: cfg.eviction_report_interval.max(MINIMUM_EVICTION_REPORT_INTERVAL),
359            warn_threshold: cfg.eviction_report_threshold,
360            state: parking_lot::Mutex::new(IntervalState {
361                started_at: Instant::now(),
362                counts: Default::default(),
363            }),
364        }
365    }
366
367    fn record(&self, cache: EvictedCache, cause: RemovalCause) {
368        if let Some(report) = self.record_at(cache, cause, Instant::now()) {
369            report.log();
370        }
371    }
372
373    /// Closes the interval if `now` is past it, then counts one eviction; returns the closed interval's counts.
374    fn record_at(&self, cache: EvictedCache, cause: RemovalCause, now: Instant) -> Option<EvictionReport> {
375        // One lock for check, drain and count, so a rollover cannot split an eviction from its interval.
376        let mut state = self.state.lock();
377        let report = self.close_interval_if_elapsed(&mut state, now);
378        state.counts[cache as usize].bump(cause);
379        drop(state);
380
381        #[cfg(all(feature = "telemetry", not(test)))]
382        if cause != RemovalCause::Explicit {
383            METRIC_SURB_STORE_EVICTIONS.increment(&[cache.into(), cause_label(cause)]);
384        }
385        report
386    }
387
388    /// No runtime here, so the first eviction past the interval closes it: a quiet store reports late, not never.
389    fn close_interval_if_elapsed(&self, state: &mut IntervalState, now: Instant) -> Option<EvictionReport> {
390        if now.duration_since(state.started_at) < self.interval {
391            return None;
392        }
393        state.started_at = now;
394        let counts = std::mem::take(&mut state.counts);
395        Some(EvictionReport {
396            interval: self.interval,
397            warn_threshold: self.warn_threshold,
398            per_cache: EvictedCache::ALL.map(|cache| (cache, counts[cache as usize])),
399        })
400    }
401}

403/// The counts of one closed report interval, ready to be logged.
404#[derive(Debug)]
405struct EvictionReport {
406    interval: Duration,
407    warn_threshold: u64,
408    per_cache: [(EvictedCache, EvictionCounts); 4],
409}
410
411impl EvictionReport {
412    fn counts(&self, cache: EvictedCache) -> EvictionCounts {
413        self.per_cache[cache as usize].1
414    }
415
416    fn exceeds_threshold(&self, cache: EvictedCache) -> bool {
417        self.counts(cache).total() > self.warn_threshold
418    }
419
420    fn log(&self) {
421        let interval_secs = self.interval.as_secs();
422        for (cache, counts) in self.per_cache {
423            #[cfg(all(feature = "telemetry", not(test)))]
424            for (cause, value) in [
425                (RemovalCause::Expired, counts.expired),
426                (RemovalCause::Size, counts.size),
427                (RemovalCause::Replaced, counts.replaced),
428            ] {
429                METRIC_SURB_STORE_EVICTIONS_LAST_INTERVAL.set(&[cache.into(), cause_label(cause)], value as f64);
430            }
431
432            if counts.total() == 0 {
433                continue;
434            }
435            if self.exceeds_threshold(cache) {
436                tracing::warn!(
437                    %cache,
438                    expired = counts.expired,
439                    size = counts.size,
440                    replaced = counts.replaced,
441                    interval_secs,
442                    "SURB store evicted entries in the last interval"
443                );
444            } else {
445                tracing::debug!(
446                    %cache,
447                    expired = counts.expired,
448                    size = counts.size,
449                    replaced = counts.replaced,
450                    interval_secs,
451                    "SURB store evicted entries in the last interval"
452                );
453            }
454        }
455    }
456}

458impl MemorySurbStore {
459    /// Creates a new instance with the given configuration.
460    pub fn new(cfg: SurbStoreConfig) -> Self {
461        #[cfg(all(feature = "telemetry", not(test)))]
462        {
463            lazy_static::initialize(&METRIC_SURB_STORE_EVICTIONS);
464            lazy_static::initialize(&METRIC_SURB_STORE_EVICTIONS_LAST_INTERVAL);
465        }
466        let stats = Arc::new(EvictionStats::new(&cfg));
467        Self {
468            // Reply openers are indexed by entire Sender IDs (Pseudonym + SURB ID)
469            // in a cascade fashion, allowing the entire batches (by Pseudonym) to be evicted
470            // if not used.
471            pseudonym_openers: moka::sync::Cache::builder()
472                .time_to_idle(cfg.pseudonyms_lifetime.max(MINIMUM_SURB_LIFETIME))
473                .eviction_policy(moka::policy::EvictionPolicy::lru())
474                .eviction_listener({
475                    let stats = stats.clone();
476                    move |_pseudonym, _openers, cause| stats.record(EvictedCache::ReplyOpenerBatch, cause)
477                })
478                .max_capacity(cfg.max_openers_per_pseudonym.max(MINIMUM_OPENER_PSEUDONYMS) as u64)
479                .build(),
480            // SURBs are indexed only by Pseudonyms, which have longer lifetimes.
481            // For each Pseudonym, there's an RB of SURBs and their IDs.
482            surbs_per_pseudonym: moka::sync::Cache::builder()
483                .time_to_idle(cfg.pseudonyms_lifetime.max(MINIMUM_SURB_LIFETIME))
484                .eviction_policy(moka::policy::EvictionPolicy::lru())
485                .eviction_listener({
486                    let stats = stats.clone();
487                    move |_pseudonym, _surbs, cause| stats.record(EvictedCache::SurbRing, cause)
488                })
489                .max_capacity(cfg.max_pseudonyms.max(MINIMUM_SURBS_PER_PSEUDONYM) as u64)
490                .build(),
491            invalidated_relayers: Default::default(),
492            generations: moka::sync::Cache::builder()
493                // Sending-side state, so it is sized and aged like the sibling sending-side
494                // `pseudonym_openers` cache — NOT like the receiving-side `surbs_per_pseudonym`.
495                // Retained for the reply-opener window (>= the peer's SURB idle window) so the serial
496                // cannot reset to 0 while the peer still holds SURBs it numbers, and bounded by the
497                // reply-opener pseudonym capacity (which covers `maximum_managed_sessions`) so LRU
498                // pressure from the receiving-side `max_pseudonyms` cannot evict a live sender's
499                // generation. See the field doc for why an early eviction would strand the reply path.
500                .time_to_idle(cfg.reply_opener_lifetime.max(MINIMUM_OPENER_LIFETIME))
501                .eviction_policy(moka::policy::EvictionPolicy::lru())
502                .eviction_listener({
503                    let stats = stats.clone();
504                    move |pseudonym, _generation, cause| {
505                        // Minting keeps this warm, so each eviction earns a line: a reset serial has the peer reject
506                        // SURBs.
507                        tracing::warn!(%pseudonym, ?cause, "evicting SURB generation for pseudonym");
508                        stats.record(EvictedCache::Generation, cause);
509                    }
510                })
511                .max_capacity(cfg.max_openers_per_pseudonym.max(MINIMUM_OPENER_PSEUDONYMS) as u64)
512                .build(),
513            stats,
514            cfg: cfg.into(),
515        }
516    }

518    /// Whether `relayer` is currently unusable as a return path's first hop.
519    pub fn is_relayer_invalidated(&self, relayer: &HoprKeyIdent) -> bool {
520        self.invalidated_relayers.read().contains(relayer)
521    }
522
523    /// Whether a stored SURB can still be used to reply: its first relayer must still be payable.
524    ///
525    /// A direct return path is exempt — its "first relayer" is the final recipient, which needs no
526    /// channel (RFC-0003 §3.2, RFC-0006 §6.1). Without that exemption, closing an unrelated channel
527    /// to a session's originator would discard perfectly good SURBs.
528    fn is_surb_usable(&self, surb: &HoprSurb) -> bool {
529        match surb.additional_data_receiver.proof_of_relay_values().chain_length() {
530            // Direct return path: the "first relayer" is the final recipient, which needs no channel.
531            1 => true,
532            // A chain length is hops + 1, so 0 cannot occur on a well-formed SURB. Refuse it rather
533            // than let a malformed value pass as "direct" and bypass the check below.
534            //
535            // The length is an unvalidated byte off a SURB minted by the counterparty, so this is a
536            // statement about that peer, not a local fault we could act on: `warn`, not `error`.
537            0 => {
538                tracing::warn!(
539                    first_relayer = %surb.first_relayer,
540                    "refusing a malformed SURB declaring a zero-length return path"
541                );
542                false
543            }
544            _ => {
545                let usable = !self.invalidated_relayers.read().contains(&surb.first_relayer);
546                if !usable {
547                    tracing::trace!(
548                        first_relayer = %surb.first_relayer,
549                        "refusing a SURB whose first relayer is invalidated"
550                    );
551                }
552                usable
553            }
554        }
555    }

565    #[tracing::instrument(skip_all, level = "trace", fields(?matcher), ret)]
566    fn find_surb(&self, matcher: SurbMatcher) -> Option<FoundSurb> {
567        let pseudonym = matcher.pseudonym();
568        let surbs_for_pseudonym = self.surbs_per_pseudonym.get(&pseudonym)?;
569
570        match matcher {
571            // SURBs whose return path no longer has a usable first edge are dropped on the way,
572            // rather than handed out only to have the reply fail to be paid for.
573            SurbMatcher::Pseudonym(_) => surbs_for_pseudonym
574                .pop_next_valid(|_, surb| self.is_surb_usable(surb))
575                .map(|popped_surb| FoundSurb {
576                    sender_id: HoprSenderId::from_pseudonym_and_id(&pseudonym, popped_surb.id),
577                    surb: popped_surb.surb,
578                    remaining: popped_surb.remaining,
579                }),
580            // The following code intentionally only checks the SURB at the popping end of the
581            // ring buffer and does not search the entire RB.
582            // This is because the exact match use-case is suited only for situations
583            // when there is a single SURB in the RB.
584            SurbMatcher::Exact(id) => {
585                surbs_for_pseudonym
586                    .pop_one_if_has_id(&id.surb_id())
587                    .map(|popped_surb| FoundSurb {
588                        sender_id: HoprSenderId::from_pseudonym_and_id(&pseudonym, popped_surb.id),
589                        surb: popped_surb.surb,
590                        remaining: popped_surb.remaining, // = likely 0
591                    })
592            }
593        }
594    }
595
596    #[tracing::instrument(skip_all, level = "trace", fields(%pseudonym, num_surbs = surbs.len()))]
597    fn insert_surbs(&self, pseudonym: HoprPseudonym, mut surbs: Vec<(HoprSurbId, HoprSurb)>) -> SurbInsertOutcome {
598        // A batch is one packet's worth of SURBs, minted by the creator at a single generation, so
599        // the generation of the first stands for the whole batch. An empty batch carries no
600        // generation and must not create or disturb the buffer.
601        let Some(generation) = surbs
602            .first()
603            .map(|(_, surb)| surb.additional_data_receiver.generation())
604        else {
605            return SurbInsertOutcome {
606                retained: self.surbs_per_pseudonym.get(&pseudonym).map(|rb| rb.len()).unwrap_or(0),
607                evicted: 0,
608            };
609        };
610
611        // That "single generation" holds by construction only for a batch minted by an honest
612        // creator: `PacketRouting::ForwardPath` stamps one generation into every SURB it mints. On
613        // this side the batch is parsed out of a counterparty-controlled payload, so enforce it
614        // rather than assume it — otherwise a mixed batch smuggles SURBs for a superseded return
615        // path into a buffer the push below labels with the newer generation, where no later push
616        // can clear them. The first SURB always survives, so the buffer is never created empty.
617        let mixed = surbs.len();
618        surbs.retain(|(_, surb)| surb.additional_data_receiver.generation() == generation);
619        let dropped = mixed - surbs.len();
620        if dropped > 0 {
621            // A statement about the peer that minted the batch, not a local fault: `warn`, not
622            // `error`. These are not capacity pressure, so they stay out of the `evicted` count.
623            tracing::warn!(
624                %pseudonym,
625                dropped,
626                generation,
627                "discarding SURBs whose generation disagrees with the rest of their batch"
628            );
629        }
630
631        self.surbs_per_pseudonym
632            .entry_by_ref(&pseudonym)
633            .or_insert_with(|| SurbRingBuffer::new(self.cfg.rb_capacity.max(MIN_SURB_RB_CAPACITY), self.cfg.pop_order))
634            .value()
635            .push(surbs, generation)
636    }
637
638    #[tracing::instrument(skip_all, level = "trace", fields(?sender_id))]
639    fn insert_reply_opener(&self, sender_id: HoprSenderId, opener: ReplyOpener) {
640        let opener_lifetime = self.cfg.reply_opener_lifetime.max(MINIMUM_OPENER_LIFETIME);
641        let max_openers_per_pseudonym = self.cfg.max_openers_per_pseudonym.max(MINIMUM_OPENERS_PER_PSEUDONYM);
642        let stats = self.stats.clone();
643        self.pseudonym_openers
644            .get_with(sender_id.pseudonym(), move || {
645                moka::sync::Cache::builder()
646                    .time_to_live(opener_lifetime)
647                    // Keep the newest openers, not the stalest. Reply openers are written once and
648                    // never read before they are used, so under the default TinyLFU policy every
649                    // entry ties at frequency zero and incumbents win admission: a full cache then
650                    // freezes the oldest openers and drops every newer one. But the counterparty's
651                    // SURB ring always hands back the newest SURBs, whose openers are exactly those
652                    // dropped — so replies stop decrypting once the cache fills. LRU on this
653                    // write-only workload evicts by insertion order, i.e. it sheds the oldest and
654                    // keeps the newest, mirroring the SURB ring buffer (and the outer cache).
655                    .eviction_policy(moka::policy::EvictionPolicy::lru())
656                    .eviction_listener(move |_id, _opener, cause| stats.record(EvictedCache::ReplyOpener, cause))
657                    .max_capacity(max_openers_per_pseudonym as u64)
658                    .build()
659            })
660            .insert(sender_id.surb_id(), opener);
661    }

663    #[tracing::instrument(skip_all, level = "trace")]
664    fn invalidate_relayer(&self, relayer: &HoprKeyIdent) {
665        if self.invalidated_relayers.write().insert(*relayer) {
666            tracing::info!(
667                %relayer,
668                "invalidating stored SURBs whose return path starts with this relayer"
669            );
670        }
671    }

673    #[tracing::instrument(skip_all, level = "trace")]
674    fn revalidate_relayer(&self, relayer: &HoprKeyIdent) {
675        if self.invalidated_relayers.write().remove(relayer) {
676            tracing::info!(%relayer, "relayer is usable again for SURB return paths");
677        }
678    }
679
680    #[tracing::instrument(skip_all, level = "trace", fields(?sender_id), ret)]
681    fn find_reply_opener(&self, sender_id: &HoprSenderId) -> Option<ReplyOpener> {
682        self.pseudonym_openers
683            .get(&sender_id.pseudonym())
684            .and_then(|cache| cache.remove(&sender_id.surb_id()))
685    }
686
687    fn current_generation(&self, pseudonym: &HoprPseudonym) -> u8 {
688        self.generations
689            .get(pseudonym)
690            .map(|g| g.load(std::sync::atomic::Ordering::Relaxed))
691            .unwrap_or(0)
692    }
693
694    #[tracing::instrument(skip_all, level = "trace", fields(%pseudonym), ret)]
695    fn bump_generation(&self, pseudonym: &HoprPseudonym) -> u8 {
696        // `fetch_add` returns the previous value; the new generation is one past it. A `u8` serial
697        // wraps cleanly (255 -> 0) and the replying side compares with RFC-1982 arithmetic.
698        self.generations
699            .get_with(*pseudonym, || Arc::new(std::sync::atomic::AtomicU8::new(0)))
700            .fetch_add(1, std::sync::atomic::Ordering::Relaxed)
701            .wrapping_add(1)
702    }

705/// Represents a single SURB along with its ID popped from the [`SurbRingBuffer`].
706#[derive(Debug, Clone)]
707pub struct PoppedSurb<S> {
708    /// Complete SURB sender ID.
709    pub id: HoprSurbId,
710    /// The popped SURB.
711    pub surb: S,
712    /// Number of SURBs left in the RB after the pop.
713    pub remaining: usize,
714}
715
716/// RFC-1982 serial-number comparison over a `u8` generation: is `a` strictly newer than `b`?
717///
718/// SURB generations are minted monotonically by the creator and only ever compared across the two
719/// adjacent generations that can be in flight at once, so a `u8` serial (window 128) is ample and
720/// wraps cleanly: 255 → 0 reads as newer.
721fn generation_is_newer(a: u8, b: u8) -> bool {
722    a != b && a.wrapping_sub(b) < 128
723}
724
725/// Ring buffer of SURBs and their IDs, all belonging to one pseudonym and therefore identified only
726/// by [`HoprSurbId`].
727///
728/// Backed by a [`VecDeque`] that is never allowed to exceed `capacity`: a push into a full buffer
729/// evicts the oldest element first. [`SurbPopOrder`] picks which end a pop consumes from; overflow
730/// always evicts the oldest, in either order.
731///
732/// ## Generations: dropping SURBs for a superseded return path
733///
734/// A return path that dies deep in a multi-hop route is invisible to this replying side — nothing
735/// here can tell a stale SURB from a live one. The SURB creator can: it stamps every SURB of a
736/// batch with a generation (`SurbReceiverInfo::generation`) and bumps it whenever it changes the
737/// return path. This buffer keeps only the highest generation it has seen: the first push carrying a
738/// newer generation **clears the buffer wholesale** before inserting, so a return-path change takes
739/// effect on the very next reply rather than only once the stale SURBs drain — and stale SURBs are
740/// never handed out. A push carrying an older generation (a late/reordered batch) is discarded.
741/// Clearing is a per-path-change O(n) sweep, so pops need no per-SURB generation check.
742///
743/// ## Why the capacity is a ceiling, not a reservation
744///
745/// The deque grows with occupancy rather than being sized at construction. The distinction matters
746/// because the pseudonym a buffer is filed under is chosen by whoever sent the packet: any
747/// `HoprPacket::Final` carrying a SURB reaches `insert_surbs`, which mints a buffer for a pseudonym
748/// it has never seen, with no Session or handshake behind it.
749///
750/// A structure that took its whole capacity upfront therefore let an unauthenticated peer reserve
751/// `rb_capacity` × `size_of::<(HoprSurbId, S)>()` of address space per pseudonym it invented —
752/// ~16.8 MB each at the default capacity, and `max_pseudonyms` of those. Resident memory was never
753/// the problem (untouched pages cost nothing), but the reservation is real to anything that
754/// accounts address space: strict overcommit, `ulimit -v`, `vm.max_map_count`.
755///
756/// Growth is geometric and amortised, and the deque never shrinks below its high-water mark, so a
757/// pseudonym that genuinely fills up still ends at the same footprint — and stops reallocating
758/// there, however long the steady-state overflow runs. It just has to earn it.
759#[derive(Clone, Debug)]
760pub struct SurbRingBuffer<S> {
761    inner: Arc<parking_lot::Mutex<GenerationalBuffer<S>>>,
762    /// Ceiling on the number of retained SURBs; the oldest are dropped once a push would exceed it.
763    capacity: usize,
764    /// Which end a pop consumes from; overflow always evicts the oldest regardless.
765    pop_order: SurbPopOrder,
766}
767
768/// The mutex-protected state of a [`SurbRingBuffer`]: the SURBs and the highest generation seen.
769///
770/// Both live under one lock so that clearing the deque and advancing the generation on a newer
771/// batch is atomic against a concurrent pop.
772#[derive(Debug)]
773struct GenerationalBuffer<S> {
774    surbs: VecDeque<(HoprSurbId, S)>,
775    /// Highest generation seen; `None` until the first push.
776    generation: Option<u8>,
777}
778
779impl<S> GenerationalBuffer<S> {
780    /// The end a pop consumes from under `order` (front for FIFO, back for LIFO). The single place
781    /// the order-to-end polarity is decided, so [`pop_end`](Self::pop_end) and its peek stay in step.
782    fn peek_end(&self, order: SurbPopOrder) -> Option<&(HoprSurbId, S)> {
783        match order {
784            SurbPopOrder::Fifo => self.surbs.front(),
785            SurbPopOrder::Lifo => self.surbs.back(),
786        }
787    }
788
789    /// Removes and returns the SURB at the end `order` consumes from.
790    fn pop_end(&mut self, order: SurbPopOrder) -> Option<(HoprSurbId, S)> {
791        match order {
792            SurbPopOrder::Fifo => self.surbs.pop_front(),
793            SurbPopOrder::Lifo => self.surbs.pop_back(),
794        }
795    }
796}
797
798impl<S> SurbRingBuffer<S> {
799    /// Creates a buffer holding at most `capacity` (min 1, so a push is never a no-op) SURBs,
800    /// popped in the given order.
801    pub fn new(capacity: usize, pop_order: SurbPopOrder) -> Self {
802        Self {
803            inner: Arc::new(parking_lot::Mutex::new(GenerationalBuffer {
804                surbs: VecDeque::new(),
805                generation: None,
806            })),
807            // A zero capacity would make the eviction below pop from an empty deque and then push
808            // past the bound. Callers already clamp to `MIN_SURB_RB_CAPACITY`; this is belt-and-braces.
809            capacity: capacity.max(1),
810            pop_order,
811        }
812    }
813
814    /// Pushes all SURBs of one batch, stamped with `generation`, evicting the oldest past capacity.
815    ///
816    /// A batch is minted at a single generation by the creator (one packet's worth of SURBs), so a
817    /// single `generation` covers the whole `surbs` iterator. Relative to the highest generation
818    /// seen so far:
819    /// - **newer** → the buffer is cleared before inserting, so SURBs for the superseded return path are dropped at
820    ///   once rather than lingering until they drain;
821    /// - **equal** → the batch is appended (an ordinary refill);
822    /// - **older** → the batch is discarded as a late/reordered leftover.
823    ///
824    /// Once at capacity, each insert evicts the oldest SURB. Under PIX that is a lost SSA share, not
825    /// merely a lost SURB — see [`SurbStoreConfig::rb_capacity`].
826    ///
827    /// Returns what the push did; the eviction count is what lets a caller notice the overflow at
828    /// all, since dropping the oldest entry is otherwise indistinguishable from a clean insert.
829    /// It counts *capacity* overflow only: SURBs dropped because a newer generation superseded them
830    /// were already unusable, and reporting them as pressure would ask the creator to slow down
831    /// because it re-planned its own return path.
832    pub fn push<I: IntoIterator<Item = (HoprSurbId, S)>>(&self, surbs: I, generation: u8) -> SurbInsertOutcome {
833        let mut inner = self.inner.lock();
834
835        match inner.generation {
836            Some(current) if generation == current => {} // ordinary refill: append below
837            Some(current) if generation_is_newer(generation, current) => {
838                // Return path changed: everything held is for the superseded path. Drop it wholesale
839                // so the newer batch is all that remains and the next reply uses the live path.
840                inner.surbs.clear();
841                inner.generation = Some(generation);
842            }
843            Some(_) => {
844                // Older than what we already hold: a late or reordered batch for a path the creator
845                // has already moved on from. Discard it rather than reintroduce stale SURBs.
846                return SurbInsertOutcome {
847                    retained: inner.surbs.len(),
848                    evicted: 0,
849                };
850            }
851            None => inner.generation = Some(generation),
852        }
853
854        let mut evicted = 0;
855        for surb in surbs {
856            // Evict before inserting, so the length never exceeds the ceiling and the backing
857            // allocation stops growing once the high-water mark is reached.
858            if inner.surbs.len() >= self.capacity {
859                inner.surbs.pop_front();
860                evicted += 1;
861            }
862            inner.surbs.push_back(surb);
863        }
864        SurbInsertOutcome {
865            retained: inner.surbs.len(),
866            evicted,
867        }
868    }
869
870    /// Pops the next SURB that `is_valid` accepts, in the buffer's [`SurbPopOrder`].
871    ///
872    /// **Destructive:** rejected entries are discarded, not skipped, so an unusable SURB neither is
873    /// handed out nor blocks those behind it. Pass only a validity test — a selective predicate
874    /// (say, a routing preference) would drain the buffer. `None` once it is exhausted without a
875    /// match.
876    ///
877    /// `is_valid` runs *outside* the lock: it is caller-supplied and may take locks of its own, so
878    /// calling it inside the critical section would invite lock-order inversion.
879    pub fn pop_next_valid<F>(&self, is_valid: F) -> Option<PoppedSurb<S>>
880    where
881        F: Fn(&HoprSurbId, &S) -> bool,
882    {
883        loop {
884            let (id, surb, remaining) = {
885                let mut inner = self.inner.lock();
886                let (id, surb) = inner.pop_end(self.pop_order)?;
887                (id, surb, inner.surbs.len())
888            };
889
890            if is_valid(&id, &surb) {
891                return Some(PoppedSurb { id, surb, remaining });
892            }
893        }
894    }
895
896    /// Number of SURBs currently held.
897    fn len(&self) -> usize {
898        self.inner.lock().surbs.len()
899    }
900
901    /// Pops the next SURB (in the buffer's [`SurbPopOrder`]) only if it has the given ID.
902    pub fn pop_one_if_has_id(&self, id: &HoprSurbId) -> Option<PoppedSurb<S>> {
903        let mut inner = self.inner.lock();
904
905        if inner.peek_end(self.pop_order).is_some_and(|(surb_id, _)| surb_id == id) {
906            let (id, surb) = inner.pop_end(self.pop_order)?;
907            Some(PoppedSurb {
908                id,
909                surb,
910                remaining: inner.surbs.len(),
911            })
912        } else {
913            None
914        }
915    }
916}

1401    ///
1402    /// One session pseudonym uploads hard: it keeps minting SURBs (one reply opener stored per SURB
1403    /// on the *client*, one SURB stored on the *exit*) while the exit barely replies, so nothing is
1404    /// consumed and both stores run to capacity and start evicting. `flood` pairs are pushed, then

1406    /// to open it on the client. Returns how many of those replies land on a reply opener the client
1407    /// has already evicted, i.e. how many replies the client cannot decrypt.

1410    /// client cares only about `max_openers` (its reply-opener cache), the exit only about
1411    /// `rb_capacity` + `order` (its SURB ring).

1413        max_openers: usize,

1416        flood: usize,
1417        replies: usize,
1418    ) -> usize {

1421            rb_capacity,
1422            pop_order: order,
1423            ..Default::default()
1424        });
